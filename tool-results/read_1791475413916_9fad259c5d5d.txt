   192→   192→function hashT(i: number, k: number): number {
   193→   193→  let h = (i + k * 57) * 2654435761;
   194→   194→  h = (h ^ (h >> 13)) * 1274126177;
   195→   195→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;

   203→   203→function hoverCorners(g: Game, x: number, y: number, w: number, h: number, col = COL.gold) {
   204→   204→  const hov = g.mouse.x >= x && g.mouse.x <= x + w && g.mouse.y >= y && g.mouse.y <= y + h;
   205→   205→  if (!hov) return;
   206→   206→  const ctx = g.ctx;

   208→   208→  ctx.fillStyle = col;
   209→   209→  ctx.fillRect(x - 2, y - 2, c, 2); ctx.fillRect(x - 2, y - 2, 2, c);
   210→   210→  ctx.fillRect(x + w + 2 - c, y - 2, c, 2); ctx.fillRect(x + w, y - 2, 2, c);
   211→   211→  ctx.fillRect(x - 2, y + h, c, 2); ctx.fillRect(x - 2, y + h - c, 2, c);
   212→   212→  ctx.fillRect(x + w + 2 - c, y + h, c, 2); ctx.fillRect(x + w, y + h - c, 2, c);
   213→   213→}
