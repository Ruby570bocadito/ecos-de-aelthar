    24→const WORLD_FILTER: Record<string, string> = {
    25→  presente: 'saturate(0.74) contrast(0.98)',

    29→export function drawGame(g: Game) {
    30→  const ctx = g.ctx;
    31→  clearHits(g);
    32→  const dtF = fxFrame(g); // dt de dibujado + decaimiento seguro del shake
    33→  ctx.imageSmoothingEnabled = false;
    34→  ctx.clearRect(0, 0, VIEW_W, VIEW_H);
    35→
    36→  if (g.state === 'title' || g.state === 'controls') {
    37→    drawScreens(g);
    38→    return;
    39→  }
    40→
    41→  // árbol de habilidades: pantalla propia a pantalla completa (agente 12-b)
    42→  if (g.state === 'skills') {
    43→    drawSkillTree(g);
    44→    return;
    45→  }
    46→
    47→  if (g.state === 'play' || g.state === 'dialogue') {
    48→    updateAmbient(g, dtF);
    49→  }
    50→
    51→  drawWorld(g);
    52→  drawHud(g);
    53→  drawOverlays(g); // destello, memoria, banner de jefe (sobre el HUD)
    54→  drawScreens(g); // intro, pause, dialogue, dead, end (overlays)
    55→}

    59→function drawWorld(g: Game) {
    60→  const ctx = g.ctx;
    61→  // sacudida de cámara (g.shake decae en update y, por seguridad, en fxFrame)
    62→  const shx = g.shake > 0.2 ? (Math.random() - 0.5) * g.shake * 2 : 0;
    63→  const shy = g.shake > 0.2 ? (Math.random() - 0.5) * g.shake * 2 : 0;
    64→  ctx.save();
    65→  ctx.translate(shx, shy);
    66→  const camX = Math.round(g.camX), camY = Math.round(g.camY);
    67→
    68→  // suelo (prerenderizado por época) — el canvas está en píxeles de mundo (1x)
    69→  const ground = g.epoch === 'pasado' && g.groundPastCanvas ? g.groundPastCanvas : g.groundCanvas;
    70→  if (ground) {
    71→    ctx.save();
    72→    ctx.filter = WORLD_FILTER[g.epoch] ?? 'none';
    73→    const gx = Math.floor(camX / ZOOM), gy = Math.floor(camY / ZOOM);
    74→    ctx.drawImage(ground, gx, gy, VIEW_W / ZOOM, VIEW_H / ZOOM, 0, 0, VIEW_W, VIEW_H);
    75→    ctx.restore();
    76→  }
    77→
    78→  const sx = (wx: number) => wx * ZOOM - camX;
    79→  const sy = (wy: number) => wy * ZOOM - camY;
    80→
    81→  // agua viva: brillos especulares sobre los tiles '~' visibles (R3-c)
    82→  drawWaterGlints(g, sx, sy);
    83→
    84→  ctx.save();
    85→  ctx.filter = WORLD_FILTER[g.epoch] ?? 'none';
    86→
    87→  // oro perdido
    88→  for (const dgl of g.deadGolds) {
    89→    if (dgl.map !== g.mapId) continue;
    90→    const bob = Math.sin(g.globalT * 3) * 2;
    91→    ctx.drawImage(getSpr('goldbag')[0], sx(dgl.x - 5), sy(dgl.y - 8 + bob), 10 * ZOOM, 8 * ZOOM);
    92→  }
    93→
    94→  // ecos menores (susurros)
    95→  for (const ec of g.map.echoes) {
    96→    if (g.takenEchoes.has(ec.id)) continue;
    97→    const bob = Math.sin(g.globalT * 2 + ec.x) * 2;
    98→    const fr = Math.floor(g.globalT * 3) % 2;
    99→    ctx.globalAlpha = 0.75;
   100→    ctx.drawImage(getSpr('wisp')[fr], sx(ec.x * TILE + 8 - 5), sy(ec.y * TILE + 2 + bob), 10 * ZOOM, 10 * ZOOM);
   101→    ctx.globalAlpha = 1;
   102→  }
   103→
   104→  // props
   105→  drawProps(g, sx, sy);
   106→
   107→  // estelas de esquiva (afterimages del jugador, bajo las entidades)
   108→  drawRollTrail(g, sx, sy);
   109→
   110→  // entidades ordenadas por Y
   111→  type Drawable = { e: Entity; y: number };
   112→  const ents: Drawable[] = [];
   113→  for (const n of g.npcs) ents.push({ e: n, y: n.y });
   114→  if (g.companion && g.companion.downT <= 0) ents.push({ e: g.companion, y: g.companion.y });
   115→  for (const e of g.enemies) if (!e.dead) ents.push({ e, y: e.y });
   116→  if (g.player) ents.push({ e: g.player, y: g.player.y });
   117→  ents.sort((a, b) => a.y - b.y);
   118→
   119→  for (const { e } of ents) {
   120→    drawEntity(g, e, sx, sy);
   121→  }
   122→
   123→  ctx.restore();
   124→
   125→  // cofres (encima, siempre visibles)
   126→  for (const ch of g.map.chests) {
   127→    if (ch.needPast && g.epoch !== 'pasado') continue;
   128→    const opened = g.openedChests.has(ch.id);
   129→    const sprC = getSpr(opened ? 'chest_open' : 'chest')[0];
   130→    ctx.drawImage(sprC, sx(ch.x * TILE), sy(ch.y * TILE - 2), 16 * ZOOM, 14 * ZOOM);
   131→  }
   132→
   133→  // fauna y viajante del mundo vivo (13-b) — capa ambiente bajo las entidades
   134→  drawWorldLife(g, sx, sy);
   135→
   136→  drawCombatFx(g, sx, sy);
   137→
   138→  // partículas
   139→  for (const p of g.particles) {
   140→    ctx.globalAlpha = Math.max(0, p.t / p.maxT);
   141→    ctx.fillStyle = p.color;
   142→    ctx.fillRect(sx(p.x) - p.size, sy(p.y) - p.size, p.size * 2 * ZOOM * 0.6, p.size * 2 * ZOOM * 0.6);
   143→  }
   144→  ctx.globalAlpha = 1;
   145→
   146→  // partículas ambientales del mapa (motas, hojas, ceniza, niebla...)
   147→  drawAmbient(g, 'world');
   148→
   149→  // personalidad cromática por bioma (R3-c): siempre tras suelo/entidades
   150→  // y ANTES de la iluminación — la técnica offscreen de Task 4 queda intacta.
   151→  drawBiomeTint(g);
   152→
   153→  drawLighting(g);
   154→
   155→  // capa de cielo: estrellas, luna, antorchas (sobre la iluminación)
   156→  drawAmbient(g, 'sky');
   157→
   158→  // textos flotantes (después de la luz: siempre legibles)
   159→  drawFloats(g, sx, sy);
   160→
   161→  // prompt de interacción
   162→  drawInteractPrompt(g, sx, sy);
   163→
   164→  // banner del mapa
   165→  if (g.mapTitleT > 0) {
   166→    const a = Math.min(1, g.mapTitleT);
   167→    ctx.globalAlpha = Math.min(1, a * 1.5);
   168→    textShadow(g, g.map.name, VIEW_W / 2, 90, 18, COL.goldSoft, '#000', 'center', true);
   169→    text(g, g.map.subtitle, VIEW_W / 2, 120, 16, COL.dim, 'center');
   170→    ctx.globalAlpha = 1;
   171→  }
   172→
   173→  // aviso de época
   174→  if (g.epochFx > 0) {
   175→    const a = g.epochFx / 0.8;
   176→    ctx.fillStyle = g.epoch === 'pasado' ? `rgba(255, 216, 138, ${a * 0.5})` : `rgba(120, 140, 190, ${a * 0.5})`;
   177→    ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   178→    if (a > 0.4) {
   179→      textShadow(g, g.epoch === 'pasado' ? '◆ EL PASADO ◆' : '◆ EL PRESENTE ◆', VIEW_W / 2, 180, 16, g.epoch === 'pasado' ? COL.epochPast : COL.epochNow, '#000', 'center', true);
   180→    }
   181→  }
   182→
   183→  // fade de transición
   184→  if (g.fadeT > 0) {
   185→    ctx.fillStyle = `rgba(4,4,10,${g.fadeT})`;
   186→    ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   187→  }
   188→  ctx.restore();
   189→}

   193→function drawRollTrail(g: Game, sx: (n: number) => number, sy: (n: number) => number) {
   194→  const ctx = g.ctx;
   195→  const p = g.player;
   196→  if (!p) return;
   197→  const trail = getRollTrail();
   198→  if (trail.length === 0) return;
   199→  const spr = getSpr(p.sprite);
   200→  for (const tp of trail) {
   201→    const a = Math.max(0, tp.life / TRAIL_LIFE) * 0.35;
   202→    if (a <= 0.02) continue;
   203→    const fr = Math.min(entityFrame(spr, tp.dir, true, tp.anim), spr.length - 1);
   204→    const dx = sx(tp.x) - (spr[0].width * ZOOM) / 2;
   205→    const dy = sy(tp.y + 4) - spr[0].height * ZOOM;
   206→    ctx.save();
   207→    ctx.filter = WORLD_FILTER[g.epoch] ?? 'none';
   208→    ctx.globalAlpha = a;
   209→    if (tp.dir === 'left') {
   210→      ctx.translate(dx + spr[0].width * ZOOM, 0);
   211→      ctx.scale(-1, 1);
   212→      ctx.drawImage(spr[fr], 0, dy, spr[0].width * ZOOM, spr[0].height * ZOOM);
   213→    } else {
   214→      ctx.drawImage(spr[fr], dx, dy, spr[0].width * ZOOM, spr[0].height * ZOOM);
   215→    }
   216→    ctx.restore();
   217→  }
   218→  ctx.globalAlpha = 1;
   219→}

   223→function drawProps(g: Game, sx: (n: number) => number, sy: (n: number) => number) {
   224→  const ctx = g.ctx;
   225→  for (const pr of g.map.props) {
   226→    if (pr.needPast && g.epoch !== 'pasado') continue;
   227→    if (pr.needPresent && g.epoch !== 'presente') continue;
   228→    const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   229→    if (pr.kind === 'sanctuary') {
   230→      const fr = Math.floor(g.globalT * 2) % 2;
   231→      const s = getSpr('sanctuary')[fr];
   232→      ctx.drawImage(s, sx(px - 10), sy(py - 22), 20 * ZOOM, 30 * ZOOM);
   233→      // aura (R3-c: en mapas oscuros el pulso llega un poco más lejos, r 26→30)
   234→      const auraR = g.map.dark ? 28 + Math.sin(g.globalT * 2) * 2 : 26;
   235→      ctx.globalAlpha = 0.18 + Math.sin(g.globalT * 2) * 0.08;
   236→      ctx.fillStyle = '#8ef0ff';
   237→      ctx.beginPath();
   238→      ctx.arc(sx(px), sy(py - 6), auraR, 0, Math.PI * 2);
   239→      ctx.fill();
   240→      ctx.globalAlpha = 1;
   241→    } else if (pr.kind === 'forge') {
   242→      // yunque + brasa
   243→      ctx.fillStyle = '#4a4a58';
   244→      ctx.fillRect(sx(px - 6), sy(py - 4), 12 * ZOOM, 6 * ZOOM);
   245→      ctx.fillStyle = '#5a5a6a';
   246→      ctx.fillRect(sx(px - 8), sy(py - 6), 6 * ZOOM, 4 * ZOOM);
   247→      const fl = 0.6 + Math.sin(g.globalT * 7) * 0.3;
   248→      ctx.fillStyle = `rgba(255,120,40,${fl})`;
   249→      ctx.fillRect(sx(px - 3), sy(py - 9), 6 * ZOOM, 5 * ZOOM);
   250→      ctx.fillStyle = `rgba(255,200,80,${fl})`;
   251→      ctx.fillRect(sx(px - 2), sy(py - 8), 4 * ZOOM, 3 * ZOOM);
   252→    } else if (pr.kind === 'fragment') {
   253→      const fr = Math.floor(g.globalT * 4) % 3;
   254→      const bob = Math.sin(g.globalT * 2.4) * 3;
   255→      ctx.globalAlpha = 0.25;
   256→      ctx.fillStyle = '#ffe9a0';
   257→      ctx.beginPath();
   258→      ctx.arc(sx(px), sy(py - 6 + bob), 18, 0, Math.PI * 2);
   259→      ctx.fill();
   260→      ctx.globalAlpha = 1;
   261→      ctx.drawImage(getSpr('fragment')[fr], sx(px - 6), sy(py - 12 + bob), 12 * ZOOM, 12 * ZOOM);
   262→    } else if (pr.kind === 'altarEcho') {
   263→      // altar con el Eco (si no recogido) — generalizado Acto II: cada altar
   264→      // consulta sus propias flags de eco/custodio según su id
   265→      const ecoFlag = pr.id === 'altar_mareas' ? 'ecoMareas' : pr.id === 'altar_cumbres' ? 'ecoCumbres' : 'ecoVoz';
   266→      const bossFlag = pr.id === 'altar_mareas' ? 'sirenaDefeated' : pr.id === 'altar_cumbres' ? 'golemDefeated' : 'guardianDefeated';
   267→      const ecoColor = pr.id === 'altar_mareas' ? '#8ef0ff' : pr.id === 'altar_cumbres' ? '#a8d8ff' : '#ffe9a0';
   268→      ctx.fillStyle = '#6a6a7a';
   269→      ctx.fillRect(sx(px - 7), sy(py - 2), 14 * ZOOM, 8 * ZOOM);
   270→      ctx.fillStyle = '#8a8a9a';
   271→      ctx.fillRect(sx(px - 5), sy(py - 5), 10 * ZOOM, 4 * ZOOM);
   272→      if (!g.flags[ecoFlag]) {
   273→        const bob = Math.sin(g.globalT * 2.6) * 3;
   274→        const gl = g.flags[bossFlag] ? 0.7 : 0.25;
   275→        ctx.globalAlpha = gl;
   276→        ctx.fillStyle = ecoColor;
   277→        ctx.beginPath();
   278→        ctx.arc(sx(px), sy(py - 12 + bob), 12, 0, Math.PI * 2);
   279→        ctx.fill();
   280→        ctx.globalAlpha = 1;
   281→        ctx.drawImage(getSpr('fragment')[Math.floor(g.globalT * 3) % 3], sx(px - 5), sy(py - 18 + bob), 10 * ZOOM, 10 * ZOOM);
   282→      }
   283→    } else if (pr.kind === 'sign') {
   284→      ctx.fillStyle = '#6d4520';
   285→      ctx.fillRect(sx(px - 1), sy(py - 6), 2 * ZOOM, 12 * ZOOM);
   286→      ctx.fillStyle = '#8a5a2b';
   287→      ctx.fillRect(sx(px - 7), sy(py - 13), 14 * ZOOM, 8 * ZOOM);
   288→      ctx.fillStyle = '#5c3a1e';
   289→      ctx.fillRect(sx(px - 6), sy(py - 11), 12 * ZOOM, 1.5 * ZOOM);
   290→      ctx.fillRect(sx(px - 6), sy(py - 9), 9 * ZOOM, 1.5 * ZOOM);
   291→    } else if (pr.kind === 'wreck' || pr.kind === 'faro' || pr.kind === 'lamp') {
   292→      // Acto II: props de la expansión (nave naufragada, faro, faroles de Merrow)
   293→      drawExpansionProp(ctx, pr.kind, sx(px), sy(py), ZOOM, g.globalT, !!g.flags[pr.id]);
   294→    }
   295→  }
   296→}

   300→function drawEntity(g: Game, e: Entity, sx: (n: number) => number, sy: (n: number) => number) {
   301→  const ctx = g.ctx;
   302→  const spr = getSpr(e.sprite);
   303→  const zoomW = spr[0].width, zoomH = spr[0].height;
   304→
   305→  // parpadeo suave del jugador durante i-frames (esquiva/daño reciente)
   306→  let spriteAlpha = 1;
   307→  if (e.kind === 'player') {
   308→    const pp = g.player!;
   309→    if (pp.iframes > 0 && pp.rollT <= 0) spriteAlpha = 0.55 + Math.abs(Math.sin(g.globalT * 24)) * 0.35;
   310→  }
   311→  // Acto II: fase intangible de enemigos (espectro/sirena sumergida)
   312→  if (e.kind === 'enemy' && (e as Enemy).invulT !== undefined && (e as Enemy).invulT! > 0) {
   313→    spriteAlpha = 0.3 + Math.abs(Math.sin(g.globalT * 14)) * 0.18;
   314→  }
   315→
   316→  // sombra (estable)
   317→  ctx.globalAlpha = 0.3;
   318→  ctx.fillStyle = '#000';
   319→  ctx.beginPath();
   320→  ctx.ellipse(sx(e.x), sy(e.y + 4), (e.w / 2 + 3) * ZOOM, 3 * ZOOM, 0, 0, Math.PI * 2);
   321→  ctx.fill();
   322→
   323→  const fi = entityFrame(spr, e.dir, e.moving, e.anim);
   324→  const idx = Math.min(fi, spr.length - 1);
   325→
   326→  // R3-c: poses de combate del Portador (contrato con 9-b). getAttackFrames /
   327→  // getCastFrames devuelven [frameAnticipación, frameGolpe] (mismo tamaño que
   328→  // los frames de andar) o null. Llamada opcional vía namespace: mientras 9-b
   329→  // no exista el fallback es el frame de andar — nunca rompe la compilación.
   330→  let poseCv: HTMLCanvasElement | null = null;
   331→  if (e.kind === 'player' && g.player && g.player.attackT > 0) {
   332→    const pl = g.player;
   333→    const dur = pl.chargedHit ? 0.4 : 0.26;
   334→    const anticip = pl.attackT > dur * 0.5; // 1ª mitad: preparación · 2ª: golpe
   335→    const SP = SPRITES as unknown as {
   336→      getAttackFrames?: (base: string, dir: string) => HTMLCanvasElement[] | null;
   337→      getCastFrames?: (base: string, dir: string) => HTMLCanvasElement[] | null;
   338→    };
   339→    const frames = pl.discipline === 'tejedor' && SP.getCastFrames
   340→      ? (SP.getCastFrames(pl.sprite, pl.dir) ?? (SP.getAttackFrames ? SP.getAttackFrames(pl.sprite, pl.dir) : null))
   341→      : (SP.getAttackFrames ? SP.getAttackFrames(pl.sprite, pl.dir) : null);
   342→    if (frames && frames.length > 1) poseCv = frames[anticip ? 0 : 1] ?? null;
   343→  }
   344→
   345→  const flip = e.dir === 'left';
   346→  const dx = sx(e.x) - (zoomW * ZOOM) / 2;
   347→  const dy = sy(e.y + 4) - zoomH * ZOOM;
   348→
   349→  const frCv = poseCv ?? spr[idx];
   350→  ctx.globalAlpha = spriteAlpha;
   351→  if (flip) {
   352→    ctx.save();
   353→    ctx.translate(dx + zoomW * ZOOM, 0);
   354→    ctx.scale(-1, 1);
   355→    ctx.drawImage(frCv, 0, dy, zoomW * ZOOM, zoomH * ZOOM);
   356→    ctx.restore();
   357→  } else {
   358→    ctx.drawImage(frCv, dx, dy, zoomW * ZOOM, zoomH * ZOOM);
   359→  }
   360→  ctx.globalAlpha = 1;
   361→
   362→  // flash de daño
   363→  const en = e as Enemy;
   364→  if (en.hitFlash && en.hitFlash > 0) {
   365→    ctx.globalAlpha = 0.5;
   366→    ctx.fillStyle = '#fff';
   367→    ctx.fillRect(dx + 4, dy + 4, zoomW * ZOOM - 8, zoomH * ZOOM - 8);
   368→    ctx.globalAlpha = 1;
   369→  }
   370→
   371→  // telegrafía de ataque
   372→  if (en.kind === 'enemy' && en.ai === 'carga' && en.windup > 0) {
   373→    textShadow(g, '!', sx(e.x), dy - 16, 14, '#ff5040', '#000', 'center', true);
   374→  }
   375→  if (en.kind === 'enemy' && en.ai === 'aturdido') {
   376→    const t = g.globalT * 6;
   377→    text(g, '✦', sx(e.x) + Math.sin(t) * 6, dy - 14, 10, '#ffe86a', 'center');
   378→    if (en.maxSta > 0) text(g, 'QUEBRADO', sx(e.x), dy - 30, 8, '#ffe86a', 'center');
   379→  }
   380→
   381→  // barras de vida de enemigos dañados
   382→  if (en.kind === 'enemy' && en.hp < en.maxHp && !en.dead && en.etype !== 'guardian') {
   383→    bar(g, sx(e.x) - 12, dy - 6, 24, 3, en.hp / en.maxHp, COL.hp, COL.hpBg);
   384→  }
   385→
   386→  // iconos de estado con mini-barra de tiempo
   387→  if (en.kind === 'enemy' && !en.dead) {
   388→    drawStatusIcons(g, en, sx(e.x), dy - 12);
   389→  }
   390→
   391→  // indicador del jugador
   392→  if (e.kind === 'player') {
   393→    const p = g.player!;
   394→    // aura de época si puede cambiar
   395→    if (p.hasEcho && g.map.epochDiffs.length > 0) {
   396→      ctx.globalAlpha = 0.14 + Math.sin(g.globalT * 3) * 0.06;
   397→      ctx.fillStyle = g.epoch === 'pasado' ? '#ffd88a' : '#8ab8d8';
   398→      ctx.beginPath();
   399→      ctx.arc(sx(p.x), sy(p.y - 5), 15, 0, Math.PI * 2);
   400→      ctx.fill();
   401→      ctx.globalAlpha = 1;
   402→    }
   403→    if (p.parryFx > 0) {
   404→      ctx.globalAlpha = Math.min(1, p.parryFx * 3);
   405→      ctx.strokeStyle = '#fff8c0';
   406→      ctx.lineWidth = 2;
   407→      ctx.beginPath();
   408→      ctx.arc(sx(p.x), sy(p.y - 6), 14 + (0.32 - p.parryFx) * 30, 0, Math.PI * 2);
   409→      ctx.stroke();
   410→      ctx.globalAlpha = 1;
   411→    }
   412→    if (p.charging && p.chargeT > 0.35) {
   413→      // indicador de carga: aro que crece + flecha
   414→      const k = Math.min(1, p.chargeT / 0.8);
   415→      ctx.globalAlpha = 0.5 + k * 0.4;
   416→      ctx.strokeStyle = '#ffe86a';

   699→function drawLighting(g: Game) {
   700→  const ctx = g.ctx;
   701→  const p = g.player;
   702→  if (!p) return;
   703→  // oscuridad por noche (rampa suave)
   704→  const dayLight = Math.max(0.1, Math.sin(g.dayT * Math.PI * 2) * 1.25 + 0.25);
   705→  let darkness = (1 - Math.min(1, dayLight)) * 0.62;
   706→  if (g.map.dark) {
   707→    // la cripta respira como brasas lejanas: dos senos superpuestos de
   708→    // frecuencias distintas (más orgánico que un solo parpadeo) — R3-c
   709→    darkness = 0.8 + 0.02 * Math.sin(g.globalT * 7) + 0.015 * Math.sin(g.globalT * 13);
   710→  }
   711→  if (darkness > 0.02) {
   712→    const lc = getLightCanvas();
   713→    const lx = lc.ctx;
   714→    // 1) capa de oscuridad con alpha real (0.62 noche / 0.8 cripta)
   715→    lx.globalCompositeOperation = 'source-over';
   716→    lx.clearRect(0, 0, VIEW_W, VIEW_H);
   717→    lx.globalAlpha = Math.min(1, darkness);
   718→    lx.fillStyle = g.map.dark ? '#060810' : '#0a1030';
   719→    lx.fillRect(0, 0, VIEW_W, VIEW_H);
   720→    lx.globalAlpha = 1;
   721→    // 2) agujeros de luz sobre la capa (borran oscuridad, no mundo)
   722→    lx.globalCompositeOperation = 'destination-out';
   723→    const px = p.x * ZOOM - g.camX, py = (p.y - 6) * ZOOM - g.camY;
   724→    const grad = lx.createRadialGradient(px, py, 10, px, py, g.map.dark ? 130 : 170);
   725→    grad.addColorStop(0, 'rgba(0,0,0,0.95)');
   726→    grad.addColorStop(1, 'rgba(0,0,0,0)');
   727→    lx.fillStyle = grad;
   728→    lx.fillRect(px - 200, py - 200, 400, 400);
   729→    // santuarios iluminan (con pulso suave)
   730→    for (const pr of g.map.props) {
   731→      if (pr.kind !== 'sanctuary') continue;
   732→      const sxp = pr.x * TILE * ZOOM + 8 - g.camX, syp = pr.y * TILE * ZOOM - g.camY;
   733→      if (sxp < -100 || syp < -100 || sxp > VIEW_W + 100 || syp > VIEW_H + 100) continue;
   734→      const rad = 80 + Math.sin(g.globalT * 2 + pr.x) * 5;
   735→      const grad2 = lx.createRadialGradient(sxp, syp, 4, sxp, syp, rad);
   736→      grad2.addColorStop(0, 'rgba(0,0,0,0.7)');
   737→      grad2.addColorStop(1, 'rgba(0,0,0,0)');
   738→      lx.fillStyle = grad2;
   739→      lx.fillRect(sxp - 95, syp - 95, 190, 190);
   740→    }
   741→    lx.globalCompositeOperation = 'source-over';
   742→    // 3) componer la capa sobre el mundo: los huecos revelan el escenario
   743→    ctx.drawImage(lc.canvas, 0, 0);
   744→    if (g.map.dark) {
   745→      ctx.fillStyle = 'rgba(30,20,60,0.18)';
   746→      ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   747→    } else if (darkness > 0.25) {
   748→      // noche exterior: tinte azul frío
   749→      ctx.fillStyle = `rgba(16,22,52,${darkness * 0.22})`;
   750→      ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   751→    }
   752→  }
   753→
   754→  // amanecer/atardecer: tinte cálido en las transiciones del ciclo
   755→  if (!g.map.dark) {
   756→    const warm = Math.exp(-Math.pow(g.dayT - 0.52, 2) / (2 * 0.075 * 0.075));
   757→    if (warm > 0.05) {
   758→      ctx.fillStyle = `rgba(255,148,74,${warm * 0.13})`;
   759→      ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   760→    }
   761→  }
   762→
   763→  // niebla del presente (además de las bandas de fx.ts)
   764→  if (g.epoch === 'presente' && g.map.epochDiffs.length > 0) {
   765→    ctx.save();

   767→    ctx.fillStyle = '#9ec4b4';

   769→      const fx = ((g.globalT * 12 + i * 260) % (VIEW_W + 300)) - 150;
   770→      const fy = 80 + i * 110 + Math.sin(g.globalT * 0.8 + i) * 18;
   771→      ctx.beginPath();
   772→      ctx.ellipse(fx, fy, 130, 24, 0, 0, Math.PI * 2);
   773→      ctx.fill();
   774→    }
   775→    ctx.restore();
   776→  }
   777→  // destello de daño
   778→  const pl = g.player!;
   779→  if (pl.lastHitT > 0) {
   780→    ctx.fillStyle = `rgba(200,40,40,${pl.lastHitT * 0.7})`;
   781→    ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   782→  }

   784→  const vg = ctx.createRadialGradient(VIEW_W / 2, VIEW_H / 2, VIEW_H * 0.45, VIEW_W / 2, VIEW_H / 2, VIEW_H * 0.85);
   785→  vg.addColorStop(0, 'rgba(0,0,0,0)');
   786→  vg.addColorStop(1, 'rgba(0,0,0,0.32)');
   787→  ctx.fillStyle = vg;
   788→  ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   789→

   791→  const hpPct = pl.hp / pl.maxHp;
   792→  if (g.state === 'play' && hpPct < 0.3 && pl.hp > 0) {
   793→    const a = 0.18 + 0.06 * Math.sin(g.globalT * 5);
   794→    const rg = ctx.createRadialGradient(VIEW_W / 2, VIEW_H / 2, VIEW_H * 0.3, VIEW_W / 2, VIEW_H / 2, VIEW_H * 0.78);
   795→    rg.addColorStop(0, 'rgba(180,40,40,0)');

   797→    ctx.fillStyle = rg;

   799→  }

   806→function drawFloats(g: Game, sx: (n: number) => number, sy: (n: number) => number) {
   807→  const ctx = g.ctx;
   808→  for (const f of g.floats) {
   809→    const alpha = Math.min(1, f.t * 2.2);
   810→    if (alpha <= 0) continue;
   811→    // pop: más grande los primeros ~0.15 s
   812→    const pop = f.t > 0.74 ? 1.38 : 1;
   813→    const special = FLOAT_SPECIAL.has(f.text);
   814→    let size = f.size * 1.6 * pop;
   815→    let color = f.color;
   816→    if (special) { color = '#ffe86a'; size *= 1.3; }
   817→    else if (f.color === '#ffd24a') size *= 1.15; // críticos dorados
   818→    else if (f.color === '#ff7060') size *= 1.08; // daño propio
   819→    const x = sx(f.x), y = sy(f.y);
   820→    ctx.globalAlpha = alpha;
   821→    ctx.font = fBody(size);
   822→    ctx.textAlign = 'center';
   823→    ctx.textBaseline = 'top';
   824→    // contorno oscuro (4 direcciones + diagonal)
   825→    ctx.fillStyle = 'rgba(14,10,20,0.9)';
   826→    ctx.fillText(f.text, x + 1, y);
   827→    ctx.fillText(f.text, x - 1, y);
   828→    ctx.fillText(f.text, x, y + 1);
   829→    ctx.fillText(f.text, x, y - 1);
   830→    ctx.fillText(f.text, x + 1, y + 1);
   831→    ctx.fillStyle = color;
   832→    ctx.fillText(f.text, x, y);
   833→    if (special) {
   834→      // subrayado brillante para los remates
   835→      ctx.fillStyle = 'rgba(255,232,106,0.8)';
   836→      ctx.fillRect(x - size * f.text.length * 0.22, y + size * 1.05, size * f.text.length * 0.44, 1.5);
   837→    }
   838→  }
   839→  ctx.globalAlpha = 1;
   840→}
