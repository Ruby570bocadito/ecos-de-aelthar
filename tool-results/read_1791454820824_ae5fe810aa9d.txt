   128→  start() {
   129→    if (this.running) return;
   130→    this.running = true;
   131→    this.lastTs = performance.now();
   132→    const loop = (ts: number) => {
   133→      if (!this.running) return;
   134→      try {
   135→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   136→        this.lastTs = ts;
   137→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   138→        this.globalT += dt;
   139→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   140→        drawGame(this);
   141→        this.loopError = null;
   142→      } catch (err) {
   143→        this.loopError = String(err);
   144→        console.error('[EcosAelthar loop]', err);
   145→        try {
   146→          this.ctx.fillStyle = '#000';
   147→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   148→          this.ctx.font = '12px monospace';
   149→          this.ctx.fillStyle = '#ff7060';
   150→          this.ctx.textAlign = 'left';
   151→          this.ctx.textBaseline = 'top';
   152→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   153→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   154→        } catch { /* noop */ }
   155→      }
   156→      this.raf = requestAnimationFrame(loop);
   157→    };
   158→    this.raf = requestAnimationFrame(loop);
   159→  }
   160→
   161→  stop() {
   162→    this.running = false;
   163→    cancelAnimationFrame(this.raf);
   164→    audio.stopMusic();
   165→  }

   297→  buildGround() {
   298→    const { w, h } = this.map;
   299→    const mk = (ep: Epoch) => {
   300→      const c = document.createElement('canvas');
   301→      c.width = w * TILE; c.height = h * TILE;
   302→      const x = c.getContext('2d')!;
   303→      x.imageSmoothingEnabled = false;
   304→      // suelo
   305→      for (let ty = 0; ty < h; ty++) {
   306→        for (let tx = 0; tx < w; tx++) {
   307→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   308→          drawTile(x, ch, tx, ty, this.mapId, 0);
   309→        }
   310→      }
   311→      // objetos altos por orden de fila
   312→      for (let ty = 0; ty < h; ty++) {
   313→        for (let tx = 0; tx < w; tx++) {
   314→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   315→          if (SOLID_CHARS.has(ch) && (ch === 't' || ch === 'p')) drawTallTile(x, ch, tx, ty, this.mapId);
   316→        }
   317→      }
   318→      return c;
   319→    };
   320→    this.groundCanvas = mk('presente');
   321→    this.groundPastCanvas = this.map.epochDiffs.length ? mk('pasado') : this.groundCanvas;
   322→    // minimapa
   323→    const mini = document.createElement('canvas');
   324→    mini.width = w; mini.height = h;
   325→    const mx = mini.getContext('2d')!;
   326→    for (let ty = 0; ty < h; ty++) {
   327→      for (let tx = 0; tx < w; tx++) {
   328→        const ch = tileAt(this.map, this.rows, tx, ty, 'presente');
   329→        mx.fillStyle =
   330→          ch === '~' ? '#3a6a9a' : ch === '=' ? '#b89a6a' :
   331→          ch === ':' ? '#6a6a7a' : ch === 't' || ch === 'p' ? '#24512a' :
   332→          ch === '#' || ch === 'H' || ch === 'r' ? '#7a7a8a' :
   333→          ch === 'n' ? '#7ea49a' : '#4a8a44';
   334→        mx.fillRect(tx, ty, 1, 1);
   335→      }
   336→    }
   337→    this.miniCanvas = mini;
   338→  }

   388→  boxFree(x: number, y: number, w: number, h: number): boolean {
   389→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   390→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   391→  }

   393→  moveEntity(e: { x: number; y: number; w: number; h: number }, dx: number, dy: number) {
   394→    if (dx !== 0 && this.boxFree(e.x + dx, e.y, e.w, e.h)) e.x += dx;
   395→    if (dy !== 0 && this.boxFree(e.x, e.y + dy, e.w, e.h)) e.y += dy;

   398→  updateCamera(snap = false) {
   399→    if (!this.player) return;
   400→    const targetX = this.player.x * ZOOM - VIEW_W / 2;
   401→    const targetY = this.player.y * ZOOM - VIEW_H / 2;
   402→    const maxX = this.map.w * TILE * ZOOM - VIEW_W;
   403→    const maxY = this.map.h * TILE * ZOOM - VIEW_H;
   404→    const cx = Math.max(0, Math.min(maxX, targetX));
   405→    const cy = Math.max(0, Math.min(maxY, targetY));
   406→    if (snap) { this.camX = cx; this.camY = cy; }
   407→    else { this.camX += (cx - this.camX) * 0.14; this.camY += (cy - this.camY) * 0.14; }
   408→  }

   420→    this.epochFx = 0.8;
   421→    audio.sfx('epoch');

   427→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   428→    const p = this.player;
   429→    if (!p) return null;
   430→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   431→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   432→      const d = Math.hypot(x - p.x, y - p.y);
   433→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   434→    };
   435→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   436→    for (const ch of this.map.chests) {
   437→      if (this.openedChests.has(ch.id)) continue;
   438→      if (ch.needPast && this.epoch !== 'pasado') continue;
   439→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   440→    }
   441→    for (const pr of this.map.props) {
   442→      if (pr.needPast && this.epoch !== 'pasado') continue;
   443→      if (pr.needPresent && this.epoch !== 'presente') continue;
   444→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   445→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   446→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   447→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   448→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   449→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   450→    }

   552→  applyAction(action: string) {
   553→    const p = this.player;
   554→    if (!p) return;
   555→    switch (true) {
   556→      case action === 'accept_q2':
   557→        this.questIdx = 1; this.questStep = 0; this.flags.q2 = true;
   558→        audio.sfx('quest'); this.toast('Nueva misión: Lobos en la Niebla', '#8ef0b0');

   560→      case action === 'accept_q3':
   561→        this.questIdx = 2; this.questStep = this.flags.visitedBosque ? 1 : 0;
   562→        p.gold += 50; p.potions += 2; p.repGuardianes += 10;
   563→        audio.sfx('quest'); this.toast('Misión completada: Lobos en la Niebla (+50 coronas, +2 pociones, +10 reputación)', '#8ef0b0');
   564→        this.toast('Nueva misión: El Susurro del Bosque', '#8ef0b0');
   565→        break;
   566→      case action === 'accept_q4':
   567→        this.questIdx = 3; this.questStep = 0;
   568→        audio.sfx('quest'); this.toast('Nueva misión: La Cripta del Primer Canto', '#8ef0b0');
   569→        break;
   570→      case action === 'accept_q5':
   571→        this.questIdx = 4; this.questStep = 0;
   572→        audio.sfx('quest'); this.toast('Nueva misión: Ecos de Esperanza', '#8ef0b0');
   573→        break;
   574→      case action === 'fragment_touched':
   575→        this.flags.fragmentTouched = true;
   576→        p.hasEcho = true;
   577→        audio.sfx('echo');
   578→        this.burst(p.x, p.y - 8, '#ffe9a0', 24, 80);
   579→        if (this.questIdx === 2) this.questAdvance();
   580→        this.toast('Resonancia despierta: pulsa Q para alternar entre presente y pasado', '#ffe9a0');
   581→        break;
   582→      case action === 'eco_taken':
   583→        this.flags.ecoVoz = true;
   584→        p.points += 1; p.repGuardianes += 10;
   585→        p.hp = p.maxHp;
   586→        if (this.questIdx === 3 && this.questStep === 2) this.questAdvance();
   587→        this.toast('Eco de la Voz recuperado: +1 punto de habilidad, +10 reputación', '#ffe9a0');
   588→        break;
   589→      case action === 'fragment':
   590→        break;
   591→      case action === 'forge': {
   592→        const cost = 30 + p.weaponPlus * 25;
   593→        if (p.weaponPlus >= 5) { this.toast('Toln: «+5 es lo que da de sí esta forja, Portador.»', '#e8a'); audio.sfx('error'); }
   594→        else if (p.gold >= cost) {
   595→          p.gold -= cost; p.weaponPlus++;
   596→          audio.sfx('confirm');
   597→          this.toast(`Arma mejorada a +${p.weaponPlus} (${cost} coronas)`, '#f0c84a');
   598→        } else { this.toast(`Te faltan coronas (${cost} necesarias)`, '#e88'); audio.sfx('error'); }
   599→        break;
   600→      }
   601→      case action === 'buy_potion':
   602→        if (p.gold >= 15) { p.gold -= 15; p.potions++; audio.sfx('coin'); this.toast('Poción comprada (15 coronas)', '#f0c84a'); }
   603→        else { this.toast('Te faltan coronas', '#e88'); audio.sfx('error'); }
   604→        break;
   605→      case action === 'recruit_ilwen':
   606→        if (!this.companion) {
   607→          this.companion = this.makeCompanion();
   608→          this.toast('Ilwen se une al grupo: cubre tu espalda con su arco', '#8ef0b0');
   609→          audio.sfx('confirm');
   610→        }
   611→        break;
   612→      case action === 'rest':
   613→        p.hp = p.maxHp; p.sta = p.maxSta;
   614→        this.save();
   615→        audio.sfx('save');
   616→        this.toast('Descansas junto al cristal. Partida guardada. Vida restaurada.', '#8ef0ff');
   617→        break;
   618→      case action.startsWith('travel_'): {
   619→        const dest = action.slice(7) as MapId;
   620→        if (MAPS[dest]) {
   621→          this.closeDialogue();
   622→          this.fadeTo(dest, ...this.sanctuaryPos(dest));
   623→        }
   624→        break;
   625→      }
   626→      case action === 'end_demo': {
   627→        const mins = Math.floor(p.playTime / 60), secs = Math.floor(p.playTime % 60);
   628→        this.endStats =
   629→          `Nivel ${p.level} · ${p.kills} enemigos derrotados · ${p.gold} coronas\n` +
   630→          `Reputación Guardianes: ${p.repGuardianes >= 0 ? '+' : ''}${p.repGuardianes} · Muertes: ${p.deaths}\n` +
   631→          `Tiempo de juego: ${mins}m ${secs}s · Época favorita: la que tú elijas`;
   632→        this.save();
   633→        this.setState('end');
   634→        audio.playTrack('title');
   635→        break;
   636→      }
   637→      case action === 'close':
   638→        this.closeDialogue();
   639→        break;
   640→    }
   641→  }

   734→  private onKeyDown = (e: KeyboardEvent) => {
   735→    const k = e.key.toLowerCase();
   736→    if ([' ', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(k)) e.preventDefault();
   737→    if (e.repeat) return;
   738→    this.keys.add(k);
   739→    audio.resume();
   740→
   741→    if (this.state === 'title') {
   742→      if (k === 'enter' || k === 'e' || k === ' ') { /* botones por click; enter = nueva partida */ }
   743→    } else if (this.state === 'intro') {
   744→      if (k === 'e' || k === ' ' || k === 'enter') {
   745→        this.introIdx++;
   746→        audio.sfx('blip');
   747→        if (this.introIdx >= 3) this.startPlay();
   748→      }
   749→    } else if (this.state === 'dialogue') {
   750→      if (k === 'e' || k === ' ' || k === 'enter') this.advanceDialogue();
   751→      else if (k === 'arrowup' || k === 'w') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + this.dlgNode.options.length - 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
   752→      else if (k === 'arrowdown' || k === 's') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
   753→    } else if (this.state === 'play') {
   754→      if (k === 'e') this.tryInteract();
   755→      else if (k === 'q') this.epochSwitch();
   756→      else if (k === 'escape' || k === 'm') { this.setState('pause'); audio.sfx('uiOpen'); }
   757→      else if (k === 'f') this.drinkPotion();
   758→      else if (k === 't' && e.shiftKey) {
   759→        // accesibilidad / QoL: esperar hasta el alba
   760→        this.dayT = 0.22;
   761→        this.toast('Esperas junto al camino hasta el alba...', '#ffe86a');
   762→        audio.sfx('save');
   763→      }
   764→      else if (['1', '2', '3', '4'].includes(k)) this.useSkill(parseInt(k, 10) - 1);
   765→    } else if (this.state === 'pause') {
   766→      if (k === 'escape' || k === 'm') this.setState('play');
   767→    } else if (this.state === 'dead') {
   768→      if (k === 'e' || k === 'enter') this.respawn();
   769→    } else if (this.state === 'end') {
   770→      if (k === 'enter' || k === 'e') this.setState('title');
   771→    }
   772→  };

   791→  private onMouseDown = (e: MouseEvent) => {
   792→    audio.resume();
   793→    const p = this.canvasPos(e);
   794→    this.mouse.x = p.x; this.mouse.y = p.y;
   795→    // UI hits primero
   796→    for (const h of this.uiHit) {
   797→      if (p.x >= h.x && p.x <= h.x + h.w && p.y >= h.y && p.y <= h.y + h.h) {
   798→        audio.sfx('select');
   799→        h.cb();
   800→        return;
   801→      }
   802→    }
   803→    if (e.button === 2) { this.mouse.rdown = true; this.startParry(); return; }
   804→    if (e.button !== 0) return;
   805→    this.mouse.down = true;
   806→    if (this.state === 'dialogue') this.advanceDialogue();
   807→    else if (this.state === 'intro') { this.introIdx++; if (this.introIdx >= 3) this.startPlay(); else audio.sfx('blip'); }
   808→    else if (this.state === 'end') this.setState('title');
   809→    else if (this.state === 'play') this.startAttack();
   810→  };

   845→  releaseCharge() {
   846→    const p = this.player!;
   847→    const wasCharging = p.charging;
   848→    p.charging = false;
   849→    const charged = wasCharging && p.chargeT > 0.35;
   850→    const chargeTime = p.chargeT;
   851→    p.chargeT = 0;
   852→    if (!wasCharging) return;
   853→    if (p.attackT > 0 || p.rollT > 0 || this.state !== 'play') return;
   854→    if (p.sta < 8) { audio.sfx('parryFail'); return; }
   855→    p.attackT = charged ? 0.4 : 0.26;
   856→    p.combo = (p.combo + 1) % 3;
   857→    p.sta -= charged ? 18 : 8;
   858→    p.chargedHit = charged;
   859→    this.aimAtMouse();
   860→    audio.sfx(charged ? 'swing2' : 'swing');
   861→    // daño del golpe
   862→    const comboMult = [1, 1.12, 1.28][p.combo];
   863→    const dmg = playerMeleeDmg(p) * comboMult * (charged ? 2.1 : 1);
   864→    if (charged) {
   865→      const dirs: Record<Dir, [number, number]> = { down: [0, 1], up: [0, -1], left: [-1, 0], right: [1, 0] };
   866→      const [dx, dy] = dirs[p.dir];
   867→      this.aoeHit(p.x + dx * 20, p.y + dy * 20, 34, dmg, 'ninguno', false);
   868→      this.shake = 3;
   869→      this.burst(p.x + dx * 18, p.y + dy * 18, '#ffe86a', 12, 70);
   870→    } else {
   871→      this.meleeHit(24, 20, dmg, 'ninguno', 50);
   872→    }
   873→    void chargeTime;
   874→  }

   876→  startParry() {
   877→    const p = this.player!;
   878→    if (this.state !== 'play' || p.rollT > 0) return;
   879→    if (p.sta < 12) { audio.sfx('parryFail'); return; }
   880→    p.sta -= 12;
   881→    p.parryT = 0.2;
   882→    p.parryFx = 0.32;
   883→    audio.sfx('swing');
   884→  }

   886→  aimAtMouse() {
   887→    const p = this.player!;
   888→    const wx = this.mouse.x / ZOOM + this.camX / ZOOM;
   889→    const wy = this.mouse.y / ZOOM + this.camY / ZOOM;
   890→    const dx = wx - p.x, dy = wy - p.y;
   891→    if (Math.abs(dx) > Math.abs(dy)) p.dir = dx > 0 ? 'right' : 'left';
   892→    else p.dir = dy > 0 ? 'down' : 'up';
   893→  }

   895→  useSkill(i: number) {
   896→    const p = this.player!;
   897→    const disc = SKILLS[p.discipline];
   898→    if (i < 0 || i >= disc.length) return;

   903→    p.cds[i] = sk.cd;
   904→    this.castSkill(sk.id, sk.element);
   905→  }

   973→  meleeHit(reach: number, width: number, dmg: number, element: Element, kb: number) {
   974→    const p = this.player!;
   975→    const dirs: Record<Dir, [number, number]> = { down: [0, 1], up: [0, -1], left: [-1, 0], right: [1, 0] };
   976→    const [dx, dy] = dirs[p.dir];
   977→    const cx = p.x + dx * reach * 0.6, cy = p.y + dy * reach * 0.6;
   978→    for (const e of this.enemies) {
   979→      if (e.dead) continue;
   980→      if (Math.hypot(e.x - cx, e.y - cy) < reach + e.w / 2 + 4) {
   981→        this.damageEnemy(e, dmg, element, kb, dx, dy);
   982→      }
   983→    }
   984→    void width;
   985→  }

   987→  aoeHit(x: number, y: number, r: number, dmg: number, element: Element, stun: boolean) {
   988→    for (const e of this.enemies) {
   989→      if (e.dead) continue;
   990→      if (Math.hypot(e.x - x, e.y - y) < r + e.w / 2) {
   991→        this.damageEnemy(e, dmg, element, 60, Math.sign(e.x - x), Math.sign(e.y - y));
   992→        if (stun) { e.ai = 'aturdido'; e.aiT = 1.5; }
   993→      }
   994→    }
   995→  }

   997→  chainLightning(x: number, y: number, dmg: number, jumps: number, nx: number, ny: number) {
   998→    let cx = x + nx * 14, cy = y + ny * 14;
   999→    const hitSet = new Set<Enemy>();
  1000→    for (let j = 0; j < jumps; j++) {
  1001→      let best: Enemy | null = null, bd = 90;
  1002→      for (const e of this.enemies) {
  1003→        if (e.dead || hitSet.has(e)) continue;
  1004→        const d = Math.hypot(e.x - cx, e.y - cy);
  1005→        if (d < bd) { bd = d; best = e; }
  1006→      }
  1007→      if (!best) break;
  1008→      hitSet.add(best);
  1009→      this.damageEnemy(best, dmg * (1 - j * 0.2), 'rayo', 20, 0, 0);
  1010→      this.lightningFx(cx, cy, best.x, best.y);
  1011→      cx = best.x; cy = best.y;
  1012→    }
  1013→  }

  1015→  lightningFx(x0: number, y0: number, x1: number, y1: number) {
  1016→    const steps = 6;
  1017→    for (let i = 0; i <= steps; i++) {
  1018→      const t = i / steps;
  1019→      this.particles.push({
  1020→        x: x0 + (x1 - x0) * t + (Math.random() - 0.5) * 6,
  1021→        y: y0 + (y1 - y0) * t + (Math.random() - 0.5) * 6,
  1022→        vx: 0, vy: 0, t: 0.18, maxT: 0.18, color: '#ffe86a', size: 2, grav: 0,
  1023→      });
  1024→    }
  1025→  }

  1027→  damageEnemy(e: Enemy, dmg: number, element: Element, kb: number, kbx = 0, kby = 0) {
  1028→    if (e.dead) return;
  1029→    const def = ENEMY_DEFS[e.etype];
  1030→    e.aggro = true;
  1031→    let final = dmg;
  1032→    let crit = false;
  1033→    const p = this.player!;
  1034→    if (Math.random() * 100 < Math.min(40, 5 + p.attrs.des * 2)) { final *= 2; crit = true; }
  1035→    if (def.weakTo !== 'ninguno' && element === def.weakTo) final *= 1.5;
  1036→    if (e.ai === 'aturdido' && e.etype === 'guardian') final *= 1.5;
  1037→    final = Math.max(1, Math.round(final));
  1038→    e.hp -= final;
  1039→    e.hitFlash = 0.12;
  1040→    // quiebre
  1041→    if (e.maxSta > 0) {
  1042→      e.sta -= final * 1.1;
  1043→      if (e.sta <= 0 && e.ai !== 'aturdido') {
  1044→        e.sta = 0;
  1045→        e.ai = 'aturdido';
  1046→        e.aiT = 4;
  1047→        this.toast('¡QUEBRADO! El enemigo queda vulnerable', '#ffe86a');
  1048→        this.floatAt(e.x, e.y - 20, 'QUEBRADO', '#ffe86a', 7);
  1049→        audio.sfx('roar');
  1050→      }
  1051→    }
  1052→    // estados
  1053→    if (element === 'fuego') e.statuses.push({ kind: 'quemado', t: 3, power: 4 });
  1054→    if (element === 'hielo') e.statuses.push({ kind: 'congelado', t: 2.5, power: 0.5 });
  1055→    // reacciones elementales
  1056→    if (element === 'hielo' && e.statuses.some(s => s.kind === 'quemado' && s.t > 1)) {
  1057→      this.aoeHit(e.x, e.y, 30, 12, 'ninguno', false);
  1058→      this.floatAt(e.x, e.y - 16, 'VAPOR', '#d0f0f8', 6);
  1059→      this.burst(e.x, e.y - 6, '#d0f0f8', 12);
  1060→    }
  1061→    // knockback
  1062→    if (kb > 0) this.moveEntity(e, kbx * kb * 0.06, kby * kb * 0.06);
  1063→    // feedback
  1064→    this.floatAt(e.x + (Math.random() - 0.5) * 8, e.y - 14, `${final}`, crit ? '#ffd24a' : '#fff');
  1065→    if (crit) this.floatAt(e.x, e.y - 22, '¡CRÍTICO!', '#ffd24a', 6);
  1066→    this.burst(e.x, e.y - 4, crit ? '#ffd24a' : '#f0e8e0', crit ? 10 : 5);
  1067→    audio.sfx(crit ? 'crit' : 'hit');
  1068→    this.hitStop = crit ? 0.09 : 0.04;
  1069→    // resonancia
  1070→    const espMult = 1 + p.attrs.esp * 0.1;
  1071→    p.res = Math.min(p.maxRes, p.res + 6 * espMult);
  1072→    if (e.hp <= 0) this.killEnemy(e);
  1073→  }

  1075→  killEnemy(e: Enemy) {
  1076→    e.dead = true;
  1077→    const def = ENEMY_DEFS[e.etype];
  1078→    const p = this.player!;
  1079→    p.kills++;
  1080→    const espMult = 1 + p.attrs.esp * 0.1;
  1081→    p.res = Math.min(p.maxRes, p.res + 10 * espMult);
  1082→    this.gainXp(def.xp);
  1083→    const gold = Math.round(def.gold[0] + Math.random() * (def.gold[1] - def.gold[0]));
  1084→    p.gold += gold;
  1085→    this.floatAt(e.x, e.y - 20, `+${gold} coronas`, '#f0c84a');
  1086→    this.burst(e.x, e.y - 4, e.etype === 'guardian' ? '#7ee8ff' : '#9ec4b4', e.etype === 'guardian' ? 40 : 14, e.etype === 'guardian' ? 120 : 60);
  1087→    audio.sfx('enemyDie');
  1088→    // cuenta de lobos para la misión
  1089→    if (e.etype === 'lobo' && this.questIdx === 1 && this.questStep === 0) {

  1091→      this.flags.wolfKills = n;

  1094→    }

  1096→      this.flags.guardianDefeated = true;
  1097→      this.bossActive = false;
  1098→      audio.setCombat(false);
  1099→      audio.playTrack('crypt');

  1101→      this.toast('El Guardián Hueco se deshace en notas de silencio...', '#7ee8ff');
  1102→      this.toast('El altar del Eco brilla al norte', '#ffe9a0');
  1103→      if (this.questIdx === 3 && this.questStep === 1) this.questAdvance();
  1104→    }
  1105→  }

  1107→  gainXp(xp: number) {
  1108→    const p = this.player!;
  1109→    p.xp += xp;
  1110→    let next = this.xpNext(p.level);
  1111→    while (p.xp >= next) {
  1112→      p.xp -= next;
  1113→      p.level++;
  1114→      p.maxHp += 7;
  1115→      p.hp = p.maxHp;
  1116→      p.points += 3;
  1117→      audio.sfx('levelup');
  1118→      this.toast(`¡Nivel ${p.level}! +3 puntos de atributo (menú > Estado)`, '#ffe86a');
  1119→      this.burst(p.x, p.y - 8, '#ffe86a', 20, 70);
  1120→      next = this.xpNext(p.level);
  1121→    }
  1122→  }

  1124→  xpNext(level: number): number {
  1125→    return Math.round(36 * Math.pow(level, 1.45));

  1128→  damagePlayer(dmg: number, fromX: number, fromY: number) {
  1129→    const p = this.player!;
  1130→    if (p.iframes > 0 || p.rollT > 0 || this.state !== 'play') return;
  1131→    if (p.parryT > 0) {
  1132→      // ¡parada perfecta!
  1133→      audio.sfx('parry');
  1134→      p.res = Math.min(p.maxRes, p.res + 20);
  1135→      p.parryFx = 0.4;
  1136→      this.hitStop = 0.12;
  1137→      this.burst(p.x + (fromX - p.x) * 0.3, p.y - 8 + (fromY - p.y) * 0.3, '#fff8c0', 14, 100);
  1138→      this.floatAt(p.x, p.y - 22, '¡PARADA!', '#fff8c0', 7);
  1139→      // aturde al atacante cercano
  1140→      let best: Enemy | null = null, bd = 34;
  1141→      for (const e of this.enemies) {
  1142→        if (e.dead) continue;
  1143→        const d = Math.hypot(e.x - p.x, e.y - p.y);
  1144→        if (d < bd) { bd = d; best = e; }
  1145→      }
  1146→      if (best) { best.ai = 'aturdido'; best.aiT = 1.6; best.windup = 0; }
  1147→      p.parryT = 0;
  1148→      return;
  1149→    }
  1150→    const red = Math.min(0.5, p.attrs.vig * 0.01);
  1151→    const final = Math.max(1, Math.round(dmg * (1 - red)));
  1152→    p.hp -= final;
  1153→    p.iframes = 0.6;
  1154→    p.lastHitT = 0.3;
  1155→    this.shake = 4;
  1156→    audio.sfx('hurt');
  1157→    this.floatAt(p.x, p.y - 16, `-${final}`, '#ff7060');
  1158→    const dx = p.x - fromX, dy = p.y - fromY;
  1159→    const l = Math.max(1, Math.hypot(dx, dy));
  1160→    this.moveEntity(p, (dx / l) * 8, (dy / l) * 8);
  1161→    if (p.hp <= 0) this.playerDied();
  1162→  }

  1167→export function playerMeleeDmg(p: Player): number {
  1168→  const base = p.discipline === 'alba' ? 12 : 8;

  1170→  const buff = (p.buffT ?? 0) > 0 ? 1.5 : 1;
  1171→  return (base + stat + p.level * 1 + p.weaponPlus * 2.5) * buff;
  1172→}
