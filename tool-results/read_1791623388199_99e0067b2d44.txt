     2→// ECOS DE AELTHAR — ACTO II · Arte pixel procedural de la
     3→// EXPANSIÓN (AGENTE 7-b · enemigos + sprites).
     4→// 100% procedural: canvas + fillRect píxel a píxel, sin assets.
     5→// NO toca sprites.ts: usa registerSpr()/hash2() (contrato).
     6→//
     7→// Contenido:
     8→//   · initExpansionSprites(): registra 'neumo', 'espectro', 'arpi',
     9→//     'sirena' (jefa 32×32), 'golem' (jefe 32×32) y los proyectiles
    10→//     'orb', 'shard', 'nota' (1 frame).
    11→//   · drawExpansionTile(): tiles de suelo nuevos ('s' arena, 'S' nieve,
    12→//     'i' hielo) + variantes de '.' ',' ':' '=' por mapa (costa,
    13→//     aldea, cumbres). Devuelve true si dibujó el tile.
    14→//   · drawExpansionTallTile(): palmera ('p' en costa) y pino nevado
    15→//     ('t'/'p' en cumbres). Devuelve true si dibujó.
    16→//   · drawExpansionProp(): 'wreck', 'faro', 'lamp' (centrados en cx,cy).
    17→//   · drawExpansionProjectile(): 'orb', 'shard', 'nota' con animación.
    18→//
    19→// Estilo: copia la técnica de sprites.ts (buildWolf/buildGuardian/
    20→// buildFragment/drawTile): outline oscuro, sombreado simple por
    21→// rects, hash2(tx,ty) para moteado determinista, zoom lo aplica el
    22→// render (los canvas van en píxeles de arte).
    23→// ============================================================

    28→// ---------------- utilidades locales ----------------
    29→
    30→function cv(w: number, h: number): { c: HTMLCanvasElement; x: CanvasRenderingContext2D } {
    31→  const c = document.createElement('canvas');
    32→  c.width = w; c.height = h;
    33→  const x = c.getContext('2d')!;
    34→  x.imageSmoothingEnabled = false;
    35→  return { c, x };
    36→}
    37→
    38→/** Rect pintado (equivalente al px() interno de sprites.ts). */
    39→function rc(x: CanvasRenderingContext2D, rx: number, ry: number, rw: number, rh: number, col: string) {
    40→  x.fillStyle = col;
    41→  x.fillRect(rx, ry, rw, rh);
    42→}
    43→
    44→/** Disco pixelado determinista (para burbujas, orbes y halos). */
    45→function disc(x: CanvasRenderingContext2D, cx: number, cy: number, r: number, col: string, alpha = 1) {
    46→  x.globalAlpha = alpha;
    47→  x.fillStyle = col;
    48→  const top = Math.ceil(r);
    49→  for (let dy = -top; dy <= top; dy++) {
    50→    const w = Math.sqrt(Math.max(0, r * r - dy * dy));
    51→    if (w < 0.35 && Math.abs(dy) > 0) continue;
    52→    x.fillRect(Math.round(cx - w), cy + dy, Math.max(1, Math.round(w * 2 + 1)), 1);
    53→  }
    54→  x.globalAlpha = 1;
    55→}

    61→// ---------------- Neumo de Marea (16×16 · 2 frames) ----------------
    62→// Criatura burbuja de espuma: flota y tiembla. Aguamarina translúcida
    63→// con brillo y un ojo oscuro pequeño.
    64→
    65→function buildNeumo(): Frames {
    66→  const frames: Frames = [];
    67→  for (let f = 0; f < 2; f++) {
    68→    const { c, x } = cv(16, 16);
    69→    const O = '#1c5a5e', BD = '#4ec2b8', B = '#7fe8d8', HL = '#d8fcf4', EYE = '#173038';
    70→    const oy = f; // tiembla: el cuerpo sube/baja 1 px
    71→    // contorno + masa
    72→    disc(x, 8, 8, f === 0 ? 6 : 6.4, O, 0.85);
    73→    disc(x, 8, 8 + oy * 0.4, 5.2, BD, 0.72);
    74→    disc(x, 8, 8 + oy * 0.4, 4.1, B, 0.68);
    75→    // brillo (espalda superior)
    76→    x.globalAlpha = 0.92;
    77→    rc(x, 5, 4 - f, 3, 2, HL);
    78→    rc(x, 4, 5 - f, 1, 1, HL);
    79→    x.globalAlpha = 0.55;
    80→    rc(x, 9, 5 - f, 1, 2, HL);
    81→    // ojo oscuro pequeño
    82→    x.globalAlpha = 1;
    83→    rc(x, 9, 7 - f, 2, 2, EYE);
    84→    rc(x, 9, 7 - f, 1, 1, '#ffffff');
    85→    // burbujitas de espuma en la base
    86→    x.globalAlpha = 0.7;
    87→    rc(x, 3, 12, 2, 2, B);
    88→    rc(x, 11, 12 + f, 2, 1, B);
    89→    rc(x, 7, 13, 1, 1, HL);
    90→    x.globalAlpha = 1;
    91→    frames.push(c);
    92→  }
    93→  return frames;
    94→}

   100→function buildEspectro(): Frames {
   101→  const frames: Frames = [];
   102→  for (let f = 0; f < 2; f++) {
   103→    const { c, x } = cv(16, 16);
   104→    const O = '#3a4a5e', B = '#c9d4e4', S = '#a2b2c8', EYE = '#232e42';
   105→    const dy = f === 0 ? 0 : -1; // sube/baja
   106→    // cuerpo (semivapor)
   107→    x.globalAlpha = 0.88;
   108→    rc(x, 4, 7 + dy, 8, 4, B);
   109→    rc(x, 5, 11 + dy, 6, 2, B);
   110→    rc(x, 4, 7 + dy, 1, 4, S);
   111→    rc(x, 11, 7 + dy, 1, 4, S);
   112→    // cabeza
   113→    x.globalAlpha = 0.95;
   114→    rc(x, 5, 2 + dy, 6, 5, B);
   115→    rc(x, 4, 3 + dy, 8, 3, B);
   116→    rc(x, 5, 6 + dy, 6, 1, S);
   117→    // borde difuminado: costados y rabo translúcidos
   118→    x.globalAlpha = 0.35;
   119→    rc(x, 3, 4 + dy, 1, 6, B);
   120→    rc(x, 12, 4 + dy, 1, 6, B);
   121→    rc(x, 4, 13 + dy, 8, 1, B);
   122→    // jirones finales (alternan con el frame)
   123→    x.globalAlpha = 0.55;
   124→    if (f === 0) { rc(x, 6, 13, 2, 2, S); rc(x, 9, 13, 1, 1, S); }
   125→    else { rc(x, 5, 12, 1, 2, S); rc(x, 8, 13, 2, 1, S); }
   126→    // ojos huecos
   127→    x.globalAlpha = 1;
   128→    rc(x, 6, 4 + dy, 1, 2, EYE);
   129→    rc(x, 9, 4 + dy, 1, 2, EYE);
   130→    x.globalAlpha = 0.5;
   131→    rc(x, 7, 6 + dy, 2, 1, EYE);
   132→    // contorno parcial (difuminado, nunca cerrado)
   133→    x.globalAlpha = 0.38;
   134→    rc(x, 4, 2 + dy, 8, 1, O);
   135→    rc(x, 4, 2 + dy, 1, 4, O);
   136→    rc(x, 11, 2 + dy, 1, 4, O);
   137→    x.globalAlpha = 1;
   138→    frames.push(c);
   139→  }
   140→  return frames;
   141→}

   348→function buildOrb(): Frames {
   349→  const { c, x } = cv(12, 12);
   350→  disc(x, 6, 6, 5, '#2e78b8', 0.9);   // borde de agua profunda
   351→  disc(x, 6, 6, 4.1, '#58a8e0', 0.82);
   352→  disc(x, 6, 6, 2.6, '#9ad0f0', 0.8);
   353→  x.globalAlpha = 1;
   354→  rc(x, 4, 3, 2, 2, '#f4fbff');       // brillo
   355→  rc(x, 8, 9, 1, 1, '#2e78b8');
   356→  return [c];
   357→}

   359→function buildShard(): Frames {
   360→  const { c, x } = cv(12, 12);
   361→  // rombo afilado vertical
   362→  const rows: [number, number, number][] = [
   363→    // [y, x0, ancho]
   364→    [0, 5, 2], [1, 4, 4], [2, 4, 4], [3, 3, 6], [4, 3, 6],
   365→    [5, 2, 8], [6, 2, 8], [7, 3, 6], [8, 3, 6], [9, 4, 4], [10, 4, 4], [11, 5, 2],
   366→  ];
   367→  for (const [y, x0, w] of rows) {
   368→    rc(x, x0, y, w, 1, w <= 2 ? '#dff4fc' : '#9fd4ec');
   369→    if (w >= 6) { rc(x, x0, y, 1, 1, '#5aa8cc'); rc(x, x0 + w - 1, y, 1, 1, '#5aa8cc'); }
   370→  }
   371→  rc(x, 5, 2, 1, 6, '#eaf8ff');       // filo de luz
   372→  rc(x, 6, 3, 1, 3, '#ffffff');
   373→  return [c];
   374→}

   387→  rc(x, 10, 1, 1, 1, '#ffffff');      // destello

   412→// ============================================================
   413→// TILES DE SUELO (Acto II)
   414→// El motor llama a drawTile() al pre-renderizar el suelo; el
   415→// integrador debe probar PRIMERO esta función y caer al dibujo
   416→// genérico si devuelve false:
   417→//   if (drawExpansionTile(x, ch, tx, ty, this.mapId)) continue;
   418→//   drawTile(x, ch, tx, ty, this.mapId, 0);
   419→// ============================================================
   420→
   421→export function drawExpansionTile(
   422→  x: CanvasRenderingContext2D, ch: string, tx: number, ty: number, mapId: string,
   423→): boolean {
   424→  const px0 = tx * 16, py0 = ty * 16;
   425→  const r = hash2(tx, ty);
   426→  const r2 = hash2(tx * 7 + 3, ty * 11 + 5);
   427→  const r3 = hash2(tx * 13 + 1, ty * 3 + 7);
   428→  switch (ch) {
   429→    case 's': { // arena con moteado determinista
   430→      rc(x, px0, py0, 16, 16, r < 0.5 ? '#dcc590' : '#d2bb84');
   431→      for (let i = 0; i < 5; i++) {
   432→        const hx = hash2(tx * 5 + i * 3, ty * 7 + i);
   433→        const hy = hash2(tx * 11 + i, ty * 5 + i * 7);
   434→        if (hx < 0.45) {
   435→          x.fillStyle = i % 2 === 0 ? '#c4ab74' : '#e6d2a2';
   436→          x.fillRect(px0 + Math.floor(hx * 30) % 15, py0 + Math.floor(hy * 30) % 15, 1, 1);
   437→        }
   438→      }
   439→      if (r2 > 0.86) { // concha suelta
   440→        rc(x, px0 + 4 + Math.floor(r * 6), py0 + 5 + Math.floor(r3 * 6), 2, 1, '#f0e8d4');
   441→      }
   442→      if (r3 < 0.1) rc(x, px0 + 9, py0 + 3, 2, 1, '#e8d4a4');
   443→      return true;
   444→    }
   445→    case 'S': { // nieve con destellos
   446→      rc(x, px0, py0, 16, 16, r < 0.5 ? '#e9eef5' : '#e2e9f1');
   447→      const n = r2 > 0.5 ? 2 : 1;
   448→      for (let i = 0; i < n; i++) {
   449→        const hx = hash2(tx * 3 + i * 11, ty * 9 + i * 5);
   450→        const hy = hash2(tx * 17 + i, ty * 13 + i * 3);
   451→        rc(x, px0 + 2 + Math.floor(hx * 12), py0 + 2 + Math.floor(hy * 12), 1, 1, '#ffffff');
   452→      }
   453→      if (r3 > 0.72) rc(x, px0 + 4 + Math.floor(r * 8), py0 + 9, 3, 1, '#d2dde9');
   454→      return true;
   455→    }
   456→    case 'i': { // hielo liso con grietas claras + reflejo
   457→      rc(x, px0, py0, 16, 16, r < 0.5 ? '#a9d2ea' : '#9fc9e4');
   458→      // reflejo diagonal
   459→      x.globalAlpha = 0.2;
   460→      rc(x, px0 + 2 + Math.floor(r2 * 4), py0, 3, 16, '#eaf7ff');
   461→      x.globalAlpha = 1;
   462→      // grietas (polilínea determinista)
   463→      if (r2 > 0.4) {
   464→        const cx0 = px0 + 3 + Math.floor(r * 6), cy0 = py0 + 3 + Math.floor(r3 * 8);
   465→        rc(x, cx0, cy0, 4, 1, '#d6eefc');
   466→        rc(x, cx0 + 3, cy0 + 1, 1, 3, '#d6eefc');
   467→        rc(x, cx0 + 2, cy0 + 3, 2, 1, '#c2e2f4');
   468→      }
   469→      if (r3 < 0.25) rc(x, px0 + 10, py0 + 10, 3, 1, '#c2e2f4');
   470→      rc(x, px0, py0, 16, 1, '#b8dcf0');
   471→      return true;
   472→    }
   473→    case '.': {
   474→      if (mapId === 'costa') { // hierba salada amarillenta-verdosa
   475→        rc(x, px0, py0, 16, 16, r < 0.5 ? '#8aa860' : '#7e9c56');
   476→        for (let i = 0; i < 4; i++) {
   477→          const hx = hash2(tx * 4 + i, ty * 9 + i);
   478→          if (hx < 0.4) {
   479→            rc(x, px0 + Math.floor(hx * 14), py0 + Math.floor(hash2(tx + i, ty * 3 + i) * 14), 1, 2,
   480→              i % 2 === 0 ? '#a8c070' : '#6e8c4c');
   481→          }
   482→        }
   483→        if (r2 > 0.8) rc(x, px0 + 6, py0 + 8, 2, 1, '#c8b878'); // brizna seca
   484→        return true;
   485→      }
   486→      if (mapId === 'cumbres') { // nieve corta verdosa
   487→        rc(x, px0, py0, 16, 16, r < 0.5 ? '#a8c098' : '#9cb48c');
   488→        for (let i = 0; i < 4; i++) {
   489→          const hx = hash2(tx * 6 + i, ty * 5 + i * 3);
   490→          if (hx < 0.4) {
   491→            rc(x, px0 + Math.floor(hx * 14), py0 + Math.floor(hash2(tx + i * 5, ty + i) * 14), 1, 1,
   492→              i % 2 === 0 ? '#c4d8b4' : '#8aa07c');
   493→          }
   494→        }
   495→        if (r2 > 0.62) rc(x, px0 + 3 + Math.floor(r * 8), py0 + 4 + Math.floor(r3 * 8), 2, 1, '#eef4f0');
   496→        return true;
   497→      }
   498→      return false; // lunaris/bosque: dibujo genérico del motor
   499→    }
   500→    case ',': {
   501→      if (mapId === 'costa') { // flores de sal
   502→        rc(x, px0, py0, 16, 16, r < 0.5 ? '#8aa860' : '#7e9c56');
   503→        // matas de sal cristalizada
   504→        const bx = px0 + 3 + Math.floor(r * 7), by = py0 + 3 + Math.floor(r2 * 7);
   505→        rc(x, bx, by, 2, 2, '#f0f4f0');
   506→        rc(x, bx + 1, by - 1, 1, 1, '#ffffff');
   507→        rc(x, bx - 1, by + 1, 1, 1, '#d8e8e0');
   508→        if (r3 > 0.5) {
   509→          const cx2 = px0 + 9 - Math.floor(r * 5), cy2 = py0 + 9 - Math.floor(r2 * 5);
   510→          rc(x, cx2, cy2, 2, 1, '#e4efe6');
   511→          rc(x, cx2, cy2 + 1, 1, 1, '#c8dcd4');
   512→        }
   513→        return true;
   514→      }
   515→      return false;
   516→    }
   517→    case ':': {
   518→      if (mapId === 'aldea') { // empedrado viejo más oscuro/agrietado
   519→        rc(x, px0, py0, 16, 16, r < 0.5 ? '#585864' : '#525260');
   520→        rc(x, px0, py0, 16, 1, '#6a6a78');
   521→        // juntas de losas
   522→        x.fillStyle = '#40404c';
   523→        x.fillRect(px0, py0, 1, 16);
   524→        x.fillRect(px0, py0 + (r > 0.5 ? 7 : 9), 16, 1);
   525→        x.fillRect(px0 + (r2 > 0.5 ? 6 : 10), py0, 1, 8);
   526→        // grietas
   527→        if (r2 > 0.7) {
   528→          rc(x, px0 + 3 + Math.floor(r * 6), py0 + 3, 1, 5, '#3c3c48');
   529→          rc(x, px0 + 4 + Math.floor(r * 6), py0 + 7, 2, 1, '#3c3c48');
   530→        }
   531→        if (r3 > 0.85) rc(x, px0 + 11, py0 + 11, 2, 1, '#6e6e7c'); // piedra clara
   532→        return true;
   533→      }
   534→      return false; // cripta/lunaris: dibujo genérico
   535→    }
   536→    case '=': {
   537→      if (mapId === 'cumbres') { // camino pisado moreno
   538→        rc(x, px0, py0, 16, 16, r < 0.5 ? '#8a7048' : '#7c6440');
   539→        rc(x, px0, py0, 16, 1, '#9a8056');
   540→        for (let i = 0; i < 5; i++) {
   541→          const hx = hash2(tx * 13 + i, ty * 5 + i * 3);
   542→          if (hx < 0.4) {
   543→            rc(x, px0 + Math.floor(hx * 13), py0 + Math.floor(hash2(tx + i * 7, ty + i) * 13), 2, 1, '#6a5436');
   544→          }
   545→        }
   546→        if (r2 > 0.75) rc(x, px0 + 5 + Math.floor(r * 6), py0 + 6, 2, 1, '#a8906a'); // piedrecilla
   547→        return true;
   548→      }
   549→      return false; // lunaris: dibujo genérico
   550→    }
   551→    default:
   552→      return false;
   553→  }
   554→}

   556→// ============================================================
   557→// OBJETOS ALTOS (se dibujan por fila sobre el suelo, igual que
   558→// drawTallTile del motor). Devuelve true si dibujó el tile.
   559→// ============================================================
   560→
   561→export function drawExpansionTallTile(
   562→  x: CanvasRenderingContext2D, ch: string, tx: number, ty: number, mapId: string,
   563→): boolean {
   564→  const px0 = tx * 16, py0 = ty * 16;
   565→  const r = hash2(tx * 3 + 1, ty * 5 + 7);
   566→  const r2 = hash2(tx * 11 + 9, ty * 7 + 3);
   567→
   568→  if (ch === 'p' && mapId === 'costa') {
   569→    // ---- PALMERA: tronco curvado + hojas ----
   570→    const bend = r > 0.5 ? 1 : -1; // curva determinista por tile
   571→    // tronco (segmentos que se curvan)
   572→    const seg = [
   573→      [7, 13, 2], [7, 11, 2], [8, 9, 2], [8 + bend, 7, 2], [8 + bend, 5, 2], [9 + bend, 3, 2],
   574→    ] as [number, number, number][];
   575→    for (let i = 0; i < seg.length; i++) {
   576→      const [sx, sy, sw] = seg[i];
   577→      rc(x, px0 + sx + (bend > 0 ? 0 : 0), py0 + sy, sw, 2, i % 2 === 0 ? '#8a6a3e' : '#6d5430');
   578→    }
   579→    // anillos del tronco
   580→    rc(x, px0 + 7, py0 + 12, 2, 1, '#5c462a');
   581→    rc(x, px0 + 8 + bend, py0 + 8, 2, 1, '#5c462a');
   582→    const crownX = px0 + 9 + bend, crownY = py0 + 2;
   583→    // hojas (palmas) — 5 frondas
   584→    rc(x, crownX - 6, crownY + 1, 7, 2, '#3e8a4a');   // izquierda
   585→    rc(x, crownX - 7, crownY + 2, 4, 2, '#2e6a38');
   586→    rc(x, crownX + 1, crownY + 1, 7, 2, '#3e8a4a');   // derecha
   587→    rc(x, crownX + 5, crownY + 2, 4, 2, '#2e6a38');
   588→    rc(x, crownX - 2, crownY - 2, 5, 2, '#5aa858');   // arriba
   589→    rc(x, crownX - 1, crownY - 3, 3, 1, '#5aa858');
   590→    rc(x, crownX - 5, crownY + 3, 3, 1, '#5aa858');   // caídas
   591→    rc(x, crownX + 3, crownY + 3, 3, 1, '#5aa858');
   592→    if (r2 > 0.55) rc(x, crownX - 8, crownY + 3, 2, 1, '#2e6a38'); // fronda extra
   593→    // cocos
   594→    rc(x, crownX - 2, crownY + 3, 2, 2, '#6d4520');
   595→    rc(x, crownX + 1, crownY + 4, 2, 2, '#6d4520');
   596→    return true;
   597→  }
   598→
   599→  if ((ch === 't' || ch === 'p') && mapId === 'cumbres') {
   600→    // ---- PINO NEVADO: verde oscuro con capa blanca ----
   601→    // tronco
   602→    rc(x, px0 + 7, py0 + 12, 2, 4, '#54381e');
   603→    rc(x, px0 + 7, py0 + 15, 2, 1, '#3e2a16');
   604→    // tres pisos de copa, cada uno con gorro de nieve
   605→    const layers: [number, number, number][] = [
   606→      // [x0, y0, ancho]
   607→      [3, 7, 10], [4, 4, 8], [5, 1, 6],
   608→    ];
   609→    for (let i = 0; i < layers.length; i++) {
   610→      const [lx, ly, lw] = layers[i];
   611→      const deep = i % 2 === 0 ? '#1e4a34' : '#21503a';
   612→      rc(x, px0 + lx, py0 + ly, lw, 4, deep);
   613→      rc(x, px0 + lx + 1, py0 + ly + 1, lw - 3, 1, '#2e5e42'); // brillo interior
   614→      // capa de nieve
   615→      const snowW = r2 > 0.5 ? lw : lw - 2;
   616→      rc(x, px0 + lx, py0 + ly, snowW, 1, '#e8f0f4');
   617→      rc(x, px0 + lx + 2, py0 + ly + 1, Math.max(1, snowW - 5), 1, '#d4e4ea');
   618→    }
   619→    // copa encendida
   620→    rc(x, px0 + 7, py0 - 1, 2, 2, '#e8f0f4');
   621→    if (r > 0.7) rc(x, px0 + 5, py0 + 3, 2, 1, '#f4fafc');
   622→    return true;
   623→  }
   624→
   625→  return false;
   626→}

   628→// ============================================================
   629→// PROPS DE LA EXPANSIÓN (dibujados a pantalla, centrados en
   630→// (cx, cy) = coordenadas de pantalla del centro del prop — el
   631→// motor pasará sx(px), sy(py) como hace drawProps()).
   632→// ============================================================
   633→
   634→export function drawExpansionProp(
   635→  ctx: CanvasRenderingContext2D, kind: string, cx: number, cy: number,
   636→  zoom: number, t: number, lit: boolean,
   637→): void {
   638→  const Z = zoom;
   639→  switch (kind) {
   640→    case 'wreck': {
   641→      // ---- casco de nave naufragada (~40×24) ----
   642→      ctx.save();
   643→      ctx.translate(cx, cy);
   644→      ctx.rotate(-0.16);
   645→      // sombra
   646→      ctx.globalAlpha = 0.25;
   647→      ctx.fillStyle = '#000';
   648→      ctx.beginPath();
   649→      ctx.ellipse(0, 11 * Z, 22 * Z, 4 * Z, 0, 0, Math.PI * 2);
   650→      ctx.fill();
   651→      ctx.globalAlpha = 1;
   652→      // casco: franjas apiladas (quilla arriba, panza abajo)
   653→      for (let i = 0; i < 5; i++) {
   654→        const w = (17 - i * 2.4) * Z;
   655→        ctx.fillStyle = i % 2 === 0 ? '#6d4a2a' : '#54381e';
   656→        ctx.fillRect(-w, (6 - i * 4) * Z, w * 2, 4 * Z);
   657→      }
   658→      // borde superior desbordado (cubierta rota)
   659→      ctx.fillStyle = '#8a6438';
   660→      ctx.fillRect(-14 * Z, -12 * Z, 10 * Z, 2 * Z);
   661→      ctx.fillStyle = '#54381e';
   662→      ctx.fillRect(2 * Z, -11 * Z, 7 * Z, 2 * Z);
   663→      // tablón suelto
   664→      ctx.fillStyle = '#7d5630';
   665→      ctx.fillRect(-4 * Z, -9 * Z, 9 * Z, 1.5 * Z);
   666→      // musgo
   667→      ctx.fillStyle = '#4a7a3e';
   668→      ctx.fillRect(-9 * Z, -6 * Z, 5 * Z, 2 * Z);
   669→      ctx.fillRect(6 * Z, 1 * Z, 4 * Z, 2 * Z);
   670→      // mástil roto
   671→      ctx.fillStyle = '#54381e';
   672→      ctx.fillRect(6 * Z, -26 * Z, 2.4 * Z, 22 * Z);
   673→      ctx.fillStyle = '#3e2a16';
   674→      ctx.fillRect(7.4 * Z, -26 * Z, 1 * Z, 22 * Z);
   675→      // vela desgarrada ondeando con sin(t)
   676→      const wav = Math.sin(t * 1.7) * 3;
   677→      ctx.globalAlpha = 0.92;
   678→      ctx.fillStyle = '#cfc8b4';
   679→      ctx.beginPath();
   680→      ctx.moveTo(8.6 * Z, -25 * Z);
   681→      ctx.quadraticCurveTo((26 + wav) * Z, (-20 + wav * 0.6) * Z, (22 + wav) * Z, (-5 + wav) * Z);
   682→      ctx.lineTo(15 * Z, (-9 + wav * 0.4) * Z);
   683→      ctx.lineTo(10.5 * Z, (-15 + wav * 0.2) * Z);
   684→      ctx.closePath();
   685→      ctx.fill();
   686→      // jirones (desgarrada)
   687→      ctx.globalAlpha = 0.85;
   688→      ctx.fillStyle = '#a89f8a';
   689→      ctx.fillRect((17 + wav * 0.5) * Z, (-13 + wav) * Z, 4 * Z, 2 * Z);
   690→      ctx.fillRect(11 * Z, -19 * Z, 3 * Z, 2 * Z);
   691→      ctx.globalAlpha = 1;
   692→      ctx.restore();
   693→      break;
   694→    }
   695→    case 'faro': {
   696→      // ---- faro en ruinas (~20×48): torre de piedra con franjas ----
   697→      // 8 segmentos de 6 px → 48 px de alto, se estrecha hacia arriba
   698→      for (let i = 0; i < 8; i++) {
   699→        const yTop = cy + (14 - i * 6) * Z - 6 * Z;
   700→        const halfW = (10 - i * 0.5) * Z;
   701→        ctx.fillStyle = i % 2 === 0 ? '#c8c2b4' : '#9a5a4a';
   702→        ctx.fillRect(cx - halfW, yTop, halfW * 2, 6 * Z + 0.6);
   703→      }
   704→      // piedras caídas / ruina
   705→      ctx.fillStyle = '#8a8478';
   706→      ctx.fillRect(cx - 11 * Z, cy + 8 * Z, 3 * Z, 2 * Z);
   707→      ctx.fillRect(cx + 7 * Z, cy + 2 * Z, 4 * Z, 2 * Z);
   708→      // grieta
   709→      ctx.fillStyle = 'rgba(40,36,30,0.5)';
   710→      ctx.fillRect(cx - 2 * Z, cy - 10 * Z, 1.2 * Z, 10 * Z);
   711→      ctx.fillRect(cx - 1 * Z, cy - 2 * Z, 1.2 * Z, 6 * Z);
   712→      // linterna
   713→      const gy = cy - 34 * Z;
   714→      ctx.fillStyle = '#3a3e48';
   715→      ctx.fillRect(cx - 6 * Z, gy, 12 * Z, 8 * Z);
   716→      if (lit) {
   717→        // luz superior que pulsa con sin(t*2.4)
   718→        const pulse = 0.55 + Math.sin(t * 2.4) * 0.45;
   719→        ctx.globalAlpha = pulse;
   720→        ctx.fillStyle = '#ffe9a0';
   721→        ctx.fillRect(cx - 4 * Z, gy + 2 * Z, 8 * Z, 4 * Z);
   722→        ctx.globalAlpha = 0.25 + pulse * 0.3;
   723→        ctx.fillStyle = '#ffe9a0';
   724→        ctx.beginPath();
   725→        ctx.arc(cx, gy + 4 * Z, 10 * Z, 0, Math.PI * 2);
   726→        ctx.fill();
   727→        // haz cónico semitransparente rotando lento
   728→        ctx.globalAlpha = 0.12;
   729→        ctx.save();
   730→        ctx.translate(cx, gy + 4 * Z);
   731→        ctx.rotate(t * 0.35);
   732→        ctx.fillStyle = '#fff3c8';
   733→        const L = 64 * Z;
   734→        ctx.beginPath();
   735→        ctx.moveTo(0, -3 * Z); ctx.lineTo(L, -12 * Z); ctx.lineTo(L, 12 * Z); ctx.closePath();
   736→        ctx.fill();
   737→        ctx.beginPath();
   738→        ctx.moveTo(0, -3 * Z); ctx.lineTo(-L, -12 * Z); ctx.lineTo(-L, 12 * Z); ctx.closePath();
   739→        ctx.fill();
   740→        ctx.restore();
   741→        ctx.globalAlpha = 1;
   742→      } else {
   743→        // ventana oscura
   744→        ctx.fillStyle = '#1c2028';
   745→        ctx.fillRect(cx - 4 * Z, gy + 2 * Z, 8 * Z, 4 * Z);
   746→      }
   747→      // cúpula rota
   748→      ctx.fillStyle = '#6a6458';
   749→      ctx.fillRect(cx - 6 * Z, gy - 3 * Z, 5 * Z, 3 * Z);
   750→      ctx.fillRect(cx + 1 * Z, gy - 2 * Z, 5 * Z, 2 * Z);
   751→      break;
   752→    }
   753→    case 'lamp': {
   754→      // ---- farol de pie (~10×18) ----
   755→      // peana + poste
   756→      ctx.fillStyle = '#2a2a32';
   757→      ctx.fillRect(cx - 3 * Z, cy + 6 * Z, 6 * Z, 2 * Z);
   758→      ctx.fillStyle = '#3a3a44';
   759→      ctx.fillRect(cx - 1 * Z, cy - 8 * Z, 2 * Z, 15 * Z);
   760→      // caja del farol
   761→      ctx.fillStyle = '#3a3a44';
   762→      ctx.fillRect(cx - 3.5 * Z, cy - 16 * Z, 7 * Z, 9 * Z);
   763→      ctx.fillStyle = '#2a2a32';
   764→      ctx.fillRect(cx - 4.5 * Z, cy - 18 * Z, 9 * Z, 2.5 * Z);
   765→      if (lit) {
   766→        // llama cálida parpadeante + halo dorado
   767→        const fl = 0.7 + Math.sin(t * 9) * 0.2 + Math.sin(t * 23.7) * 0.1;
   768→        ctx.globalAlpha = 0.22 + fl * 0.18;
   769→        ctx.fillStyle = '#ffd88a';
   770→        ctx.beginPath();
   771→        ctx.arc(cx, cy - 11.5 * Z, 10 * Z, 0, Math.PI * 2);
   772→        ctx.fill();
   773→        ctx.globalAlpha = 1;
   774→        ctx.fillStyle = `rgba(255,216,138,${0.7 + fl * 0.3})`;
   775→        ctx.fillRect(cx - 2.5 * Z, cy - 14.5 * Z, 5 * Z, 6 * Z);
   776→        ctx.fillStyle = '#fff6d8';
   777→        ctx.fillRect(cx - 1 * Z, cy - 13 * Z, 2 * Z, 3 * Z);
   778→      } else {
   779→        // cristal apagado gris
   780→        ctx.fillStyle = '#8a8f98';
   781→        ctx.fillRect(cx - 2.5 * Z, cy - 14.5 * Z, 5 * Z, 6 * Z);
   782→        ctx.fillStyle = '#6a6f78';
   783→        ctx.fillRect(cx - 1 * Z, cy - 12 * Z, 2 * Z, 3 * Z);
   784→      }
   785→      break;
   786→    }
   787→    default:
   788→      break; // prop no gestionado: el motor usa su dibujo propio
   789→  }
   790→}

   798→/** Corchea pixelada reutilizable (nota musical). */
   799→function notaGlyph(ctx: CanvasRenderingContext2D, x: number, y: number, s: number, col: string) {
   800→  const u = s / 6;
   801→  ctx.fillStyle = col;
   802→  // cabeza
   803→  ctx.fillRect(x - 1.6 * u, y + 1.2 * u, 2.6 * u, 1.8 * u);
   804→  // mástil
   805→  ctx.fillRect(x + 0.4 * u, y - 2.6 * u, 0.8 * u, 4.2 * u);
   806→  // banderín
   807→  ctx.fillRect(x + 1.2 * u, y - 2.6 * u, 0.8 * u, 1.6 * u);
   808→  ctx.fillRect(x + 2 * u, y - 1.6 * u, 0.6 * u, 1.2 * u);
   809→}
   810→
   811→export function drawExpansionProjectile(
   812→  ctx: CanvasRenderingContext2D, sprite: string, x: number, y: number,
   813→  r: number, zoom: number, t: number,
   814→): boolean {
   815→  switch (sprite) {
   816→    case 'orb': {
   817→      // esfera de agua con highlight y ondulación
   818→      const wob = 1 + Math.sin(t * 6) * 0.12;
   819→      const R = Math.max(3, r) * zoom * wob;
   820→      ctx.globalAlpha = 0.28;
   821→      ctx.fillStyle = '#2e78b8';
   822→      ctx.beginPath(); ctx.arc(x, y, R + 1.5, 0, Math.PI * 2); ctx.fill();
   823→      ctx.globalAlpha = 0.85;
   824→      ctx.fillStyle = '#58a8e0';
   825→      ctx.beginPath(); ctx.arc(x, y, R, 0, Math.PI * 2); ctx.fill();
   826→      ctx.globalAlpha = 0.9;
   827→      ctx.fillStyle = '#9ad0f0';
   828→      ctx.beginPath(); ctx.arc(x - R * 0.2, y - R * 0.25, R * 0.55, 0, Math.PI * 2); ctx.fill();
   829→      ctx.globalAlpha = 1;
   830→      ctx.fillStyle = '#f4fbff';
   831→      ctx.fillRect(x - R * 0.45, y - R * 0.55, Math.max(1.5, R * 0.35), Math.max(1.5, R * 0.35));
   832→      return true;
   833→    }
   834→    case 'shard': {
   835→      // cristal de hielo afilado girando lento
   836→      ctx.save();
   837→      ctx.translate(x, y);
   838→      ctx.rotate(t * 3);
   839→      const s = Math.max(3, r + 1) * zoom;
   840→      ctx.fillStyle = '#5aa8cc';
   841→      ctx.beginPath();
   842→      ctx.moveTo(0, -s); ctx.lineTo(s * 0.55, 0); ctx.lineTo(0, s); ctx.lineTo(-s * 0.55, 0);
   843→      ctx.closePath(); ctx.fill();
   844→      ctx.fillStyle = '#9fd4ec';
   845→      ctx.beginPath();
   846→      ctx.moveTo(0, -s * 0.8); ctx.lineTo(s * 0.3, 0); ctx.lineTo(0, s * 0.8); ctx.lineTo(-s * 0.3, 0);
   847→      ctx.closePath(); ctx.fill();
   848→      ctx.fillStyle = '#eaf8ff';
   849→      ctx.beginPath();
   850→      ctx.moveTo(0, -s * 0.45); ctx.lineTo(s * 0.12, 0); ctx.lineTo(0, s * 0.45); ctx.lineTo(-s * 0.12, 0);
   851→      ctx.closePath(); ctx.fill();
   852→      ctx.restore();
   853→      return true;
   854→    }
   855→    case 'nota': {
   856→      // nota musical luminosa con estela parpadeante
   857→      const bob = Math.sin(t * 4) * 1.5;
   858→      const pulse = 0.7 + Math.sin(t * 8) * 0.3;
   859→      const s = Math.max(3, r + 1) * zoom;
   860→      ctx.globalAlpha = 0.22 * pulse;
   861→      ctx.fillStyle = '#bfe0ff';
   862→      ctx.beginPath(); ctx.arc(x, y + bob, s * 1.6, 0, Math.PI * 2); ctx.fill();
   863→      for (let i = 1; i <= 2; i++) {
   864→        ctx.globalAlpha = (0.3 / i) * pulse;
   865→        notaGlyph(ctx, x - i * 2, y + bob - i * 3, s * (1 - i * 0.18), '#9fc8e8');
   866→      }
   867→      ctx.globalAlpha = 1;
   868→      notaGlyph(ctx, x, y + bob, s, '#f2f9ff');
   869→      return true;
   870→    }
   871→    default:
   872→      return false; // no gestionado: fallback del motor
   873→  }
   874→}

   949→// ---------------- El Coro Roto (30×30 · 2 frames × 3 máscaras · JEFE) ----
   950→// MASA de tres máscaras flotantes unidas por hilos de bruma: una central
   951→// grande cantando al revés (boca vertical) y dos menores a los flancos.
   952→// buildCoro(fase) devuelve la variante de la fase:
   953→//   1 · MÁSCARA DEL PULSO    — hueso claro, glow aguamarina, intacta.
   954→//   2 · MÁSCARA DEL VERA     — hueso frío, glow violeta, grietas finas.
   955→//   3 · MÁSCARA DEL SILENCIO — hueso apagado, glow dorado, cuarteadas.
   956→
   957→function buildCoro(fase: 1 | 2 | 3): Frames {
   958→  const pal = fase === 1
   959→    ? { HUESO: '#d8d4c4', HUESO2: '#bdb8a6', GLOW: '#7ee8ff', HUECO: '#1a2030', FILO: '#eef2e8' }
   960→    : fase === 2
   961→      ? { HUESO: '#c8c2b2', HUESO2: '#aaa494', GLOW: '#b48fff', HUECO: '#1c1830', FILO: '#e4e2da' }
   962→      : { HUESO: '#c4bcae', HUESO2: '#9c9486', GLOW: '#ffd88a', HUECO: '#241f28', FILO: '#dcd8cc' };
   963→  const frames: Frames = [];
   964→  for (let f = 0; f < 2; f++) {
   965→    const { c, x } = cv(30, 30);
   966→    const bob = f === 0 ? 0 : 1;
   967→    const { HUESO, HUESO2, GLOW, HUECO, FILO } = pal;
   968→    // ---- hilos de bruma que unen la masa ----
   969→    x.globalAlpha = 0.4;
   970→    rc(x, 8, 14 + bob, 4, 1, GLOW);
   971→    rc(x, 18, 14 - bob, 4, 1, GLOW);
   972→    x.globalAlpha = 0.22;
   973→    disc(x, 15, 13 + bob, 11, GLOW);                 // aura compartida
   974→    x.globalAlpha = 1;
   975→    // ---- máscara central (boca vertical = canto al revés) ----
   976→    rc(x, 10, 5 + bob, 10, 13, HUESO);
   977→    rc(x, 9, 6 + bob, 12, 11, HUESO);
   978→    rc(x, 10, 17 + bob, 10, 2, HUESO2);              // mentón
   979→    rc(x, 9, 6 + bob, 1, 9, HUESO2);
   980→    rc(x, 20, 6 + bob, 1, 9, HUESO2);
   981→    // ojos huecos (el central mayor según la fase: se apaga el coro)
   982→    const eyeH = fase === 1 ? 2 : fase === 2 ? 3 : 4;
   983→    rc(x, 11, 8 + bob, 2, eyeH, HUECO);
   984→    rc(x, 17, 8 + bob, 2, eyeH, HUECO);
   985→    rc(x, 11, 8 + bob, 2, 1, FILO);
   986→    rc(x, 17, 8 + bob, 2, 1, FILO);
   987→    // boca vertical (la nota al revés) + glow interior
   988→    rc(x, 14, 11 + bob, 2, 5, HUECO);
   989→    x.globalAlpha = 0.85;
   990→    rc(x, 14, 11 + bob, 1, 4, GLOW);
   991→    x.globalAlpha = 1;
   992→    // grietas: finas en F2, cuarteadas en F3
   993→    if (fase >= 2) {
   994→      rc(x, 12, 5 + bob, 1, 3, HUESO2);
   995→      rc(x, 18, 13 + bob, 1, 4, HUESO2);
   996→    }
   997→    if (fase === 3) {
   998→      rc(x, 10, 12 + bob, 3, 1, HUECO);
   999→      rc(x, 16, 5 + bob, 1, 4, HUECO);
  1000→      rc(x, 13, 18 + bob, 4, 1, HUECO);
  1001→      rc(x, 19, 9 + bob, 1, 3, HUESO2);
  1002→    }
  1003→    // ---- máscara izquierda (perfil, cuelga más baja) ----
  1004→    rc(x, 3, 12 + bob, 6, 8, HUESO2);
  1005→    rc(x, 2, 13 + bob, 1, 6, HUESO2);
  1006→    rc(x, 4, 15 + bob, 2, 2, HUECO);                 // ojo ladeado
  1007→    rc(x, 5, 20 + bob, 1, 2, HUECO);                 // boca pequeña
  1008→    // ---- máscara derecha (perfil, más alta) ----
  1009→    rc(x, 21, 9 + bob, 6, 8, HUESO2);
  1010→    rc(x, 27, 10 + bob, 1, 6, HUESO2);
  1011→    rc(x, 23, 12 + bob, 2, 2, HUECO);
  1012→    rc(x, 24, 17 + bob, 1, 2, HUECO);
  1013→    // destello del glow (parpadeo por frame)
  1014→    if (f === 1) {
  1015→      x.globalAlpha = 0.7;
  1016→      rc(x, 14, 15 + bob, 1, 1, FILO);
  1017→      rc(x, 6, 11 + bob, 1, 1, GLOW);
  1018→      rc(x, 23, 8 + bob, 1, 1, GLOW);
  1019→      x.globalAlpha = 1;
  1020→    }
  1021→    frames.push(c);
  1022→  }
  1023→  return frames;
