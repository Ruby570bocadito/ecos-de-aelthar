    90→function drawNote(ctx: CanvasRenderingContext2D, x: number, y: number, s: number, color: string, variant: number) {
    91→  ctx.fillStyle = color;
    92→  ctx.fillRect(x, y + s * 3, s * 2, s * 1.5);          // cabeza
    93→  ctx.fillRect(x + s * 1.5, y, s * 0.75, s * 3.6);     // palo
    94→  if (variant === 0) {
    95→    ctx.fillRect(x + s * 2, y, s * 1.25, s * 1.1);     // banderín
    96→  } else {
    97→    ctx.fillRect(x + s * 2.6, y + s * 2.4, s * 2, s * 1.5); // segunda cabeza
    98→    ctx.fillRect(x + s * 1.5, y, s * 3, s * 0.8);      // barra
    99→  }

   209→// helpers deterministas locales (evitan importar hash2 aquí)
   210→function hashT(i: number, k: number): number {
   211→  let h = (i + k * 57) * 2654435761;
   212→  h = (h ^ (h >> 13)) * 1274126177;
   213→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;
   214→}
   215→function getSprWisp(t: number, i: number): HTMLCanvasElement {
   216→  const frames = getSpr('wisp');
   217→  const f = frames?.length ? frames[Math.floor(t * 3 + i) % frames.length] : (undefined as unknown as HTMLCanvasElement);
   218→  if (!f) {
   219→    const w = window as unknown as { __wispMiss?: unknown[] };
   220→    if (!w.__wispMiss) w.__wispMiss = [];
   221→    if (w.__wispMiss.length < 3) {
   222→      w.__wispMiss.push({
   223→        t: Math.round(performance.now()), i,
   224→        hasFrames: !!frames, len: frames?.length ?? -1,
   225→        hasG: !!(window as unknown as { __g?: unknown }).__g,
   226→        heroAlba: !!getSpr('hero_alba'),
   227→      });
   228→    }
   229→  }
   230→  return f;
   231→}

   233→// esquinas doradas al pasar el ratón (hover más vivo)
   234→function hoverCorners(g: Game, x: number, y: number, w: number, h: number, col = COL.gold) {
   235→  const hov = g.mouse.x >= x && g.mouse.x <= x + w && g.mouse.y >= y && g.mouse.y <= y + h;
   236→  if (!hov) return;
   237→  const ctx = g.ctx;
   238→  const c = 7;
   239→  ctx.fillStyle = col;
   240→  ctx.fillRect(x - 2, y - 2, c, 2); ctx.fillRect(x - 2, y - 2, 2, c);
   241→  ctx.fillRect(x + w + 2 - c, y - 2, c, 2); ctx.fillRect(x + w, y - 2, 2, c);
   242→  ctx.fillRect(x - 2, y + h, c, 2); ctx.fillRect(x - 2, y + h - c, 2, c);
   243→  ctx.fillRect(x + w + 2 - c, y + h, c, 2); ctx.fillRect(x + w, y + h - c, 2, c);
   244→}

   530→function drawDialogue(g: Game) {
   531→  const ctx = g.ctx;
   532→  const node = g.dlgNode;
   533→  if (!node) return;
   534→  const bw = VIEW_W - 80, bx = 40;
   535→  // la caja crece según texto y opciones para que nunca se solapen
   536→  const fullLines = wrapText(node.text, 76).length;
   537→  const nOpts = node.options?.length ?? 0;
   538→  const bh = Math.max(150, 46 + fullLines * 19 + (nOpts > 0 ? nOpts * 26 + 14 : 0));
   539→  const by = VIEW_H - bh - 24;
   540→  ctx.fillStyle = 'rgba(4,5,12,0.4)';
   541→  ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   542→  panel(g, bx, by, bw, bh, '#6a5a38', 'rgba(10,12,22,0.96)');
   543→  // borde doble pulido + remaches
   544→  ctx.strokeStyle = 'rgba(240,200,74,0.4)';
   545→  ctx.lineWidth = 1;
   546→  ctx.strokeRect(bx + 4.5, by + 4.5, bw - 9, bh - 9);
   547→  ctx.fillStyle = COL.gold;
   548→  for (const [cxx, cyy] of [[bx + 7, by + 7], [bx + bw - 7, by + 7], [bx + 7, by + bh - 7], [bx + bw - 7, by + bh - 7]]) {
   549→    ctx.fillRect(cxx - 1.5, cyy - 1.5, 3, 3);
   550→  }
   551→
   552→  // retrato pixel (respira 1px y parpadea de vez en cuando)
   553→  const pkey = mapPortrait(node.portrait);
   554→  const blink = (g.globalT % 3.7) < 0.14;
   555→  const bob = Math.round(Math.sin(g.globalT * 2.3) * 0.9);
   556→  const frS = 74;
   557→  const frX = bx + 14, frY = by + 14;
   558→  ctx.fillStyle = '#0e0c18';
   559→  ctx.fillRect(frX, frY, frS, frS);
   560→  drawPortrait(ctx, pkey, frX + (frS - 28 * 1.72) / 2, frY + (frS - 40 * 1.72) / 2 + bob, 1.72, blink);
   561→  ctx.strokeStyle = '#6a5a38';
   562→  ctx.strokeRect(frX, frY, frS, frS);
   563→  ctx.strokeStyle = 'rgba(240,200,74,0.35)';
   564→  ctx.strokeRect(frX + 2.5, frY + 2.5, frS - 5, frS - 5);
   565→
   566→  // nombre y texto
   567→  text(g, node.name, bx + 104, by + 12, 17, COL.goldSoft);
   568→  const shown = node.text.slice(0, Math.floor(g.dlgCharT));
   569→  const lines = wrapText(shown, 76);
   570→  lines.forEach((l, i) => text(g, l, bx + 104, by + 36 + i * 19, 17, COL.text));
   571→
   572→  // opciones (debajo de la última línea de texto, sin solaparse nunca)
   573→  if (node.options && node.options.length > 0 && g.dlgCharT >= node.text.length) {
   574→    const oy = by + 36 + fullLines * 19 + 12;
   575→    for (let i = 0; i < node.options.length; i++) {
   576→      const opt = node.options[i];
   577→      const selected = g.dlgSel === i;
   578→      const ox = bx + 104;
   579→      const hover = addHit(g, ox, oy + i * 26, 400, 24, () => {
   580→        g.dlgSel = i;
   581→        g.advanceDialogue();
   582→      });
   583→      if (selected || hover) {
   584→        ctx.fillStyle = 'rgba(240,200,74,0.14)';
   585→        ctx.fillRect(ox - 4, oy + i * 26, 404, 24);
   586→      }
   587→      text(g, `${selected ? '▸' : ' '} ${opt.text}`, ox, oy + i * 26 + 3, 16, selected || hover ? COL.gold : COL.text);
   588→    }
   589→  } else if (g.dlgCharT >= node.text.length) {
   590→    if (Math.sin(g.globalT * 5) > -0.2) text(g, 'E ▸', bx + bw - 44, by + bh - 30, 16, COL.dim, 'center');
   591→  }
   592→}
