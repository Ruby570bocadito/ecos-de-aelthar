    96→function drawNote(ctx: CanvasRenderingContext2D, x: number, y: number, s: number, color: string, variant: number) {
    97→  ctx.fillStyle = color;
    98→  ctx.fillRect(x, y + s * 3, s * 2, s * 1.5);          // cabeza
    99→  ctx.fillRect(x + s * 1.5, y, s * 0.75, s * 3.6);     // palo
   100→  if (variant === 0) {
   101→    ctx.fillRect(x + s * 2, y, s * 1.25, s * 1.1);     // banderín
   102→  } else {
   103→    ctx.fillRect(x + s * 2.6, y + s * 2.4, s * 2, s * 1.5); // segunda cabeza
   104→    ctx.fillRect(x + s * 1.5, y, s * 3, s * 0.8);      // barra
   105→  }
   106→}

   225→// helpers deterministas locales (evitan importar hash2 aquí)
   226→function hashT(i: number, k: number): number {
   227→  let h = (i + k * 57) * 2654435761;
   228→  h = (h ^ (h >> 13)) * 1274126177;
   229→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;
   230→}
   231→function getSprWisp(t: number, i: number): HTMLCanvasElement {
   232→  const frames = getSpr('wisp');
   233→  const f = frames?.length ? frames[Math.floor(t * 3 + i) % frames.length] : (undefined as unknown as HTMLCanvasElement);
   234→  if (!f) {
   235→    const w = window as unknown as { __wispMiss?: unknown[] };
   236→    if (!w.__wispMiss) w.__wispMiss = [];
   237→    if (w.__wispMiss.length < 3) {
   238→      w.__wispMiss.push({
   239→        t: Math.round(performance.now()), i,
   240→        hasFrames: !!frames, len: frames?.length ?? -1,
   241→        hasG: !!(window as unknown as { __g?: unknown }).__g,
   242→        heroAlba: !!getSpr('hero_alba'),
   243→      });
   244→    }
   245→  }
   246→  return f;
   247→}

   250→function hoverCorners(g: Game, x: number, y: number, w: number, h: number, col = COL.gold) {
   251→  const hov = g.mouse.x >= x && g.mouse.x <= x + w && g.mouse.y >= y && g.mouse.y <= y + h;
   252→  if (!hov) return;
   253→  const ctx = g.ctx;
   254→  const c = 7;
   255→  ctx.fillStyle = col;
   256→  ctx.fillRect(x - 2, y - 2, c, 2); ctx.fillRect(x - 2, y - 2, 2, c);
   257→  ctx.fillRect(x + w + 2 - c, y - 2, c, 2); ctx.fillRect(x + w, y - 2, 2, c);
   258→  ctx.fillRect(x - 2, y + h, c, 2); ctx.fillRect(x - 2, y + h - c, 2, c);
   259→  ctx.fillRect(x + w + 2 - c, y + h, c, 2); ctx.fillRect(x + w, y + h - c, 2, c);
   260→}

   548→function drawDialogue(g: Game) {
   549→  const ctx = g.ctx;
   550→  const node = g.dlgNode;
   551→  if (!node) return;
   552→  const bw = VIEW_W - 80, bx = 40;
   553→  // la caja crece según texto y opciones para que nunca se solapen
   554→  const fullLines = wrapText(node.text, 76).length;
   555→  const nOpts = node.options?.length ?? 0;
   556→  const bh = Math.max(150, 46 + fullLines * 19 + (nOpts > 0 ? nOpts * 26 + 14 : 0));
   557→  const by = VIEW_H - bh - 24;
   558→  ctx.fillStyle = 'rgba(4,5,12,0.4)';
   559→  ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   560→  panel(g, bx, by, bw, bh, '#6a5a38', 'rgba(10,12,22,0.96)');
   561→  // borde doble pulido + remaches
   562→  ctx.strokeStyle = 'rgba(240,200,74,0.4)';
   563→  ctx.lineWidth = 1;
   564→  ctx.strokeRect(bx + 4.5, by + 4.5, bw - 9, bh - 9);
   565→  ctx.fillStyle = COL.gold;
   566→  for (const [cxx, cyy] of [[bx + 7, by + 7], [bx + bw - 7, by + 7], [bx + 7, by + bh - 7], [bx + bw - 7, by + bh - 7]]) {
   567→    ctx.fillRect(cxx - 1.5, cyy - 1.5, 3, 3);
   568→  }
   569→
   570→  // retrato pixel (respira 1px y parpadea de vez en cuando)
   571→  const pkey = mapPortrait(node.portrait);
   572→  const blink = (g.globalT % 3.7) < 0.14;
   573→  const bob = Math.round(Math.sin(g.globalT * 2.3) * 0.9);
   574→  const frS = 74;
   575→  const frX = bx + 14, frY = by + 14;
   576→  ctx.fillStyle = '#0e0c18';
   577→  ctx.fillRect(frX, frY, frS, frS);
   578→  drawPortrait(ctx, pkey, frX + (frS - 28 * 1.72) / 2, frY + (frS - 40 * 1.72) / 2 + bob, 1.72, blink);
   579→  ctx.strokeStyle = '#6a5a38';
   580→  ctx.strokeRect(frX, frY, frS, frS);
   581→  ctx.strokeStyle = 'rgba(240,200,74,0.35)';
   582→  ctx.strokeRect(frX + 2.5, frY + 2.5, frS - 5, frS - 5);
   583→
   584→  // nombre y texto
   585→  text(g, node.name, bx + 104, by + 12, 17, COL.goldSoft);
   586→  const shown = node.text.slice(0, Math.floor(g.dlgCharT));
   587→  const lines = wrapText(shown, 76);
   588→  lines.forEach((l, i) => text(g, l, bx + 104, by + 36 + i * 19, 17, COL.text));
   589→
   590→  // opciones (debajo de la última línea de texto, sin solaparse nunca)
   591→  if (node.options && node.options.length > 0 && g.dlgCharT >= node.text.length) {
   592→    const oy = by + 36 + fullLines * 19 + 12;
   593→    for (let i = 0; i < node.options.length; i++) {
   594→      const opt = node.options[i];
   595→      const selected = g.dlgSel === i;
   596→      const ox = bx + 104;
   597→      const hover = addHit(g, ox, oy + i * 26, 400, 24, () => {
   598→        g.dlgSel = i;
   599→        g.advanceDialogue();
   600→      });
   601→      if (selected || hover) {
   602→        ctx.fillStyle = 'rgba(240,200,74,0.14)';
   603→        ctx.fillRect(ox - 4, oy + i * 26, 404, 24);
   604→      }
   605→      text(g, `${selected ? '▸' : ' '} ${opt.text}`, ox, oy + i * 26 + 3, 16, selected || hover ? COL.gold : COL.text);
   606→    }
   607→  } else if (g.dlgCharT >= node.text.length) {
   608→    if (Math.sin(g.globalT * 5) > -0.2) text(g, 'E ▸', bx + bw - 44, by + bh - 30, 16, COL.dim, 'center');
   609→  }
