   143→/** RNG determinista por oleada (convención del proyecto: sin azar de ejecución). */
   144→function mulberry32(seed: number): () => number {
   145→  let a = seed >>> 0;
   146→  return () => {
   147→    a |= 0; a = (a + 0x6d2b79f5) | 0;
   148→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   149→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   150→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   151→  };
   152→}

   179→function snapshotCampaign(g: Game): ChallengeSnap {
   180→  const p = g.player;
   181→  let saveRaw: string | null = null;
   182→  try { saveRaw = localStorage.getItem(SAVE_KEY); } catch { /* noop */ }
   183→  return {
   184→    fromCampaign: !!p,
   185→    gold: p?.gold ?? 0,
   186→    potions: p?.potions ?? 0,
   187→    deaths: p?.deaths ?? 0,
   188→    kills: p?.kills ?? 0,
   189→    playTime: p?.playTime ?? 0,
   190→    hp: p?.hp ?? 0, sta: p?.sta ?? 0, res: p?.res ?? 0,
   191→    questIdx: g.questIdx, questStep: g.questStep,
   192→    flags: { ...g.flags },
   193→    memories: [...(p?.memories ?? [])],
   194→    repFacciones: p?.repFacciones ? { ...p.repFacciones } : undefined,
   195→    deadGolds: g.deadGolds.slice(),
   196→    saveRaw,
   197→  };
   198→}

   200→/**
   201→ * Devuelve la campaña a su estado exacto de antes del reto. Se llama en
   202→ * finish() (victoria/derrota/aborto). La XP y el nivel ganados en la arena
   203→ * NO se restauran: son la recompensa persistente de haber sobrevivido.
   204→ */
   205→function restoreCampaign(g: Game, run: ChallengeRun): void {
   206→  const s = run.snap;
   207→  if (!s) return;
   208→  const p = g.player;
   209→  if (s.fromCampaign && p) {
   210→    p.gold = s.gold;
   211→    p.potions = s.potions;
   212→    p.deaths = s.deaths;
   213→    p.kills = s.kills;
   214→    p.playTime = s.playTime;
   215→    p.hp = s.hp; p.sta = s.sta; p.res = s.res;
   216→    p.memories = s.memories;
   217→    p.repFacciones = s.repFacciones;
   218→    g.questIdx = s.questIdx;
   219→    g.questStep = s.questStep;
   220→    g.flags = { ...s.flags };
   221→    g.deadGolds = s.deadGolds;
   222→  } else if (!s.fromCampaign) {
   223→    // Portador temporal: el oro/pociones que acumuló en la arena no significan
   224→    // nada; endChallenge lo retira al volver al título (aquí aún lo usa el HUD).
   225→    g.deadGolds = [];
   226→  }
   227→  // ¿el jugador guardó desde la pausa estando DENTRO de la arena? El fichero
   228→  // habría serializado mapId 'arena' + estado a medio reto. Marcamos la
   229→  // reparación y endChallenge la ejecuta sobre lunaris ya cargado.
   230→  let raw: string | null = null;
   231→  try { raw = localStorage.getItem(SAVE_KEY); } catch { /* noop */ }
   232→  if (raw !== s.saveRaw) run.resave = s.saveRaw === null ? 'remove' : 'rewrite';
   233→}

   237→/** Escalado de la oleada: hp ×(1 + 0.12·(oleada−1)) — la espec del encargo. */
   238→function waveMult(w: number): number { return 1 + 0.12 * (w - 1); }
   239→
   240→/**
   241→ * Mezcla de enemigos por oleada. Tablas fijas 1–7 (rampa legible: Acto I,
   242→ * mini-jefe cada 3, tipos del Acto II desde la 4) y mezcla determinista
   243→ * (semilla = nº de oleada) a partir de la 8. El daño NO se escala por
   244→ * instancia (los cerebros leen ENEMY_DEFS): la presión sube por HP, por
   245→ * mezcla (Acto II pega más) y por el Centinela.
   246→ */
   247→function waveComposition(w: number): EnemyType[] {
   248→  if (w === 1) return ['lobo', 'lobo', 'lobo'];
   249→  if (w === 2) return ['lobo', 'sombra', 'lobo', 'sombra'];
   250→  if (w === 3) return ['guardian', 'esqueleto', 'esqueleto'];
   251→  if (w === 4) return ['neumo', 'neumo', 'espectro', 'espectro'];
   252→  if (w === 5) return ['arpi', 'arpi', 'espectro', 'neumo'];
   253→  if (w === 6) return ['guardian', 'arpi', 'arpi', 'espectro'];
   254→  if (w === 7) return ['espectro', 'espectro', 'neumo', 'arpi', 'sombra', 'lobo'];
   255→  const pool: EnemyType[] = ['lobo', 'sombra', 'esqueleto', 'neumo', 'espectro', 'arpi'];
   256→  const rng = mulberry32(w * 7919);
   257→  const n = Math.min(10, 5 + Math.floor((w - 7) / 2));
   258→  const out: EnemyType[] = [];
   259→  for (let i = 0; i < n; i++) out.push(pool[Math.floor(rng() * pool.length)]);
   260→  if (w % 3 === 0) out.unshift('guardian');
   261→  return out;
   262→}
   263→
   264→/**
   265→ * Instancia un enemigo de oleada alrededor del Portador: anillo de 8–14
   266→ * tiles (nunca encima), dentro del recinto y solo sobre suelo libre
   267→ * (tileSolidAt + boxFree con el hitbox del tipo). Reintenta 40 ángulos; si
   268→ * el jugador está acorralado, entra por el sur del anillo.
   269→ */
   270→function spawnArenaEnemy(g: Game, run: ChallengeRun, type: EnemyType, mult: number): void {
   271→  const p = g.player!;
   272→  const def = ENEMY_DEFS[type];
   273→  // hitbox idéntica a makeEnemy (guardian/sirena/golem grandes)
   274→  const big = type === 'guardian' || type === 'sirena' || type === 'golem';
   275→  const bw = big ? 22 : 12;
   276→  const bh = big ? 16 : 10;
   277→  const push = (x: number, y: number) => {
   278→    const e = g.makeEnemy(type, x, y, 2, 'arena');
   279→    // mini-jefe (guardián) un 25% más blando: es un escollo de oleada, no la pelea de campaña
   280→    const scaled = Math.max(1, Math.round(def.hp * (type === 'guardian' ? mult * 0.75 : mult)));
   281→    e.hp = scaled; e.maxHp = scaled;
   282→    g.enemies.push(e);
   283→    run.spawnedTotal = (run.spawnedTotal ?? 0) + 1;
   284→  };
   285→  for (let tries = 0; tries < 40; tries++) {
   286→    const a = Math.random() * Math.PI * 2;
   287→    const d = (8 + Math.random() * 6) * TILE;
   288→    const x = p.x + Math.cos(a) * d;
   289→    const y = p.y + Math.sin(a) * d;
   290→    if (x < 2.5 * TILE || y < 2.5 * TILE) continue;
   291→    if (x > (g.map.w - 2.5) * TILE || y > (g.map.h - 2.5) * TILE) continue;
   292→    if (g.tileSolidAt(x, y)) continue;
   293→    if (!g.boxFree(x, y, bw, bh)) continue;
   294→    push(x, y);
   295→    return;
   296→  }
   297→  push((CENTER_TX + 0.5) * TILE, (CENTER_TY + 6) * TILE);
   298→}
   299→
   300→function startWave(g: Game, run: ChallengeRun): void {
   301→  run.wave++;
   302→  run.spawnQueue = waveComposition(run.wave);
   303→  run.spawnT = 0.2;
   304→  run.waveTime = 0;
   305→  run.waveHadBoss = run.spawnQueue.includes('guardian');
   306→  run.waveBossDown = !run.waveHadBoss;
   307→  run.bannerT = 2.2;
   308→  run.bannerText = run.waveHadBoss ? `OLEADA ${run.wave} · CENTINELA DEL ECO` : `OLEADA ${run.wave}`;
   309→  audio.sfx('banner');
   310→  if (run.waveHadBoss) {
   311→    g.bossActive = true;
   312→    g.toast('Un Centinela del Eco despierta en la arena', COL.boss);
   313→  }
   314→}

   316→/** Spawnea el jefe del duelo 1v1 (versión "de arena", sin flags de campaña). */
   317→function spawnDuelBoss(g: Game, run: ChallengeRun, id: string): void {
   318→  const info = BOSS_INFO[id] ?? BOSS_INFO.guardian;
   319→  const type = id as EnemyType;
   320→  const bx = (CENTER_TX + 0.5) * TILE;
   321→  const by = (CENTER_TY - 8) * TILE; // 8 tiles al norte: entra en aggro con el primer paso
   322→  const e = g.makeEnemy(type, bx, by, 0, 'boss'); // spawnGuard 0,5 s de cortesía
   323→  run.spawnedTotal = (run.spawnedTotal ?? 0) + 1;
   324→  run.bossEnemy = e;
   325→  g.enemies.push(e);
   326→  g.bossRef = e;
   327→  g.bossActive = true;
   328→  g.bossBannerT = 2.6;
   329→  g.bossBannerText = info.name;
   330→  g.bossBannerSub = info.sub;
   331→  audio.sfx('roar');
   332→  g.toast(`DUELO 1v1 — ${info.name}: una sola vida, sin pociones`, COL.boss);
   333→}

   337→/** Tick del modo desafío (llamado al final de Game.update en cada frame de juego). */
   338→export function challengeTick(g: Game, dt: number): void {
   339→  const run = g.challengeRun;
   340→  if (!run || run.phase !== 'jugando') return;
   341→
   342→  // --- ABORTO 1: se salió de la arena con el reto en marcha (puerta sur o
   343→  // viaje). El loadMap del destino YA regeneró el mundo (enemigos, FX,
   344→  // bossRef): aquí solo se restaura la campaña y se cierra el reto.
   345→  if (g.state === 'play' && g.mapId !== ARENA_MAP_ID) {
   346→    restoreCampaign(g, run);
   347→    const temp = !run.snap?.fromCampaign;
   348→    g.challengeRun = null;
   349→    g.bossRef = null; g.bossActive = false;
   350→    audio.setCombat(false);
   351→    if (temp) {
   352→      // Portador temporal sin campaña: no hay mundo que seguir → título
   353→      g.player = null;
   354→      g.setState('title');
   355→    } else {
   356→      g.toast('Has abandonado el Desafío. Tu canto retoma la campaña donde estaba...', '#c8b0e8');
   357→    }
   358→    return;
   359→  }
   360→  // --- ABORTO 2: el Portador fue REEMPLAZADO (Nueva partida / Continuar
   361→  // tras un GUARDAR Y SALIR desde la pausa). El estado nuevo viene limpio del
   362→  // guardado: NO se restaura nada, solo se descarta el reto huérfano.
   363→  if (run.playerRef && g.player !== run.playerRef) {
   364→    g.challengeRun = null;
   365→    g.bossRef = null; g.bossActive = false;
   366→    return;
   367→  }
   368→  if (g.state !== 'play' || !g.player) return; // diálogo: el mundo está detenido
   369→
   370→  const p = g.player;
   371→  run.timeSec = (run.timeSec ?? 0) + dt;
   372→  if (run.bannerT && run.bannerT > 0) run.bannerT -= dt;
   373→  if (run.mode === 'oleadas' && run.wave > 0) run.waveTime += dt;
   374→
   375→  // kills observadas: todos los enemigos del mapa son nuestros →
   376→  // instanciados − vivos (los muertos ya los filtró update.ts)
   377→  run.kills = Math.max(run.kills, (run.spawnedTotal ?? 0) - g.enemies.length);
   378→  run.score = (run.wavesCleared ?? 0) + run.kills;
   379→
   380→  // pausa dramática tras la caída del jefe: deja respirar el slowmo/burst de
   381→  // su muerte (expansionDeathFx) antes de saltar al panel de resultados
   382→  if (run.endBeat !== undefined) {
   383→    run.endBeat -= dt;
   384→    if (run.endBeat <= 0) finish(g, run, true);
   385→    return;
   386→  }
   387→
   388→  if (run.mode === 'jefe') {
   389→    // VICTORIA: killEnemy marcó dead y cantó victoria de CAMPAÑA (flags,
   390→    // música, botín) — restoreCampaign dentro de finish lo deshace todo.
   391→    const be = run.bossEnemy;
   392→    if (be && be.dead) {
   393→      run.endBeat = 1.7;
   394→      g.enemies = []; g.projectiles = []; g.waves = []; g.telegraphs = []; // los neumos invocados se disuelven
   395→      g.bossRef = null; g.bossActive = false;
   396→      audio.setCombat(false);
   397→    }
   398→    return;
   399→  }
   400→
   401→  // ---------------- MODO OLEADAS ----------------
   402→
   403→  // el Centinela muerto canta victoria de CAMPAÑA vía killEnemy (música de
   404→  // cripta + toasts del altar): deshacer su rastro en cuanto cae
   405→  if (run.waveHadBoss && !run.waveBossDown && !g.enemies.some(e => e.etype === 'guardian' && !e.dead)) {
   406→    run.waveBossDown = true;
   407→    audio.playTrack('boss');
   408→    g.toasts = g.toasts.filter(t => !t.text.includes('Guardián Hueco') && !t.text.includes('altar del Eco'));
   409→    g.toast('El Centinela del Eco cae hecho añicos', COL.goldSoft);
   410→  }
   411→
   412→  // cuenta atrás entre oleadas
   413→  if (run.betweenWaves > 0) {
   414→    run.betweenWaves -= dt;
   415→    if (run.betweenWaves <= 0) startWave(g, run);
   416→    return;
   417→  }
   418→
   419→  // entrada goteada de la cola de la oleada
   420→  if (run.spawnQueue && run.spawnQueue.length > 0) {
   421→    run.spawnT = (run.spawnT ?? 0) - dt;
   422→    if (run.spawnT <= 0) {
   423→      const type = run.spawnQueue.shift();
   424→      if (type) spawnArenaEnemy(g, run, type, waveMult(run.wave));
   425→      run.spawnT = 0.5;
   426→    }
   427→    return; // la oleada sigue entrando: nada de despeje todavía
   428→  }
   429→
   430→  // oleada despejada → recompensa breve y cuenta atrás (récord se aplica al final)
   431→  if (run.wave > 0 && g.enemies.length === 0) {
   432→    run.wavesCleared = run.wave;
   433→    run.score = (run.wavesCleared ?? 0) + run.kills;
   434→    const heal = Math.round(p.maxHp * 0.15);
   435→    if (p.hp > 0 && p.hp < p.maxHp) {
   436→      p.hp = Math.min(p.maxHp, p.hp + heal);
   437→      g.floatAt(p.x, p.y - 18, `+${heal}`, '#7ef0a0');
   438→    }
   439→    audio.sfx('quest');
   440→    g.toast(`Oleada ${run.wave} superada. Respira: la Niebla vuelve en 3...`, '#8ef0b0');
   441→    run.betweenWaves = 3.0;
   442→    run.waveHadBoss = false;
   443→    run.waveBossDown = false;
   444→    run.bannerT = 0;
   445→  }
   446→}

   449→export function onChallengeDeath(g: Game): void {
   450→  const run = g.challengeRun;
   451→  if (!run) return;
   452→  if (run.phase !== 'jugando') { g.challengeRun = null; return; }
   453→  // el peaje de muerte de campaña (oro/2, deadGolds, deaths++) ya lo cobró
   454→  // playerDied: restoreCampaign dentro de finish lo devuelve intacto
   455→  finish(g, run, false);
   456→}

   458→/** Entrada desde el botón DESAFÍO del título (lo añade 12-a en screens.ts). */
   459→export function startChallenge(g: Game, mode: 'oleadas' | 'jefe', boss?: string): void {
   460→  if (g.challengeRun && g.challengeRun.phase === 'jugando') return; // ya hay un reto en marcha
   461→  menuOpen = false;
   462→
   463→  const snap = snapshotCampaign(g);           // ANTES de tocar nada del mundo
   464→  if (!g.player) g.player = makeArenaPlayer();
   465→
   466→  // arena: época única presente, aterrizaje en el centro (findSafeTile corrige
   467→  // si acaso); loadMap vacía enemigos/proyectiles y regenera suelo y minimapa
   468→  g.epoch = 'presente';
   469→  g.loadMap(ARENA_MAP_ID, CENTER_TX, CENTER_TY);
   470→  g.epoch = 'presente';
   471→
   472→  const p = g.player;
   473→  if (p) {
   474→    p.hp = p.maxHp; p.sta = p.maxSta; p.res = 0;
   475→    p.iframes = 0; p.rollT = 0; p.attackT = 0; p.charging = false; p.chargeT = 0;
   476→    p.cds = [0, 0, 0, 0];
   477→    // DUELO: 1 sola vida y SIN pociones (F no ayuda: la arena no regala curas)
   478→    if (mode === 'jefe') p.potions = 0;
   479→  }
   480→
   481→  const run: ChallengeRun = {
   482→    mode,
   483→    boss: mode === 'jefe' ? (boss && BOSS_INFO[boss] ? boss : 'guardian') : undefined,
   484→    wave: 0, waveTime: 0,
   485→    betweenWaves: mode === 'oleadas' ? 3.4 : 0,
   486→    kills: 0, score: 0, active: true,
   487→    phase: 'jugando', wavesCleared: 0, timeSec: 0,
   488→    spawnQueue: [], spawnT: 0, spawnedTotal: 0,
   489→    bossEnemy: null, bannerT: 0, bannerText: '',
   490→    best: readBest(), newBest: false,
   491→    snap, playerRef: g.player,
   492→  };
   493→  g.challengeRun = run;
   494→
   495→  g.projectiles = []; g.waves = []; g.telegraphs = []; g.toasts = [];
   496→  g.bossRef = null; g.bossActive = false;
   497→
   498→  if (mode === 'oleadas') {
   499→    run.bannerT = 3.4;
   500→    run.bannerText = 'SOBREVIVE';
   501→    g.toast('El Desafío comienza: la arena juzga, no perdona', COL.goldSoft);
   502→  } else {
   503→    spawnDuelBoss(g, run, run.boss as string);
   504→  }
   505→
   506→  audio.playTrack('boss');
   507→  g.setState('play');
   508→}

   514→export function endChallenge(g: Game, _victory: boolean): void {
   515→  const run = g.challengeRun;
   516→  g.challengeRun = null;
   517→  menuOpen = false;

   519→  // PARTIDA/CONTINUAR crean el suyo). Solo tras usarlo, para que el HUD del
   520→  // panel de resultados pudiera leerlo hasta aquí.
   521→  if (run?.snap && !run.snap.fromCampaign) g.player = null;
   522→  g.enemies = [];
   523→  g.loadMap('lunaris', 25, 19); // punto del Santuario (mismo que respawn en campaña)
   524→  if (run?.resave === 'rewrite') {
   525→    // el guardado se escribió DENTRO de la arena: reescribirlo con el estado
   526→    // de campaña ya restaurado y posición de santuario
   527→    g.save();
   528→  } else if (run?.resave === 'remove') {
   529→    try { localStorage.removeItem(SAVE_KEY); } catch { /* noop */ }
   530→  }
   531→  g.setState('title'); // setState ya encaja la música del título
   532→}

   536→/** Resultado + limpieza + vuelta al título (victoria, derrota o aborto). */
   537→function finish(g: Game, run: ChallengeRun, victory: boolean): void {
   538→  run.active = false;
   539→  run.phase = 'resultado';
   540→  run.resultWin = victory;
   541→  run.score = (run.wavesCleared ?? 0) + run.kills;
   542→
   543→  // récord de oleadas (el duelo no lleva récord: su trofeo es el logro)
   544→  if (run.mode === 'oleadas' && run.score > (run.best ?? 0)) {
   545→    try { localStorage.setItem(LS_BEST, String(run.score)); } catch { /* noop */ }
   546→    run.best = run.score;
   547→    run.newBest = true;
   548→  }
   549→
   550→  restoreCampaign(g, run);
   551→  g.enemies = []; g.projectiles = []; g.waves = []; g.telegraphs = [];
   552→  g.bossRef = null; g.bossActive = false;
   553→  audio.setCombat(false);
   554→  // killEnemy cantó victoria de CAMPAÑA (flags de jefe, altar, botín): las
   555→  // restauración ya deshizo el estado; los toasts de campaña se limpian aquí
   556→  g.toasts = [];
   557→
   558→  // recompensa campañista: primer duelo ganado a CADA jefe, con Portador de
   559→  // campaña. Se aplica DESPUÉS de restaurar (la restauración devuelve las
   560→  // pociones exactas de antes del reto; aquí se suma la recompensa nueva).
   561→  // Si ganaste con el Portador temporal no se escribe el logro: el premio
   562→  // sigue disponible para tu campaña de verdad.
   563→  if (victory && run.mode === 'jefe' && run.snap?.fromCampaign && run.boss && g.player) {
   564→    const logros = readLogros();
   565→    const key = `duelo:${run.boss}`;
   566→    if (!logros.includes(key)) {
   567→      logros.push(key);
   568→      writeLogros(logros);
   569→      g.player.potions += 1;
   570→      audio.sfx('coin');
   571→      g.toast('Recompensa del Desafío: +1 poción', '#7ef0a0');
   572→      g.floatAt(g.player.x, g.player.y - 26, '+1 poción', '#7ef0a0', 8);
   573→    }
   574→  }
   575→
   576→  audio.sfx(victory ? 'levelup' : 'die');
   577→  menuOpen = false;
   578→  g.setState('title'); // el panel de resultados se dibuja sobre el título
   579→}
