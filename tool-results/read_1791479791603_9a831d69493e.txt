    30→/**
    31→ * Ajusta el buffer del juego al aspecto de la ventana para eliminar el
    32→ * letterbox (la "barra negra" abajo/arriba). Mantiene 540px de alto base
    33→ * a 16:9 exacto; pantallas más anchas ganan vista lateral (hasta 1600px)
    34→ * y más cuadradas ganan vista vertical (hasta 800px). Clamp de seguridad
    35→ * para aspectos extremos (móvil vertical): ahí el letterbox es aceptable.
    36→ */
    37→export function fitViewToWindow(winW: number, winH: number): void {
    38→  const a = winW / Math.max(1, winH);
    39→  let vw = Math.round(540 * a);
    40→  let vh = 540;
    41→  if (vw < 840) { vw = 840; vh = Math.round(vw / a); }
    42→  else if (vw > 1600) { vw = 1600; vh = Math.round(vw / a); }
    43→  vh = Math.max(460, Math.min(800, vh));
    44→  vw = Math.max(840, Math.min(1600, Math.round(vh * a)));
    45→  if (vw !== VIEW_W || vh !== VIEW_H) { VIEW_W = vw; VIEW_H = vh; }
    46→}

   211→   * 11-a (integración): red de seguridad del Acto II contra JUEGO FUERA DE
   212→   * ORDEN. Los pasos de q7-q9 se completaban solo por EVENTO (watcher de
   213→   * aggro, killEnemy, lightLamp, eco_*_taken): si el jugador completa el
   214→   * objetivo ANTES de aceptar la misión —mata a la Sirena durante q6, mata al

   222→  private catchUpActo2() {
   223→    if (this.questIdx < 5 || this.questIdx > 9) return;
   224→    // q7 (idx 6): 0 avistada · 1 derrotada · 2 Eco de las Mareas tomado
   225→    if (this.questIdx === 6) {
   226→      if (this.questStep === 0 && this.flags.sirenaDefeated) this.questAdvance();
   227→      if (this.questIdx === 6 && this.questStep === 1 && this.flags.sirenaDefeated) this.questAdvance();
   228→      if (this.questIdx === 6 && this.questStep === 2 && this.flags.ecoMareas) this.questAdvance();
   229→    }
   230→    // q8 (idx 7): 1 los 3 faroles encendidos (orden-independiente) · 2 Eco de los Nombres
   231→    if (this.questIdx === 7) {
   232→      if (this.questStep === 1 &&
   233→          (['lamp1', 'lamp2', 'lamp3'] as const).filter(l => this.flags[l]).length >= 3) this.questAdvance();
   234→      if (this.questIdx === 7 && this.questStep === 2 && this.flags.ecoNombres) this.questAdvance();
   235→    }
   236→    // q9 (idx 8): 1 gólem derrotado · 2 Eco de las Cumbres tomado
   237→    if (this.questIdx === 8) {
   238→      if (this.questStep === 1 && this.flags.golemDefeated) this.questAdvance();
   239→      if (this.questIdx === 8 && this.questStep === 2 && this.flags.ecoCumbres) this.questAdvance();
   240→    }
   241→  }

   245→  start() {
   246→    if (this.running) return;
   247→    this.running = true;
   248→    this.lastTs = performance.now();
   249→    const loop = (ts: number) => {
   250→      if (!this.running) return;
   251→      try {
   252→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   253→        this.lastTs = ts;
   254→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   255→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   256→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   257→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   258→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   259→        this.globalT += dt;
   260→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   261→        drawGame(this);
   262→        this.loopError = null;
   263→      } catch (err) {
   264→        this.loopError = err instanceof Error ? (err.stack ?? String(err)) : String(err);
   265→        console.error('[EcosAelthar loop]', err);
   266→        try {
   267→          this.ctx.fillStyle = '#000';
   268→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   269→          this.ctx.font = '12px monospace';
   270→          this.ctx.fillStyle = '#ff7060';
   271→          this.ctx.textAlign = 'left';
   272→          this.ctx.textBaseline = 'top';
   273→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   274→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   275→        } catch { /* noop */ }
   276→      }
   277→      this.raf = requestAnimationFrame(loop);
   278→    };
   279→    this.raf = requestAnimationFrame(loop);
   280→  }
   281→
   282→  stop() {
   283→    this.running = false;
   284→    cancelAnimationFrame(this.raf);
   285→    audio.stopMusic();
   286→  }

   351→      attrs: { ...p.attrs }, points: p.points,

   353→      iframes: 0, parryT: 0, parryFx: 0, attackT: 0, combo: 0,

   361→    this.flags = { ...d.flags };

   372→    // restaura la posición fina del guardado SOLO si su tile es seguro
   373→    // (ni sólido ni dentro de una zona de salida); si no, conserva el tile

   413→  loadMap(id: MapId, tx: number, ty: number) {
   414→    // memoria del jefe: si sales de la cripta con el Guardián herido, no se
   415→    // regenera al volver (cierra el exploit de reseteo de combate)
   416→    // memoria del jefe (generalizada Acto II): si sales de un mapa con su JEFE vivo, no se regenera
   417→    if (this.bossRef && !this.bossRef.dead) {
   418→      this.flags[`bossHp_${this.mapId}`] = this.bossRef.hp;
   419→      if (this.mapId === 'cripta') this.flags.bossHp = this.bossRef.hp; // compat con saves antiguos
   420→    }
   421→    this.mapId = id;
   422→    this.map = MAPS[id];
   423→    this.rows = mapRows(this.map);
   424→    this.buildGround();
   425→    if (this.player) {

   465→  buildGround() {
   466→    const { w, h } = this.map;
   467→    const mk = (ep: Epoch) => {
   468→      const c = document.createElement('canvas');
   469→      c.width = w * TILE; c.height = h * TILE;
   470→      const x = c.getContext('2d')!;
   471→      x.imageSmoothingEnabled = false;
   472→      // suelo
   473→      for (let ty = 0; ty < h; ty++) {
   474→        for (let tx = 0; tx < w; tx++) {
   475→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   476→          // Acto II: arena/nieve/hielo y suelos por bioma los dibuja la expansión; fallback genérico
   477→          if (!drawExpansionTile(x, ch, tx, ty, this.mapId)) drawTile(x, ch, tx, ty, this.mapId, 0);
   478→        }
   479→      }
   480→      // objetos altos por orden de fila
   481→      for (let ty = 0; ty < h; ty++) {
   482→        for (let tx = 0; tx < w; tx++) {
   483→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   484→          if (SOLID_CHARS.has(ch) && (ch === 't' || ch === 'p')) {
   485→            if (!drawExpansionTallTile(x, ch, tx, ty, this.mapId)) drawTallTile(x, ch, tx, ty, this.mapId);
   486→          }
   487→        }
   488→      }
   489→      return c;
   490→    };
   491→    this.groundCanvas = mk('presente');
   492→    this.groundPastCanvas = this.map.epochDiffs.length ? mk('pasado') : this.groundCanvas;
   493→    // minimapa
   494→    const mini = document.createElement('canvas');
   495→    mini.width = w; mini.height = h;
   496→    const mx = mini.getContext('2d')!;
   497→    for (let ty = 0; ty < h; ty++) {
   498→      for (let tx = 0; tx < w; tx++) {
   499→        const ch = tileAt(this.map, this.rows, tx, ty, 'presente');
   500→        mx.fillStyle =
   501→          ch === 's' ? '#d8c07a' : ch === 'S' ? '#dfe8f2' : ch === 'i' ? '#a8cfe8' :
   502→          ch === '~' ? '#3a6a9a' : ch === '=' ? '#b89a6a' :
   503→          ch === ':' ? '#6a6a7a' : ch === 't' || ch === 'p' ? '#24512a' :
   504→          ch === '#' || ch === 'H' || ch === 'r' ? '#7a7a8a' :
   505→          ch === 'n' ? '#7ea49a' : '#4a8a44';
   506→        mx.fillRect(tx, ty, 1, 1);
   507→      }
   508→    }
   509→    this.miniCanvas = mini;
   510→  }

   568→  /** ¿(tx,ty) está dentro de alguna zona de salida del mapa actual? */
   569→  private inExitZone(tx: number, ty: number): boolean {
   570→    return this.map.exits.some(ex => {
   571→      if (ex.needPast && this.epoch !== 'pasado') return false;
   572→      return tx >= ex.x && tx < ex.x + ex.w && ty >= ex.y && ty < ex.y + ex.h;
   573→    });
   574→  }

   576→  /**
   577→   * Busca el tile más cercano (búsqueda por anillos) que sea transitable y
   578→   * NO esté dentro de una zona de salida. Garantiza que ningún viaje aterrice
   579→   * en un muro ni en un teletransporte (bucle cripta↔bosque, P0-1).
   580→   */
   581→  private findSafeTile(tx: number, ty: number): [number, number] {
   582→    const ok = (x: number, y: number) =>
   583→      x >= 0 && y >= 0 && x < this.map.w && y < this.map.h &&
   584→      !this.tileSolidAt(x * TILE + 8, y * TILE + 8) &&
   585→      !this.inExitZone(x, y);
   586→    if (ok(tx, ty)) return [tx, ty];
   587→    for (let r = 1; r <= 8; r++) {
   588→      for (let dy = -r; dy <= r; dy++) {
   589→        for (let dx = -r; dx <= r; dx++) {
   590→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   591→          if (ok(tx + dx, ty + dy)) return [tx + dx, ty + dy];
   592→        }
   593→      }
   594→    }
   595→    return [tx, ty]; // sin sitio libre: comportamiento anterior
   596→  }

   598→  boxFree(x: number, y: number, w: number, h: number): boolean {
   599→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   600→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   601→  }

   603→  moveEntity(e: { x: number; y: number; w: number; h: number }, dx: number, dy: number) {
   604→    if (dx !== 0 && this.boxFree(e.x + dx, e.y, e.w, e.h)) e.x += dx;
   605→    if (dy !== 0 && this.boxFree(e.x, e.y + dy, e.w, e.h)) e.y += dy;
   606→  }

   608→  updateCamera(snap = false) {
   609→    if (!this.player) return;
   610→    const targetX = this.player.x * ZOOM - VIEW_W / 2;
   611→    const targetY = this.player.y * ZOOM - VIEW_H / 2;

   613→    const maxY = this.map.h * TILE * ZOOM - VIEW_H;
   614→    const cx = Math.max(0, Math.min(maxX, targetX));
   615→    const cy = Math.max(0, Math.min(maxY, targetY));
   616→    if (snap) { this.camX = cx; this.camY = cy; }
   617→    else { this.camX += (cx - this.camX) * 0.14; this.camY += (cy - this.camY) * 0.14; }
   618→  }

   646→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   647→    const p = this.player;
   648→    if (!p) return null;
   649→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   650→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   651→      const d = Math.hypot(x - p.x, y - p.y);
   652→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   653→    };
   654→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   655→    for (const ch of this.map.chests) {
   656→      if (this.openedChests.has(ch.id)) continue;
   657→      if (ch.needPast && this.epoch !== 'pasado') continue;
   658→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   659→    }
   660→    for (const pr of this.map.props) {
   661→      if (pr.needPast && this.epoch !== 'pasado') continue;
   662→      if (pr.needPresent && this.epoch !== 'presente') continue;
   663→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   664→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   665→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   666→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   667→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   668→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   669→      else if (pr.kind === 'lamp' && !this.flags[pr.id]) consider(px, py, 'lamp', 'Encender el Farol del Recuerdo', () => this.lightLamp(pr.id), 28);
   670→    }
   671→    for (const ec of this.map.echoes) {
   672→      if (this.takenEchoes.has(ec.id)) continue;
   673→      consider(ec.x * TILE + 8, ec.y * TILE + 8, 'echo', 'Escuchar eco menor', () => this.takeEchoMinor(ec.id, ec.title, ec.text), 26);
   674→    }
   675→    return best;
   676→  }

   815→  advanceDialogue() {

   819→    if (this.dlgNode.onEnd) this.applyAction(this.dlgNode.onEnd);
   820→    if (this.dlgNode.options && this.dlgNode.options.length > 0) {
   821→      const opt = this.dlgNode.options[this.dlgSel];
   822→      recordDialogueTone(this, opt);
   823→      if (opt.action) this.applyAction(opt.action);
   824→      if (opt.next) { this.openDialogue(opt.next); return; }
   825→      this.closeDialogue();
   826→      return;
   827→    }
   828→    if (this.dlgNode.next) { this.openDialogue(this.dlgNode.next); return; }
   829→    this.closeDialogue();

  1143→  private onKeyDown = (e: KeyboardEvent) => {
  1144→    const k = e.key.toLowerCase();
  1145→    // no interceptar teclas cuando se escribe en un input DOM (nombre del Portador)
  1146→    const t = e.target as HTMLElement | null;
  1147→    const isDomInput = !!t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable);
  1148→    if (!isDomInput && [' ', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(k)) e.preventDefault();
  1149→    if (e.repeat) return;
  1150→    this.keys.add(k);
  1151→    audio.resume();
  1152→
  1153→    if (this.state === 'title') {
  1154→      if (k === 'enter' || k === 'e' || k === ' ') { /* botones por click; enter = nueva partida */ }
  1155→    } else if (this.state === 'intro') {
  1156→      if (k === 'e' || k === ' ' || k === 'enter') {
  1157→        this.introIdx++;
  1158→        audio.sfx('blip');
  1159→        if (this.introIdx >= 3) this.startPlay();
  1160→      }
  1161→    } else if (this.state === 'dialogue') {
  1162→      if (k === 'e' || k === ' ' || k === 'enter') this.advanceDialogue();
  1163→      else if (k === 'arrowup' || k === 'w') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + this.dlgNode.options.length - 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
  1164→      else if (k === 'arrowdown' || k === 's') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
  1165→    } else if (this.state === 'play') {
  1166→      if (k === ' ' && this.player && this.player.rollT <= 0 && this.player.attackT <= 0 && this.player.sta >= 20) {
  1167→        this.rollQueued = true; // 1 pulsación = 1 voltereta (no en cadena)
  1168→      }
  1169→      if (k === 'e') this.tryInteract();
  1170→      else if (k === 'q') this.epochSwitch();
  1171→      else if (k === 'escape' || k === 'm') { this.setState('pause'); audio.sfx('uiOpen'); }
  1172→      else if (k === 'f') this.drinkPotion();
  1173→      else if (k === 't' && e.shiftKey) {
  1174→        // accesibilidad / QoL: esperar hasta el alba
  1175→        this.dayT = 0.22;
  1176→        this.toast('Esperas junto al camino hasta el alba...', '#ffe86a');
  1177→        audio.sfx('save');
  1178→      }
  1179→      else if (['1', '2', '3', '4'].includes(k)) this.useSkill(parseInt(k, 10) - 1);
  1180→      else if (k === 'k') { this.setState('skills'); audio.sfx('uiOpen'); }
  1181→    } else if (this.state === 'pause') {
  1182→      if (k === 'escape' || k === 'm') this.setState('play');
  1183→    } else if (this.state === 'skills') {
  1184→      if (k === 'escape' || k === 'k' || k === 'm') { this.setState('play'); audio.sfx('uiOpen'); }
  1185→    } else if (this.state === 'dead') {
  1186→      if (k === 'e' || k === 'enter') this.respawn();
  1187→    } else if (this.state === 'end') {
  1188→      if (k === 'enter' || k === 'e') this.setState('title');
  1189→    }
  1190→  };

  1600→  damagePlayer(dmg: number, fromX: number, fromY: number) {
  1601→    const p = this.player!;
  1602→    // balanceador (12-c): embudo único de todo el daño enemigo (neutro en desafío)
  1603→    dmg = Math.max(1, Math.round(dmg * enemyStatMult(this).dmg));
  1604→    if (p.iframes > 0 || p.rollT > 0 || this.state !== 'play') return;
  1605→    if (p.parryT > 0) {
  1606→      // ¡parada perfecta!
  1607→      audio.sfx('parry');
  1608→      p.res = Math.min(p.maxRes, p.res + 20);
  1609→      p.parryFx = 0.4;
  1610→      this.hitStop = 0.12;
  1611→      this.burst(p.x + (fromX - p.x) * 0.3, p.y - 8 + (fromY - p.y) * 0.3, '#fff8c0', 14, 100);
  1612→      this.floatAt(p.x, p.y - 22, '¡PARADA!', '#fff8c0', 7);
  1613→      // aturde al atacante cercano
  1614→      let best: Enemy | null = null, bd = 34;
  1615→      for (const e of this.enemies) {
  1616→        if (e.dead) continue;
  1617→        const d = Math.hypot(e.x - p.x, e.y - p.y);
  1618→        if (d < bd) { bd = d; best = e; }
  1619→      }
  1620→      if (best) { best.ai = 'aturdido'; best.aiT = 1.6; best.windup = 0; }
  1621→      p.parryT = 0;
  1622→      return;
  1623→    }
  1624→    const red = Math.min(0.5, p.attrs.vig * 0.01);
  1625→    const final = Math.max(1, Math.round(dmg * (1 - red)));
  1626→    p.hp -= final;
  1627→    p.iframes = 0.6;
  1628→    p.lastHitT = 0.3;
  1629→    this.shake = 4;
  1630→    audio.sfx('hurt');
  1631→    this.floatAt(p.x, p.y - 16, `-${final}`, '#ff7060');
  1632→    const dx = p.x - fromX, dy = p.y - fromY;
  1633→    const l = Math.max(1, Math.hypot(dx, dy));
  1634→    this.moveEntity(p, (dx / l) * 8, (dy / l) * 8);
  1635→    if (p.hp <= 0) this.playerDied();
  1636→  }
