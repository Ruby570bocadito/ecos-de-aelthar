    28→/**
    29→ * Ajusta el buffer del juego al aspecto de la ventana para eliminar el
    30→ * letterbox (la "barra negra" abajo/arriba). Mantiene 540px de alto base
    31→ * a 16:9 exacto; pantallas más anchas ganan vista lateral (hasta 1600px)
    32→ * y más cuadradas ganan vista vertical (hasta 800px). Clamp de seguridad
    33→ * para aspectos extremos (móvil vertical): ahí el letterbox es aceptable.
    34→ */
    35→export function fitViewToWindow(winW: number, winH: number): void {
    36→  const a = winW / Math.max(1, winH);
    37→  let vw = Math.round(540 * a);
    38→  let vh = 540;
    39→  if (vw < 840) { vw = 840; vh = Math.round(vw / a); }
    40→  else if (vw > 1600) { vw = 1600; vh = Math.round(vw / a); }
    41→  vh = Math.max(460, Math.min(800, vh));
    42→  vw = Math.max(840, Math.min(1600, Math.round(vh * a)));
    43→  if (vw !== VIEW_W || vh !== VIEW_H) { VIEW_W = vw; VIEW_H = vh; }
    44→}

   206→  /**
   207→   * 11-a (integración): red de seguridad del Acto II contra JUEGO FUERA DE
   208→   * ORDEN. Los pasos de q7-q9 se completaban solo por EVENTO (watcher de
   209→   * aggro, killEnemy, lightLamp, eco_*_taken): si el jugador completa el
   210→   * objetivo ANTES de aceptar la misión —mata a la Sirena durante q6, mata al

   215→   * POR ESTADO: cada paso se completa si su condición final ya es cierta, en

   218→  private catchUpActo2() {
   219→    if (this.questIdx < 5 || this.questIdx > 9) return;
   220→    // q7 (idx 6): 0 avistada · 1 derrotada · 2 Eco de las Mareas tomado
   221→    if (this.questIdx === 6) {
   222→      if (this.questStep === 0 && this.flags.sirenaDefeated) this.questAdvance();
   223→      if (this.questIdx === 6 && this.questStep === 1 && this.flags.sirenaDefeated) this.questAdvance();
   224→      if (this.questIdx === 6 && this.questStep === 2 && this.flags.ecoMareas) this.questAdvance();
   225→    }
   226→    // q8 (idx 7): 1 los 3 faroles encendidos (orden-independiente) · 2 Eco de los Nombres
   227→    if (this.questIdx === 7) {
   228→      if (this.questStep === 1 &&
   229→          (['lamp1', 'lamp2', 'lamp3'] as const).filter(l => this.flags[l]).length >= 3) this.questAdvance();
   230→      if (this.questIdx === 7 && this.questStep === 2 && this.flags.ecoNombres) this.questAdvance();
   231→    }
   232→    // q9 (idx 8): 1 gólem derrotado · 2 Eco de las Cumbres tomado
   233→    if (this.questIdx === 8) {
   234→      if (this.questStep === 1 && this.flags.golemDefeated) this.questAdvance();
   235→      if (this.questIdx === 8 && this.questStep === 2 && this.flags.ecoCumbres) this.questAdvance();
   236→    }
   237→  }

   241→  start() {
   242→    if (this.running) return;
   243→    this.running = true;
   244→    this.lastTs = performance.now();
   245→    const loop = (ts: number) => {
   246→      if (!this.running) return;
   247→      try {
   248→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   249→        this.lastTs = ts;
   250→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   251→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   252→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   253→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   254→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   255→        this.globalT += dt;
   256→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   257→        drawGame(this);
   258→        this.loopError = null;
   259→      } catch (err) {
   260→        this.loopError = err instanceof Error ? (err.stack ?? String(err)) : String(err);
   261→        console.error('[EcosAelthar loop]', err);
   262→        try {
   263→          this.ctx.fillStyle = '#000';
   264→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   265→          this.ctx.font = '12px monospace';
   266→          this.ctx.fillStyle = '#ff7060';
   267→          this.ctx.textAlign = 'left';
   268→          this.ctx.textBaseline = 'top';
   269→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   270→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   271→        } catch { /* noop */ }
   272→      }
   273→      this.raf = requestAnimationFrame(loop);
   274→    };
   275→    this.raf = requestAnimationFrame(loop);
   276→  }
   277→
   278→  stop() {
   279→    this.running = false;
   280→    cancelAnimationFrame(this.raf);
   281→    audio.stopMusic();
   282→  }

   409→  loadMap(id: MapId, tx: number, ty: number) {
   410→    // memoria del jefe: si sales de la cripta con el Guardián herido, no se
   411→    // regenera al volver (cierra el exploit de reseteo de combate)
   412→    // memoria del jefe (generalizada Acto II): si sales de un mapa con su JEFE vivo, no se regenera
   413→    if (this.bossRef && !this.bossRef.dead) {
   414→      this.flags[`bossHp_${this.mapId}`] = this.bossRef.hp;
   415→      if (this.mapId === 'cripta') this.flags.bossHp = this.bossRef.hp; // compat con saves antiguos
   416→    }
   417→    this.mapId = id;
   418→    this.map = MAPS[id];
   419→    this.rows = mapRows(this.map);
   420→    this.buildGround();
   421→    if (this.player) {
   422→      // defensa anti-softlock: nunca aterrizar sobre un tile sólido ni dentro
   423→      // de una zona de salida del destino (provocaba el bucle cripta↔bosque)
   424→      const [sx, sy] = this.findSafeTile(tx, ty);
   425→      this.player.x = sx * TILE + 8;
   426→      this.player.y = sy * TILE + 8;

   561→  /** ¿(tx,ty) está dentro de alguna zona de salida del mapa actual? */
   562→  private inExitZone(tx: number, ty: number): boolean {
   563→    return this.map.exits.some(ex => {
   564→      if (ex.needPast && this.epoch !== 'pasado') return false;
   565→      return tx >= ex.x && tx < ex.x + ex.w && ty >= ex.y && ty < ex.y + ex.h;
   566→    });

   569→  /**
   570→   * Busca el tile más cercano (búsqueda por anillos) que sea transitable y
   571→   * NO esté dentro de una zona de salida. Garantiza que ningún viaje aterrice
   572→   * en un muro ni en un teletransporte (bucle cripta↔bosque, P0-1).
   573→   */
   574→  private findSafeTile(tx: number, ty: number): [number, number] {
   575→    const ok = (x: number, y: number) =>
   576→      x >= 0 && y >= 0 && x < this.map.w && y < this.map.h &&
   577→      !this.tileSolidAt(x * TILE + 8, y * TILE + 8) &&
   578→      !this.inExitZone(x, y);
   579→    if (ok(tx, ty)) return [tx, ty];
   580→    for (let r = 1; r <= 8; r++) {
   581→      for (let dy = -r; dy <= r; dy++) {
   582→        for (let dx = -r; dx <= r; dx++) {
   583→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   584→          if (ok(tx + dx, ty + dy)) return [tx + dx, ty + dy];
   585→        }
   586→      }
   587→    }
   588→    return [tx, ty]; // sin sitio libre: comportamiento anterior
   589→  }

   591→  boxFree(x: number, y: number, w: number, h: number): boolean {
   592→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   593→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   594→  }

   598→    if (dy !== 0 && this.boxFree(e.x, e.y + dy, e.w, e.h)) e.y += dy;

   637→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   638→    const p = this.player;
   639→    if (!p) return null;
   640→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   641→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   642→      const d = Math.hypot(x - p.x, y - p.y);
   643→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   644→    };
   645→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   646→    for (const ch of this.map.chests) {
   647→      if (this.openedChests.has(ch.id)) continue;
   648→      if (ch.needPast && this.epoch !== 'pasado') continue;
   649→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   650→    }
   651→    for (const pr of this.map.props) {
   652→      if (pr.needPast && this.epoch !== 'pasado') continue;
   653→      if (pr.needPresent && this.epoch !== 'presente') continue;
   654→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   655→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   656→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   657→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   658→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   659→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   660→      else if (pr.kind === 'lamp' && !this.flags[pr.id]) consider(px, py, 'lamp', 'Encender el Farol del Recuerdo', () => this.lightLamp(pr.id), 28);
   661→    }
   662→    for (const ec of this.map.echoes) {
   663→      if (this.takenEchoes.has(ec.id)) continue;
   664→      consider(ec.x * TILE + 8, ec.y * TILE + 8, 'echo', 'Escuchar eco menor', () => this.takeEchoMinor(ec.id, ec.title, ec.text), 26);
   665→    }
   666→    return best;
   667→  }

  1134→  private onKeyDown = (e: KeyboardEvent) => {
  1135→    const k = e.key.toLowerCase();
  1136→    // no interceptar teclas cuando se escribe en un input DOM (nombre del Portador)
  1137→    const t = e.target as HTMLElement | null;
  1138→    const isDomInput = !!t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable);
  1139→    if (!isDomInput && [' ', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(k)) e.preventDefault();
  1140→    if (e.repeat) return;
  1141→    this.keys.add(k);
  1142→    audio.resume();
  1143→
  1144→    if (this.state === 'title') {
  1145→      if (k === 'enter' || k === 'e' || k === ' ') { /* botones por click; enter = nueva partida */ }
  1146→    } else if (this.state === 'intro') {
  1147→      if (k === 'e' || k === ' ' || k === 'enter') {
  1148→        this.introIdx++;
  1149→        audio.sfx('blip');
  1150→        if (this.introIdx >= 3) this.startPlay();
  1151→      }
  1152→    } else if (this.state === 'dialogue') {
  1153→      if (k === 'e' || k === ' ' || k === 'enter') this.advanceDialogue();
  1154→      else if (k === 'arrowup' || k === 'w') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + this.dlgNode.options.length - 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
  1155→      else if (k === 'arrowdown' || k === 's') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
  1156→    } else if (this.state === 'play') {
  1157→      if (k === ' ' && this.player && this.player.rollT <= 0 && this.player.attackT <= 0 && this.player.sta >= 20) {
  1158→        this.rollQueued = true; // 1 pulsación = 1 voltereta (no en cadena)
  1159→      }
  1160→      if (k === 'e') this.tryInteract();
  1161→      else if (k === 'q') this.epochSwitch();
  1162→      else if (k === 'escape' || k === 'm') { this.setState('pause'); audio.sfx('uiOpen'); }
  1163→      else if (k === 'f') this.drinkPotion();
  1164→      else if (k === 't' && e.shiftKey) {
  1165→        // accesibilidad / QoL: esperar hasta el alba
  1166→        this.dayT = 0.22;
  1167→        this.toast('Esperas junto al camino hasta el alba...', '#ffe86a');
  1168→        audio.sfx('save');
  1169→      }
  1170→      else if (['1', '2', '3', '4'].includes(k)) this.useSkill(parseInt(k, 10) - 1);
  1171→      else if (k === 'k') { this.setState('skills'); audio.sfx('uiOpen'); }
  1172→    } else if (this.state === 'pause') {
  1173→      if (k === 'escape' || k === 'm') this.setState('play');
  1174→    } else if (this.state === 'skills') {
  1175→      if (k === 'escape' || k === 'k' || k === 'm') { this.setState('play'); audio.sfx('uiOpen'); }
  1176→    } else if (this.state === 'dead') {
  1177→      if (k === 'e' || k === 'enter') this.respawn();
  1178→    } else if (this.state === 'end') {
  1179→      if (k === 'enter' || k === 'e') this.setState('title');
  1180→    }
  1181→  };

  1595→      // ¡parada perfecta!
  1596→      audio.sfx('parry');
  1597→      p.res = Math.min(p.maxRes, p.res + 20);
  1598→      p.parryFx = 0.4;
  1599→      this.hitStop = 0.12;
  1600→      this.burst(p.x + (fromX - p.x) * 0.3, p.y - 8 + (fromY - p.y) * 0.3, '#fff8c0', 14, 100);
  1601→      this.floatAt(p.x, p.y - 22, '¡PARADA!', '#fff8c0', 7);
  1602→      // aturde al atacante cercano
  1603→      let best: Enemy | null = null, bd = 34;
  1604→      for (const e of this.enemies) {
  1605→        if (e.dead) continue;
  1606→        const d = Math.hypot(e.x - p.x, e.y - p.y);
  1607→        if (d < bd) { bd = d; best = e; }
  1608→      }
  1609→      if (best) { best.ai = 'aturdido'; best.aiT = 1.6; best.windup = 0; }
  1610→      p.parryT = 0;
  1611→      return;
  1612→    }
  1613→    const red = Math.min(0.5, p.attrs.vig * 0.01);
  1614→    const final = Math.max(1, Math.round(dmg * (1 - red)));
  1615→    p.hp -= final;
  1616→    p.iframes = 0.6;
  1617→    p.lastHitT = 0.3;
  1618→    this.shake = 4;
  1619→    audio.sfx('hurt');
  1620→    this.floatAt(p.x, p.y - 16, `-${final}`, '#ff7060');
  1621→    const dx = p.x - fromX, dy = p.y - fromY;
  1622→    const l = Math.max(1, Math.hypot(dx, dy));
  1623→    this.moveEntity(p, (dx / l) * 8, (dy / l) * 8);
  1624→    if (p.hp <= 0) this.playerDied();
  1625→  }
  1626→}
