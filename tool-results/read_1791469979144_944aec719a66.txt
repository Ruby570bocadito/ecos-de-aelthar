    25→//   El tick es ROBUSTO a la colocación: detecta (vía WeakMap) si el

    47→function dist(ax: number, ay: number, bx: number, by: number): number {
    48→  return Math.hypot(bx - ax, by - ay);
    49→}

    51→function isNightG(g: Game): boolean {
    52→  return g.dayT > 0.7 || g.dayT < 0.08;
    53→}

    55→/** Mueve hacia (dx,dy) normalizado con colisión de tiles y fija dir/moving. */
    56→function moveDir(g: Game, e: Enemy, dx: number, dy: number, spd: number, dt: number): void {
    57→  const l = Math.hypot(dx, dy);
    58→  if (l < 1e-4) { e.moving = false; return; }
    59→  g.moveEntity(e, (dx / l) * spd * dt, (dy / l) * spd * dt);
    60→  e.moving = true;
    61→  if (Math.abs(dx) > 0.01) e.dir = dx > 0 ? 'right' : 'left';
    62→}

    98→// ---------------- bloque común (réplica robusta del prep de update.ts) ----------------
    99→
   100→/**
   101→ * Devuelve TRUE si el frame quedó consumido (aturdido/spawnGuard).
   102→ * Efectos: freshHit, decaimiento de hitFlash/invulT/atkCd, aggro, anim.
   103→ * Todo es idempotente con el bloque común del motor gracias a la
   104→ * detección por anim/atkCd (ver ExpMem.lastAnim/lastAtkCd).
   105→ */
   106→function commonTick(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   107→  const p = g.player;
   108→
   109→  // ¿el motor ya pasó por updateEnemy este frame? (anim solo cambia ahí)
   110→  const motorRan = m.lastAnim !== undefined && e.anim !== m.lastAnim;
   111→  m.lastAnim = e.anim;
   112→
   113→  // spawnGuard (invocaciones: sin aggro inicial)
   114→  if (e.spawnGuard && e.spawnGuard > 0) {
   115→    if (!motorRan) e.spawnGuard = Math.max(0, e.spawnGuard - dt);
   116→    if (e.spawnGuard > 0) return true;
   117→  }
   118→
   119→  // golpe nuevo: hitFlash SUBIÓ desde el frame pasado (damageEnemy pone 0.12)
   120→  const prev = m.prevHitFlash ?? 0;
   121→  m.freshHit = e.hitFlash > prev + 5e-4 && e.hitFlash > 0.06;
   122→  // decaimiento de hitFlash si el motor aún no lo hizo (colocación temprana)
   123→  if (!motorRan && e.hitFlash > 0 && Math.abs(e.hitFlash - prev) < 1e-9) {
   124→    e.hitFlash = Math.max(0, e.hitFlash - dt);
   125→  }
   126→  m.prevHitFlash = e.hitFlash;
   127→
   128→  // knockback (contrato fxcore) si el motor no lo aplicó
   129→  if (!motorRan) stepKnockback(g, e, dt);
   130→
   131→  // estados (quemado/congelado) + marca — réplica del bloque común
   132→  if (!motorRan) {
   133→    for (let i = e.statuses.length - 1; i >= 0; i--) {
   134→      const s = e.statuses[i];
   135→      s.t -= dt;
   136→      if (s.kind === 'quemado') {
   137→        e.hp -= s.power * dt;
   138→        if (Math.random() < 0.15) {
   139→          g.particles.push({ x: e.x + (Math.random() - 0.5) * 8, y: e.y - 6, vx: 0, vy: -24, t: 0.3, maxT: 0.3, color: '#ff9040', size: 1.5, grav: 0 });
   140→        }
   141→        if (e.hp <= 0) { g.killEnemy(e); return true; }
   142→      }
   143→      if (s.t <= 0) e.statuses.splice(i, 1);
   144→    }
   145→    if (e.marked !== undefined) {
   146→      e.marked -= dt;
   147→      if (e.marked <= 0) e.marked = undefined;
   148→    }
   149→  }
   150→  if (e.hp < e.maxHp) e.aggro = true;
   151→
   152→  // invulT: campo nuevo de la expansión — SIEMPRE lo decaemos aquí
   153→  if (e.invulT && e.invulT > 0) e.invulT = Math.max(0, e.invulT - dt);
   154→
   155→  // atkCd: decae solo si el motor no lo hizo ya este frame
   156→  if (m.lastAtkCd === undefined || Math.abs(e.atkCd - m.lastAtkCd) < 1e-9) e.atkCd -= dt;
   157→
   158→  // anim (cosmético) si el motor no lo hizo
   159→  if (!motorRan) e.anim += dt * (e.ai === 'persigue' ? 1.4 : 0.6);
   160→
   161→  // timer auxiliar de la expansión
   162→  e.subT = (e.subT ?? 0) + dt;
   163→
   164→  // aggro (idempotente: si el motor ya lo fijó, no-op)
   165→  if (p && !e.aggro && g.state === 'play') {
   166→    const nightMult = isNightG(g) ? 1.3 : 1;
   167→    if (dist(e.x, e.y, p.x, p.y) < def.aggroR * nightMult) {
   168→      e.aggro = true;
   169→      if (e.etype !== 'sirena' && e.etype !== 'golem') audio.sfx('blip');
   170→      // el banner de jefes lo disparan sus cerebros (flags sirenaIntro/golemIntro)
   171→    }
   172→  }
   173→
   174→  // QUEBRADO/ATURDIDO — mismo patrón que update.ts (líneas 517-522):
   175→  // cuenta atrás y, al salir, recupera la barra de quiebre completa
   176→  // (clave para el REMATE del Portador). El motor fija aiT=4 genérico
   177→  // en damageEnemy al quebrar; los jefes lo ajustan a su duración.
   178→  if (e.ai === 'aturdido') {
   179→    const stunT = e.etype === 'sirena' ? 2.6 : e.etype === 'golem' ? 2.8 : 1.6;
   180→    if (e.aiT > stunT) e.aiT = stunT;
   181→    e.aiT -= dt;
   182→    e.moving = false;
   183→    e.windup = 0;
   184→    if (e.aiT <= 0) {
   185→      e.ai = 'persigue';
   186→      if (e.maxSta > 0) e.sta = e.maxSta;
   187→    }
   188→    return true;
   189→  }
   190→  return false;
   191→}

   193→/** Daño por contacto durante dashes (una vez por dash). */
   194→function contactHit(g: Game, e: Enemy, def: EnemyDef, m: ExpMem): void {
   195→  if (m.dashHit) return;
   196→  const p = g.player;
   197→  if (!p) return;
   198→  if (dist(e.x, e.y, p.x, p.y) < 8 + e.w / 2 + 4) {
   199→    m.dashHit = true;
   200→    // damagePlayer gestiona i-frames y parada perfecta (que aturde al atacante)
   201→    g.damagePlayer(def.dmg, e.x, e.y);
   202→  }
   203→}

   205→/** Flotación de espera sin Portador presente. */
   206→function idleFloat(g: Game, e: Enemy, dt: number): void {
   207→  e.moving = false;
   208→  e.anim += dt * 0.4;
   209→}

   212→// NEUMO DE MAREA — esquiva-tirador
   213→// ============================================================
   214→
   215→function tickNeumo(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   216→  if (commonTick(g, e, dt, def, m)) return true;
   217→  const st = e.subT ?? 0; // timer auxiliar (commonTick lo incrementa)
   218→  const p = g.player;
   219→  if (!p) { idleFloat(g, e, dt); return true; }
   220→  const d = dist(e.x, e.y, p.x, p.y);
   221→
   222→  // huida por proximidad: el Portador a <40 px lo espanta 1.5 s
   223→  if (e.aggro && e.ai !== 'huye' && e.ai !== 'carga' && d < 40) {
   224→    e.ai = 'huye';
   225→    e.aiT = 1.5;
   226→  }
   227→
   228→  switch (e.ai) {
   229→    case 'huye': {
   230→      e.aiT -= dt;
   231→      const ang = Math.atan2(e.y - p.y, e.x - p.x) + Math.sin(st * 5) * 0.6;
   232→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 1.25, dt);
   233→      if (e.aiT <= 0) e.ai = e.aggro ? 'persigue' : 'patrulla';
   234→      break;
   235→    }
   236→    case 'patrulla': {
   237→      // errando cerca de su hogar, con deriva de burbuja
   238→      e.aiT -= dt;
   239→      if (e.aiT <= 0) {
   240→        e.aiT = 1.5 + Math.random() * 1.5;
   241→        e.patrolAngle = Math.random() * Math.PI * 2;
   242→      }
   243→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 60;
   244→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   245→      const drift = Math.sin(st * 2) * 8;
   246→      const tx = Math.cos(ang) * 20 + Math.cos(ang + Math.PI / 2) * drift * 0.4;
   247→      const ty = Math.sin(ang) * 20 + Math.sin(ang + Math.PI / 2) * drift * 0.4;
   248→      g.moveEntity(e, tx * dt, ty * dt);
   249→      e.moving = true;
   250→      e.dir = tx > 0 ? 'right' : 'left';
   251→      if (e.aggro) e.ai = 'persigue';
   252→      break;
   253→    }
   254→    case 'persigue':
   255→    case 'recupera': {
   256→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   257→      if (d > def.aggroR * 2.6) { e.aggro = false; e.ai = 'patrulla'; break; }
   258→      // banda de flotación 90-130 px
   259→      const ang = Math.atan2(p.y - e.y, p.x - e.x);
   260→      if (d < 80) moveDir(g, e, -Math.cos(ang), -Math.sin(ang), def.speed, dt);
   261→      else if (d > 140) moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed, dt);
   262→      else {
   263→        // deriva lateral sinusoidal (subT)
   264→        const tang = ang + Math.PI / 2;
   265→        const s = Math.sin(st * 1.4);
   266→        moveDir(g, e, Math.cos(tang) * s, Math.sin(tang) * s, def.speed * 0.6, dt);
   267→      }
   268→      e.dir = p.x > e.x ? 'right' : 'left';
   269→      if (e.ai === 'recupera') {
   270→        e.aiT -= dt;
   271→        if (e.aiT > 0) break;
   272→        e.ai = 'persigue';
   273→      }
   274→      // disparo telegrafiado
   275→      if (e.atkCd <= 0 && d < def.atkR) {
   276→        e.ai = 'carga';
   277→        e.windup = def.windup; // 0.7: el motor dibuja '!' por windup>0 en 'carga'
   278→      }
   279→      break;
   280→    }
   281→    case 'carga': {
   282→      e.windup -= dt;
   283→      e.moving = false;
   284→      e.dir = p.x > e.x ? 'right' : 'left';
   285→      if (e.windup <= 0) {
   286→        // escupe 1 'orb' hacia el Portador
   287→        const dx = p.x - e.x, dy = p.y - 4 - e.y;
   288→        const l = Math.max(1, Math.hypot(dx, dy));
   289→        g.projectiles.push({
   290→          x: e.x, y: e.y - 2, vx: (dx / l) * 120, vy: (dy / l) * 120, t: 1.8,
   291→          dmg: def.dmg, element: 'ninguno', from: 'enemy', sprite: 'orb', radius: 5, pierce: 0,
   292→        });
   293→        // sfx 'splash' lo añade el agente de pistas; si aún no existe, no-op seguro
   294→        audio.sfx('splash');
   295→        e.atkCd = def.atkCd * (0.9 + Math.random() * 0.2);
   296→        e.ai = 'recupera';
   297→        e.aiT = 0.4;
   298→      }
   299→      break;
   300→    }
   301→    default:
   302→      break;
   303→  }
   304→  m.lastAtkCd = e.atkCd;
   305→  return true;
   306→}

   312→function tickEspectro(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   313→  if (commonTick(g, e, dt, def, m)) return true;
   314→  const p = g.player;
   315→  if (!p) { idleFloat(g, e, dt); return true; }
   316→  const d = dist(e.x, e.y, p.x, p.y);
   317→
   318→  // DESVANECERSE bajo los golpes: invulnerable 0.9 s en bruma
   319→  // (el integrador cablea invulT en damageEnemy; aquí lo fijamos y decaemos)
   320→  if (m.freshHit && (e.invulT ?? 0) <= 0) {
   321→    e.invulT = 0.9;
   322→    for (let i = 0; i < 3; i++) {
   323→      g.particles.push({
   324→        x: e.x + (i - 1) * 5, y: e.y - 4 - i * 3,
   325→        vx: (Math.random() - 0.5) * 24, vy: -16 - Math.random() * 16,
   326→        t: 0.5, maxT: 0.5, color: '#c9d4e4', size: 2, grav: -8,
   327→      });
   328→    }
   329→    audio.sfx('shadow');
   330→  }
   331→
   332→  switch (e.ai) {
   333→    case 'patrulla': {
   334→      // erra y a veces se DETIENE 1-2 s (susurra)
   335→      if ((m.pauseT ?? 0) > 0) {
   336→        m.pauseT = (m.pauseT ?? 0) - dt;
   337→        e.moving = false;
   338→        if (Math.random() < 0.06) {
   339→          g.particles.push({ x: e.x + (Math.random() - 0.5) * 10, y: e.y - 8, vx: 0, vy: -12, t: 0.7, maxT: 0.7, color: '#c9d4e4', size: 1.5, grav: -4 });
   340→        }
   341→        break;
   342→      }
   343→      e.aiT -= dt;
   344→      if (e.aiT <= 0) {
   345→        e.aiT = 1.6 + Math.random() * 2;
   346→        e.patrolAngle = Math.random() * Math.PI * 2;
   347→        if (Math.random() < 0.45) { m.pauseT = 1 + Math.random(); break; }
   348→      }
   349→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 60;
   350→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   351→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 0.45, dt);
   352→      if (e.aggro) e.ai = 'persigue';
   353→      break;
   354→    }
   355→    case 'persigue': {
   356→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   357→      if (d > def.aggroR * 2.4) { e.aggro = false; e.ai = 'patrulla'; break; }
   358→      // flota persiguiendo lento (colisión de tiles normal)
   359→      if (d > 18) moveDir(g, e, p.x - e.x, p.y - e.y, def.speed, dt);
   360→      else e.moving = false;
   361→      e.dir = p.x > e.x ? 'right' : 'left';
   362→      // arremetida desde la bruma
   363→      if (e.atkCd <= 0 && d < 58) {
   364→        e.ai = 'carga';
   365→        e.windup = def.windup; // 0.5
   366→      }
   367→      break;
   368→    }
   369→    case 'carga': {
   370→      e.windup -= dt;
   371→      e.moving = false;
   372→      e.dir = p.x > e.x ? 'right' : 'left';
   373→      if (e.windup <= 0) {
   374→        const l = Math.max(1, d);
   375→        m.dashDx = (p.x - e.x) / l;
   376→        m.dashDy = (p.y - 4 - e.y) / l;
   377→        m.dashHit = false;
   378→        e.ai = 'ataca';
   379→        e.aiT = 0.35; // dash ×3 durante 0.35 s
   380→        audio.sfx('whoosh');
   381→      }
   382→      break;
   383→    }
   384→    case 'ataca': {
   385→      e.aiT -= dt;
   386→      g.moveEntity(e, m.dashDx! * def.speed * 3 * dt, m.dashDy! * def.speed * 3 * dt);
   387→      e.moving = true;
   388→      if (Math.random() < 0.4) {
   389→        g.particles.push({ x: e.x, y: e.y - 4, vx: 0, vy: -6, t: 0.3, maxT: 0.3, color: '#c9d4e4', size: 2, grav: 0 });
   390→      }
   391→      contactHit(g, e, def, m);
   392→      if (e.aiT <= 0) {
   393→        e.ai = 'recupera';
   394→        e.aiT = 0.5;
   395→        e.atkCd = def.atkCd * (0.85 + Math.random() * 0.3);
   396→      }
   397→      break;
   398→    }
   399→    case 'recupera': {
   400→      e.aiT -= dt;
   401→      e.moving = false;
   402→      if (e.aiT <= 0) e.ai = 'persigue';
   403→      break;
   404→    }
   405→    default:
   406→      break;
   407→  }
   408→  m.lastAtkCd = e.atkCd;
   409→  return true;
   410→}

   415→
   416→function tickArpi(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   417→  if (commonTick(g, e, dt, def, m)) return true;
   418→  const st = e.subT ?? 0; // acumulador del picado (commonTick lo incrementa)
   419→  const p = g.player;
   420→  if (!p) { idleFloat(g, e, dt); return true; }
   421→  const d = dist(e.x, e.y, p.x, p.y);
   422→
   423→  // frágil: herida grave → huida errática volando
   424→  if (e.aggro && e.ai !== 'huye' && e.hp < e.maxHp * 0.3) {
   425→    e.ai = 'huye';
   426→    e.aiT = 2.5;
   427→  }
   428→
   429→  switch (e.ai) {
   430→    case 'huye': {
   431→      e.aiT -= dt;
   432→      const ang = Math.atan2(e.y - p.y, e.x - p.x) + Math.sin(st * 6) * 0.7;
   433→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 1.35, dt);
   434→      if (e.aiT <= 0) e.ai = 'persigue';
   435→      break;
   436→    }
   437→    case 'patrulla': {
   438→      e.aiT -= dt;
   439→      if (e.aiT <= 0) {
   440→        e.aiT = 1.2 + Math.random() * 1.6;
   441→        e.patrolAngle = Math.random() * Math.PI * 2;
   442→      }
   443→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 70;
   444→      const wob = Math.sin(st * 3) * 0.4;
   445→      const base = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   446→      moveDir(g, e, Math.cos(base + wob), Math.sin(base + wob), def.speed * 0.55, dt);
   447→      if (e.aggro) e.ai = 'persigue';
   448→      break;
   449→    }
   450→    case 'persigue': {
   451→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   452→      if (d > def.aggroR * 2.4) { e.aggro = false; e.ai = 'patrulla'; break; }
   453→      // órbita a ~75 px: tangencial + corrección radial
   454→      if (m.orbitDir === undefined) m.orbitDir = e.patrolAngle > Math.PI ? -1 : 1;
   455→      if (Math.random() < 0.002) m.orbitDir *= -1;
   456→      const ang = Math.atan2(e.y - p.y, e.x - p.x); // del Portador a mí
   457→      const corr = Math.max(-1, Math.min(1, (d - 75) / 26));
   458→      const tang = ang + (Math.PI / 2) * m.orbitDir;
   459→      const vx = Math.cos(tang) * def.speed - Math.cos(ang) * corr * def.speed * 0.95;
   460→      const vy = Math.sin(tang) * def.speed - Math.sin(ang) * corr * def.speed * 0.95;
   461→      g.moveEntity(e, vx * dt, vy * dt);
   462→      e.moving = true;
   463→      e.dir = vx > 0 ? 'right' : 'left';
   464→      // PICADO cada ~2.5 s (subT)
   465→      if (st >= 2.5 && d < 150 && d > 30) {
   466→        e.subT = 0;
   467→        e.ai = 'carga';
   468→        e.windup = def.windup; // 0.35
   469→      }
   470→      break;
   471→    }
   472→    case 'carga': {
   473→      e.windup -= dt;
   474→      e.moving = false;
   475→      e.dir = p.x > e.x ? 'right' : 'left';
   476→      if (e.windup <= 0) {
   477→        // dash hacia la posición actual del Portador (240 px/s, 0.45 s)
   478→        const l = Math.max(1, d);
   479→        m.dashDx = (p.x - e.x) / l;
   480→        m.dashDy = (p.y - 4 - e.y) / l;
   481→        m.dashHit = false;
   482→        e.ai = 'ataca';
   483→        e.aiT = 0.45;
   484→        audio.sfx('whoosh');
   485→      }
   486→      break;
   487→    }
   488→    case 'ataca': {
   489→      e.aiT -= dt;
   490→      g.moveEntity(e, m.dashDx! * 240 * dt, m.dashDy! * 240 * dt);
   491→      e.moving = true;
   492→      if (Math.random() < 0.5) {
   493→        g.particles.push({
   494→          x: e.x - m.dashDx! * 6, y: e.y - m.dashDy! * 6,
   495→          vx: (Math.random() - 0.5) * 20, vy: (Math.random() - 0.5) * 20,
   496→          t: 0.3, maxT: 0.3, color: '#e8f4fc', size: 1.5, grav: 0,
   497→        });
   498→      }
   499→      contactHit(g, e, def, m);
   500→      if (e.aiT <= 0) {
   501→        e.ai = 'recupera';
   502→        e.aiT = 0.8; // tras el picado: se aleja 0.8 s
   503→      }
   504→      break;
   505→    }
   506→    case 'recupera': {
   507→      e.aiT -= dt;
   508→      const ang = Math.atan2(e.y - p.y, e.x - p.x);
   509→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 0.9, dt);
   510→      if (e.aiT <= 0) {
   511→        e.ai = 'persigue';
   512→        e.atkCd = def.atkCd * (0.85 + Math.random() * 0.3);
   513→      }
   514→      break;
   515→    }
   516→    default:
   517→      break;
   518→  }
   519→  m.lastAtkCd = e.atkCd;
   520→  return true;
   521→}

   608→function tickSirena(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   609→  if (commonTick(g, e, dt, def, m)) return true;
   610→  const st = e.subT ?? 0; // deriva de flotación (commonTick lo incrementa)
   611→  const p = g.player;
   612→  if (!p) { idleFloat(g, e, dt); return true; }
   613→  const d = dist(e.x, e.y, p.x, p.y);
   614→
   615→  // ---- banner de la jefa (misma técnica que guardianBrain) ----
   616→  if (e.aggro && !g.flags.sirenaIntro) {
   617→    g.flags.sirenaIntro = true;
   618→    g.bossBannerT = 3.2;
   619→    g.bossBannerText = 'SIRENA ABISAL';
   620→    g.bossBannerSub = 'La que olvidó su nombre';
   621→    addFlash(g, '#8ef0ff', 0.25);
   622→    addShake(g, 4);
   623→    audio.sfx('banner');
   624→    e.sumT = 12;
   625→    g.toast('La Sirena Abisal alza su canto entre la bruma', '#8ef0ff');
   626→  }
   627→
   628→  // ---- fases (1: >60% · 2: 60-30% · 3: <30%) ----
   629→  const hpPct = e.hp / e.maxHp;
   630→  const newPhase = hpPct > 0.6 ? 1 : hpPct > 0.3 ? 2 : 3;
   631→  if (newPhase > e.phase) {
   632→    e.phase = newPhase;
   633→    addFlash(g, newPhase === 3 ? '#b48fff' : '#8ef0ff', 0.2);
   634→    addShake(g, 5);
   635→    requestSlowmo(g, 0.2);
   636→    audio.sfx('roar');
   637→    g.toast(`La Sirena cambia de fase (${e.phase}/3)`, '#8ef0ff');
   638→    e.windup = 0;
   639→    m.pendingSalva = 0;
   640→    if (e.ai === 'carga') e.ai = 'persigue';
   641→    if (e.phase === 2) { m.tpT = 0; e.sumT = Math.min(e.sumT, 10); }
   642→  }
   643→
   644→  switch (e.ai) {
   645→    case 'patrulla': {
   646→      // flota en calma cerca de su hogar hasta el aggro
   647→      e.aiT -= dt;
   648→      if (e.aiT <= 0) {
   649→        e.aiT = 2 + Math.random() * 2;
   650→        e.patrolAngle = Math.random() * Math.PI * 2;
   651→      }
   652→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 50;
   653→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   654→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 0.4, dt);
   655→      if (e.aggro) e.ai = 'persigue';
   656→      break;
   657→    }
   658→    case 'persigue': {
   659→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   660→      // reposicionamiento por fase (banda de distancia)
   661→      let minD = 90, maxD = 140;
   662→      if (e.phase === 2) { minD = 100; maxD = 150; }
   663→      if (e.phase === 3) { minD = 60; maxD = 100; }
   664→      const ang = Math.atan2(p.y - e.y, p.x - e.x);
   665→      if (d > maxD) moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed, dt);
   666→      else if (d < minD) moveDir(g, e, -Math.cos(ang), -Math.sin(ang), def.speed, dt);
   667→      else if (e.phase === 3) {
   668→        // órbita lenta manteniendo 60-100 px
   669→        if (m.orbitDir === undefined) m.orbitDir = 1;
   670→        const tang = ang + (Math.PI / 2) * m.orbitDir;
   671→        moveDir(g, e, Math.cos(tang), Math.sin(tang), def.speed * 0.55, dt);
   672→      } else {
   673→        const tang = ang + Math.PI / 2;
   674→        const s = Math.sin(st * 0.8);
   675→        moveDir(g, e, Math.cos(tang) * s, Math.sin(tang) * s, def.speed * 0.4, dt);
   676→      }
   677→      e.dir = p.x > e.x ? 'right' : 'left';
   678→      // ataque cuando toca
   679→      if (e.atkCd <= 0) {
   680→        e.ai = 'carga';
   681→        e.windup = e.phase === 3 ? 0.6 : 0.55;
   682→        m.pendingSalva = e.phase === 3 ? 0 : e.phase === 2 ? 4 : 3;
   683→        if (e.phase < 3) {
   684→          // 'aro' de marea en su posición: el motor lo explota (daño si
   685→          // el Portador se pega a ella) al expirar el telegraph
   686→          e.telegraphKind = 'aro';
   687→          g.telegraphs.push({ x: e.x, y: e.y, r: 26, t: 0.55, maxT: 0.55, dmg: def.dmg, kind: 'aro' });
   688→        } else {
   689→          e.telegraphKind = 'salva';
   690→        }
   691→      }
   692→      break;
   693→    }
   694→    case 'carga': {
   695→      e.windup -= dt;
   696→      e.moving = false;
   697→      e.dir = p.x > e.x ? 'right' : 'left';
   698→      if (e.windup <= 0) {
   699→        e.telegraphKind = undefined;
   700→        fireNotas(g, e, p, m);
   701→        e.atkCd = (e.phase === 3 ? 2.2 : 2.0) * (0.9 + Math.random() * 0.2);
   702→        e.ai = 'recupera';
   703→        e.aiT = 0.35;
   704→      }
   705→      break;
   706→    }
   707→    case 'recupera': {
   708→      e.aiT -= dt;
   709→      e.moving = false;
   710→      if (e.aiT <= 0) e.ai = 'persigue';
   711→      break;
   712→    }
   713→    default:
   714→      break;
   715→  }
   716→
   717→  // ---- coro de neumos (fases 1-2, cadencia 12 s / 10 s) ----
   718→  if (e.aggro && e.phase < 3) {
   719→    e.sumT -= dt;
   720→    if (e.sumT <= 0) summonNeumo(g, e, e.phase, m);
   721→  }
   722→
   723→  // ---- teletransporte acuático (solo fase 2, cada 6 s) ----
   724→  if (e.aggro && e.phase === 2) {
   725→    m.tpT = (m.tpT ?? 0) + dt;
   726→    if (m.tpT >= 6) teleportSirena(g, e, p, m);
   727→  }
   728→
   729→  m.lastAtkCd = e.atkCd;
   730→  return true;
   731→  // NOTA: su muerte la gestiona el motor (killEnemy genérico + XP/oro).
   732→  // El integrador pondrá g.flags.sirenaDefeated en killEnemy (o watcher).
   733→}

   855→      e.moving = false;
   856→      e.dir = p.x > e.x ? 'right' : 'left';
   857→      if (e.windup <= 0) {
   858→        e.telegraphKind = undefined;
   859→        if (m.act === 'lanza') {
   860→          // 2 cristales 'shard' en arco hacia el Portador
   861→          const a = Math.atan2(p.y - 4 - e.y, p.x - e.x);
   862→          for (const off of [-0.13, 0.13]) {
   863→            g.projectiles.push({
   864→              x: e.x, y: e.y - 6, vx: Math.cos(a + off) * 150, vy: Math.sin(a + off) * 150,
   865→              t: 1.7, dmg: 12, element: 'hielo', from: 'enemy', sprite: 'shard', radius: 5, pierce: 0,
   866→            });
   867→          }
   868→          audio.sfx('ice');
   869→          e.atkCd = def.atkCd * (0.9 + Math.random() * 0.2);
   870→          e.ai = 'recupera';
   871→          e.aiT = 0.5;
   872→        } else if (m.act === 'embiste') {
   873→          // embestida en línea hacia el Portador
   874→          const l = Math.max(1, d);
   875→          m.dashDx = (p.x - e.x) / l;
   876→          m.dashDy = (p.y - e.y) / l;
   877→          m.dashHit = false;
   878→          e.ai = 'ataca';
   879→          e.aiT = 0.9;
   880→          audio.sfx('whoosh');
   881→        } else {
   882→          // slam: el telegraph ya colocada hace el trabajo; solo respira
   883→          e.atkCd = def.atkCd * (0.9 + Math.random() * 0.2);
   884→          e.ai = 'recupera';
   885→          e.aiT = 0.55;
   886→        }
   887→      }
   888→      break;
   889→    }
   890→    case 'ataca': {
   891→      // EMBESTIDA (180 px/s · 0.9 s) con rebote contra muros
   892→      e.aiT -= dt;
   893→      const stepX = m.dashDx! * 180 * dt;
   894→      const stepY = m.dashDy! * 180 * dt;
   895→      const freeX = g.boxFree(e.x + stepX, e.y, e.w, e.h);
   896→      const freeY = g.boxFree(e.x, e.y + stepY, e.w, e.h);
   897→      if (!freeX && !freeY) {
   898→        // ¡rebote! detiene la embestida
   899→        addShake(g, 3);
   900→        g.burst(e.x, e.y, '#dceef8', 12, 60);
   901→        e.ai = 'recupera';
   902→        e.aiT = 0.6;
   903→        e.atkCd = def.atkCd * (0.9 + Math.random() * 0.2);
   904→        break;
   905→      }
   906→      g.moveEntity(e, stepX, stepY);
   907→      e.moving = true;
   908→      // nieve levantada al embestir
   909→      if (Math.random() < 0.6) {
   910→        g.particles.push({
   911→          x: e.x + (Math.random() - 0.5) * 14, y: e.y + 4,
   912→          vx: -m.dashDx! * 30, vy: -20 - Math.random() * 20,
   913→          t: 0.4, maxT: 0.4, color: '#eef6fc', size: 2, grav: 30,
   914→        });
   915→      }
   916→      contactHit(g, e, def, m);
   917→      if (e.aiT <= 0) {
   918→        e.ai = 'recupera';
   919→        e.aiT = 0.6;
   920→        e.atkCd = def.atkCd * (0.9 + Math.random() * 0.2);
   921→      }
   922→      break;
   923→    }
   924→    case 'recupera': {
   925→      e.aiT -= dt;
   926→      e.moving = false;
   927→      if (e.aiT <= 0) e.ai = e.aggro ? 'persigue' : 'patrulla';
   928→      break;
   929→    }
   930→    default:
   931→      break;
   932→  }
   933→  m.lastAtkCd = e.atkCd;
   934→  return true;
   935→  // NOTA: su muerte la gestiona el motor (killEnemy genérico + XP/oro).
   936→  // El integrador pondrá g.flags.golemDefeated en killEnemy (o watcher).
   937→}
   938→
   939→// ============================================================
   940→// ENTRADA ÚNICA — se llama AL PASAR por los 5 tipos nuevos;
   941→// true = frame gestionado por completo (el motor salta su IA).
   942→// ============================================================
   943→
   944→export function expansionTick(g: Game, e: Enemy, dt: number, def: EnemyDef): boolean {
   945→  switch (e.etype) {
   946→    case 'neumo': return tickNeumo(g, e, dt, def, mem(e));
   947→    case 'espectro': return tickEspectro(g, e, dt, def, mem(e));
   948→    case 'arpi': return tickArpi(g, e, dt, def, mem(e));
   949→    case 'sirena': return tickSirena(g, e, dt, def, mem(e));
   950→    case 'golem': return tickGolem(g, e, dt, def, mem(e));
   951→    default: return false; // tipos base: IA genérica del motor
   952→  }
   953→}
