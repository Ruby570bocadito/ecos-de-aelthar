   165→   165→  start() {
   166→   166→    if (this.running) return;
   167→   167→    this.running = true;
   168→   168→    this.lastTs = performance.now();
   169→   169→    const loop = (ts: number) => {
   170→   170→      if (!this.running) return;
   171→   171→      try {
   172→   172→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   173→   173→        this.lastTs = ts;
   174→   174→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   175→   175→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   176→   176→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   177→   177→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   178→   178→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   179→   179→        this.globalT += dt;
   180→   180→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   181→   181→        drawGame(this);
   182→   182→        this.loopError = null;
   183→   183→      } catch (err) {
   184→   184→        this.loopError = err instanceof Error ? (err.stack ?? String(err)) : String(err);
   185→   185→        console.error('[EcosAelthar loop]', err);
   186→   186→        try {
   187→   187→          this.ctx.fillStyle = '#000';
   188→   188→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   189→   189→          this.ctx.font = '12px monospace';
   190→   190→          this.ctx.fillStyle = '#ff7060';
   191→   191→          this.ctx.textAlign = 'left';
   192→   192→          this.ctx.textBaseline = 'top';
   193→   193→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   194→   194→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   195→   195→        } catch { /* noop */ }
   196→   196→      }
   197→   197→      this.raf = requestAnimationFrame(loop);
   198→   198→    };
   199→   199→    this.raf = requestAnimationFrame(loop);
   200→   200→  }

   333→   333→  loadMap(id: MapId, tx: number, ty: number) {
   334→   334→    // memoria del jefe: si sales de la cripta con el Guardián herido, no se
   335→   335→    // regenera al volver (cierra el exploit de reseteo de combate)
   336→   336→    // memoria del jefe (generalizada Acto II): si sales de un mapa con su JEFE vivo, no se regenera
   337→   337→    if (this.bossRef && !this.bossRef.dead) {
   338→   338→      this.flags[`bossHp_${this.mapId}`] = this.bossRef.hp;
   339→   339→      if (this.mapId === 'cripta') this.flags.bossHp = this.bossRef.hp; // compat con saves antiguos
   340→   340→    }
   341→   341→    this.mapId = id;
   342→   342→    this.map = MAPS[id];
   343→   343→    this.rows = mapRows(this.map);
   344→   344→    this.buildGround();
   345→   345→    if (this.player) {
   346→   346→      // defensa anti-softlock: nunca aterrizar sobre un tile sólido ni dentro
   347→   347→      // de una zona de salida del destino (provocaba el bucle cripta↔bosque)
   348→   348→      const [sx, sy] = this.findSafeTile(tx, ty);
   349→   349→      this.player.x = sx * TILE + 8;
   350→   350→      this.player.y = sy * TILE + 8;
   351→   351→    }
   352→   352→    this.exitCd = 0.9;
   353→   353→    this.spawnEnemies();
   354→   354→    // restaura la vida memorizada del Guardián si la partida sigue en curso
   355→   355→    const savedBossHp = typeof this.flags[`bossHp_${id}`] === 'number' ? (this.flags[`bossHp_${id}`] as number)
   356→   356→      : id === 'cripta' && typeof this.flags.bossHp === 'number' ? (this.flags.bossHp as number) : null;
   357→   357→    if (savedBossHp !== null) {
   358→   358→      const boss = this.enemies.find(e => BOSS_DEFEAT_FLAG[e.etype] !== undefined);
   359→   359→      if (boss) boss.hp = Math.max(1, Math.min(boss.maxHp, savedBossHp));
   360→   360→    }
   361→   361→    this.spawnNpcs();
   362→   362→    this.projectiles = []; this.waves = []; this.telegraphs = [];
   363→   363→    this.bossRef = null; this.bossActive = false;
   364→   364→    audio.setCombat(false);
   365→   365→    this.visitedMaps[id] = true;
   366→   366→    if (id === 'bosque' && !this.flags.visitedBosque) {
   367→   367→      this.flags.visitedBosque = true;
   368→   368→      if (this.questIdx === 2 && this.questStep === 0) this.questAdvance();
   369→   369→    }
   370→   370→    if (id === 'cripta' && !this.flags.visitedCripta) {
   371→   371→      this.flags.visitedCripta = true;
   372→   372→      if (this.questIdx === 3 && this.questStep === 0) this.questAdvance();
   373→   373→    }
   374→   374→    // Acto II: primera visita a los mapas nuevos avanza el objetivo de viaje
   375→   375→    if ((id === 'costa' || id === 'aldea' || id === 'cumbres') && !this.flags[`visited_${id}`]) {
   376→   376→      this.flags[`visited_${id}`] = true;
