   106→// nota musical pixel (sin depender de glifos de fuente)
   107→function drawNote(ctx: CanvasRenderingContext2D, x: number, y: number, s: number, color: string, variant: number) {
   108→  ctx.fillStyle = color;
   109→  ctx.fillRect(x, y + s * 3, s * 2, s * 1.5);          // cabeza
   110→  ctx.fillRect(x + s * 1.5, y, s * 0.75, s * 3.6);     // palo
   111→  if (variant === 0) {
   112→    ctx.fillRect(x + s * 2, y, s * 1.25, s * 1.1);     // banderín
   113→  } else {
   114→    ctx.fillRect(x + s * 2.6, y + s * 2.4, s * 2, s * 1.5); // segunda cabeza
   115→    ctx.fillRect(x + s * 1.5, y, s * 3, s * 0.8);      // barra
   116→  }
   117→}

   236→// helpers deterministas locales (evitan importar hash2 aquí)
   237→function hashT(i: number, k: number): number {
   238→  let h = (i + k * 57) * 2654435761;
   239→  h = (h ^ (h >> 13)) * 1274126177;
   240→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;
   241→}
   242→function getSprWisp(t: number, i: number): HTMLCanvasElement {
   243→  const frames = getSpr('wisp');
   244→  const f = frames?.length ? frames[Math.floor(t * 3 + i) % frames.length] : (undefined as unknown as HTMLCanvasElement);
   245→  if (!f) {
   246→    const w = window as unknown as { __wispMiss?: unknown[] };
   247→    if (!w.__wispMiss) w.__wispMiss = [];
   248→    if (w.__wispMiss.length < 3) {
   249→      w.__wispMiss.push({
   250→        t: Math.round(performance.now()), i,
   251→        hasFrames: !!frames, len: frames?.length ?? -1,
   252→        hasG: !!(window as unknown as { __g?: unknown }).__g,
   253→        heroAlba: !!getSpr('hero_alba'),
   254→      });
   255→    }
   256→  }
   257→  return f;
   258→}

   261→function hoverCorners(g: Game, x: number, y: number, w: number, h: number, col = COL.gold) {
   262→  const hov = g.mouse.x >= x && g.mouse.x <= x + w && g.mouse.y >= y && g.mouse.y <= y + h;
   263→  if (!hov) return;
   264→  const ctx = g.ctx;
   265→  const c = 7;
   266→  ctx.fillStyle = col;
   267→  ctx.fillRect(x - 2, y - 2, c, 2); ctx.fillRect(x - 2, y - 2, 2, c);
   268→  ctx.fillRect(x + w + 2 - c, y - 2, c, 2); ctx.fillRect(x + w, y - 2, 2, c);
   269→  ctx.fillRect(x - 2, y + h, c, 2); ctx.fillRect(x - 2, y + h - c, 2, c);
   270→  ctx.fillRect(x + w + 2 - c, y + h, c, 2); ctx.fillRect(x + w, y + h - c, 2, c);
   271→}

   580→function drawDialogue(g: Game) {
   581→  const ctx = g.ctx;
   582→  const node = g.dlgNode;
   583→  if (!node) return;
   584→  const bw = VIEW_W - 80, bx = 40;
   585→  // la caja crece según texto y opciones para que nunca se solapen
   586→  const fullLines = wrapText(node.text, 76).length;
   587→  const nOpts = node.options?.length ?? 0;
   588→  const bh = Math.max(150, 46 + fullLines * 19 + (nOpts > 0 ? nOpts * 26 + 14 : 0));
   589→  const by = VIEW_H - bh - 24;
   590→  ctx.fillStyle = 'rgba(4,5,12,0.4)';
   591→  ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   592→  panel(g, bx, by, bw, bh, '#6a5a38', 'rgba(10,12,22,0.96)');
   593→  // borde doble pulido + remaches
   594→  ctx.strokeStyle = 'rgba(240,200,74,0.4)';
   595→  ctx.lineWidth = 1;
   596→  ctx.strokeRect(bx + 4.5, by + 4.5, bw - 9, bh - 9);
   597→  ctx.fillStyle = COL.gold;
   598→  for (const [cxx, cyy] of [[bx + 7, by + 7], [bx + bw - 7, by + 7], [bx + 7, by + bh - 7], [bx + bw - 7, by + bh - 7]]) {
   599→    ctx.fillRect(cxx - 1.5, cyy - 1.5, 3, 3);
   600→  }
   601→
   602→  // retrato pixel (respira 1px y parpadea de vez en cuando)
   603→  const pkey = mapPortrait(node.portrait);
   604→  const blink = (g.globalT % 3.7) < 0.14;
   605→  const bob = Math.round(Math.sin(g.globalT * 2.3) * 0.9);
   606→  const frS = 74;
   607→  const frX = bx + 14, frY = by + 14;
   608→  ctx.fillStyle = '#0e0c18';

   610→  drawPortrait(ctx, pkey, frX + (frS - 28 * 1.72) / 2, frY + (frS - 40 * 1.72) / 2 + bob, 1.72, blink);
   611→  ctx.strokeStyle = '#6a5a38';
   612→  ctx.strokeRect(frX, frY, frS, frS);
   613→  ctx.strokeStyle = 'rgba(240,200,74,0.35)';
   614→  ctx.strokeRect(frX + 2.5, frY + 2.5, frS - 5, frS - 5);
   615→
   616→  // nombre y texto
   617→  text(g, node.name, bx + 104, by + 12, 17, COL.goldSoft);
   618→  const shown = node.text.slice(0, Math.floor(g.dlgCharT));
   619→  const lines = wrapText(shown, 76);
   620→  lines.forEach((l, i) => text(g, l, bx + 104, by + 36 + i * 19, 17, COL.text));
   621→
   622→  // opciones (debajo de la última línea de texto, sin solaparse nunca)
   623→  if (node.options && node.options.length > 0 && g.dlgCharT >= node.text.length) {
   624→    const oy = by + 36 + fullLines * 19 + 12;
   625→    for (let i = 0; i < node.options.length; i++) {

   630→        g.dlgSel = i;
   631→        g.advanceDialogue();

   634→        ctx.fillStyle = 'rgba(240,200,74,0.14)';
   635→        ctx.fillRect(ox - 4, oy + i * 26, 404, 24);

   637→      text(g, `${selected ? '▸' : ' '} ${opt.text}`, ox, oy + i * 26 + 3, 16, selected || hover ? COL.gold : COL.text);
   638→    }
