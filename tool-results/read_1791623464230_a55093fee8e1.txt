   101→/** Mueve hacia (dx,dy) normalizado con colisión de tiles y fija dir/moving. */
   102→function moveDir(g: Game, e: Enemy, dx: number, dy: number, spd: number, dt: number): void {
   103→  const l = Math.hypot(dx, dy);
   104→  if (l < 1e-4) { e.moving = false; return; }
   105→  g.moveEntity(e, (dx / l) * spd * dt, (dy / l) * spd * dt);
   106→  e.moving = true;
   107→  if (Math.abs(dx) > 0.01) e.dir = dx > 0 ? 'right' : 'left';
   108→}

   176→// ---------------- bloque común (réplica robusta del prep de update.ts) ----------------
   177→
   178→/**
   179→ * Devuelve TRUE si el frame quedó consumido (aturdido/spawnGuard).
   180→ * Efectos: freshHit, decaimiento de hitFlash/invulT/atkCd, aggro, anim.
   181→ * Todo es idempotente con el bloque común del motor gracias a la
   182→ * detección por anim/atkCd (ver ExpMem.lastAnim/lastAtkCd).
   183→ */
   184→function commonTick(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   185→  const p = g.player;
   186→
   187→  // ¿el motor ya pasó por updateEnemy este frame? (anim solo cambia ahí)
   188→  const motorRan = m.lastAnim !== undefined && e.anim !== m.lastAnim;
   189→  m.lastAnim = e.anim;
   190→
   191→  // spawnGuard (invocaciones: sin aggro inicial)
   192→  if (e.spawnGuard && e.spawnGuard > 0) {
   193→    if (!motorRan) e.spawnGuard = Math.max(0, e.spawnGuard - dt);
   194→    if (e.spawnGuard > 0) return true;
   195→  }
   196→
   197→  // golpe nuevo: hitFlash SUBIÓ desde el frame pasado (damageEnemy pone 0.12)
   198→  const prev = m.prevHitFlash ?? 0;
   199→  m.freshHit = e.hitFlash > prev + 5e-4 && e.hitFlash > 0.06;
   200→  // decaimiento de hitFlash si el motor aún no lo hizo (colocación temprana)
   201→  if (!motorRan && e.hitFlash > 0 && Math.abs(e.hitFlash - prev) < 1e-9) {
   202→    e.hitFlash = Math.max(0, e.hitFlash - dt);
   203→  }
   204→  m.prevHitFlash = e.hitFlash;
   205→
   206→  // knockback (contrato fxcore) si el motor no lo aplicó
   207→  if (!motorRan) stepKnockback(g, e, dt);
   208→
   209→  // estados (quemado/congelado) + marca — réplica del bloque común
   210→  if (!motorRan) {
   211→    for (let i = e.statuses.length - 1; i >= 0; i--) {
   212→      const s = e.statuses[i];
   213→      s.t -= dt;
   214→      if (s.kind === 'quemado') {
   215→        e.hp -= s.power * dt;
   216→        if (Math.random() < 0.15) {
   217→          g.particles.push({ x: e.x + (Math.random() - 0.5) * 8, y: e.y - 6, vx: 0, vy: -24, t: 0.3, maxT: 0.3, color: '#ff9040', size: 1.5, grav: 0 });
   218→        }
   219→        if (e.hp <= 0) { g.killEnemy(e); return true; }
   220→      }
   221→      if (s.t <= 0) e.statuses.splice(i, 1);
   222→    }
   223→    if (e.marked !== undefined) {
   224→      e.marked -= dt;
   225→      if (e.marked <= 0) e.marked = undefined;
   226→    }
   227→  }
   228→  if (e.hp < e.maxHp) e.aggro = true;
   229→
   230→  // invulT: campo nuevo de la expansión — SIEMPRE lo decaemos aquí
   231→  if (e.invulT && e.invulT > 0) e.invulT = Math.max(0, e.invulT - dt);
   232→
   233→  // atkCd: decae solo si el motor no lo hizo ya este frame
   234→  if (m.lastAtkCd === undefined || Math.abs(e.atkCd - m.lastAtkCd) < 1e-9) e.atkCd -= dt;
   235→
   236→  // anim (cosmético) si el motor no lo hizo
   237→  if (!motorRan) e.anim += dt * (e.ai === 'persigue' ? 1.4 : 0.6);
   238→
   239→  // timer auxiliar de la expansión
   240→  e.subT = (e.subT ?? 0) + dt;
   241→
   242→  // aggro (idempotente: si el motor ya lo fijó, no-op)
   243→  if (p && !e.aggro && g.state === 'play') {
   244→    const nightMult = isNightG(g) ? 1.3 : 1;
   245→    if (dist(e.x, e.y, p.x, p.y) < def.aggroR * nightMult) {
   246→      e.aggro = true;
   247→      // 14-a: los jefes nuevos NO pitan (sus cerebros disparan banner+sfx propio,
   248→      // misma técnica que sirena/golem)
   249→      if (e.etype !== 'sirena' && e.etype !== 'golem' && e.etype !== T_VULT && e.etype !== T_CORO) audio.sfx('blip');
   250→      // el banner de jefes lo disparan sus cerebros (flags sirenaIntro/golemIntro/vultIntro/coroIntro)
   251→    }
   252→  }
   253→
   254→  // QUEBRADO/ATURDIDO — mismo patrón que update.ts (líneas 517-522):
   255→  // cuenta atrás y, al salir, recupera la barra de quiebre completa
   256→  // (clave para el REMATE del Portador). El motor fija aiT=4 genérico
   257→  // en damageEnemy al quebrar; los jefes lo ajustan a su duración.
   258→  if (e.ai === 'aturdido') {
   259→    // 14-a: Vult se sacude más rápido (2.0 s); El Coro Roto tiene su
   260→    // propia mecánica: quebrar = CAE LA MÁSCARA (ver abajo, 1.4 s).
   261→    const stunT = e.etype === 'sirena' ? 2.6 : e.etype === 'golem' ? 2.8 : e.etype === T_VULT ? 2.0 : 1.6;
   262→    if (e.aiT > stunT) e.aiT = stunT;
   263→    // 14-a EL CORO ROTO: al quebrar la barra de quiebre cae la máscara
   264→    // actual y entra la siguiente con patrón nuevo (UNA vez por periodo).
   265→    if (e.etype === T_CORO && !m.maskRoto) {
   266→      m.maskRoto = true;
   267→      e.aiT = 1.4; // el trastabilleo de la caída es más corto que un quebrado
   268→      caeMascara(g, e, m, true);
   269→    }
   270→    e.aiT -= dt;
   271→    e.moving = false;
   272→    e.windup = 0;
   273→    // EL CANTO LA ROMPE / CRISTALES ROTOS (9-a): mientras dura el aturdimiento
   274→    // los jefes del Acto II sueltan su esencia (2 partículas por frame).
   275→    if (e.etype === 'sirena') {
   276→      // notas liberadas que ascienden (el canto se le escapa)
   277→      for (let i = 0; i < 2; i++) {
   278→        g.particles.push({
   279→          x: e.x + (Math.random() - 0.5) * 16, y: e.y - 4 - Math.random() * 10,
   280→          vx: (Math.random() - 0.5) * 18, vy: -26 - Math.random() * 18,
   281→          t: 0.55, maxT: 0.55, color: '#ffe9a0', size: 1.6, grav: 0,
   282→        });
   283→      }
   284→    } else if (e.etype === 'golem') {
   285→      // cristales rotos que caen
   286→      for (let i = 0; i < 2; i++) {
   287→        g.particles.push({
   288→          x: e.x + (Math.random() - 0.5) * 20, y: e.y - 10 - Math.random() * 8,
   289→          vx: (Math.random() - 0.5) * 14, vy: 8 + Math.random() * 14,
   290→          t: 0.5, maxT: 0.5, color: '#cfe8ff', size: 1.6, grav: 260,
   291→        });
   292→      }
   293→    }
   294→    if (e.aiT <= 0) {
   295→      e.ai = 'persigue';
   296→      if (e.maxSta > 0) e.sta = e.maxSta;
   297→      if (e.etype === T_CORO) m.maskRoto = false; // la siguiente máscara puede volver a caer
   298→    }
   299→    return true;
   300→  }
   301→  return false;
   302→}

   304→/** Daño por contacto durante dashes (una vez por dash). */
   305→function contactHit(g: Game, e: Enemy, def: EnemyDef, m: ExpMem): void {
   306→  if (m.dashHit) return;
   307→  const p = g.player;
   308→  if (!p) return;
   309→  if (dist(e.x, e.y, p.x, p.y) < 8 + e.w / 2 + 4) {
   310→    m.dashHit = true;
   311→    // damagePlayer gestiona i-frames y parada perfecta (que aturde al atacante)
   312→    g.damagePlayer(def.dmg, e.x, e.y);
   313→  }
   314→}

   325→
   326→function tickNeumo(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   327→  if (commonTick(g, e, dt, def, m)) return true;
   328→  const st = e.subT ?? 0; // timer auxiliar (commonTick lo incrementa)
   329→  const p = g.player;
   330→  if (!p) { idleFloat(g, e, dt); return true; }
   331→  const d = dist(e.x, e.y, p.x, p.y);
   332→
   333→  // huida por proximidad: el Portador a <40 px lo espanta 1.5 s
   334→  if (e.aggro && e.ai !== 'huye' && e.ai !== 'carga' && d < 40) {
   335→    e.ai = 'huye';
   336→    e.aiT = 1.5;
   337→  }
   338→
   339→  switch (e.ai) {
   340→    case 'huye': {
   341→      e.aiT -= dt;
   342→      const ang = Math.atan2(e.y - p.y, e.x - p.x) + Math.sin(st * 5) * 0.6;
   343→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 1.25, dt);
   344→      if (e.aiT <= 0) e.ai = e.aggro ? 'persigue' : 'patrulla';
   345→      break;
   346→    }
   347→    case 'patrulla': {
   348→      // errando cerca de su hogar, con deriva de burbuja
   349→      e.aiT -= dt;
   350→      if (e.aiT <= 0) {
   351→        e.aiT = 1.5 + Math.random() * 1.5;
   352→        e.patrolAngle = Math.random() * Math.PI * 2;
   353→      }
   354→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 60;
   355→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   356→      const drift = Math.sin(st * 2) * 8;
   357→      const tx = Math.cos(ang) * 20 + Math.cos(ang + Math.PI / 2) * drift * 0.4;
   358→      const ty = Math.sin(ang) * 20 + Math.sin(ang + Math.PI / 2) * drift * 0.4;
   359→      g.moveEntity(e, tx * dt, ty * dt);
   360→      e.moving = true;
   361→      e.dir = tx > 0 ? 'right' : 'left';
   362→      if (e.aggro) e.ai = 'persigue';
   363→      break;
   364→    }
   365→    case 'persigue':
   366→    case 'recupera': {
   367→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   368→      if (d > def.aggroR * 2.6) { e.aggro = false; e.ai = 'patrulla'; break; }
   369→      // banda de flotación 90-130 px
   370→      const ang = Math.atan2(p.y - e.y, p.x - e.x);
   371→      if (d < 80) moveDir(g, e, -Math.cos(ang), -Math.sin(ang), def.speed, dt);
   372→      else if (d > 140) moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed, dt);
   373→      else {
   374→        // deriva lateral sinusoidal (subT)
   375→        const tang = ang + Math.PI / 2;
   376→        const s = Math.sin(st * 1.4);
   377→        moveDir(g, e, Math.cos(tang) * s, Math.sin(tang) * s, def.speed * 0.6, dt);
   378→      }
   379→      e.dir = p.x > e.x ? 'right' : 'left';
   380→      if (e.ai === 'recupera') {
   381→        e.aiT -= dt;
   382→        if (e.aiT > 0) break;
   383→        e.ai = 'persigue';
   384→      }
   385→      // disparo telegrafiado
   386→      if (e.atkCd <= 0 && d < def.atkR) {
   387→        e.ai = 'carga';
   388→        e.windup = def.windup; // 0.7: el motor dibuja '!' por windup>0 en 'carga'
   389→      }
   390→      break;
   391→    }
   392→    case 'carga': {
   393→      e.windup -= dt;
   394→      e.moving = false;
   395→      e.dir = p.x > e.x ? 'right' : 'left';
   396→      if (e.windup <= 0) {
   397→        // escupe 'orb' hacia el Portador — NEUMO AGÓNICO (9-a): con hp<35%
   398→        // el escupitajo se abre en abanico de 3 orbes (±15°)
   399→        const dx = p.x - e.x, dy = p.y - 4 - e.y;
   400→        const base = Math.atan2(dy, dx);
   401→        const agoniza = e.hp < e.maxHp * 0.35;
   402→        const offs = agoniza ? [-0.2618, 0, 0.2618] : [0];
   403→        for (const off of offs) {
   404→          g.projectiles.push({
   405→            x: e.x, y: e.y - 2, vx: Math.cos(base + off) * 120, vy: Math.sin(base + off) * 120, t: 1.8,
   406→            dmg: def.dmg, element: 'ninguno', from: 'enemy', sprite: 'orb', radius: 5, pierce: 0,
   407→          });
   408→        }
   409→        // sfx 'splash' lo añade el agente de pistas; si aún no existe, no-op seguro
   410→        audio.sfx('splash');
   411→        e.atkCd = def.atkCd * (0.9 + Math.random() * 0.2);
   412→        e.ai = 'recupera';
   413→        e.aiT = 0.4;
   414→      }
   415→      break;
   416→    }
   417→    default:
   418→      break;
   419→  }
   420→  m.lastAtkCd = e.atkCd;
   421→  return true;
   422→}

   428→function tickEspectro(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   429→  if (commonTick(g, e, dt, def, m)) return true;
   430→  const p = g.player;
   431→  if (!p) { idleFloat(g, e, dt); return true; }
   432→  const d = dist(e.x, e.y, p.x, p.y);
   433→
   434→  // DESVANECERSE bajo los golpes: invulnerable 0.9 s en bruma
   435→  // (el integrador cablea invulT en damageEnemy; aquí lo fijamos y decaemos)
   436→  if (m.freshHit && (e.invulT ?? 0) <= 0) {
   437→    e.invulT = 0.9;
   438→    for (let i = 0; i < 3; i++) {
   439→      g.particles.push({
   440→        x: e.x + (i - 1) * 5, y: e.y - 4 - i * 3,
   441→        vx: (Math.random() - 0.5) * 24, vy: -16 - Math.random() * 16,
   442→        t: 0.5, maxT: 0.5, color: '#c9d4e4', size: 2, grav: -8,
   443→      });
   444→    }
   445→    audio.sfx('shadow');
   446→  }
   447→
   448→  switch (e.ai) {
   449→    case 'patrulla': {
   450→      // erra y a veces se DETIENE 1-2 s (susurra)
   451→      if ((m.pauseT ?? 0) > 0) {
   452→        m.pauseT = (m.pauseT ?? 0) - dt;
   453→        e.moving = false;
   454→        if (Math.random() < 0.06) {
   455→          g.particles.push({ x: e.x + (Math.random() - 0.5) * 10, y: e.y - 8, vx: 0, vy: -12, t: 0.7, maxT: 0.7, color: '#c9d4e4', size: 1.5, grav: -4 });
   456→        }
   457→        break;
   458→      }
   459→      e.aiT -= dt;
   460→      if (e.aiT <= 0) {
   461→        e.aiT = 1.6 + Math.random() * 2;
   462→        e.patrolAngle = Math.random() * Math.PI * 2;
   463→        if (Math.random() < 0.45) { m.pauseT = 1 + Math.random(); break; }
   464→      }
   465→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 60;
   466→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   467→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 0.45, dt);
   468→      if (e.aggro) e.ai = 'persigue';
   469→      break;
   470→    }
   471→    case 'persigue': {
   472→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   473→      if (d > def.aggroR * 2.4) { e.aggro = false; e.ai = 'patrulla'; break; }
   474→      // flota persiguiendo lento (colisión de tiles normal)
   475→      if (d > 18) moveDir(g, e, p.x - e.x, p.y - e.y, def.speed, dt);
   476→      else e.moving = false;
   477→      e.dir = p.x > e.x ? 'right' : 'left';
   478→      // arremetida desde la bruma
   479→      if (e.atkCd <= 0 && d < 58) {
   480→        e.ai = 'carga';
   481→        e.windup = def.windup; // 0.5
   482→      }
   483→      break;
   484→    }
   485→    case 'carga': {
   486→      e.windup -= dt;
   487→      e.moving = false;
   488→      e.dir = p.x > e.x ? 'right' : 'left';
   489→      if (e.windup <= 0) {
   490→        const l = Math.max(1, d);
   491→        m.dashDx = (p.x - e.x) / l;
   492→        m.dashDy = (p.y - 4 - e.y) / l;
   493→        m.dashHit = false;
   494→        e.ai = 'ataca';
   495→        e.aiT = 0.35; // dash ×3 durante 0.35 s
   496→        audio.sfx('whoosh');
   497→      }
   498→      break;
   499→    }
   500→    case 'ataca': {
   501→      e.aiT -= dt;
   502→      g.moveEntity(e, m.dashDx! * def.speed * 3 * dt, m.dashDy! * def.speed * 3 * dt);
   503→      e.moving = true;
   504→      if (Math.random() < 0.4) {
   505→        g.particles.push({ x: e.x, y: e.y - 4, vx: 0, vy: -6, t: 0.3, maxT: 0.3, color: '#c9d4e4', size: 2, grav: 0 });
   506→      }
   507→      contactHit(g, e, def, m);
   508→      if (e.aiT <= 0) {
   509→        e.ai = 'recupera';
   510→        e.aiT = 0.5;
   511→        e.atkCd = def.atkCd * (0.85 + Math.random() * 0.3);
   512→      }
   513→      break;
   514→    }
   515→    case 'recupera': {
   516→      e.aiT -= dt;
   517→      e.moving = false;
   518→      if (e.aiT <= 0) e.ai = 'persigue';
   519→      break;
   520→    }
   521→    default:
   522→      break;
   523→  }
   524→  m.lastAtkCd = e.atkCd;
   525→  return true;
   526→}

   532→function tickArpi(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   533→  if (commonTick(g, e, dt, def, m)) return true;
   534→  const st = e.subT ?? 0; // acumulador del picado (commonTick lo incrementa)
   535→  const p = g.player;
   536→  if (!p) { idleFloat(g, e, dt); return true; }
   537→  const d = dist(e.x, e.y, p.x, p.y);
   538→
   539→  // frágil: herida grave → huida errática volando
   540→  if (e.aggro && e.ai !== 'huye' && e.hp < e.maxHp * 0.3) {
   541→    e.ai = 'huye';
   542→    e.aiT = 2.5;
   543→  }
   544→
   545→  switch (e.ai) {
   546→    case 'huye': {
   547→      e.aiT -= dt;
   548→      const ang = Math.atan2(e.y - p.y, e.x - p.x) + Math.sin(st * 6) * 0.7;
   549→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 1.35, dt);
   550→      if (e.aiT <= 0) e.ai = 'persigue';
   551→      break;
   552→    }
   553→    case 'patrulla': {
   554→      e.aiT -= dt;
   555→      if (e.aiT <= 0) {
   556→        e.aiT = 1.2 + Math.random() * 1.6;
   557→        e.patrolAngle = Math.random() * Math.PI * 2;
   558→      }
   559→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 70;
   560→      const wob = Math.sin(st * 3) * 0.4;
   561→      const base = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   562→      moveDir(g, e, Math.cos(base + wob), Math.sin(base + wob), def.speed * 0.55, dt);
   563→      if (e.aggro) e.ai = 'persigue';
   564→      break;
   565→    }
   566→    case 'persigue': {
   567→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   568→      if (d > def.aggroR * 2.4) { e.aggro = false; e.ai = 'patrulla'; break; }
   569→      // órbita a ~75 px: tangencial + corrección radial
   570→      if (m.orbitDir === undefined) m.orbitDir = e.patrolAngle > Math.PI ? -1 : 1;
   571→      if (Math.random() < 0.002) m.orbitDir *= -1;
   572→      const ang = Math.atan2(e.y - p.y, e.x - p.x); // del Portador a mí
   573→      const corr = Math.max(-1, Math.min(1, (d - 75) / 26));
   574→      const tang = ang + (Math.PI / 2) * m.orbitDir;
   575→      const vx = Math.cos(tang) * def.speed - Math.cos(ang) * corr * def.speed * 0.95;
   576→      const vy = Math.sin(tang) * def.speed - Math.sin(ang) * corr * def.speed * 0.95;
   577→      g.moveEntity(e, vx * dt, vy * dt);
   578→      e.moving = true;
   579→      e.dir = vx > 0 ? 'right' : 'left';
   580→      // PICADO cada ~2.5 s (subT) — ARPÍAS EN PANDILLA (9-a): si otra arpi
   581→      // de la bandada picó hace <0.4 s y esta está en rango, se suma al
   582→      // asalto al instante (los windup de 0.35 s hacen que los picados
   583→      // caigan casi a la vez sobre el Portador).
   584→      const packT = g.globalT - lastDiveT;
   585→      if ((st >= 2.5 || packT < 0.4) && d < 150 && d > 30) {
   586→        e.subT = 0;
   587→        lastDiveT = g.globalT;
   588→        e.ai = 'carga';
   589→        e.windup = def.windup; // 0.35
   590→      }
   591→      break;
   592→    }
   593→    case 'carga': {
   594→      e.windup -= dt;
   595→      e.moving = false;
   596→      e.dir = p.x > e.x ? 'right' : 'left';
   597→      if (e.windup <= 0) {
   598→        // dash hacia la posición actual del Portador (240 px/s, 0.45 s)
   599→        const l = Math.max(1, d);
   600→        m.dashDx = (p.x - e.x) / l;
   601→        m.dashDy = (p.y - 4 - e.y) / l;
   602→        m.dashHit = false;
   603→        e.ai = 'ataca';
   604→        e.aiT = 0.45;
   605→        audio.sfx('whoosh');
   606→      }
   607→      break;
   608→    }
   609→    case 'ataca': {
   610→      e.aiT -= dt;
   611→      g.moveEntity(e, m.dashDx! * 240 * dt, m.dashDy! * 240 * dt);
   612→      e.moving = true;
   613→      if (Math.random() < 0.5) {
   614→        g.particles.push({
   615→          x: e.x - m.dashDx! * 6, y: e.y - m.dashDy! * 6,
   616→          vx: (Math.random() - 0.5) * 20, vy: (Math.random() - 0.5) * 20,
   617→          t: 0.3, maxT: 0.3, color: '#e8f4fc', size: 1.5, grav: 0,
   618→        });
   619→      }
   620→      contactHit(g, e, def, m);
   621→      if (e.aiT <= 0) {
   622→        e.ai = 'recupera';
   623→        e.aiT = 0.8; // tras el picado: se aleja 0.8 s
   624→      }
   625→      break;
   626→    }
   627→    case 'recupera': {
   628→      e.aiT -= dt;
   629→      const ang = Math.atan2(e.y - p.y, e.x - p.x);
   630→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 0.9, dt);
   631→      if (e.aiT <= 0) {
   632→        e.ai = 'persigue';
   633→        e.atkCd = def.atkCd * (0.85 + Math.random() * 0.3);
   634→      }
   635→      break;
   636→    }
   637→    default:
   638→      break;
   639→  }
   640→  m.lastAtkCd = e.atkCd;
   641→  return true;
   642→}

   734→function tickSirena(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
   735→  // ---- guarda de muerte (9-a): última salva visual ANTES de que el motor
   736→  //      la filtre del array (el killEnemy ya habrá puesto e.dead) ----
   737→  if (e.hp <= 0 && !e.dead) { deathSirena(g, e); return true; }
   738→  if (commonTick(g, e, dt, def, m)) return true;
   739→  const p = g.player;
   740→  if (!p) { idleFloat(g, e, dt); return true; }
   741→  const d = dist(e.x, e.y, p.x, p.y);
   742→
   743→  // ---- banner de la jefa (misma técnica que guardianBrain) ----
   744→  if (e.aggro && !g.flags.sirenaIntro) {
   745→    g.flags.sirenaIntro = true;
   746→    g.bossBannerT = 3.2;
   747→    g.bossBannerText = 'SIRENA ABISAL';
   748→    g.bossBannerSub = 'La que olvidó su nombre';
   749→    addFlash(g, '#8ef0ff', 0.25);
   750→    addShake(g, 4);
   751→    audio.sfx('banner');
   752→    e.sumT = 12;
   753→    g.toast('La Sirena Abisal alza su canto entre la bruma', '#8ef0ff');
   754→  }
   755→
   756→  // ---- fases (1: >60% · 2: 60-30% · 3: <30%) ----
   757→  const hpPct = e.hp / e.maxHp;
   758→  const newPhase = hpPct > 0.6 ? 1 : hpPct > 0.3 ? 2 : 3;
   759→  if (newPhase > e.phase) {
   760→    e.phase = newPhase;
   761→    addFlash(g, newPhase === 3 ? '#b48fff' : '#8ef0ff', 0.2);
   762→    addShake(g, 5);
   763→    requestSlowmo(g, 0.2);
   764→    audio.sfx('roar');
   765→    g.toast(`La Sirena cambia de fase (${e.phase}/3)`, '#8ef0ff');
   766→    e.windup = 0;
   767→    m.pendingSalva = 0;
   768→    if (e.ai === 'carga') e.ai = 'persigue';
   769→    if (e.phase === 2) { m.tpT = 0; e.sumT = Math.min(e.sumT, 10); }
   770→  }
   771→
   772→  switch (e.ai) {
   773→    case 'patrulla': {
   774→      // flota en calma cerca de su hogar hasta el aggro
   775→      e.aiT -= dt;
   776→      if (e.aiT <= 0) {
   777→        e.aiT = 2 + Math.random() * 2;
   778→        e.patrolAngle = Math.random() * Math.PI * 2;
   779→      }

   781→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   782→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 0.4, dt);
   783→      if (e.aggro) e.ai = 'persigue';
   784→      break;
   785→    }
   786→    case 'persigue': {
   787→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   788→      // reposicionamiento por fase (banda de distancia)
   789→      let minD = 90, maxD = 140;
   790→      if (e.phase === 2) { minD = 100; maxD = 150; }
   791→      if (e.phase === 3) { minD = 60; maxD = 100; }
   792→      const ang = Math.atan2(p.y - e.y, p.x - e.x);
   793→      if (d > maxD) moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed, dt);
   794→      else if (d < minD) moveDir(g, e, -Math.cos(ang), -Math.sin(ang), def.speed, dt);
   795→      else {
   796→        // MAREA BAJA (9-a): entre salvas strafea en arco orbital elegante
   797→        // alrededor del Portador (fase 2 amplia y serena, fase 3 cerrada)
   798→        if (m.orbitDir === undefined) m.orbitDir = 1;
   799→        if (Math.random() < 0.003) m.orbitDir *= -1; // invierte el arco de vez en cuando
   800→        const tang = ang + (Math.PI / 2) * m.orbitDir;
   801→        moveDir(g, e, Math.cos(tang), Math.sin(tang), def.speed * (e.phase === 3 ? 0.55 : 0.5), dt);
   802→      }
   803→      e.dir = p.x > e.x ? 'right' : 'left';
   804→      // ataque cuando toca
   805→      if (e.atkCd <= 0) {

  1624→// SÁTIRO DE LA NIEBLA — enemigo de mapa (14-a): silba BALADAS CURVAS

  1626→// marca de caída + orbe real con el daño); si te acercas, huye silbando.

  1628→function tickSatiro(g: Game, e: Enemy, dt: number, def: EnemyDef, m: ExpMem): boolean {
  1629→  if (commonTick(g, e, dt, def, m)) return true;
  1630→  const p = g.player;
  1631→  if (!p) { idleFloat(g, e, dt); return true; }
  1632→  const d = dist(e.x, e.y, p.x, p.y);
  1633→
  1634→  switch (e.ai) {
  1635→    case 'patrulla': {
  1636→      e.aiT -= dt;
  1637→      if (e.aiT <= 0) { e.aiT = 2 + Math.random() * 2; e.patrolAngle = Math.random() * Math.PI * 2; }
  1638→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 44;
  1639→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
  1640→      moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed * 0.4, dt);
  1641→      if (e.aggro) e.ai = 'persigue';
  1642→      break;
  1643→    }
  1644→    case 'persigue': {
  1645→      if (!e.aggro) { e.ai = 'patrulla'; break; }
  1646→      const ang = Math.atan2(p.y - e.y, p.x - e.x);
  1647→      if (d < 75) moveDir(g, e, -Math.cos(ang), -Math.sin(ang), def.speed * 0.95, dt);
  1648→      else if (d > 165) moveDir(g, e, Math.cos(ang), Math.sin(ang), def.speed, dt);
  1649→      else {
  1650→        if (m.orbitDir === undefined) m.orbitDir = 1;
  1651→        if (Math.random() < 0.004) m.orbitDir *= -1;
  1652→        const tang = ang + (Math.PI / 2) * m.orbitDir;
  1653→        moveDir(g, e, Math.cos(tang), Math.sin(tang), def.speed * 0.6, dt);
  1654→      }
  1655→      e.dir = p.x > e.x ? 'right' : 'left';
  1656→      if (e.atkCd <= 0 && d < def.atkR + 20) {
  1657→        // BALADA CURVA: marca de caída en la posición PREDICHA (0.5 s)
  1658→        const v = getPortadorVel();
  1659→        const px = p.x + v.x * 0.5, py = p.y + v.y * 0.5;
  1660→        m.saltoX = px; m.saltoY = py;
  1661→        e.ai = 'carga';
  1662→        e.windup = 0.8;
  1663→        e.telegraphKind = 'aro';
  1664→        // el telegraph SOLO avisa (dmg 0): el daño lo lleva el orbe
  1665→        g.telegraphs.push({ x: px, y: py, r: 24, t: 0.8, maxT: 0.8, dmg: 0, kind: 'aro' });
  1666→      }
  1667→      break;
  1668→    }
  1669→    case 'carga': {
  1670→      e.windup -= dt;
  1671→      e.moving = false;
  1672→      e.dir = p.x > e.x ? 'right' : 'left';
  1673→      if (e.windup <= 0) {
  1674→        e.telegraphKind = undefined;
  1675→        // silba el orbe hacia el punto marcado (llega cuando el 'aro' cierra)
  1676→        if (m.saltoX !== undefined && m.saltoY !== undefined) {
  1677→          const a = Math.atan2(m.saltoY - e.y, m.saltoX - e.x);
  1678→          const dd = Math.hypot(m.saltoX - e.x, m.saltoY - e.y);
  1679→          // el punto de spawn NUNCA dentro de pared (el bucle de proyectiles
  1680→          // elimina al instante los orbes nacidos en tile sólido): prueba el
  1681→          // offset de boca, el centro del cuerpo y un empujón en la dirección
  1682→          let ox = e.x, oy = e.y - 6;
  1683→          if (g.tileSolidAt(ox, oy)) oy = e.y;
  1684→          if (g.tileSolidAt(ox, oy)) { ox = e.x + Math.cos(a) * 10; oy = e.y + Math.sin(a) * 10; }
  1685→          g.projectiles.push({
  1686→            x: ox, y: oy, vx: Math.cos(a) * Math.max(60, dd / 0.8), vy: Math.sin(a) * Math.max(60, dd / 0.8),
  1687→            t: 1.1, dmg: def.dmg, element: 'ninguno', from: 'enemy', sprite: 'orb', radius: 4, pierce: 0,
  1688→          });
  1689→        }
  1690→        audio.sfx('blip');
  1691→        e.atkCd = 2.2 * (0.9 + Math.random() * 0.2);
  1692→        e.ai = 'recupera'; e.aiT = 0.4;
  1693→      }
  1694→      break;
  1695→    }
  1696→    case 'recupera': {
  1697→      e.aiT -= dt;
  1698→      e.moving = false;
  1699→      if (e.aiT <= 0) e.ai = 'persigue';
  1700→      break;
  1701→    }
  1702→    default: break;
  1703→  }
  1704→  m.lastAtkCd = e.atkCd;
  1705→  return true;
  1706→}
