    61→  return Math.hypot(bx - ax, by - ay);

    76→/** Color de chispa según el contexto: arma del Portador (acero/dorado
    77→ *  cargado) o elemento del golpe si viene de habilidad. */
    78→function sparkColorFor(p: Player, e: Enemy): string {
    79→  if (p.attackT > 0) return p.chargedHit ? '#ffe86a' : '#f5f2ea';
    80→  if (e.statuses.some(s => s.kind === 'quemado')) return '#ff9040';
    81→  if (e.statuses.some(s => s.kind === 'congelado')) return '#a0e8ff';
    82→  return '#e8e2d4';
    83→}

    85→/** Escanea enemigos buscando golpes recién conectados y suelta chispas
    86→ *  direccionales opuestas al swing (+ destello dorado si fue crítico:
    87→ *  el motor fija hitStop 0.09 en crítico frente a 0.04 normal).
    88→ *  Debe correr ANTES del bucle de enemigos (hitFlash aún sin decaer). */
    89→function updateCombatFx(g: Game): void {
    90→  fxF++;
    91→  const p = g.player;
    92→  if (!p) return;
    93→  for (const e of g.enemies) {
    94→    if (e.dead) continue;
    95→    const seen = fxHitSeen.get(e) ?? 0;
    96→    fxHitSeen.set(e, e.hitFlash);
    97→    if (e.hitFlash <= seen + 5e-4 || e.hitFlash <= 0.06) continue; // no es golpe nuevo
    98→    if (fxF - (fxOwnFx.get(e) ?? -99) <= 2) continue;              // ya tiene chispas propias
    99→    // rebote del impacto: chispas en dirección opuesta al swing
   100→    combatSparks(g, e.x, e.y - 4, p.x - e.x, p.y - e.y, sparkColorFor(p, e), 5, 70);
   101→    if (g.hitStop > 0.05) critGlint(g, e.x, e.y - 6);
   102→  }
   103→}

   105→export function updateGame(g: Game, dt: number) {
   106→  // cosmética siempre
   107→  if (g.state === 'dialogue' && g.dlgNode) g.dlgCharT += dt * 45;
   108→  g.mapTitleT = Math.max(0, g.mapTitleT - dt);
   109→  g.epochFx = Math.max(0, g.epochFx - dt);
   110→  g.shake = Math.max(0, g.shake - dt * 22);
   111→  for (const t of g.toasts) t.t -= dt;
   112→  g.toasts = g.toasts.filter(t => t.t > 0);
   113→  for (const f of g.floats) { f.t -= dt; f.y += f.vy * dt; }
   114→  g.floats = g.floats.filter(f => f.t > 0);
   115→  for (const p of g.particles) { p.t -= dt; p.x += p.vx * dt; p.y += p.vy * dt; p.vy += p.grav * dt; }
   116→  g.particles = g.particles.filter(p => p.t > 0);
   117→  if (g.state !== 'play') { g.updateCamera(); return; }
   118→
   119→  const p = g.player;
   120→  if (!p) return;
   121→  p.playTime += dt;
   122→  g.dayT = (g.dayT + dt / 240) % 1;
   123→  if (p.buffT) { p.buffT -= dt; if (p.buffT <= 0) p.buffT = undefined; }
   124→
   125→  // ---------------- fades y transición de mapa ----------------
   126→  // enfriamiento de salidas: tras cargar un mapa se ignoran sus zonas de
   127→  // salida un instante (defensa extra anti-bucle de teletransporte)
   128→  if (g.exitCd > 0) g.exitCd -= dt;
   129→  if (g.fadeDir !== 0) {
   130→    g.fadeT += g.fadeDir * dt * 2.4;
   131→    if (g.fadeT >= 1 && g.fadeDir > 0) {
   132→      g.fadeT = 1;
   133→      if (g.pendingMap) {
   134→        const { to, tx, ty } = g.pendingMap;
   135→        g.pendingMap = null;
   136→        g.loadMap(to, tx, ty);
   137→        audio.playTrack(g.map.music);
   138→        g.save();
   139→        g.toast('Autoguardado', '#8ef0ff');
   140→      }
   141→      g.fadeDir = -1;
   142→    }
   143→    if (g.fadeT <= 0 && g.fadeDir < 0) { g.fadeT = 0; g.fadeDir = 0; }
   144→    if (g.fadeDir !== 0) { g.updateCamera(); return; }
   145→  }
   146→
   147→  // ---------------- input del jugador ----------------
   148→  const k = g.keys;
   149→  let mx = 0, my = 0;
   150→  if (k.has('a') || k.has('arrowleft')) mx -= 1;
   151→  if (k.has('d') || k.has('arrowright')) mx += 1;
   152→  if (k.has('w') || k.has('arrowup')) my -= 1;
   153→  if (k.has('s') || k.has('arrowdown')) my += 1;
   154→  const mlen = Math.hypot(mx, my) || 1;
   155→
   156→  p.sta = Math.min(p.maxSta, p.sta + (p.rollT > 0 ? 0 : 26) * dt);
   157→  if (p.iframes > 0) p.iframes -= dt;
   158→  if (p.parryT > 0) p.parryT -= dt;
   159→  if (p.parryFx > 0) p.parryFx -= dt;
   160→  if (p.lastHitT > 0) p.lastHitT -= dt;
   161→  for (let i = 0; i < p.cds.length; i++) if (p.cds[i] > 0) p.cds[i] -= dt;
   162→
   163→  // knockback suave del jugador (ondas del jefe, slams) — contrato fxcore
   164→  stepKnockback(g, p, dt);
   165→
   166→  // carga de ataque
   167→  if (p.charging) {
   168→    p.chargeT += dt;
   169→    if (p.chargeT > 0.35 && Math.random() < 0.4) {
   170→      g.particles.push({ x: p.x + (Math.random() - 0.5) * 14, y: p.y - 10, vx: 0, vy: -30, t: 0.3, maxT: 0.3, color: '#ffe86a', size: 1.5, grav: 0 });
   171→    }
   172→  }
   173→
   174→  if (p.rollT > 0) {
   175→    // rodar
   176→    p.rollT -= dt;
   177→    const rd = (p as unknown as { rollDir?: Dir }).rollDir ?? p.dir;
   178→    const [rx, ry] = DIRS[rd];
   179→    const spd = 168;
   180→    g.moveEntity(p, rx * spd * dt, ry * spd * dt);
   181→    p.moving = true;
   182→    p.anim += dt * 1.4;
   183→    if (Math.random() < 0.5) g.particles.push({ x: p.x, y: p.y, vx: 0, vy: 0, t: 0.25, maxT: 0.25, color: '#d8dce4', size: 2, grav: 0 });
   184→  } else if (p.attackT > 0) {
   185→    p.attackT -= dt;
   186→    p.moving = false;
   187→    p.anim += dt;
   188→  } else {
   189→    // HIELO RESBALADIZO (Acto II, 9-a): caminar sobre el lago helado de las
   190→    // Cumbres (tile 'i') conserva la INERCIA. Fuera de hielo, control normal
   191→    // 1:1 (intacto). Nunca aplica con voltereta (otra rama) ni knockback.
   192→    const im = iceMem.get(p) ?? { vx: 0, vy: 0 };
   193→    iceMem.set(p, im);
   194→    const kbActive = (p.kbVx ?? 0) !== 0 || (p.kbVy ?? 0) !== 0;
   195→    const onIce = !kbActive &&
   196→      tileAt(g.map, g.rows, Math.floor(p.x / TILE), Math.floor((p.y + p.h / 2) / TILE), g.epoch) === 'i';
   197→    const spd = 74;
   198→    if (onIce) {
   199→      if (mx !== 0 || my !== 0) {
   200→        // al pisar hielo con velocidad, se parte de la inercia real del frame anterior
   201→        if (im.vx === 0 && im.vy === 0) { im.vx = plVX; im.vy = plVY; }
   202→        // aceleración reducida (×0.55): la velocidad persigue la objetivo más despacio
   203→        const k = 1 - Math.pow(0.55, dt * 60);
   204→        im.vx += ((mx / mlen) * spd - im.vx) * k;
   205→        im.vy += ((my / mlen) * spd - im.vy) * k;
   206→      } else {
   207→        // sin input: conserva la última velocidad con decaimiento 0.90^(dt·60)
   208→        // hasta bajar de 5 px/s (deslizamiento suave, no frenazo seco)
   209→        const dec = Math.pow(0.9, dt * 60);
   210→        im.vx *= dec;
   211→        im.vy *= dec;
   212→        if (im.vx * im.vx + im.vy * im.vy < 25) { im.vx = 0; im.vy = 0; }
   213→      }
   214→      if (im.vx !== 0 || im.vy !== 0) {
   215→        g.moveEntity(p, im.vx * dt, im.vy * dt);
   216→        p.moving = true;
   217→        p.anim += dt * ((mx !== 0 || my !== 0) ? 1 : 0.7); // derrapa sin input
   218→        if ((mx !== 0 || my !== 0) && (!p.charging || p.chargeT < 0.2)) {
   219→          if (Math.abs(mx) > Math.abs(my)) p.dir = mx > 0 ? 'right' : 'left';
   220→          else p.dir = my > 0 ? 'down' : 'up';
   221→        }
   222→      } else {
   223→        p.moving = false;
   224→        p.anim += dt * 0.4;
   225→      }
   226→    } else {
   227→      // control normal (tiles no helados): la memoria de hielo se resetea
   228→      im.vx = 0;
   229→      im.vy = 0;
   230→      if (mx !== 0 || my !== 0) {
   231→        g.moveEntity(p, (mx / mlen) * spd * dt, (my / mlen) * spd * dt);
   232→        p.moving = true;
   233→        p.anim += dt;
   234→        if (!p.charging || p.chargeT < 0.2) {
   235→          if (Math.abs(mx) > Math.abs(my)) p.dir = mx > 0 ? 'right' : 'left';
   236→          else p.dir = my > 0 ? 'down' : 'up';
   237→        }
   238→      } else {
   239→        p.moving = false;
   240→        p.anim += dt * 0.4;
   241→      }
   242→    }
   243→  }
   244→
   245→  // velocidad real del Portador (9-a): desplazamiento de este frame / dt.
   246→  // La usan la inercia del hielo (semilla al pisarlo) y la ventisca del
   247→  // gólem (getPortadorVel). Saltos de teleporte (>160 px) → velocidad 0.
   248→  if (plLastOk) {
   249→    const ddx = p.x - plLastX, ddy = p.y - plLastY;
   250→    if (ddx * ddx + ddy * ddy < 160 * 160) {
   251→      plVX = ddx / Math.max(1e-4, dt);
   252→      plVY = ddy / Math.max(1e-4, dt);
   253→      const vm = Math.hypot(plVX, plVY);
   254→      if (vm > 240) { plVX *= 240 / vm; plVY *= 240 / vm; } // tope (rodar+retroceso)
   255→    } else { plVX = 0; plVY = 0; }
   256→  }
   257→  plLastX = p.x; plLastY = p.y; plLastOk = true;
   258→
   259→  // ---------------- ventana de combo · golpe recién liberado · REMATE ----------------
   260→  // FX de impacto (8-c): detecta golpes conectados el frame anterior
   261→  // (input/parcelas) y suelta chispas; antes del bucle de enemigos.
   262→  updateCombatFx(g);
   263→
   264→  const prevAttackT = lastAttackT.get(p) ?? 0;
   265→  const attackStarted = p.attackT > 0 && prevAttackT <= 0;
   266→  lastAttackT.set(p, p.attackT);
   267→  // p.chargedHit es del motor (la usa render para el arco cargado); se limpia al
   268→  // terminar el golpe para que el siguiente ataque no herede el flag.
   269→  if (p.attackT <= 0 && p.chargedHit) p.chargedHit = false;
   270→
   271→  // ventana de combo: se mantiene mientras encadenas; al caducar el combo se reinicia
   272→  if (p.attackT > 0) {
   273→    p.comboT = 1.2;
   274→  } else if ((p.comboT ?? 0) > 0) {
   275→    p.comboT = (p.comboT ?? 0) - dt;
   276→    if ((p.comboT ?? 0) <= 0) { p.comboT = 0; p.combo = 0; }
   277→  }
   278→
   279→  if (attackStarted) {
   280→    // EMBESTIDA (8-c): el peso del tajo empuja al Portador un paso en su
   281→    // dirección. applyKnockback + stepKnockback (ya en marcha, línea 100)
   282→    // mueven con g.moveEntity → respeta tiles; fuerza suave que decae sola.
   283→    if (p.rollT <= 0) {
   284→      const [lgx, lgy] = DIRS[p.dir];
   285→      applyKnockback(p, lgx, lgy, p.chargedHit ? 88 : 62);
   286→    }
   287→    // impacto cargado: onda de empuje frontal (el peso del Portador)
   288→    if (p.chargedHit) {
   289→      const [fdx, fdy] = DIRS[p.dir];
   290→      const cx2 = p.x + fdx * 20, cy2 = p.y + fdy * 20;
   291→      for (const e of g.enemies) {
   292→        if (e.dead) continue;
   293→        if (Math.hypot(e.x - cx2, e.y - cy2) < 36 + e.w / 2) {
   294→          applyKnockback(e, e.x - p.x, e.y - p.y, 200);
   295→        }
   296→      }
   297→    }
   298→    // REMATE: enemigo quebrado cerca + golpe recién liberado → bonus único por quebrado
   299→    for (const e of g.enemies) {
   300→      if (e.dead || e.ai !== 'aturdido' || e.maxSta <= 0) continue;
   301→      if (finisherUsed.has(e)) continue;
   302→      if (dist(p.x, p.y, e.x, e.y) < 30) {
   303→        finisherUsed.add(e);
   304→        const [rdx, rdy] = DIRS[p.dir];
   305→        g.damageEnemy(e, playerMeleeDmg(p) * 0.6, 'ninguno', 90, rdx, rdy);
   306→        // chispas del REMATE (8-c): doradas, opuestas al swing; marcado para
   307→        // que updateCombatFx no las duplique al ver el hitFlash subir
   308→        fxOwnFx.set(e, fxF);
   309→        combatSparks(g, e.x, e.y - 4, -rdx, -rdy, '#ffe86a', 6, 90);
   310→        critGlint(g, e.x, e.y - 6);
   311→        g.floatAt(e.x, e.y - 26, '¡REMATE!', '#ffe86a', 12);
   312→        addShake(g, 4);
   313→        requestSlowmo(g, 0.2);
   314→        p.res = Math.min(p.maxRes, p.res + 15);
   315→        // sfx 'break' es nuevo del contrato (lo añade 3-a); audio.sfx no tiene
   316→        // default: nombre desconocido = no-op seguro.
   317→        audio.sfx('break');
   318→        break;
   319→      }
   320→    }
   321→  }
   322→
   323→  // esquiva (detección de borde: la pulsación se pone en cola en keydown;
   324→  // mantener Espacio ya no encadena volteretas con invulnerabilidad continua)
   325→  if (g.rollQueued && p.rollT <= 0 && p.attackT <= 0 && p.sta >= 20) {
   326→    g.rollQueued = false;
   327→    p.rollT = 0.3;
   328→    p.iframes = 0.34;
   329→    p.sta -= 20;
   330→    let rd: Dir = p.dir;
   331→    if (mx !== 0 || my !== 0) rd = Math.abs(mx) > Math.abs(my) ? (mx > 0 ? 'right' : 'left') : (my > 0 ? 'down' : 'up');
   332→    (p as unknown as { rollDir?: Dir }).rollDir = rd;
   333→    // soplo de esquiva (8-c): anillo de partículas que se abre con la voltereta
   334→    dodgeRing(g, p.x, p.y);
   335→    audio.sfx('dodge');
   336→  }
   337→
   338→  // impacto del ataque (instantáneo en la liberación, ver releaseCharge)
   339→
   340→  // ---------------- recoger oro perdido ----------------
   341→  for (let i = g.deadGolds.length - 1; i >= 0; i--) {
   342→    const dgl = g.deadGolds[i];
   343→    if (dgl.map !== g.mapId) continue;
   344→    if (dist(p.x, p.y, dgl.x, dgl.y) < 14) {
   345→      p.gold += dgl.amount;
   346→      g.deadGolds.splice(i, 1);
   347→      audio.sfx('coin');
   348→      g.floatAt(dgl.x, dgl.y - 10, `+${dgl.amount}`, '#f0c84a');
   349→      g.toast(`Recuperas tu eco de oro (+${dgl.amount})`, '#f0c84a');
   350→    }
   351→  }
   352→
   353→  // ---------------- salidas de mapa ----------------
   354→  if (g.fadeDir === 0 && g.exitCd <= 0) {
   355→    const ptx = Math.floor(p.x / TILE), pty = Math.floor(p.y / TILE);
   356→    for (const ex of g.map.exits) {
   357→      if (ex.needPast && g.epoch !== 'pasado') continue;
   358→      if (ptx >= ex.x && ptx < ex.x + ex.w && pty >= ex.y && pty < ex.y + ex.h) {
   359→        g.fadeTo(ex.to, ex.tx, ex.ty);
   360→        audio.sfx('echo');
   361→        break;
   362→      }
   363→    }
   364→  }
   365→
   366→  // ---------------- watchers de contenido (baratos, O(1) por frame) ----------------
   367→  // q2_done: Brisa da por completada la misión de los lobos → Ilwen aparece en el Bosque
   368→  // (la bandera no la fijaba nadie y su showFlag dejaba a la elfa invisible para siempre)
   369→  if (g.questIdx >= 2 && !g.flags.q2_done) g.flags.q2_done = true;
   370→  // Memoria II: tras recuperar el Eco de la Voz, la casa junto al río aflora al pisar el Bosque
   371→  if (g.flags.ecoVoz && g.mapId === 'bosque' && !g.flags.mem_casa) {
   372→    g.flags.mem_casa = true;
   373→    if (!(p.memories ?? []).includes('mem_casa')) g.applyAction('memory_mem_casa');
   374→  }
   375→  // Memoria III: la madre sin rostro aflora al vencer al Guardián Hueco
   376→  if (g.flags.guardianDefeated && !g.flags.mem_madre) {
   377→    g.flags.mem_madre = true;
   378→    if (!(p.memories ?? []).includes('mem_madre')) g.applyAction('memory_mem_madre');
   379→  }
   380→
   381→  // ---------------- compañera ----------------
   382→  if (g.companion) updateCompanion(g, dt);

   561→  // ORIGINAL de abajo, intacta.

   568→      c.moving = true; c.anim += dt;
   569→      c.dir = dx > 0 ? 'right' : dx < 0 ? 'left' : dy > 0 ? 'down' : 'up';

   621→      && (!defensivo16b || dist(e.x, e.y, p.x, p.y) < 2 * TILE));

   670→function updateEnemy(g: Game, e: Enemy, dt: number) {
   671→  const p = g.player!;
   672→  const def = ENEMY_DEFS[e.etype];
   673→  if (e.hitFlash > 0) e.hitFlash -= dt;
   674→  if (e.spawnGuard && e.spawnGuard > 0) { e.spawnGuard -= dt; return; }
   675→
   676→  // REMATE disponible al entrar en aturdimiento/quebrado (una vez por periodo)
   677→  const prevStun = wasAturdido.get(e) ?? false;
   678→  if (e.ai === 'aturdido' && !prevStun) finisherUsed.delete(e);
   679→  wasAturdido.set(e, e.ai === 'aturdido');
   680→
   681→  // knockback suave (proyectiles, cargas, ondas) — contrato fxcore
   682→  stepKnockback(g, e, dt);
   683→
   684→  // estados
   685→  for (let i = e.statuses.length - 1; i >= 0; i--) {
   686→    const s = e.statuses[i];
   687→    s.t -= dt;
   688→    if (s.kind === 'quemado') {
   689→      e.hp -= s.power * dt;
   690→      if (Math.random() < 0.15) g.particles.push({ x: e.x + (Math.random() - 0.5) * 8, y: e.y - 6, vx: 0, vy: -24, t: 0.3, maxT: 0.3, color: '#ff9040', size: 1.5, grav: 0 });
   691→      if (e.hp <= 0) { g.killEnemy(e); return; }
   692→    }
   693→    if (s.t <= 0) e.statuses.splice(i, 1);
   694→  }
   695→  // decae la marca de Ilwen (el icono lo dibuja 3-a)
   696→  if (e.marked !== undefined) {
   697→    e.marked -= dt;
   698→    if (e.marked <= 0) e.marked = undefined;
   699→  }
   700→  if (e.hp < e.maxHp) e.aggro = true;
   701→
   702→  // Acto II: decae la fase intangible y delega en el cerebro propio si lo maneja
   703→  if (e.invulT !== undefined && e.invulT > 0) e.invulT -= dt;
   704→  if (EXPANSION_TYPES.has(e.etype) && expansionTick(g, e, dt, def)) return;
   705→
   706→  const d = dist(e.x, e.y, p.x, p.y);
   707→  const nightMult = isNight(g) ? 1.3 : 1;
   708→  const aggroR = def.aggroR * nightMult * (e.etype === 'guardian' ? (g.bossActive ? 99 : 1) : 1);
   709→  if (!e.aggro && d < aggroR && g.state === 'play') {
   710→    e.aggro = true;
   711→    if (e.etype === 'guardian') {
   712→      // primera vez que entra en aggro: banner del jefe (render lo dibuja)
   713→      if (!g.flags.bossIntro) {
   714→        g.flags.bossIntro = true;
   715→        g.bossBannerT = 3.2;
   716→        g.bossBannerText = 'GUARDIÁN HUECO';
   717→        g.bossBannerSub = 'Custodio del Eco de la Voz';
   718→        addFlash(g, '#7ee8ff', 0.25);
   719→        // sfx 'banner' es nuevo del contrato (3-a); no-op seguro hasta entonces
   720→        audio.sfx('banner');
   721→      }
   722→    }
   723→    else audio.sfx('blip');
   724→  }
   725→  e.atkCd -= dt;
   726→  e.anim += dt * (e.ai === 'persigue' ? 1.4 : 0.6);
   727→
   728→  if (e.ai === 'aturdido') {
   729→    e.aiT -= dt;
   730→    e.moving = false;
   731→    if (e.aiT <= 0) { e.ai = 'persigue'; if (e.maxSta > 0) e.sta = e.maxSta; }
   732→    return;
   733→  }
   734→
   735→  switch (e.ai) {
   736→    case 'patrulla': {
   737→      e.aiT -= dt;
   738→      if (e.aiT <= 0) {
   739→        e.aiT = 1.5 + Math.random() * 2;
   740→        e.patrolAngle = Math.random() * Math.PI * 2;
   741→      }
   742→      // si se aleja de su hogar, vuelve; si no, deambula
   743→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 60;
   744→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   745→      const vx = Math.cos(ang) * 22, vy = Math.sin(ang) * 22;
   746→      const before = { x: e.x, y: e.y };
   747→      g.moveEntity(e, vx * dt * speedMult(e), vy * dt * speedMult(e));
   748→      if (Math.abs(e.x - before.x) < 0.01 && Math.abs(e.y - before.y) < 0.01) {
   749→        e.patrolAngle += Math.PI * (0.5 + Math.random());
   750→      }
   751→      e.moving = true;
   752→      e.dir = Math.cos(ang) > 0 ? 'right' : 'left';
   753→      if (e.aggro) e.ai = 'persigue';
   754→      break;
   755→    }
   756→    case 'persigue': {
   757→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   758→      if (d > aggroR * 2.4) { e.aggro = false; e.ai = 'patrulla'; break; }
   759→      // ==== 16-b (señuelo): enemigo atraído camina hacia el señuelo y NO
   760→      // ataca al Portador; al expirar el timer retoma la persecución normal ====
   761→      if (lureActive(g, e)) { sennoChase(g, e, dt, def.speed * speedMult(e)); break; }
   762→      const dx = (p.x - e.x) / (d || 1), dy = (p.y - e.y) / (d || 1);
   763→      const spd = def.speed * speedMult(e);
   764→      if (d > def.atkR * 0.8) {
   765→        g.moveEntity(e, dx * spd * dt, dy * spd * dt);
   766→        e.moving = true;
   767→        e.dir = dx > 0 ? 'right' : 'left';
   768→      } else e.moving = false;
   769→      if (d <= def.atkR && e.atkCd <= 0) {
   770→        e.ai = 'carga';
   771→        e.windup = def.windup * (e.etype === 'guardian' && e.phase >= 3 ? 0.7 : 1);
   772→        e.telegraphKind = undefined;
   773→      }
   774→      // jefe: acciones especiales
   775→      if (e.etype === 'guardian') guardianBrain(g, e, dt, d);
   776→      break;
   777→    }
   778→    case 'carga': {
   779→      e.windup -= dt;
   780→      e.moving = false;
   781→      // mirar al jugador
   782→      e.dir = p.x > e.x ? 'right' : 'left';
   783→      if (e.windup <= 0) {
   784→        if (e.telegraphKind === 'onda') {
   785→          // onda expansiva del Guardián
   786→          g.waves.push({ x: e.x, y: e.y, r: 10, maxR: 150, speed: 130, dmg: 13 * e.phase, hit: false });
   787→          audio.sfx('slam');
   788→          g.shake = 5;
   789→          e.telegraphKind = undefined;
   790→          e.ai = 'recupera';
   791→          e.aiT = 0.6;
   792→          e.atkCd = 2.6;
   793→        } else if (e.etype === 'guardian') {
   794→          // golpe en área
   795→          e.ai = 'ataca';
   796→          e.aiT = 0.22;
   797→          audio.sfx('slam');
   798→          if (d < 46) {
   799→            // el slam del Guardián arrea al Portador
   800→            if (p.parryT <= 0 && p.iframes <= 0 && p.rollT <= 0) applyKnockback(p, p.x - e.x, p.y - e.y, 260);
   801→            g.damagePlayer(def.dmg * e.phase, e.x, e.y);
   802→          }
   803→          g.burst(e.x, e.y, '#8a84a8', 14, 80);
   804→          g.shake = 5;
   805→        } else {
   806→          // estampida del lobo / mandoble del esqueleto
   807→          e.ai = 'ataca';
   808→          e.aiT = 0.22;
   809→          const dx = (p.x - e.x) / (d || 1), dy = (p.y - e.y) / (d || 1);
   810→          g.moveEntity(e, dx * 14, dy * 14);
   811→          if (d < def.atkR + 10) {
   812→            g.damagePlayer(def.dmg, e.x, e.y);
   813→          }
   814→        }
   815→      }
   816→      break;
   817→    }
   818→    case 'ataca': {
   819→      e.aiT -= dt;
   820→      if (e.aiT <= 0) {
   821→        e.ai = 'recupera';
   822→        e.aiT = 0.45;
   823→        e.atkCd = def.atkCd * (0.85 + Math.random() * 0.3);
   824→      }
   825→      break;
   826→    }
   827→    case 'recupera': {
   828→      e.aiT -= dt;
   829→      e.moving = false;
   830→      if (e.aiT <= 0) e.ai = e.aggro ? 'persigue' : 'patrulla';
   831→      break;
   832→    }
   833→    default: break;
   834→  }
   835→
   836→  // huida de lobos heridos
   837→  if (e.etype === 'lobo' && e.hp < e.maxHp * 0.25 && e.ai !== 'huye' && Math.random() < 0.01) {
   838→    e.ai = 'huye'; e.aiT = 2.2;
   839→  }
   840→  if (e.ai === 'huye') {
   841→    e.aiT -= dt;
   842→    const dx = (e.x - p.x) / (d || 1), dy = (e.y - p.y) / (d || 1);
   843→    g.moveEntity(e, dx * def.speed * 1.2 * dt, dy * def.speed * 1.2 * dt);
   844→    if (e.aiT <= 0) e.ai = 'persigue';
   845→  }
   846→}

   848→function guardianBrain(g: Game, e: Enemy, dt: number, d: number) {
   849→  const p = g.player!;
   850→  const hpPct = e.hp / e.maxHp;
   851→  const newPhase = hpPct > 0.6 ? 1 : hpPct > 0.3 ? 2 : 3;
   852→  if (newPhase !== e.phase) {
   853→    e.phase = newPhase;
   854→    audio.sfx('roar');
   855→    g.toast(`El Guardián cambia de fase (${e.phase}/3)`, '#7ee8ff');
   856→    addShake(g, 5);
   857→    addFlash(g, e.phase === 3 ? '#e8a0ff' : '#7ee8ff', 0.2);
   858→    requestSlowmo(g, 0.25);
   859→    if (e.phase === 2) {
   860→      // invoca sombras
   861→      for (let i = 0; i < 2; i++) {
   862→        const s = g.makeEnemy('sombra', e.x + (i === 0 ? -24 : 24), e.y + 12, 0, 'boss');
   863→        s.aggro = true;
   864→        g.enemies.push(s);
   865→      }
   866→      g.toast('El Guardián llama a sombras sin rostro', '#b48fff');
   867→    }
   868→  }
   869→  e.sumT -= dt;
   870→  if (e.atkCd <= 0) {
   871→    // elegir acción: slam a distancia, onda en fase 2+, swipe si muy cerca
   872→    if (d < 40) {
   873→      e.ai = 'carga';
   874→      e.windup = 0.55;
   875→      e.atkCd = 2.2;
   876→    } else if (e.phase >= 2 && Math.random() < 0.5) {
   877→      // onda expansiva
   878→      e.windup = 0.9;
   879→      e.ai = 'carga';
   880→      e.telegraphKind = 'onda';
   881→      e.atkCd = 3.2;
   882→    } else {
   883→      // slam telegrafiado sobre el jugador
   884→      g.telegraphs.push({ x: p.x, y: p.y, r: 34, t: 0.85, maxT: 0.85, dmg: 16 * e.phase, kind: 'slam' });
   885→      e.atkCd = 2.4 / (e.phase >= 3 ? 1.4 : 1);
   886→      e.ai = 'recupera';
   887→      e.aiT = 0.5;
   888→    }
   889→  }
   890→}
