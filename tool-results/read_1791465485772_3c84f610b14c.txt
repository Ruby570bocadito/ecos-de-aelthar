    10→    10→export function hash2(x: number, y: number): number {
    11→    11→  let h = x * 374761393 + y * 668265263;
    12→    12→  h = (h ^ (h >> 13)) * 1274126177;
    13→    13→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;
    14→    14→}

    20→    20→function mkCanvas(w: number, h: number): { c: HTMLCanvasElement; x: CanvasRenderingContext2D } {
    21→    21→  const c = document.createElement('canvas');

    23→    23→  const x = c.getContext('2d')!;
    24→    24→  x.imageSmoothingEnabled = false;

    59→    59→interface Layout {

    68→    68→function layoutFor(pal: HumanPal): Layout {
    69→    69→  if (pal.small) return { H: 15, headTop: 0, faceTop: 3, bodyTop: 7, bodyH: 4, armLen: 3 };
    70→    70→  return { H: 18, headTop: 1, faceTop: 4, bodyTop: 8, bodyH: 6, armLen: 4 };
    71→    71→}

    73→    73→/**
    74→    74→ * Dibuja un fotograma humanoide. f = fase de andar (0 zancada A,
    75→    75→ * 1 pase —cuerpo elevado 1px—, 2 zancada B). En reposo se usa f=1
    76→    76→ * sin bob para una pose neutra de pie.

    78→    78→function drawHumanFrame(
    79→    79→  x: CanvasRenderingContext2D, pal: HumanPal,
    80→    80→  dir: 'down' | 'up' | 'side', f: number, bob: number, L: Layout,
    81→    81→): void {
    82→    82→  const hs = pal.hairS ?? pal.hair;
    83→    83→  const ls = pal.legsS ?? pal.legs;
    84→    84→  const hT = L.headTop + bob;
    85→    85→  const fT = L.faceTop + bob;
    86→    86→  const bT = L.bodyTop + bob;
    87→    87→  const bodyBot = bT + L.bodyH - 1;           // última fila del torso (contorno)
    88→    88→  const legY = bodyBot + 1;                    // piernas ancladas al suelo
    89→    89→  const bootY = L.H - 1;
    90→    90→  const legH = bootY - legY;                   // filas de pierna antes de la bota
    91→    91→
    92→    92→  // balanceo de brazos (contrario entre izquierda y derecha)
    93→    93→  const offL = [1, 0, -1][f];
    94→    94→  const offR = [-1, 0, 1][f];
    95→    95→
    96→    96→  if (dir === 'down' || dir === 'up') {
    97→    97→    // ---------- cabeza ----------
    98→    98→    px(x, 4, hT, 8, 1, pal.outline);
    99→    99→    px(x, 3, hT + 1, 10, 5, pal.hair);
   100→   100→    px(x, 3, hT + 1, 10, 1, hs);
   101→   101→    if (pal.hood) { px(x, 3, fT + 2, 10, 2, pal.hair); px(x, 4, fT + 3, 8, 1, hs); }
   102→   102→    if (dir === 'down') {
   103→   103→      px(x, 5, fT, 6, 4, pal.mask ? (pal.maskC ?? pal.skin) : pal.skin);
   104→   104→      if (!pal.mask) {
   105→   105→        px(x, 6, fT + 1, 1, 2, pal.eye); px(x, 9, fT + 1, 1, 2, pal.eye);
   106→   106→        if (pal.beard) px(x, 5, fT + 3, 6, 2, pal.beard);
   107→   107→      } else {
   108→   108→        // máscara lisa: solo un brillo horizontal, sin boca ni ojos
   109→   109→        px(x, 6, fT + 1, 4, 1, '#eae6dc');
   110→   110→      }
   111→   111→      px(x, 3, fT + 2, 1, 2, pal.hair); px(x, 12, fT + 2, 1, 2, pal.hair);
   112→   112→      if (pal.ears === 'elf') { px(x, 2, fT + 1, 1, 2, pal.skin); px(x, 13, fT + 1, 1, 2, pal.skin); }
   113→   113→      if (pal.hairLong) { px(x, 3, fT + 4, 2, Math.max(1, bT - fT - 3), pal.hair); px(x, 11, fT + 4, 2, Math.max(1, bT - fT - 3), pal.hair); }
   114→   114→    }
   115→   115→    if (pal.ears === 'cat') {
   116→   116→      // orejas felinas sobre la cabeza (filas fijas para que no "floten")
   117→   117→      px(x, 3, 0, 1, 1, hs); px(x, 12, 0, 1, 1, hs);
   118→   118→      px(x, 3, 1, 2, 1, pal.hair); px(x, 11, 1, 2, 1, pal.hair);
   119→   119→    }
   120→   120→    if (pal.messy) {
   121→   121→      // pelo revuelto (mechones sueltos)
   122→   122→      px(x, 2, hT + 1, 1, 1, pal.hair); px(x, 13, hT + 1, 1, 1, pal.hair);
   123→   123→      px(x, 5, Math.max(0, hT - 1), 2, 1, pal.hair); px(x, 9, Math.max(0, hT - 1), 3, 1, pal.hair);
   124→   124→    }
   125→   125→
   126→   126→    // ---------- torso ----------
   127→   127→    px(x, 4, bT, 8, L.bodyH, pal.body);
   128→   128→    px(x, 4, bT, 8, 1, pal.bodyS);
   129→   129→    px(x, 5, bT + L.bodyH - 2, 6, 1, pal.accent);
   130→   130→    px(x, 4, bodyBot, 8, 1, pal.outline);
   131→   131→    if (pal.ribs) { px(x, 6, bT + 1, 4, 1, pal.bodyS); px(x, 6, bT + 2, 4, 1, pal.bodyS); }
   132→   132→    if (pal.pauldrons) { px(x, 3, bT, 2, 2, pal.accent); px(x, 11, bT, 2, 2, pal.accent); }
   133→   133→    if (pal.leafy) { px(x, 3, bT, 2, 1, '#8ac05a'); px(x, 11, bT, 2, 1, '#8ac05a'); }
   134→   134→
   135→   135→    // ---------- brazos (oscilación) ----------
   136→   136→    px(x, 2, bT + offL, 2, L.armLen, pal.body); px(x, 2, bT + offL + L.armLen, 2, 1, pal.skin);
   137→   137→    px(x, 12, bT + offR, 2, L.armLen, pal.body); px(x, 12, bT + offR + L.armLen, 2, 1, pal.skin);
   138→   138→    if (pal.hammer) {
   139→   139→      // martillo en la mano derecha
   140→   140→      px(x, 13, bT + offR - 2, 1, L.armLen + 1, '#7a5c3a');
   141→   141→      px(x, 12, bT + offR - 4, 3, 2, '#9aa4b4');
   142→   142→    }
   143→   143→
   144→   144→    // ---------- piernas ----------
   145→   145→    if (f === 0) {
   146→   146→      // zancada A: izquierda adelante, derecha atrás
   147→   147→      px(x, 4, legY, 3, legH, pal.legs); px(x, 9, legY, 3, legH, ls);
   148→   148→      px(x, 4, bootY, 3, 1, pal.boots); px(x, 9, bootY, 3, 1, pal.boots);
   149→   149→    } else if (f === 1) {
   150→   150→      // pase: piernas juntas, cuerpo elevado
   151→   151→      px(x, 5, legY, 3, legH, pal.legs); px(x, 8, legY, 3, legH, pal.legs);
   152→   152→      px(x, 5, bootY, 3, 1, pal.boots); px(x, 8, bootY, 3, 1, pal.boots);
   153→   153→    } else {
   154→   154→      // zancada B: derecha adelante, izquierda atrás
   155→   155→      px(x, 4, legY + 1, 3, Math.max(1, legH - 1), ls); px(x, 9, legY, 3, legH, pal.legs);
   156→   156→      px(x, 4, bootY, 3, 1, pal.boots); px(x, 9, bootY, 3, 1, pal.boots);
   157→   157→    }
   158→   158→    return;
   159→   159→  }
   160→   160→
   161→   161→  // ---------- dir === 'side' (mirando a la derecha) ----------
   162→   162→  px(x, 5, hT, 8, 1, pal.outline);
   163→   163→  px(x, 4, hT + 1, 9, 5, pal.hair);
   164→   164→  px(x, 4, hT + 1, 9, 1, hs);
   165→   165→  if (pal.hood) { px(x, 4, fT + 2, 8, 2, pal.hair); }
   166→   166→  px(x, 7, fT, 5, 4, pal.mask ? (pal.maskC ?? pal.skin) : pal.skin);
   167→   167→  if (!pal.mask) {
   168→   168→    px(x, 10, fT + 1, 1, 2, pal.eye);
   169→   169→    if (pal.beard) px(x, 8, fT + 3, 4, 2, pal.beard);
   170→   170→  } else {
   171→   171→    px(x, 8, fT + 1, 3, 1, '#eae6dc');
   172→   172→  }
   173→   173→  px(x, 5, fT + 2, 1, 2, pal.hair);
   174→   174→  if (pal.ears === 'elf') px(x, 6, fT + 1, 1, 2, pal.skin);
   175→   175→  if (pal.ears === 'cat') { px(x, 3, 0, 1, 1, hs); px(x, 4, 1, 2, 1, pal.hair); }
   176→   176→  if (pal.hairLong) px(x, 4, fT + 3, 2, Math.max(1, bT - fT - 2), pal.hair);
   177→   177→  if (pal.messy) px(x, 5, Math.max(0, hT - 1), 3, 1, pal.hair);
   178→   178→
   179→   179→  // torso
   180→   180→  px(x, 5, bT, 6, L.bodyH, pal.body);
   181→   181→  px(x, 5, bT, 6, 1, pal.bodyS);
   182→   182→  px(x, 6, bT + L.bodyH - 2, 4, 1, pal.accent);
   183→   183→  px(x, 5, bodyBot, 6, 1, pal.outline);
   184→   184→  if (pal.ribs) { px(x, 6, bT + 1, 3, 1, pal.bodyS); px(x, 6, bT + 2, 3, 1, pal.bodyS); }
   185→   185→  if (pal.pauldrons) px(x, 4, bT, 2, 2, pal.accent);
   186→   186→  if (pal.leafy) px(x, 4, bT, 2, 1, '#8ac05a');
   187→   187→
   188→   188→  // brazo delantero con balanceo + brazo trasero al tono sombreado
   189→   189→  const offF = [1, 0, -1][f];
   190→   190→  px(x, 8, bT + offF, 3, 3, pal.body);
   191→   191→  px(x, 10, bT + offF + 3, 2, 1, pal.skin);
   192→   192→  px(x, 5, bT - offF, 2, 3, pal.bodyS);
   193→   193→  if (pal.hammer) {
   194→   194→    px(x, 11, bT + offF - 2, 1, 4, '#7a5c3a');
   195→   195→    px(x, 10, bT + offF - 4, 3, 2, '#9aa4b4');
   196→   196→  }
   197→   197→
   198→   198→  // piernas en zancada (3 fases)
   199→   199→  if (f === 0) {
   200→   200→    px(x, 5, legY, 2, legH, ls); px(x, 8, legY, 2, legH, pal.legs);
   201→   201→    px(x, 5, bootY, 2, 1, pal.boots); px(x, 8, bootY, 2, 1, pal.boots);
   202→   202→  } else if (f === 1) {
   203→   203→    px(x, 6, legY, 2, legH, pal.legs); px(x, 8, legY + 1, 2, Math.max(1, legH - 1), ls);
   204→   204→    px(x, 6, bootY, 2, 1, pal.boots); px(x, 8, bootY, 2, 1, pal.boots);
   205→   205→  } else {
   206→   206→    px(x, 4, legY, 2, legH, ls); px(x, 7, legY, 2, legH, pal.legs);
   207→   207→    px(x, 4, bootY, 2, 1, pal.boots); px(x, 7, bootY, 2, 1, pal.boots);
   208→   208→  }
   209→   209→}

   211→   211→function buildHumanoid(pal: HumanPal): Frames {
   212→   212→  const frames: Frames = [];
   213→   213→  const dirs: ('down' | 'up' | 'side')[] = ['down', 'up', 'side'];
   214→   214→  const L = layoutFor(pal);
   215→   215→  for (const dir of dirs) {
   216→   216→    for (let f = 0; f < 3; f++) {
   217→   217→      const bob = f === 1 ? -1 : 0; // el cuerpo se eleva en la fase de pase
   218→   218→      const { c, x } = mkCanvas(16, L.H);
   219→   219→      drawHumanFrame(x, pal, dir, f, bob, L);
   220→   220→      frames.push(c);
   221→   221→    }
   222→   222→  }
   223→   223→  if (pal.big) {
   224→   224→    // El Gran Inquisidor se dibuja a escala ×2 (32×36): imponente como el Guardián
   225→   225→    return frames.map(fr => {
   226→   226→      const { c, x } = mkCanvas(32, L.H * 2);
   227→   227→      x.imageSmoothingEnabled = false;
   228→   228→      x.drawImage(fr, 0, 0, 32, L.H * 2);
   229→   229→      return c;
   230→   230→    });
   231→   231→  }
   232→   232→  return frames;
   233→   233→}

   235→   235→/** Índice de fotograma legacy (compatible con re-export de engine.ts). */
   236→   236→export function frameIndex(dir: string, moving: boolean, anim: number): number {
   237→   237→  const base = dir === 'down' ? 0 : dir === 'up' ? 2 : 4;
   238→   238→  if (!moving) return base;
   239→   239→  return base + (Math.floor(anim * 6) % 2);
   240→   240→}
   241→   241→
   242→   242→/**
   243→   243→ * Selección robusta de fotograma para cualquier sprite:
   244→   244→ * humanoids nuevos tienen 9 frames (3 por dirección), los demás
   245→   245→ * (lobo, guardián, wisp...) alternan su ciclo propio al moverse.
   246→   246→ */
   247→   247→export function entityFrame(spr: Frames, dir: string, moving: boolean, anim: number): number {
   248→   248→  const n = spr.length;
   249→   249→  if (n === 9) {
   250→   250→    const base = dir === 'up' ? 3 : dir === 'side' ? 6 : 0;
   251→   251→    if (!moving) return base + 1; // pose de pie = fase de pase sin bob
   252→   252→    return base + (Math.floor(anim * 8) % 3);
   253→   253→  }
   254→   254→  if (n >= 2 && moving) return Math.floor(anim * 6) % Math.min(2, n);
   255→   255→  return 0;
   256→   256→}

   330→   330→function buildChest(): { closed: Frames; open: Frames } {
   331→   331→  const mk = (open: boolean) => {
   332→   332→    const { c, x } = mkCanvas(16, 14);
   333→   333→    const O = '#3a2414', W = '#8a5a2b', W2 = '#6d4520', G = '#f0c84a';
   334→   334→    px(x, 2, open ? 5 : 3, 12, open ? 8 : 9, W);

   543→   543→// ---------------- Arco de ataque (slash) ----------------
   544→   544→
   545→   545→/**
   546→   546→ * Dibuja el arco de tajo del jugador. prog: 0→1 progreso del golpe,
   547→   547→ * combo: 0..2 (blanco → amarillo → dorado), charged: ataque cargado
   548→   548→ * (más grande, dorado y con estela). Todos los trazos son aditivos
   549→   549→ * suaves y restauran el estado del contexto.
   550→   550→ */
   551→   551→export function drawSlashArc(
   552→   552→  ctx: CanvasRenderingContext2D, cx: number, cy: number,
   553→   553→  dir: string, prog: number, combo: number, charged: boolean, zoom: number,
   554→   554→): void {
   555→   555→  const base = dir === 'down' ? Math.PI / 2 : dir === 'up' ? -Math.PI / 2 : dir === 'left' ? Math.PI : 0;
   556→   556→  const colors = charged ? '#ffd24a' : ['#f2f6fa', '#ffe86a', '#ffc040'][Math.max(0, Math.min(2, combo))];
   557→   557→  const r = (charged ? 27 : 19) * zoom;
   558→   558→  const span = 1.2;
   559→   559→  const a0 = base - 1.1 + prog * 1.45;
   560→   560→  ctx.save();
   561→   561→  // relleno en cuña (ataque cargado: más presencia)
   562→   562→  if (charged) {
   563→   563→    ctx.globalAlpha = 0.14;
   564→   564→    ctx.fillStyle = '#ffd24a';
   565→   565→    ctx.beginPath();
   566→   566→    ctx.moveTo(cx, cy);
   567→   567→    ctx.arc(cx, cy, r, a0, a0 + span);
   568→   568→    ctx.closePath();
   569→   569→    ctx.fill();
   570→   570→  }
   571→   571→  // estela: dos arcos anteriores con alpha decreciente
   572→   572→  for (let i = 2; i >= 1; i--) {
   573→   573→    const p2 = prog - i * 0.17;
   574→   574→    if (p2 <= 0) continue;
   575→   575→    ctx.globalAlpha = charged ? 0.34 - i * 0.12 : 0.26 - i * 0.1;
   576→   576→    ctx.strokeStyle = colors;
   577→   577→    ctx.lineWidth = (charged ? 5 : 3) - i;
   578→   578→    ctx.beginPath();
   579→   579→    ctx.arc(cx, cy, r * (1 - i * 0.05), base - 1.1 + p2 * 1.45, base - 1.1 + p2 * 1.45 + span * 0.8);
   580→   580→    ctx.stroke();
   581→   581→  }
   582→   582→  // arco principal
   583→   583→  ctx.globalAlpha = 0.95;
   584→   584→  ctx.strokeStyle = colors;
   585→   585→  ctx.lineWidth = charged ? 5 : 3;
   586→   586→  ctx.lineCap = 'round';
   587→   587→  ctx.beginPath();
   588→   588→  ctx.arc(cx, cy, r, a0, a0 + span);
   589→   589→  ctx.stroke();
   590→   590→  // filo brillante en el borde delantero
   591→   591→  ctx.globalAlpha = 1;
   592→   592→  ctx.strokeStyle = '#ffffff';
   593→   593→  ctx.lineWidth = 1;
   594→   594→  ctx.beginPath();
   595→   595→  ctx.arc(cx, cy, r, a0 + span - 0.22, a0 + span);
   596→   596→  ctx.stroke();
   597→   597→  ctx.restore();
   598→   598→}
