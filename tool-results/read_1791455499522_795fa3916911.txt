   138→  start() {
   139→    if (this.running) return;
   140→    this.running = true;
   141→    this.lastTs = performance.now();
   142→    const loop = (ts: number) => {
   143→      if (!this.running) return;
   144→      try {
   145→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   146→        this.lastTs = ts;
   147→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   148→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   149→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   150→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   151→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   152→        this.globalT += dt;
   153→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   154→        drawGame(this);
   155→        this.loopError = null;
   156→      } catch (err) {
   157→        this.loopError = String(err);
   158→        console.error('[EcosAelthar loop]', err);
   159→        try {
   160→          this.ctx.fillStyle = '#000';
   161→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   162→          this.ctx.font = '12px monospace';
   163→          this.ctx.fillStyle = '#ff7060';
   164→          this.ctx.textAlign = 'left';
   165→          this.ctx.textBaseline = 'top';
   166→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   167→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   168→        } catch { /* noop */ }
   169→      }
   170→      this.raf = requestAnimationFrame(loop);
   171→    };
   172→    this.raf = requestAnimationFrame(loop);
   173→  }
   174→
   175→  stop() {
   176→    this.running = false;
   177→    cancelAnimationFrame(this.raf);
   178→    audio.stopMusic();
   179→  }

   318→  buildGround() {
   319→    const { w, h } = this.map;
   320→    const mk = (ep: Epoch) => {
   321→      const c = document.createElement('canvas');
   322→      c.width = w * TILE; c.height = h * TILE;
   323→      const x = c.getContext('2d')!;
   324→      x.imageSmoothingEnabled = false;
   325→      // suelo
   326→      for (let ty = 0; ty < h; ty++) {
   327→        for (let tx = 0; tx < w; tx++) {
   328→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   329→          drawTile(x, ch, tx, ty, this.mapId, 0);
   330→        }
   331→      }
   332→      // objetos altos por orden de fila
   333→      for (let ty = 0; ty < h; ty++) {
   334→        for (let tx = 0; tx < w; tx++) {
   335→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   336→          if (SOLID_CHARS.has(ch) && (ch === 't' || ch === 'p')) drawTallTile(x, ch, tx, ty, this.mapId);
   337→        }
   338→      }
   339→      return c;
   340→    };
   341→    this.groundCanvas = mk('presente');
   342→    this.groundPastCanvas = this.map.epochDiffs.length ? mk('pasado') : this.groundCanvas;
   343→    // minimapa
   344→    const mini = document.createElement('canvas');
   345→    mini.width = w; mini.height = h;
   346→    const mx = mini.getContext('2d')!;
   347→    for (let ty = 0; ty < h; ty++) {
   348→      for (let tx = 0; tx < w; tx++) {
   349→        const ch = tileAt(this.map, this.rows, tx, ty, 'presente');
   350→        mx.fillStyle =
   351→          ch === '~' ? '#3a6a9a' : ch === '=' ? '#b89a6a' :
   352→          ch === ':' ? '#6a6a7a' : ch === 't' || ch === 'p' ? '#24512a' :
   353→          ch === '#' || ch === 'H' || ch === 'r' ? '#7a7a8a' :
   354→          ch === 'n' ? '#7ea49a' : '#4a8a44';
   355→        mx.fillRect(tx, ty, 1, 1);
   356→      }
   357→    }
   358→    this.miniCanvas = mini;
   359→  }

   409→  boxFree(x: number, y: number, w: number, h: number): boolean {
   410→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   411→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   412→  }

   414→  moveEntity(e: { x: number; y: number; w: number; h: number }, dx: number, dy: number) {
   415→    if (dx !== 0 && this.boxFree(e.x + dx, e.y, e.w, e.h)) e.x += dx;
   416→    if (dy !== 0 && this.boxFree(e.x, e.y + dy, e.w, e.h)) e.y += dy;
   417→  }

   419→  updateCamera(snap = false) {
   420→    if (!this.player) return;
   421→    const targetX = this.player.x * ZOOM - VIEW_W / 2;
   422→    const targetY = this.player.y * ZOOM - VIEW_H / 2;
   423→    const maxX = this.map.w * TILE * ZOOM - VIEW_W;
   424→    const maxY = this.map.h * TILE * ZOOM - VIEW_H;
   425→    const cx = Math.max(0, Math.min(maxX, targetX));
   426→    const cy = Math.max(0, Math.min(maxY, targetY));
   427→    if (snap) { this.camX = cx; this.camY = cy; }
   428→    else { this.camX += (cx - this.camX) * 0.14; this.camY += (cy - this.camY) * 0.14; }
   429→  }

   437→      this.toast('La Cripta existe fuera del tiempo.', '#9aa0b8');
   438→      return;
   439→    }
   440→    this.epoch = this.epoch === 'presente' ? 'pasado' : 'presente';
   441→    this.epochFx = 0.8;
   442→    audio.sfx('epoch');
   443→    this.toast(this.epoch === 'pasado' ? 'El pasado canta a tu alrededor' : 'Vuelves al presente en ruinas', '#ffe9a0');
   444→  }

   448→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   449→    const p = this.player;
   450→    if (!p) return null;
   451→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   452→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   453→      const d = Math.hypot(x - p.x, y - p.y);
   454→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   455→    };
   456→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   457→    for (const ch of this.map.chests) {
   458→      if (this.openedChests.has(ch.id)) continue;
   459→      if (ch.needPast && this.epoch !== 'pasado') continue;
   460→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   461→    }
   462→    for (const pr of this.map.props) {
   463→      if (pr.needPast && this.epoch !== 'pasado') continue;
   464→      if (pr.needPresent && this.epoch !== 'presente') continue;
   465→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   466→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   467→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   468→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   469→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   470→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   471→    }
   472→    for (const ec of this.map.echoes) {
   473→      if (this.takenEchoes.has(ec.id)) continue;
   474→      consider(ec.x * TILE + 8, ec.y * TILE + 8, 'echo', 'Escuchar eco menor', () => this.takeEchoMinor(ec.id, ec.title, ec.text), 26);
   475→    }
   476→    return best;
   477→  }

   574→  applyAction(action: string) {
   575→    const p = this.player;
   576→    if (!p) return;
   577→    switch (true) {
   578→      case action === 'accept_q2':
   579→        this.questIdx = 1; this.questStep = 0; this.flags.q2 = true;

   585→        audio.sfx('quest'); this.toast('Misión completada: Lobos en la Niebla (+50 coronas, +2 pociones, +10 reputación)', '#8ef0b0');
   586→        this.toast('Nueva misión: El Susurro del Bosque', '#8ef0b0');
   587→        break;
   588→      case action === 'accept_q4':
   589→        this.questIdx = 3; this.questStep = 0;
   590→        audio.sfx('quest'); this.toast('Nueva misión: La Cripta del Primer Canto', '#8ef0b0');
   591→        break;
   592→      case action === 'accept_q5':
   593→        this.questIdx = 4; this.questStep = 0;
   594→        audio.sfx('quest'); this.toast('Nueva misión: Ecos de Esperanza', '#8ef0b0');
   595→        break;
   596→      case action === 'fragment_touched':
   597→        this.flags.fragmentTouched = true;
   598→        p.hasEcho = true;
   599→        audio.sfx('echo');
   600→        this.burst(p.x, p.y - 8, '#ffe9a0', 24, 80);
   601→        if (this.questIdx === 2) this.questAdvance();
   602→        this.toast('Resonancia despierta: pulsa Q para alternar entre presente y pasado', '#ffe9a0');
   603→        break;
   604→      case action === 'eco_taken':
   605→        this.flags.ecoVoz = true;
   606→        p.points += 1; p.repGuardianes += 10;
   607→        p.hp = p.maxHp;
   608→        if (this.questIdx === 3 && this.questStep === 2) this.questAdvance();
   609→        this.toast('Eco de la Voz recuperado: +1 punto de habilidad, +10 reputación', '#ffe9a0');
   610→        break;
   611→      case action === 'fragment':
   612→        break;
   613→      case action === 'forge': {
   614→        const cost = 30 + p.weaponPlus * 25;
   615→        if (p.weaponPlus >= 5) { this.toast('Toln: «+5 es lo que da de sí esta forja, Portador.»', '#e8a'); audio.sfx('error'); }
   616→        else if (p.gold >= cost) {
   617→          p.gold -= cost; p.weaponPlus++;
   618→          audio.sfx('confirm');
   619→          this.toast(`Arma mejorada a +${p.weaponPlus} (${cost} coronas)`, '#f0c84a');
   620→        } else { this.toast(`Te faltan coronas (${cost} necesarias)`, '#e88'); audio.sfx('error'); }
   621→        break;
   622→      }
   623→      case action === 'buy_potion':
   624→        if (p.gold >= 15) { p.gold -= 15; p.potions++; audio.sfx('coin'); this.toast('Poción comprada (15 coronas)', '#f0c84a'); }
   625→        else { this.toast('Te faltan coronas', '#e88'); audio.sfx('error'); }
   626→        break;
   627→      case action === 'recruit_ilwen':
   628→        if (!this.companion) {
   629→          this.companion = this.makeCompanion();
   630→          this.toast('Ilwen se une al grupo: cubre tu espalda con su arco', '#8ef0b0');
   631→          audio.sfx('confirm');
   632→        }
   633→        break;
   634→      case action === 'rest':
   635→        p.hp = p.maxHp; p.sta = p.maxSta;
   636→        this.save();
   637→        audio.sfx('save');
   638→        this.toast('Descansas junto al cristal. Partida guardada. Vida restaurada.', '#8ef0ff');
   639→        break;
   640→      case action.startsWith('travel_'): {
   641→        const dest = action.slice(7) as MapId;
   642→        if (MAPS[dest]) {
   643→          this.closeDialogue();
   644→          this.fadeTo(dest, ...this.sanctuaryPos(dest));
   645→        }
   646→        break;
   647→      }
   648→      case action === 'end_demo': {
   649→        const mins = Math.floor(p.playTime / 60), secs = Math.floor(p.playTime % 60);
   650→        this.endStats =
   651→          `Nivel ${p.level} · ${p.kills} enemigos derrotados · ${p.gold} coronas\n` +
   652→          `Reputación Guardianes: ${p.repGuardianes >= 0 ? '+' : ''}${p.repGuardianes} · Muertes: ${p.deaths}\n` +
   653→          `Tiempo de juego: ${mins}m ${secs}s · Época favorita: la que tú elijas`;
   654→        this.save();
   655→        this.setState('end');
   656→        audio.playTrack('title');
   657→        break;
   658→      }
   659→      case action === 'close':
   660→        this.closeDialogue();
   661→        break;
   662→      default:
   663→        // acciones extendidas (sistema de tono, memorias, facciones...) — ver hooks.ts
   664→        handleCustomAction(this, action);
   665→        break;
   666→    }
   667→  }

   760→  private onKeyDown = (e: KeyboardEvent) => {
   761→    const k = e.key.toLowerCase();
   762→    if ([' ', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(k)) e.preventDefault();
   763→    if (e.repeat) return;
   764→    this.keys.add(k);
   765→    audio.resume();
   766→
   767→    if (this.state === 'title') {
   768→      if (k === 'enter' || k === 'e' || k === ' ') { /* botones por click; enter = nueva partida */ }
   769→    } else if (this.state === 'intro') {
   770→      if (k === 'e' || k === ' ' || k === 'enter') {
   771→        this.introIdx++;
   772→        audio.sfx('blip');
   773→        if (this.introIdx >= 3) this.startPlay();
   774→      }
   775→    } else if (this.state === 'dialogue') {
   776→      if (k === 'e' || k === ' ' || k === 'enter') this.advanceDialogue();
   777→      else if (k === 'arrowup' || k === 'w') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + this.dlgNode.options.length - 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
   778→      else if (k === 'arrowdown' || k === 's') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
   779→    } else if (this.state === 'play') {
   780→      if (k === 'e') this.tryInteract();
   781→      else if (k === 'q') this.epochSwitch();
   782→      else if (k === 'escape' || k === 'm') { this.setState('pause'); audio.sfx('uiOpen'); }
   783→      else if (k === 'f') this.drinkPotion();
   784→      else if (k === 't' && e.shiftKey) {
   785→        // accesibilidad / QoL: esperar hasta el alba
   786→        this.dayT = 0.22;
   787→        this.toast('Esperas junto al camino hasta el alba...', '#ffe86a');
   788→        audio.sfx('save');
   789→      }
   790→      else if (['1', '2', '3', '4'].includes(k)) this.useSkill(parseInt(k, 10) - 1);
   791→    } else if (this.state === 'pause') {
   792→      if (k === 'escape' || k === 'm') this.setState('play');
   793→    } else if (this.state === 'dead') {
   794→      if (k === 'e' || k === 'enter') this.respawn();
   795→    } else if (this.state === 'end') {
   796→      if (k === 'enter' || k === 'e') this.setState('title');
   797→    }
   798→  };

   817→  private onMouseDown = (e: MouseEvent) => {
   818→    audio.resume();
   819→    const p = this.canvasPos(e);
   820→    this.mouse.x = p.x; this.mouse.y = p.y;
   821→    // UI hits primero
   822→    for (const h of this.uiHit) {
   823→      if (p.x >= h.x && p.x <= h.x + h.w && p.y >= h.y && p.y <= h.y + h.h) {
   824→        audio.sfx('select');
   825→        h.cb();
   826→        return;
   827→      }
   828→    }
   829→    if (e.button === 2) { this.mouse.rdown = true; this.startParry(); return; }
   830→    if (e.button !== 0) return;
   831→    this.mouse.down = true;
   832→    if (this.state === 'dialogue') this.advanceDialogue();
   833→    else if (this.state === 'intro') { this.introIdx++; if (this.introIdx >= 3) this.startPlay(); else audio.sfx('blip'); }
   834→    else if (this.state === 'end') this.setState('title');
   835→    else if (this.state === 'play') this.startAttack();
   836→  };

   865→    const p = this.player!;

   871→  releaseCharge() {
   872→    const p = this.player!;
   873→    const wasCharging = p.charging;
   874→    p.charging = false;
   875→    const charged = wasCharging && p.chargeT > 0.35;
   876→    const chargeTime = p.chargeT;
   877→    p.chargeT = 0;
   878→    if (!wasCharging) return;
   879→    if (p.attackT > 0 || p.rollT > 0 || this.state !== 'play') return;
   880→    if (p.sta < 8) { audio.sfx('parryFail'); return; }
   881→    p.attackT = charged ? 0.4 : 0.26;
   882→    p.combo = (p.combo + 1) % 3;
   883→    p.sta -= charged ? 18 : 8;
   884→    p.chargedHit = charged;
   885→    this.aimAtMouse();
   886→    audio.sfx(charged ? 'swing2' : 'swing');
   887→    // daño del golpe
   888→    const comboMult = [1, 1.12, 1.28][p.combo];
   889→    const dmg = playerMeleeDmg(p) * comboMult * (charged ? 2.1 : 1);
   890→    if (charged) {

   999→  meleeHit(reach: number, width: number, dmg: number, element: Element, kb: number) {
  1000→    const p = this.player!;
  1001→    const dirs: Record<Dir, [number, number]> = { down: [0, 1], up: [0, -1], left: [-1, 0], right: [1, 0] };
  1002→    const [dx, dy] = dirs[p.dir];
  1003→    const cx = p.x + dx * reach * 0.6, cy = p.y + dy * reach * 0.6;
  1004→    for (const e of this.enemies) {
  1005→      if (e.dead) continue;
  1006→      if (Math.hypot(e.x - cx, e.y - cy) < reach + e.w / 2 + 4) {
  1007→        this.damageEnemy(e, dmg, element, kb, dx, dy);
  1008→      }
  1009→    }
  1010→    void width;
  1011→  }

  1013→  aoeHit(x: number, y: number, r: number, dmg: number, element: Element, stun: boolean) {
  1014→    for (const e of this.enemies) {
  1015→      if (e.dead) continue;
  1016→      if (Math.hypot(e.x - x, e.y - y) < r + e.w / 2) {
  1017→        this.damageEnemy(e, dmg, element, 60, Math.sign(e.x - x), Math.sign(e.y - y));
  1018→        if (stun) { e.ai = 'aturdido'; e.aiT = 1.5; }
  1019→      }
  1020→    }
  1021→  }

  1023→  chainLightning(x: number, y: number, dmg: number, jumps: number, nx: number, ny: number) {
  1024→    let cx = x + nx * 14, cy = y + ny * 14;
  1025→    const hitSet = new Set<Enemy>();
  1026→    for (let j = 0; j < jumps; j++) {
  1027→      let best: Enemy | null = null, bd = 90;
  1028→      for (const e of this.enemies) {
  1029→        if (e.dead || hitSet.has(e)) continue;
  1030→        const d = Math.hypot(e.x - cx, e.y - cy);
  1031→        if (d < bd) { bd = d; best = e; }
  1032→      }
  1033→      if (!best) break;
  1034→      hitSet.add(best);
  1035→      this.damageEnemy(best, dmg * (1 - j * 0.2), 'rayo', 20, 0, 0);
  1036→      this.lightningFx(cx, cy, best.x, best.y);
  1037→      cx = best.x; cy = best.y;
  1038→    }
  1039→  }

  1041→  lightningFx(x0: number, y0: number, x1: number, y1: number) {
  1042→    const steps = 6;
  1043→    for (let i = 0; i <= steps; i++) {
  1044→      const t = i / steps;
  1045→      this.particles.push({
  1046→        x: x0 + (x1 - x0) * t + (Math.random() - 0.5) * 6,
  1047→        y: y0 + (y1 - y0) * t + (Math.random() - 0.5) * 6,
  1048→        vx: 0, vy: 0, t: 0.18, maxT: 0.18, color: '#ffe86a', size: 2, grav: 0,
  1049→      });
  1050→    }
  1051→  }

  1053→  damageEnemy(e: Enemy, dmg: number, element: Element, kb: number, kbx = 0, kby = 0) {
  1054→    if (e.dead) return;
  1055→    const def = ENEMY_DEFS[e.etype];
  1056→    e.aggro = true;
  1057→    let final = dmg;
  1058→    let crit = false;
  1059→    const p = this.player!;
  1060→    if (Math.random() * 100 < Math.min(40, 5 + p.attrs.des * 2)) { final *= 2; crit = true; }
  1061→    if (def.weakTo !== 'ninguno' && element === def.weakTo) final *= 1.5;
  1062→    if (e.ai === 'aturdido' && e.etype === 'guardian') final *= 1.5;
  1063→    final = Math.max(1, Math.round(final));
  1064→    e.hp -= final;
  1065→    e.hitFlash = 0.12;
  1066→    // quiebre
  1067→    if (e.maxSta > 0) {
  1068→      e.sta -= final * 1.1;
  1069→      if (e.sta <= 0 && e.ai !== 'aturdido') {
  1070→        e.sta = 0;
  1071→        e.ai = 'aturdido';
  1072→        e.aiT = 4;
  1073→        this.toast('¡QUEBRADO! El enemigo queda vulnerable', '#ffe86a');
  1074→        this.floatAt(e.x, e.y - 20, 'QUEBRADO', '#ffe86a', 7);
  1075→        audio.sfx('roar');
  1076→      }
  1077→    }
  1078→    // estados
  1079→    if (element === 'fuego') e.statuses.push({ kind: 'quemado', t: 3, power: 4 });
  1080→    if (element === 'hielo') e.statuses.push({ kind: 'congelado', t: 2.5, power: 0.5 });
  1081→    // reacciones elementales
  1082→    if (element === 'hielo' && e.statuses.some(s => s.kind === 'quemado' && s.t > 1)) {
  1083→      this.aoeHit(e.x, e.y, 30, 12, 'ninguno', false);
  1084→      this.floatAt(e.x, e.y - 16, 'VAPOR', '#d0f0f8', 6);
  1085→      this.burst(e.x, e.y - 6, '#d0f0f8', 12);
  1086→    }
  1087→    // knockback
  1088→    if (kb > 0) this.moveEntity(e, kbx * kb * 0.06, kby * kb * 0.06);
  1089→    // feedback
  1090→    this.floatAt(e.x + (Math.random() - 0.5) * 8, e.y - 14, `${final}`, crit ? '#ffd24a' : '#fff');
  1091→    if (crit) this.floatAt(e.x, e.y - 22, '¡CRÍTICO!', '#ffd24a', 6);
  1092→    this.burst(e.x, e.y - 4, crit ? '#ffd24a' : '#f0e8e0', crit ? 10 : 5);
  1093→    audio.sfx(crit ? 'crit' : 'hit');
  1094→    this.hitStop = crit ? 0.09 : 0.04;
  1095→    // resonancia
  1096→    const espMult = 1 + p.attrs.esp * 0.1;
  1097→    p.res = Math.min(p.maxRes, p.res + 6 * espMult);
  1098→    if (e.hp <= 0) this.killEnemy(e);
  1099→  }

  1101→  killEnemy(e: Enemy) {
  1102→    e.dead = true;
  1103→    const def = ENEMY_DEFS[e.etype];
  1104→    const p = this.player!;
  1105→    p.kills++;
  1106→    const espMult = 1 + p.attrs.esp * 0.1;
  1107→    p.res = Math.min(p.maxRes, p.res + 10 * espMult);
  1108→    this.gainXp(def.xp);
  1109→    const gold = Math.round(def.gold[0] + Math.random() * (def.gold[1] - def.gold[0]));
  1110→    p.gold += gold;
  1111→    this.floatAt(e.x, e.y - 20, `+${gold} coronas`, '#f0c84a');
  1112→    this.burst(e.x, e.y - 4, e.etype === 'guardian' ? '#7ee8ff' : '#9ec4b4', e.etype === 'guardian' ? 40 : 14, e.etype === 'guardian' ? 120 : 60);
  1113→    audio.sfx('enemyDie');
  1114→    // cuenta de lobos para la misión
  1115→    if (e.etype === 'lobo' && this.questIdx === 1 && this.questStep === 0) {

  1119→      if (n >= 3) { this.questAdvance(); this.toast('Vuelve con la Anciana Brisa', '#8ef0b0'); }
  1120→    }
  1121→    if (e.etype === 'guardian') {
  1122→      this.flags.guardianDefeated = true;
  1123→      this.bossActive = false;
  1124→      audio.setCombat(false);
  1125→      audio.playTrack('crypt');

  1128→      this.toast('El altar del Eco brilla al norte', '#ffe9a0');
  1129→      if (this.questIdx === 3 && this.questStep === 1) this.questAdvance();
  1130→    }
  1131→  }

  1133→  gainXp(xp: number) {
  1134→    const p = this.player!;
  1135→    p.xp += xp;
  1136→    let next = this.xpNext(p.level);
  1137→    while (p.xp >= next) {
  1138→      p.xp -= next;
  1139→      p.level++;
  1140→      p.maxHp += 7;
  1141→      p.hp = p.maxHp;
  1142→      p.points += 3;
  1143→      audio.sfx('levelup');
  1144→      this.toast(`¡Nivel ${p.level}! +3 puntos de atributo (menú > Estado)`, '#ffe86a');
  1145→      this.burst(p.x, p.y - 8, '#ffe86a', 20, 70);
  1146→      next = this.xpNext(p.level);
  1147→    }
  1148→  }

  1150→  xpNext(level: number): number {
  1151→    return Math.round(36 * Math.pow(level, 1.45));
  1152→  }

  1154→  damagePlayer(dmg: number, fromX: number, fromY: number) {
  1155→    const p = this.player!;
  1156→    if (p.iframes > 0 || p.rollT > 0 || this.state !== 'play') return;
  1157→    if (p.parryT > 0) {
  1158→      // ¡parada perfecta!
  1159→      audio.sfx('parry');
  1160→      p.res = Math.min(p.maxRes, p.res + 20);
  1161→      p.parryFx = 0.4;
  1162→      this.hitStop = 0.12;
  1163→      this.burst(p.x + (fromX - p.x) * 0.3, p.y - 8 + (fromY - p.y) * 0.3, '#fff8c0', 14, 100);
  1164→      this.floatAt(p.x, p.y - 22, '¡PARADA!', '#fff8c0', 7);
  1165→      // aturde al atacante cercano
  1166→      let best: Enemy | null = null, bd = 34;
  1167→      for (const e of this.enemies) {
  1168→        if (e.dead) continue;
  1169→        const d = Math.hypot(e.x - p.x, e.y - p.y);
  1170→        if (d < bd) { bd = d; best = e; }
  1171→      }
  1172→      if (best) { best.ai = 'aturdido'; best.aiT = 1.6; best.windup = 0; }
  1173→      p.parryT = 0;
  1174→      return;
  1175→    }
  1176→    const red = Math.min(0.5, p.attrs.vig * 0.01);
  1177→    const final = Math.max(1, Math.round(dmg * (1 - red)));
  1178→    p.hp -= final;
  1179→    p.iframes = 0.6;
  1180→    p.lastHitT = 0.3;
  1181→    this.shake = 4;
  1182→    audio.sfx('hurt');
  1183→    this.floatAt(p.x, p.y - 16, `-${final}`, '#ff7060');
  1184→    const dx = p.x - fromX, dy = p.y - fromY;
  1185→    const l = Math.max(1, Math.hypot(dx, dy));
  1186→    this.moveEntity(p, (dx / l) * 8, (dy / l) * 8);
  1187→    if (p.hp <= 0) this.playerDied();
  1188→  }

  1193→export function playerMeleeDmg(p: Player): number {
  1194→  const base = p.discipline === 'alba' ? 12 : 8;
  1195→  const stat = p.discipline === 'alba' ? p.attrs.fue * 1.5 : p.attrs.int * 1.2;
  1196→  const buff = (p.buffT ?? 0) > 0 ? 1.5 : 1;
  1197→  return (base + stat + p.level * 1 + p.weaponPlus * 2.5) * buff;
  1198→}
