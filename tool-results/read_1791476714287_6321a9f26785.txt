    35→    35→export function fitViewToWindow(winW: number, winH: number): void {
    36→    36→  const a = winW / Math.max(1, winH);
    37→    37→  let vw = Math.round(540 * a);
    38→    38→  let vh = 540;
    39→    39→  if (vw < 840) { vw = 840; vh = Math.round(vw / a); }
    40→    40→  else if (vw > 1600) { vw = 1600; vh = Math.round(vw / a); }
    41→    41→  vh = Math.max(460, Math.min(800, vh));
    42→    42→  vw = Math.max(840, Math.min(1600, Math.round(vh * a)));
    43→    43→  if (vw !== VIEW_W || vh !== VIEW_H) { VIEW_W = vw; VIEW_H = vh; }
    44→    44→}

   241→   241→  start() {
   242→   242→    if (this.running) return;
   243→   243→    this.running = true;
   244→   244→    this.lastTs = performance.now();
   245→   245→    const loop = (ts: number) => {
   246→   246→      if (!this.running) return;
   247→   247→      try {
   248→   248→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   249→   249→        this.lastTs = ts;
   250→   250→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   251→   251→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   252→   252→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   253→   253→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   254→   254→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   255→   255→        this.globalT += dt;
   256→   256→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   257→   257→        drawGame(this);
   258→   258→        this.loopError = null;
   259→   259→      } catch (err) {
   260→   260→        this.loopError = err instanceof Error ? (err.stack ?? String(err)) : String(err);
   261→   261→        console.error('[EcosAelthar loop]', err);
   262→   262→        try {
   263→   263→          this.ctx.fillStyle = '#000';
   264→   264→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   265→   265→          this.ctx.font = '12px monospace';
   266→   266→          this.ctx.fillStyle = '#ff7060';
   267→   267→          this.ctx.textAlign = 'left';
   268→   268→          this.ctx.textBaseline = 'top';
   269→   269→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   270→   270→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   271→   271→        } catch { /* noop */ }
   272→   272→      }
   273→   273→      this.raf = requestAnimationFrame(loop);
   274→   274→    };
   275→   275→    this.raf = requestAnimationFrame(loop);
   276→   276→  }

   561→   561→  /** ¿(tx,ty) está dentro de alguna zona de salida del mapa actual? */
   562→   562→  private inExitZone(tx: number, ty: number): boolean {
   563→   563→    return this.map.exits.some(ex => {
   564→   564→      if (ex.needPast && this.epoch !== 'pasado') return false;
   565→   565→      return tx >= ex.x && tx < ex.x + ex.w && ty >= ex.y && ty < ex.y + ex.h;
   566→   566→    });

   569→   569→  /**
   570→   570→   * Busca el tile más cercano (búsqueda por anillos) que sea transitable y
   571→   571→   * NO esté dentro de una zona de salida. Garantiza que ningún viaje aterrice
   572→   572→   * en un muro ni en un teletransporte (bucle cripta↔bosque, P0-1).
   573→   573→   */
   574→   574→  private findSafeTile(tx: number, ty: number): [number, number] {
   575→   575→    const ok = (x: number, y: number) =>
   576→   576→      x >= 0 && y >= 0 && x < this.map.w && y < this.map.h &&
   577→   577→      !this.tileSolidAt(x * TILE + 8, y * TILE + 8) &&
   578→   578→      !this.inExitZone(x, y);
   579→   579→    if (ok(tx, ty)) return [tx, ty];
   580→   580→    for (let r = 1; r <= 8; r++) {
   581→   581→      for (let dy = -r; dy <= r; dy++) {
   582→   582→        for (let dx = -r; dx <= r; dx++) {
   583→   583→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   584→   584→          if (ok(tx + dx, ty + dy)) return [tx + dx, ty + dy];
   585→   585→        }
   586→   586→      }
   587→   587→    }
   588→   588→    return [tx, ty]; // sin sitio libre: comportamiento anterior
   589→   589→  }
