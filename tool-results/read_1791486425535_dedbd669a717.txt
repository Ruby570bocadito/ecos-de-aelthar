   685→export function getDialogue(nid: string, ctx: DialogueCtx): string {
   686→  const q = ctx.questIdx, s = ctx.questStep;
   687→  const td = toneOf(ctx.flags);
   688→  if (nid === 'brisa') {
   689→    // variantes por tono dominante (biblia: los PNJ tratan distinto al Portador)
   690→    const idle = td === 'empatico' ? 'brisa_idle_emp' : td === 'amenazante' ? 'brisa_idle_amenaz' : 'brisa_idle';
   691→    if (ctx.flags.demoEnded) return idle;
   692→    if (q === 0) return 'brisa_intro';
   693→    if (q === 1) return s === 0 ? 'brisa_wolves' : 'brisa_reward';
   694→    if (q === 2) return ctx.flags.fragmentTouched ? 'brisa_fragment' : 'brisa_bosque';
   695→    if (q === 3) return ctx.flags.guardianDefeated ? 'brisa_final' : 'brisa_crypt';
   696→    if (q === 4) return 'brisa_final'; // q5: el regreso con Brisa abre la oferta del sur (respaldo)
   697→    if (q >= 9) return ctx.flags.acto2Done ? idle : 'brisa_final2'; // final del Acto II
   698→    if (q === 5 && s === 0) return 'brisa_final'; // q6 sin aceptar aún: la oferta del sur (engine.talkTo ya avanzó q5→q6)
   699→    if (q >= 5) return idle; // Acto II en curso (q6..q9): Brisa acompaña desde Lunaris
   700→    return 'brisa_final';
   701→  }
   702→  if (nid === 'toln') return td === 'sarcastico' ? 'toln_intro_listillo' : 'toln_intro';
   703→  if (nid === 'ilwen') {
   704→    if (ctx.companion) return td === 'pragmatico' ? 'ilwen_chat_prag' : 'ilwen_chat';
   705→    return 'ilwen_intro';
   706→  }
   707→  if (nid === 'doran') return 'doran_intro';
   708→  if (nid === 'heraldo') return 'heraldo_intro';
   709→  if (nid === 'teo') return 'teo_intro';
   710→  // ----- Acto II · NPCs de la expansión (biblia: cada bioma guarda su duelo) -----
   711→  if (nid === 'mara') {
   712→    // La farera reacciona al destino de la Sirena (flag del motor: killEnemy)
   713→    if (ctx.flags.sirenaDefeated) return ctx.flags.maraGift ? 'mara_faro' : 'mara_react';
   714→    if (q === 5) return 'mara_intro'; // q6: la presentación de la Costa
   715→    return td === 'empatico' ? 'mara_idle_emp' : td === 'sarcastico' ? 'mara_idle_sarc' : 'mara_idle';
   716→  }
   717→  if (nid === 'vult') {
   718→    return ctx.flags.metVult ? (td === 'pragmatico' ? 'vult_idle_prag' : 'vult_idle') : 'vult_intro';
   719→  }
   720→  if (nid === 'mera') {
   721→    // Estados por faroles encendidos (flags lamp1..3 las escribe engine.lightLamp)
   722→    if (ctx.flags.ecoNombres) return 'mera_idle';
   723→    const lamps = (['lamp1', 'lamp2', 'lamp3'] as const).filter(l => !!ctx.flags[l]).length;
   724→    if (lamps >= 3) return 'mera_grateful';
   725→    if (ctx.flags.metMera) return lamps === 2 ? 'mera_wait2' : lamps === 1 ? 'mera_wait1' : 'mera_wait';
   726→    return 'mera_intro';
   727→  }
   728→  if (nid === 'ivo') {
   729→    if (ctx.flags.golemDefeated) return 'ivo_after';
   730→    if (ctx.flags.metIvo) return td === 'empatico' ? 'ivo_idle_emp' : 'ivo_idle';
   731→    return 'ivo_intro';
   732→  }
   733→  return 'brisa_idle';
   734→}

   736→/** Lee el tono dominante de las flags (hooks lo guarda como string en runtime). */
   737→function toneOf(flags: Record<string, number | boolean>): ToneKind | null {
   738→  const v = flags.tonoDominante;
   739→  return typeof v === 'string' ? (v as ToneKind) : null;
   740→}
