   383→export function getDialogue(nid: string, ctx: DialogueCtx): string {
   384→  const q = ctx.questIdx, s = ctx.questStep;
   385→  const td = toneOf(ctx.flags);
   386→  if (nid === 'brisa') {
   387→    // variantes por tono dominante (biblia: los PNJ tratan distinto al Portador)
   388→    const idle = td === 'empatico' ? 'brisa_idle_emp' : td === 'amenazante' ? 'brisa_idle_amenaz' : 'brisa_idle';
   389→    if (ctx.flags.demoEnded) return idle;
   390→    if (q === 0) return 'brisa_intro';
   391→    if (q === 1) return s === 0 ? 'brisa_wolves' : 'brisa_reward';
   392→    if (q === 2) return ctx.flags.fragmentTouched ? 'brisa_fragment' : 'brisa_bosque';
   393→    if (q === 3) return ctx.flags.guardianDefeated ? 'brisa_final' : 'brisa_crypt';
   394→    return 'brisa_final';
   395→  }
   396→  if (nid === 'toln') return td === 'sarcastico' ? 'toln_intro_listillo' : 'toln_intro';
   397→  if (nid === 'ilwen') {
   398→    if (ctx.companion) return td === 'pragmatico' ? 'ilwen_chat_prag' : 'ilwen_chat';
   399→    return 'ilwen_intro';
   400→  }
   401→  if (nid === 'doran') return 'doran_intro';
   402→  if (nid === 'heraldo') return 'heraldo_intro';
   403→  if (nid === 'teo') return 'teo_intro';
   404→  return 'brisa_idle';
   405→}

   407→/** Lee el tono dominante de las flags (hooks lo guarda como string en runtime). */
   408→function toneOf(flags: Record<string, number | boolean>): ToneKind | null {
   409→  const v = flags.tonoDominante;
   410→  return typeof v === 'string' ? (v as ToneKind) : null;
   411→}
