    91→function drawNote(ctx: CanvasRenderingContext2D, x: number, y: number, s: number, color: string, variant: number) {
    92→  ctx.fillStyle = color;
    93→  ctx.fillRect(x, y + s * 3, s * 2, s * 1.5);          // cabeza
    94→  ctx.fillRect(x + s * 1.5, y, s * 0.75, s * 3.6);     // palo
    95→  if (variant === 0) {
    96→    ctx.fillRect(x + s * 2, y, s * 1.25, s * 1.1);     // banderín
    97→  } else {
    98→    ctx.fillRect(x + s * 2.6, y + s * 2.4, s * 2, s * 1.5); // segunda cabeza
    99→    ctx.fillRect(x + s * 1.5, y, s * 3, s * 0.8);      // barra
   100→  }
   101→}

   210→// helpers deterministas locales (evitan importar hash2 aquí)
   211→function hashT(i: number, k: number): number {
   212→  let h = (i + k * 57) * 2654435761;
   213→  h = (h ^ (h >> 13)) * 1274126177;
   214→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;
   215→}
   216→function getSprWisp(t: number, i: number): HTMLCanvasElement {
   217→  const frames = getSpr('wisp');
   218→  const f = frames?.length ? frames[Math.floor(t * 3 + i) % frames.length] : (undefined as unknown as HTMLCanvasElement);
   219→  if (!f) {
   220→    const w = window as unknown as { __wispMiss?: unknown[] };
   221→    if (!w.__wispMiss) w.__wispMiss = [];
   222→    if (w.__wispMiss.length < 3) {
   223→      w.__wispMiss.push({
   224→        t: Math.round(performance.now()), i,
   225→        hasFrames: !!frames, len: frames?.length ?? -1,
   226→        hasG: !!(window as unknown as { __g?: unknown }).__g,
   227→        heroAlba: !!getSpr('hero_alba'),
   228→      });
   229→    }
   230→  }
   231→  return f;
   232→}

   234→// esquinas doradas al pasar el ratón (hover más vivo)
   235→function hoverCorners(g: Game, x: number, y: number, w: number, h: number, col = COL.gold) {
   236→  const hov = g.mouse.x >= x && g.mouse.x <= x + w && g.mouse.y >= y && g.mouse.y <= y + h;
   237→  if (!hov) return;
   238→  const ctx = g.ctx;
   239→  const c = 7;
   240→  ctx.fillStyle = col;
   241→  ctx.fillRect(x - 2, y - 2, c, 2); ctx.fillRect(x - 2, y - 2, 2, c);
   242→  ctx.fillRect(x + w + 2 - c, y - 2, c, 2); ctx.fillRect(x + w, y - 2, 2, c);
   243→  ctx.fillRect(x - 2, y + h, c, 2); ctx.fillRect(x - 2, y + h - c, 2, c);
   244→  ctx.fillRect(x + w + 2 - c, y + h, c, 2); ctx.fillRect(x + w, y + h - c, 2, c);
   245→}

   533→function drawDialogue(g: Game) {
   534→  const ctx = g.ctx;
   535→  const node = g.dlgNode;
   536→  if (!node) return;
   537→  const bw = VIEW_W - 80, bx = 40;
   538→  // la caja crece según texto y opciones para que nunca se solapen
   539→  const fullLines = wrapText(node.text, 76).length;
   540→  const nOpts = node.options?.length ?? 0;
   541→  const bh = Math.max(150, 46 + fullLines * 19 + (nOpts > 0 ? nOpts * 26 + 14 : 0));
   542→  const by = VIEW_H - bh - 24;
   543→  ctx.fillStyle = 'rgba(4,5,12,0.4)';
   544→  ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   545→  panel(g, bx, by, bw, bh, '#6a5a38', 'rgba(10,12,22,0.96)');
   546→  // borde doble pulido + remaches
   547→  ctx.strokeStyle = 'rgba(240,200,74,0.4)';
   548→  ctx.lineWidth = 1;
   549→  ctx.strokeRect(bx + 4.5, by + 4.5, bw - 9, bh - 9);
   550→  ctx.fillStyle = COL.gold;
   551→  for (const [cxx, cyy] of [[bx + 7, by + 7], [bx + bw - 7, by + 7], [bx + 7, by + bh - 7], [bx + bw - 7, by + bh - 7]]) {
   552→    ctx.fillRect(cxx - 1.5, cyy - 1.5, 3, 3);
   553→  }
   554→
   555→  // retrato pixel (respira 1px y parpadea de vez en cuando)
   556→  const pkey = mapPortrait(node.portrait);
   557→  const blink = (g.globalT % 3.7) < 0.14;
   558→  const bob = Math.round(Math.sin(g.globalT * 2.3) * 0.9);
   559→  const frS = 74;
   560→  const frX = bx + 14, frY = by + 14;
   561→  ctx.fillStyle = '#0e0c18';
   562→  ctx.fillRect(frX, frY, frS, frS);
   563→  drawPortrait(ctx, pkey, frX + (frS - 28 * 1.72) / 2, frY + (frS - 40 * 1.72) / 2 + bob, 1.72, blink);
   564→  ctx.strokeStyle = '#6a5a38';
   565→  ctx.strokeRect(frX, frY, frS, frS);
   566→  ctx.strokeStyle = 'rgba(240,200,74,0.35)';
   567→  ctx.strokeRect(frX + 2.5, frY + 2.5, frS - 5, frS - 5);
   568→
   569→  // nombre y texto
   570→  text(g, node.name, bx + 104, by + 12, 17, COL.goldSoft);
   571→  const shown = node.text.slice(0, Math.floor(g.dlgCharT));
   572→  const lines = wrapText(shown, 76);
   573→  lines.forEach((l, i) => text(g, l, bx + 104, by + 36 + i * 19, 17, COL.text));
   574→
   575→  // opciones (debajo de la última línea de texto, sin solaparse nunca)
   576→  if (node.options && node.options.length > 0 && g.dlgCharT >= node.text.length) {
   577→    const oy = by + 36 + fullLines * 19 + 12;
   578→    for (let i = 0; i < node.options.length; i++) {
   579→      const opt = node.options[i];
   580→      const selected = g.dlgSel === i;
   581→      const ox = bx + 104;
   582→      const hover = addHit(g, ox, oy + i * 26, 400, 24, () => {
   583→        g.dlgSel = i;
   584→        g.advanceDialogue();
   585→      });
   586→      if (selected || hover) {
   587→        ctx.fillStyle = 'rgba(240,200,74,0.14)';
   588→        ctx.fillRect(ox - 4, oy + i * 26, 404, 24);
   589→      }
   590→      text(g, `${selected ? '▸' : ' '} ${opt.text}`, ox, oy + i * 26 + 3, 16, selected || hover ? COL.gold : COL.text);
   591→    }
   592→  } else if (g.dlgCharT >= node.text.length) {
   593→    if (Math.sin(g.globalT * 5) > -0.2) text(g, 'E ▸', bx + bw - 44, by + bh - 30, 16, COL.dim, 'center');
   594→  }
   595→}
