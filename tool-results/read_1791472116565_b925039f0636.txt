    60→    60→    ctx.fillRect(x + s * 2, y, s * 1.25, s * 1.1);     // banderín
    61→    61→  } else {
    62→    62→    ctx.fillRect(x + s * 2.6, y + s * 2.4, s * 2, s * 1.5); // segunda cabeza
    63→    63→    ctx.fillRect(x + s * 1.5, y, s * 3, s * 0.8);      // barra
    64→    64→  }

   167→   167→// helpers deterministas locales (evitan importar hash2 aquí)
   168→   168→function hashT(i: number, k: number): number {
   169→   169→  let h = (i + k * 57) * 2654435761;
   170→   170→  h = (h ^ (h >> 13)) * 1274126177;
   171→   171→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;
   172→   172→}

   179→   179→function hoverCorners(g: Game, x: number, y: number, w: number, h: number, col = COL.gold) {
   180→   180→  const hov = g.mouse.x >= x && g.mouse.x <= x + w && g.mouse.y >= y && g.mouse.y <= y + h;
   181→   181→  if (!hov) return;
   182→   182→  const ctx = g.ctx;
   183→   183→  const c = 7;
   184→   184→  ctx.fillStyle = col;
   185→   185→  ctx.fillRect(x - 2, y - 2, c, 2); ctx.fillRect(x - 2, y - 2, 2, c);
   186→   186→  ctx.fillRect(x + w + 2 - c, y - 2, c, 2); ctx.fillRect(x + w, y - 2, 2, c);
   187→   187→  ctx.fillRect(x - 2, y + h, c, 2); ctx.fillRect(x - 2, y + h - c, 2, c);
   188→   188→  ctx.fillRect(x + w + 2 - c, y + h, c, 2); ctx.fillRect(x + w, y + h - c, 2, c);
   189→   189→}
