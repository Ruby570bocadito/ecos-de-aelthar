    78→// nota musical pixel (sin depender de glifos de fuente)
    79→function drawNote(ctx: CanvasRenderingContext2D, x: number, y: number, s: number, color: string, variant: number) {
    80→  ctx.fillStyle = color;
    81→  ctx.fillRect(x, y + s * 3, s * 2, s * 1.5);          // cabeza
    82→  ctx.fillRect(x + s * 1.5, y, s * 0.75, s * 3.6);     // palo
    83→  if (variant === 0) {
    84→    ctx.fillRect(x + s * 2, y, s * 1.25, s * 1.1);     // banderín
    85→  } else {
    86→    ctx.fillRect(x + s * 2.6, y + s * 2.4, s * 2, s * 1.5); // segunda cabeza
    87→    ctx.fillRect(x + s * 1.5, y, s * 3, s * 0.8);      // barra
    88→  }
    89→}

   194→// helpers deterministas locales (evitan importar hash2 aquí)
   195→function hashT(i: number, k: number): number {
   196→  let h = (i + k * 57) * 2654435761;
   197→  h = (h ^ (h >> 13)) * 1274126177;
   198→  return ((h ^ (h >> 16)) >>> 0) / 4294967295;
   199→}
   200→function getSprWisp(t: number, i: number): HTMLCanvasElement {
   201→  const frames = getSpr('wisp');
   202→  const f = frames?.length ? frames[Math.floor(t * 3 + i) % frames.length] : (undefined as unknown as HTMLCanvasElement);
   203→  if (!f) {
   204→    const w = window as unknown as { __wispMiss?: unknown[] };
   205→    if (!w.__wispMiss) w.__wispMiss = [];
   206→    if (w.__wispMiss.length < 3) {
   207→      w.__wispMiss.push({
   208→        t: Math.round(performance.now()), i,
   209→        hasFrames: !!frames, len: frames?.length ?? -1,
   210→        hasG: !!(window as unknown as { __g?: unknown }).__g,
   211→        heroAlba: !!getSpr('hero_alba'),
   212→      });
   213→    }
   214→  }
   215→  return f;
   216→}

   218→// esquinas doradas al pasar el ratón (hover más vivo)
   219→function hoverCorners(g: Game, x: number, y: number, w: number, h: number, col = COL.gold) {
   220→  const hov = g.mouse.x >= x && g.mouse.x <= x + w && g.mouse.y >= y && g.mouse.y <= y + h;
   221→  if (!hov) return;
   222→  const ctx = g.ctx;
   223→  const c = 7;
   224→  ctx.fillStyle = col;
   225→  ctx.fillRect(x - 2, y - 2, c, 2); ctx.fillRect(x - 2, y - 2, 2, c);
   226→  ctx.fillRect(x + w + 2 - c, y - 2, c, 2); ctx.fillRect(x + w, y - 2, 2, c);
   227→  ctx.fillRect(x - 2, y + h, c, 2); ctx.fillRect(x - 2, y + h - c, 2, c);
   228→  ctx.fillRect(x + w + 2 - c, y + h, c, 2); ctx.fillRect(x + w, y + h - c, 2, c);
   229→}

   513→function drawDialogue(g: Game) {
   514→  const ctx = g.ctx;
   515→  const node = g.dlgNode;
   516→  if (!node) return;
   517→  const bw = VIEW_W - 80, bx = 40;
   518→  // la caja crece según texto y opciones para que nunca se solapen
   519→  const fullLines = wrapText(node.text, 76).length;
   520→  const nOpts = node.options?.length ?? 0;
   521→  const bh = Math.max(150, 46 + fullLines * 19 + (nOpts > 0 ? nOpts * 26 + 14 : 0));
   522→  const by = VIEW_H - bh - 24;
   523→  ctx.fillStyle = 'rgba(4,5,12,0.4)';
   524→  ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   525→  panel(g, bx, by, bw, bh, '#6a5a38', 'rgba(10,12,22,0.96)');
   526→  // borde doble pulido + remaches
   527→  ctx.strokeStyle = 'rgba(240,200,74,0.4)';
   528→  ctx.lineWidth = 1;
   529→  ctx.strokeRect(bx + 4.5, by + 4.5, bw - 9, bh - 9);
   530→  ctx.fillStyle = COL.gold;
   531→  for (const [cxx, cyy] of [[bx + 7, by + 7], [bx + bw - 7, by + 7], [bx + 7, by + bh - 7], [bx + bw - 7, by + bh - 7]]) {
   532→    ctx.fillRect(cxx - 1.5, cyy - 1.5, 3, 3);
   533→  }
   534→
   535→  // retrato pixel (respira 1px y parpadea de vez en cuando)
   536→  const pkey = mapPortrait(node.portrait);
   537→  const blink = (g.globalT % 3.7) < 0.14;
   538→  const bob = Math.round(Math.sin(g.globalT * 2.3) * 0.9);
   539→  const frS = 74;
   540→  const frX = bx + 14, frY = by + 14;
   541→  ctx.fillStyle = '#0e0c18';
   542→  ctx.fillRect(frX, frY, frS, frS);
   543→  drawPortrait(ctx, pkey, frX + (frS - 28 * 1.72) / 2, frY + (frS - 40 * 1.72) / 2 + bob, 1.72, blink);
   544→  ctx.strokeStyle = '#6a5a38';
   545→  ctx.strokeRect(frX, frY, frS, frS);
   546→  ctx.strokeStyle = 'rgba(240,200,74,0.35)';
   547→  ctx.strokeRect(frX + 2.5, frY + 2.5, frS - 5, frS - 5);
   548→
   549→  // nombre y texto
   550→  text(g, node.name, bx + 104, by + 12, 17, COL.goldSoft);
   551→  const shown = node.text.slice(0, Math.floor(g.dlgCharT));
   552→  const lines = wrapText(shown, 76);
   553→  lines.forEach((l, i) => text(g, l, bx + 104, by + 36 + i * 19, 17, COL.text));
   554→
   555→  // opciones (debajo de la última línea de texto, sin solaparse nunca)
   556→  if (node.options && node.options.length > 0 && g.dlgCharT >= node.text.length) {
   557→    const oy = by + 36 + fullLines * 19 + 12;
   558→    for (let i = 0; i < node.options.length; i++) {
   559→      const opt = node.options[i];
   560→      const selected = g.dlgSel === i;
   561→      const ox = bx + 104;
   562→      const hover = addHit(g, ox, oy + i * 26, 400, 24, () => {
   563→        g.dlgSel = i;
   564→        g.advanceDialogue();
   565→      });
   566→      if (selected || hover) {
   567→        ctx.fillStyle = 'rgba(240,200,74,0.14)';
   568→        ctx.fillRect(ox - 4, oy + i * 26, 404, 24);
   569→      }
   570→      text(g, `${selected ? '▸' : ' '} ${opt.text}`, ox, oy + i * 26 + 3, 16, selected || hover ? COL.gold : COL.text);
   571→    }
   572→  } else if (g.dlgCharT >= node.text.length) {
   573→    if (Math.sin(g.globalT * 5) > -0.2) text(g, 'E ▸', bx + bw - 44, by + bh - 30, 16, COL.dim, 'center');
   574→  }
   575→}

   630→function drawEnd(g: Game) {
   631→  const ctx = g.ctx;
   632→  if (!endArmed) { endArmed = true; endStartT = g.globalT; }
   633→  const elapsed = g.globalT - endStartT;
   634→  const grad = ctx.createLinearGradient(0, 0, 0, VIEW_H);
   635→  grad.addColorStop(0, '#0a0c1e');
   636→  grad.addColorStop(1, '#241a30');

   638→  ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   639→  for (let i = 0; i < 60; i++) {
   640→    const x = (i * 149.3) % VIEW_W;
   641→    const y = (i * 91.7) % VIEW_H;
   642→    ctx.fillStyle = `rgba(240,200,74,${0.2 + Math.abs(Math.sin(g.globalT + i)) * 0.5})`;
   643→    ctx.fillRect(x, y, 2, 2);

   646→  const ecoCount = (g.flags.ecoVoz ? 1 : 0) + (g.flags.ecoMareas ? 1 : 0) + (g.flags.ecoCumbres ? 1 : 0);
