   572→  /** ¿(tx,ty) está dentro de alguna zona de salida del mapa actual? */
   573→  private inExitZone(tx: number, ty: number): boolean {
   574→    return this.map.exits.some(ex => {
   575→      if (ex.needPast && this.epoch !== 'pasado') return false;
   576→      return tx >= ex.x && tx < ex.x + ex.w && ty >= ex.y && ty < ex.y + ex.h;
   577→    });

   580→  /**
   581→   * Busca el tile más cercano (búsqueda por anillos) que sea transitable y
   582→   * NO esté dentro de una zona de salida. Garantiza que ningún viaje aterrice
   583→   * en un muro ni en un teletransporte (bucle cripta↔bosque, P0-1).
   584→   */
   585→  private findSafeTile(tx: number, ty: number): [number, number] {
   586→    const ok = (x: number, y: number) =>
   587→      x >= 0 && y >= 0 && x < this.map.w && y < this.map.h &&
   588→      !this.tileSolidAt(x * TILE + 8, y * TILE + 8) &&
   589→      !this.inExitZone(x, y);
   590→    if (ok(tx, ty)) return [tx, ty];
   591→    for (let r = 1; r <= 8; r++) {
   592→      for (let dy = -r; dy <= r; dy++) {
   593→        for (let dx = -r; dx <= r; dx++) {
   594→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   595→          if (ok(tx + dx, ty + dy)) return [tx + dx, ty + dy];
   596→        }
   597→      }
   598→    }
   599→    return [tx, ty]; // sin sitio libre: comportamiento anterior
   600→  }

   602→  boxFree(x: number, y: number, w: number, h: number): boolean {
   603→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   604→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   605→  }

   609→    if (dy !== 0 && this.boxFree(e.x, e.y + dy, e.w, e.h)) e.y += dy;

   650→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   651→    const p = this.player;
   652→    if (!p) return null;
   653→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   654→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   655→      const d = Math.hypot(x - p.x, y - p.y);
   656→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   657→    };
   658→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   659→    for (const ch of this.map.chests) {
   660→      if (this.openedChests.has(ch.id)) continue;
   661→      if (ch.needPast && this.epoch !== 'pasado') continue;
   662→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   663→    }
   664→    for (const pr of this.map.props) {
   665→      if (pr.needPast && this.epoch !== 'pasado') continue;
   666→      if (pr.needPresent && this.epoch !== 'presente') continue;
   667→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   668→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   669→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   670→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   671→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   672→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   673→      else if (pr.kind === 'lamp' && !this.flags[pr.id]) consider(px, py, 'lamp', 'Encender el Farol del Recuerdo', () => this.lightLamp(pr.id), 28);
   674→    }
   675→    for (const ec of this.map.echoes) {
   676→      if (this.takenEchoes.has(ec.id)) continue;
   677→      consider(ec.x * TILE + 8, ec.y * TILE + 8, 'echo', 'Escuchar eco menor', () => this.takeEchoMinor(ec.id, ec.title, ec.text), 26);
   678→    }
   679→    return best;
   680→  }
