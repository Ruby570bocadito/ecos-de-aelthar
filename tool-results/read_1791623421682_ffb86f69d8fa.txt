    25→function mkCanvas(w: number, h: number): { c: HTMLCanvasElement; x: CanvasRenderingContext2D } {
    26→  const c = document.createElement('canvas');
    27→  c.width = w; c.height = h;
    28→  const x = c.getContext('2d')!;
    29→  x.imageSmoothingEnabled = false;
    30→  return { c, x };
    31→}

    35→export interface HumanPal {
    36→  outline: string;
    37→  hair: string; hairS?: string;
    38→  skin: string;
    39→  body: string; bodyS: string;
    40→  accent: string;
    41→  legs: string; legsS?: string;
    42→  boots: string;

    45→  ribs?: boolean;

    49→  big?: boolean;            // Gran Inquisidor: escala ×2 (32×36)
    50→  ears?: 'cat' | 'elf';     // orejas felinas / élficas
    51→  mask?: boolean;           // máscara lisa sin boca (Gran Inquisidor)
    52→  maskC?: string;           // color de la máscara
    53→  hammer?: boolean;         // martillo en la mano (Brokk)
    54→  hairLong?: boolean;       // melena larga (Maelis, Nimue)
    55→  pauldrons?: boolean;      // hombreras (Kael)
    56→  leafy?: boolean;          // hojas en hombros (Doran)
    57→  messy?: boolean;          // pelo revuelto (Teo)

    60→function px(x: CanvasRenderingContext2D, X: number, Y: number, W: number, H: number, C: string) {
    61→  x.fillStyle = C; x.fillRect(X, Y, W, H);
    62→}
    63→
    64→interface Layout {
    65→  H: number;        // alto del canvas
    66→  headTop: number;  // fila de la masa de pelo
    67→  faceTop: number;  // fila de piel
    68→  bodyTop: number;  // fila del torso
    69→  bodyH: number;    // filas de torso (incluye fila de contorno)
    70→  armLen: number;   // largo del brazo sin mano
    71→}
    72→
    73→function layoutFor(pal: HumanPal): Layout {
    74→  if (pal.small) return { H: 15, headTop: 0, faceTop: 3, bodyTop: 7, bodyH: 4, armLen: 3 };
    75→  return { H: 18, headTop: 1, faceTop: 4, bodyTop: 8, bodyH: 6, armLen: 4 };
    76→}

    78→/**
    79→ * Dibuja un fotograma humanoide. f = fase de andar (0 zancada A,
    80→ * 1 pase —cuerpo elevado 1px—, 2 zancada B). En reposo se usa f=1
    81→ * sin bob para una pose neutra de pie.
    82→ */
    83→function drawHumanFrame(
    84→  x: CanvasRenderingContext2D, pal: HumanPal,
    85→  dir: 'down' | 'up' | 'side', f: number, bob: number, L: Layout,
    86→): void {
    87→  const hs = pal.hairS ?? pal.hair;
    88→  const ls = pal.legsS ?? pal.legs;
    89→  const hT = L.headTop + bob;
    90→  const fT = L.faceTop + bob;
    91→  const bT = L.bodyTop + bob;
    92→  const bodyBot = bT + L.bodyH - 1;           // última fila del torso (contorno)
    93→  const legY = bodyBot + 1;                    // piernas ancladas al suelo
    94→  const bootY = L.H - 1;
    95→  const legH = bootY - legY;                   // filas de pierna antes de la bota
    96→
    97→  // balanceo de brazos (contrario entre izquierda y derecha)
    98→  const offL = [1, 0, -1][f];
    99→  const offR = [-1, 0, 1][f];
   100→
   101→  if (dir === 'down' || dir === 'up') {
   102→    // ---------- cabeza ----------
   103→    px(x, 4, hT, 8, 1, pal.outline);
   104→    px(x, 3, hT + 1, 10, 5, pal.hair);
   105→    px(x, 3, hT + 1, 10, 1, hs);
   106→    if (pal.hood) { px(x, 3, fT + 2, 10, 2, pal.hair); px(x, 4, fT + 3, 8, 1, hs); }
   107→    if (dir === 'down') {
   108→      px(x, 5, fT, 6, 4, pal.mask ? (pal.maskC ?? pal.skin) : pal.skin);
   109→      if (!pal.mask) {
   110→        px(x, 6, fT + 1, 1, 2, pal.eye); px(x, 9, fT + 1, 1, 2, pal.eye);
   111→        if (pal.beard) px(x, 5, fT + 3, 6, 2, pal.beard);
   112→      } else {
   113→        // máscara lisa: solo un brillo horizontal, sin boca ni ojos
   114→        px(x, 6, fT + 1, 4, 1, '#eae6dc');
   115→      }
   116→      px(x, 3, fT + 2, 1, 2, pal.hair); px(x, 12, fT + 2, 1, 2, pal.hair);
   117→      if (pal.ears === 'elf') { px(x, 2, fT + 1, 1, 2, pal.skin); px(x, 13, fT + 1, 1, 2, pal.skin); }
   118→      if (pal.hairLong) { px(x, 3, fT + 4, 2, Math.max(1, bT - fT - 3), pal.hair); px(x, 11, fT + 4, 2, Math.max(1, bT - fT - 3), pal.hair); }
   119→    }
   120→    if (pal.ears === 'cat') {
   121→      // orejas felinas sobre la cabeza (filas fijas para que no "floten")
   122→      px(x, 3, 0, 1, 1, hs); px(x, 12, 0, 1, 1, hs);
   123→      px(x, 3, 1, 2, 1, pal.hair); px(x, 11, 1, 2, 1, pal.hair);
   124→    }
   125→    if (pal.messy) {
   126→      // pelo revuelto (mechones sueltos)
   127→      px(x, 2, hT + 1, 1, 1, pal.hair); px(x, 13, hT + 1, 1, 1, pal.hair);
   128→      px(x, 5, Math.max(0, hT - 1), 2, 1, pal.hair); px(x, 9, Math.max(0, hT - 1), 3, 1, pal.hair);
   129→    }
   130→
   131→    // ---------- torso ----------
   132→    px(x, 4, bT, 8, L.bodyH, pal.body);
   133→    px(x, 4, bT, 8, 1, pal.bodyS);
   134→    px(x, 5, bT + L.bodyH - 2, 6, 1, pal.accent);
   135→    px(x, 4, bodyBot, 8, 1, pal.outline);
   136→    if (pal.ribs) { px(x, 6, bT + 1, 4, 1, pal.bodyS); px(x, 6, bT + 2, 4, 1, pal.bodyS); }
   137→    if (pal.pauldrons) { px(x, 3, bT, 2, 2, pal.accent); px(x, 11, bT, 2, 2, pal.accent); }
   138→    if (pal.leafy) { px(x, 3, bT, 2, 1, '#8ac05a'); px(x, 11, bT, 2, 1, '#8ac05a'); }
   139→
   140→    // ---------- brazos (oscilación) ----------
   141→    px(x, 2, bT + offL, 2, L.armLen, pal.body); px(x, 2, bT + offL + L.armLen, 2, 1, pal.skin);
   142→    px(x, 12, bT + offR, 2, L.armLen, pal.body); px(x, 12, bT + offR + L.armLen, 2, 1, pal.skin);
   143→    if (pal.hammer) {
   144→      // martillo en la mano derecha
   145→      px(x, 13, bT + offR - 2, 1, L.armLen + 1, '#7a5c3a');
   146→      px(x, 12, bT + offR - 4, 3, 2, '#9aa4b4');
   147→    }
   148→
   149→    // ---------- piernas ----------
   150→    if (f === 0) {
   151→      // zancada A: izquierda adelante, derecha atrás
   152→      px(x, 4, legY, 3, legH, pal.legs); px(x, 9, legY, 3, legH, ls);
   153→      px(x, 4, bootY, 3, 1, pal.boots); px(x, 9, bootY, 3, 1, pal.boots);
   154→    } else if (f === 1) {
   155→      // pase: piernas juntas, cuerpo elevado
   156→      px(x, 5, legY, 3, legH, pal.legs); px(x, 8, legY, 3, legH, pal.legs);
   157→      px(x, 5, bootY, 3, 1, pal.boots); px(x, 8, bootY, 3, 1, pal.boots);
   158→    } else {
   159→      // zancada B: derecha adelante, izquierda atrás
   160→      px(x, 4, legY + 1, 3, Math.max(1, legH - 1), ls); px(x, 9, legY, 3, legH, pal.legs);
   161→      px(x, 4, bootY, 3, 1, pal.boots); px(x, 9, bootY, 3, 1, pal.boots);
   162→    }
   163→    return;
   164→  }
   165→
   166→  // ---------- dir === 'side' (mirando a la derecha) ----------
   167→  px(x, 5, hT, 8, 1, pal.outline);
   168→  px(x, 4, hT + 1, 9, 5, pal.hair);
   169→  px(x, 4, hT + 1, 9, 1, hs);
   170→  if (pal.hood) { px(x, 4, fT + 2, 8, 2, pal.hair); }
   171→  px(x, 7, fT, 5, 4, pal.mask ? (pal.maskC ?? pal.skin) : pal.skin);
   172→  if (!pal.mask) {
   173→    px(x, 10, fT + 1, 1, 2, pal.eye);
   174→    if (pal.beard) px(x, 8, fT + 3, 4, 2, pal.beard);
   175→  } else {
   176→    px(x, 8, fT + 1, 3, 1, '#eae6dc');
   177→  }
   178→  px(x, 5, fT + 2, 1, 2, pal.hair);
   179→  if (pal.ears === 'elf') px(x, 6, fT + 1, 1, 2, pal.skin);
   180→  if (pal.ears === 'cat') { px(x, 3, 0, 1, 1, hs); px(x, 4, 1, 2, 1, pal.hair); }
   181→  if (pal.hairLong) px(x, 4, fT + 3, 2, Math.max(1, bT - fT - 2), pal.hair);
   182→  if (pal.messy) px(x, 5, Math.max(0, hT - 1), 3, 1, pal.hair);
   183→
   184→  // torso
   185→  px(x, 5, bT, 6, L.bodyH, pal.body);
   186→  px(x, 5, bT, 6, 1, pal.bodyS);
   187→  px(x, 6, bT + L.bodyH - 2, 4, 1, pal.accent);
   188→  px(x, 5, bodyBot, 6, 1, pal.outline);
   189→  if (pal.ribs) { px(x, 6, bT + 1, 3, 1, pal.bodyS); px(x, 6, bT + 2, 3, 1, pal.bodyS); }
   190→  if (pal.pauldrons) px(x, 4, bT, 2, 2, pal.accent);
   191→  if (pal.leafy) px(x, 4, bT, 2, 1, '#8ac05a');
   192→
   193→  // brazo delantero con balanceo + brazo trasero al tono sombreado
   194→  const offF = [1, 0, -1][f];
   195→  px(x, 8, bT + offF, 3, 3, pal.body);
   196→  px(x, 10, bT + offF + 3, 2, 1, pal.skin);
   197→  px(x, 5, bT - offF, 2, 3, pal.bodyS);
   198→  if (pal.hammer) {
   199→    px(x, 11, bT + offF - 2, 1, 4, '#7a5c3a');
   200→    px(x, 10, bT + offF - 4, 3, 2, '#9aa4b4');
   201→  }
   202→
   203→  // piernas en zancada (3 fases)
   204→  if (f === 0) {
   205→    px(x, 5, legY, 2, legH, ls); px(x, 8, legY, 2, legH, pal.legs);
   206→    px(x, 5, bootY, 2, 1, pal.boots); px(x, 8, bootY, 2, 1, pal.boots);
   207→  } else if (f === 1) {
   208→    px(x, 6, legY, 2, legH, pal.legs); px(x, 8, legY + 1, 2, Math.max(1, legH - 1), ls);
   209→    px(x, 6, bootY, 2, 1, pal.boots); px(x, 8, bootY, 2, 1, pal.boots);
   210→  } else {
   211→    px(x, 4, legY, 2, legH, ls); px(x, 7, legY, 2, legH, pal.legs);
   212→    px(x, 4, bootY, 2, 1, pal.boots); px(x, 7, bootY, 2, 1, pal.boots);
   213→  }
   214→}

   216→function buildHumanoid(pal: HumanPal): Frames {
   217→  const frames: Frames = [];
   218→  const dirs: ('down' | 'up' | 'side')[] = ['down', 'up', 'side'];
   219→  const L = layoutFor(pal);
   220→  for (const dir of dirs) {
   221→    for (let f = 0; f < 3; f++) {
   222→      const bob = f === 1 ? -1 : 0; // el cuerpo se eleva en la fase de pase
   223→      const { c, x } = mkCanvas(16, L.H);
   224→      drawHumanFrame(x, pal, dir, f, bob, L);
   225→      frames.push(c);
   226→    }
   227→  }
   228→  if (pal.big) {
   229→    // El Gran Inquisidor se dibuja a escala ×2 (32×36): imponente como el Guardián
   230→    return frames.map(fr => {
   231→      const { c, x } = mkCanvas(32, L.H * 2);
   232→      x.imageSmoothingEnabled = false;
   233→      x.drawImage(fr, 0, 0, 32, L.H * 2);
   234→      return c;
   235→    });
   236→  }
   237→  return frames;
   238→}

   240→/** 'side' canónico o 'left'/'right' del motor (flip lo decide render). */
   241→function esLateral(dir: string): boolean {
   242→  return dir === 'side' || dir === 'left' || dir === 'right';
   243→}
   244→
   245→/** Índice de fotograma legacy (compatible con re-export de engine.ts). */
   246→export function frameIndex(dir: string, moving: boolean, anim: number): number {
   247→  // FIX R18 (bug «al andar a los lados mira al frente»): e.dir es
   248→  // 'left'/'right' en el motor, nunca 'side' — ambas cuentan como laterales.
   249→  const base = dir === 'down' ? 0 : dir === 'up' ? 2 : esLateral(dir) ? 4 : 0;
   250→  if (!moving) return base;
   251→  return base + (Math.floor(anim * 6) % 2);
   252→}
   253→
   254→/**
   255→ * Selección robusta de fotograma para cualquier sprite:
   256→ * humanoids nuevos tienen 9 frames (3 por dirección), los demás
   257→ * (lobo, guardián, wisp...) alternan su ciclo propio al moverse.
   258→ */
   259→export function entityFrame(spr: Frames, dir: string, moving: boolean, anim: number): number {
   260→  const n = spr.length;
   261→  if (n === 9) {
   262→    // FIX R18 (bug «al andar a los lados mira al frente»): e.dir llega como
   263→    // 'left'/'right' del motor — el flip de render distingue left/right, aquí
   264→    // solo importa que sea fila lateral.
   265→    const base = dir === 'up' ? 3 : esLateral(dir) ? 6 : 0;
   266→    if (!moving) return base + 1; // pose de pie = fase de pase sin bob
   267→    return base + (Math.floor(anim * 8) % 3);
   268→  }
   269→  if (n >= 2 && moving) return Math.floor(anim * 6) % Math.min(2, n);
   270→  return 0;
   271→}

   288→    px(x, 16, 4, 1, 1, O);

   387→function buildFragment(): Frames {
   388→  const frames: Frames = [];
   389→  for (let f = 0; f < 3; f++) {

   673→/** Arma vertical: empuñadura en (cx, gy), cuerpo hacia arriba (up) o abajo. */
   674→function weaponVert(x: CanvasRenderingContext2D, W: WeaponDef, cx: number, gy: number, len: number, up: boolean): void {
   675→  pxc(x, cx, gy, 2, 1, W.grip);                       // empuñadura
   676→  pxc(x, cx - 1, up ? gy - 1 : gy + 1, 4, 1, W.trim); // guarda
   677→  const s0 = up ? gy - 2 : gy + 2;                    // primera fila del arma
   678→  if (W.kind === 'sword') {
   679→    const top = up ? s0 - len + 1 : s0;
   680→    pxc(x, cx, top, 2, len, W.metal);
   681→    pxc(x, cx, top, 1, len, W.glint);                 // filo (columna de brillo)
   682→    pxc(x, cx, up ? top : top + len - 1, 2, 1, '#ffffff'); // punta
   683→  } else if (W.kind === 'staff') {
   684→    const sl = Math.max(1, len - 1);
   685→    const top = up ? s0 - sl + 1 : s0;
   686→    pxc(x, cx, top, 1, sl, W.metal);                  // vara
   687→    const gemY = up ? top - 1 : top + sl;
   688→    pxc(x, cx, gemY, 2, 1, W.trim);                   // gema/punta
   689→    pxc(x, cx, gemY, 1, 1, W.glint);
   690→  } else { // hammer
   691→    const sl = Math.max(1, len - 2);
   692→    const top = up ? s0 - sl + 1 : s0;
   693→    pxc(x, cx, top, 1, sl, W.grip);                   // mango
   694→    const headY = up ? top - 2 : top + sl - 1;
   695→    pxc(x, cx - 1, headY, 4, 2, W.metal);             // cabeza
   696→    pxc(x, cx - 1, up ? headY : headY + 1, 4, 1, W.glint);
   697→  }
   698→}

   701→function weaponHoriz(x: CanvasRenderingContext2D, W: WeaponDef, hx: number, ry: number, len: number): void {
   702→  if (W.kind === 'sword') {
   703→    pxc(x, hx, ry - 1, 1, 4, W.trim);
   704→    pxc(x, hx + 1, ry, len, 2, W.metal);
   705→    pxc(x, hx + 1, ry, len, 1, W.glint);
   706→    pxc(x, hx + len, ry, 1, 2, '#ffffff');
   707→  } else if (W.kind === 'staff') {
   708→    pxc(x, hx + 1, ry, len, 1, W.metal);
   709→    pxc(x, hx + len, ry - 1, 1, 3, W.trim);
   710→    pxc(x, hx + len, ry, 1, 1, W.glint);
   711→  } else { // hammer
   712→    pxc(x, hx + 1, ry, Math.max(1, len - 1), 1, W.grip);
   713→    pxc(x, hx + len - 1, ry - 1, 2, 4, W.metal);
   714→    pxc(x, hx + len - 1, ry - 1, 2, 1, W.glint);
   715→  }

   718→/** Poses con arma en mano (espada/bastón/martillo). dir ya resuelto a
   719→ *  'down'|'up'|'right' (las poses laterales miran a la derecha). */
   720→function weaponPoses(pal: HumanPal, W: WeaponDef, dir3: 'down' | 'up' | 'side', dir: 'down' | 'up' | 'right'): Frames {
   721→  const L = layoutFor(pal);
   722→
   723→  // frame 0 · anticipación: arma atrás sobre el hombro, cuerpo atrás 1px
   724→  const f0 = (() => {
   725→    const P = poseCanvas(pal, dir3, -1);
   726→    if (dir3 === 'side') {
   727→      clearArmFront(P.x, P.bT);
   728→      px(P.x, 9, P.bT - 2, 2, 3, pal.body);  // brazo alzado
   729→      px(P.x, 9, P.bT - 3, 2, 1, pal.skin);
   730→      if (W.kind === 'hammer') {
   731→        // martillo descansando tras la cabeza (la cabeza no toca el borde)
   732→        pxc(P.x, 10, P.bT - 4, 1, 2, W.grip);
   733→        pxc(P.x, 9, P.bT - 6, 4, 2, W.metal);

   968→/** Punto de entrada común: resuelve paleta/arma/dirección y delega.
   969→ *  'left' reutiliza las frames de 'right' (el render ya las voltea). */
   970→function buildCombatPose(baseName: string, dir: string, kind: 'attack' | 'cast'): HTMLCanvasElement[] | null {
   971→  if (dir !== 'down' && dir !== 'up' && dir !== 'left' && dir !== 'right') return null;
   972→  if (kind === 'attack' ? !ATTACK_POSES.has(baseName) : !CAST_POSES.has(baseName)) return null;
   973→  const pal = PALS[baseName];
   974→  const W = WEAPONS[baseName];
   975→  if (!pal || !W) return null;
   976→
   977→  const dir3: 'down' | 'up' | 'side' = dir === 'down' || dir === 'up' ? dir : 'side';
   978→  const sideDir: 'down' | 'up' | 'right' = dir === 'down' ? 'down' : dir === 'up' ? 'up' : 'right';
   979→
   980→  const frames = kind === 'cast'
   981→    ? castPoses(pal, W, dir3)
   982→    : baseName === 'ilwen'
   983→      ? bowPoses(pal, W, dir3)
   984→      : weaponPoses(pal, W, dir3, sideDir);
   985→  return frames.length === 2 ? [frames[0], frames[1]] : null;
   986→}

   988→// ---------------- Arco de ataque (slash) ----------------
   989→
   990→/**
   991→ * Dibuja el arco de tajo del jugador. prog: 0→1 progreso del golpe,
   992→ * combo: 0..2 (blanco → amarillo → dorado), charged: ataque cargado
   993→ * (más grande, dorado y con estela). Todos los trazos son aditivos
   994→ * suaves y restauran el estado del contexto.
   995→ */
   996→export function drawSlashArc(
   997→  ctx: CanvasRenderingContext2D, cx: number, cy: number,
   998→  dir: string, prog: number, combo: number, charged: boolean, zoom: number,
   999→): void {
  1000→  const base = dir === 'down' ? Math.PI / 2 : dir === 'up' ? -Math.PI / 2 : dir === 'left' ? Math.PI : 0;
  1001→  const colors = charged ? '#ffd24a' : ['#f2f6fa', '#ffe86a', '#ffc040'][Math.max(0, Math.min(2, combo))];
  1002→  const r = (charged ? 27 : 19) * zoom;
  1003→  const span = 1.2;
  1004→  const a0 = base - 1.1 + prog * 1.45;
  1005→  ctx.save();
  1006→  // relleno en cuña (ataque cargado: más presencia)
  1007→  if (charged) {
  1008→    ctx.globalAlpha = 0.14;
  1009→    ctx.fillStyle = '#ffd24a';
  1010→    ctx.beginPath();
  1011→    ctx.moveTo(cx, cy);
  1012→    ctx.arc(cx, cy, r, a0, a0 + span);
  1013→    ctx.closePath();
  1014→    ctx.fill();
  1015→  }
  1016→  // estela: dos arcos anteriores con alpha decreciente
  1017→  for (let i = 2; i >= 1; i--) {
  1018→    const p2 = prog - i * 0.17;
  1019→    if (p2 <= 0) continue;
  1020→    ctx.globalAlpha = charged ? 0.34 - i * 0.12 : 0.26 - i * 0.1;
  1021→    ctx.strokeStyle = colors;
  1022→    ctx.lineWidth = (charged ? 5 : 3) - i;
  1023→    ctx.beginPath();
  1024→    ctx.arc(cx, cy, r * (1 - i * 0.05), base - 1.1 + p2 * 1.45, base - 1.1 + p2 * 1.45 + span * 0.8);
  1025→    ctx.stroke();
  1026→  }
  1027→  // arco principal
  1028→  ctx.globalAlpha = 0.95;
  1029→  ctx.strokeStyle = colors;
  1030→  ctx.lineWidth = charged ? 5 : 3;
  1031→  ctx.lineCap = 'round';
  1032→  ctx.beginPath();
  1033→  ctx.arc(cx, cy, r, a0, a0 + span);
  1034→  ctx.stroke();
  1035→  // filo brillante en el borde delantero
  1036→  ctx.globalAlpha = 1;
  1037→  ctx.strokeStyle = '#ffffff';
  1038→  ctx.lineWidth = 1;
  1039→  ctx.beginPath();
  1040→  ctx.arc(cx, cy, r, a0 + span - 0.22, a0 + span);
  1041→  ctx.stroke();
  1042→  ctx.restore();
  1043→}

  1361→/**

  1363→ * `at` (opcional) devuelve el char del tile vecino (dx,dy) — habilita

  1366→export function drawTile(
  1367→  x: CanvasRenderingContext2D, ch: string, tx: number, ty: number,
  1368→  mapId: string, t: number, at?: NeighborFn,
  1369→): void {
  1370→  switch (ch) {
  1371→    case '.': case ',': case 'c': case 'm':
  1372→      paintGrass(x, ch, tx, ty, mapId, t, at); break;
  1373→    case '~': case 'B': case 'x':
  1374→      paintWater(x, ch, tx, ty, mapId, t, at); break;
  1375→    case ':': case '_': case '#': case 'P': case 'A': case 'V':
  1376→      paintStone(x, ch, tx, ty, mapId, t, at); break;
  1377→    case 'H': case 'r': case 'd': case 'F': case 'w': case 'g': case 'R':
  1378→      paintVillage(x, ch, tx, ty, mapId, t, at); break;
  1379→    default:
  1380→      px(x, tx * TILE, ty * TILE, TILE, TILE, '#4f8a46');
  1381→  }
  1382→}
