   671→export function getDialogue(nid: string, ctx: DialogueCtx): string {
   672→  const q = ctx.questIdx, s = ctx.questStep;
   673→  const td = toneOf(ctx.flags);
   674→  if (nid === 'brisa') {
   675→    // variantes por tono dominante (biblia: los PNJ tratan distinto al Portador)
   676→    const idle = td === 'empatico' ? 'brisa_idle_emp' : td === 'amenazante' ? 'brisa_idle_amenaz' : 'brisa_idle';
   677→    if (ctx.flags.demoEnded) return idle;
   678→    if (q === 0) return 'brisa_intro';
   679→    if (q === 1) return s === 0 ? 'brisa_wolves' : 'brisa_reward';
   680→    if (q === 2) return ctx.flags.fragmentTouched ? 'brisa_fragment' : 'brisa_bosque';
   681→    if (q === 3) return ctx.flags.guardianDefeated ? 'brisa_final' : 'brisa_crypt';
   682→    if (q === 4) return 'brisa_final'; // q5: el regreso con Brisa abre la oferta del sur (respaldo)
   683→    if (q >= 9) return ctx.flags.acto2Done ? idle : 'brisa_final2'; // final del Acto II
   684→    if (q === 5 && s === 0) return 'brisa_final'; // q6 sin aceptar aún: la oferta del sur (engine.talkTo ya avanzó q5→q6)
   685→    if (q >= 5) return idle; // Acto II en curso (q6..q9): Brisa acompaña desde Lunaris
   686→    return 'brisa_final';
   687→  }
   688→  if (nid === 'toln') return td === 'sarcastico' ? 'toln_intro_listillo' : 'toln_intro';
   689→  if (nid === 'ilwen') {
   690→    if (ctx.companion) return td === 'pragmatico' ? 'ilwen_chat_prag' : 'ilwen_chat';
   691→    return 'ilwen_intro';
   692→  }
   693→  if (nid === 'doran') return 'doran_intro';
   694→  if (nid === 'heraldo') return 'heraldo_intro';
   695→  if (nid === 'teo') return 'teo_intro';
   696→  // ----- Acto II · NPCs de la expansión (biblia: cada bioma guarda su duelo) -----
   697→  if (nid === 'mara') {
   698→    // La farera reacciona al destino de la Sirena (flag del motor: killEnemy)
   699→    if (ctx.flags.sirenaDefeated) return ctx.flags.maraGift ? 'mara_faro' : 'mara_react';
   700→    if (q === 5) return 'mara_intro'; // q6: la presentación de la Costa
   701→    return td === 'empatico' ? 'mara_idle_emp' : td === 'sarcastico' ? 'mara_idle_sarc' : 'mara_idle';
   702→  }
   703→  if (nid === 'vult') {
   704→    return ctx.flags.metVult ? (td === 'pragmatico' ? 'vult_idle_prag' : 'vult_idle') : 'vult_intro';
   705→  }
   706→  if (nid === 'mera') {
   707→    // Estados por faroles encendidos (flags lamp1..3 las escribe engine.lightLamp)
   708→    if (ctx.flags.ecoNombres) return 'mera_idle';
   709→    const lamps = (['lamp1', 'lamp2', 'lamp3'] as const).filter(l => !!ctx.flags[l]).length;
   710→    if (lamps >= 3) return 'mera_grateful';
   711→    if (ctx.flags.metMera) return lamps === 2 ? 'mera_wait2' : lamps === 1 ? 'mera_wait1' : 'mera_wait';
   712→    return 'mera_intro';
   713→  }
   714→  if (nid === 'ivo') {
   715→    if (ctx.flags.golemDefeated) return 'ivo_after';
   716→    if (ctx.flags.metIvo) return td === 'empatico' ? 'ivo_idle_emp' : 'ivo_idle';
   717→    return 'ivo_intro';
   718→  }
   719→  return 'brisa_idle';
   720→}

   722→/** Lee el tono dominante de las flags (hooks lo guarda como string en runtime). */
   723→function toneOf(flags: Record<string, number | boolean>): ToneKind | null {
   724→  const v = flags.tonoDominante;
   725→  return typeof v === 'string' ? (v as ToneKind) : null;
   726→}
