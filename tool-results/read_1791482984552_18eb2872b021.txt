    30→/**
    31→ * Ajusta el buffer del juego al aspecto de la ventana para eliminar el
    32→ * letterbox (la "barra negra" abajo/arriba). Mantiene 540px de alto base
    33→ * a 16:9 exacto; pantallas más anchas ganan vista lateral (hasta 1600px)
    34→ * y más cuadradas ganan vista vertical (hasta 800px). Clamp de seguridad
    35→ * para aspectos extremos (móvil vertical): ahí el letterbox es aceptable.
    36→ */
    37→export function fitViewToWindow(winW: number, winH: number): void {
    38→  const a = winW / Math.max(1, winH);
    39→  let vw = Math.round(540 * a);
    40→  let vh = 540;
    41→  if (vw < 840) { vw = 840; vh = Math.round(vw / a); }
    42→  else if (vw > 1600) { vw = 1600; vh = Math.round(vw / a); }
    43→  vh = Math.max(460, Math.min(800, vh));
    44→  vw = Math.max(840, Math.min(1600, Math.round(vh * a)));
    45→  if (vw !== VIEW_W || vh !== VIEW_H) { VIEW_W = vw; VIEW_H = vh; }
    46→}

   226→  private catchUpActo2() {
   227→    if (this.questIdx < 5 || this.questIdx > 9) return;
   228→    // q7 (idx 6): 0 avistada · 1 derrotada · 2 Eco de las Mareas tomado
   229→    if (this.questIdx === 6) {
   230→      if (this.questStep === 0 && this.flags.sirenaDefeated) this.questAdvance();
   231→      if (this.questIdx === 6 && this.questStep === 1 && this.flags.sirenaDefeated) this.questAdvance();
   232→      if (this.questIdx === 6 && this.questStep === 2 && this.flags.ecoMareas) this.questAdvance();
   233→    }
   234→    // q8 (idx 7): 1 los 3 faroles encendidos (orden-independiente) · 2 Eco de los Nombres
   235→    if (this.questIdx === 7) {
   236→      if (this.questStep === 1 &&
   237→          (['lamp1', 'lamp2', 'lamp3'] as const).filter(l => this.flags[l]).length >= 3) this.questAdvance();
   238→      if (this.questIdx === 7 && this.questStep === 2 && this.flags.ecoNombres) this.questAdvance();
   239→    }
   240→    // q9 (idx 8): 1 gólem derrotado · 2 Eco de las Cumbres tomado
   241→    if (this.questIdx === 8) {
   242→      if (this.questStep === 1 && this.flags.golemDefeated) this.questAdvance();
   243→      if (this.questIdx === 8 && this.questStep === 2 && this.flags.ecoCumbres) this.questAdvance();
   244→    }
   245→  }

   249→  start() {
   250→    if (this.running) return;
   251→    this.running = true;
   252→    this.lastTs = performance.now();
   253→    const loop = (ts: number) => {
   254→      if (!this.running) return;
   255→      try {
   256→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   257→        this.lastTs = ts;
   258→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   259→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   260→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   261→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   262→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   263→        this.globalT += dt;
   264→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   265→        drawGame(this);
   266→        this.loopError = null;
   267→      } catch (err) {
   268→        this.loopError = err instanceof Error ? (err.stack ?? String(err)) : String(err);
   269→        console.error('[EcosAelthar loop]', err);
   270→        try {
   271→          this.ctx.fillStyle = '#000';
   272→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   273→          this.ctx.font = '12px monospace';
   274→          this.ctx.fillStyle = '#ff7060';
   275→          this.ctx.textAlign = 'left';
   276→          this.ctx.textBaseline = 'top';
   277→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   278→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   279→        } catch { /* noop */ }
   280→      }
   281→      this.raf = requestAnimationFrame(loop);
   282→    };
   283→    this.raf = requestAnimationFrame(loop);
   284→  }
   285→
   286→  stop() {
   287→    this.running = false;
   288→    cancelAnimationFrame(this.raf);
   289→    audio.stopMusic();
   290→  }

   355→      attrs: { ...p.attrs }, points: p.points,
   356→      gold: p.gold, weaponPlus: p.weaponPlus, potions: p.potions, cds: [0, 0, 0, 0],

   417→  loadMap(id: MapId, tx: number, ty: number) {
   418→    // memoria del jefe: si sales de la cripta con el Guardián herido, no se
   419→    // regenera al volver (cierra el exploit de reseteo de combate)
   420→    // memoria del jefe (generalizada Acto II): si sales de un mapa con su JEFE vivo, no se regenera

   572→  /** ¿(tx,ty) está dentro de alguna zona de salida del mapa actual? */
   573→  private inExitZone(tx: number, ty: number): boolean {
   574→    return this.map.exits.some(ex => {
   575→      if (ex.needPast && this.epoch !== 'pasado') return false;
   576→      return tx >= ex.x && tx < ex.x + ex.w && ty >= ex.y && ty < ex.y + ex.h;
   577→    });

   580→  /**
   581→   * Busca el tile más cercano (búsqueda por anillos) que sea transitable y
   582→   * NO esté dentro de una zona de salida. Garantiza que ningún viaje aterrice
   583→   * en un muro ni en un teletransporte (bucle cripta↔bosque, P0-1).
   584→   */
   585→  private findSafeTile(tx: number, ty: number): [number, number] {
   586→    const ok = (x: number, y: number) =>
   587→      x >= 0 && y >= 0 && x < this.map.w && y < this.map.h &&
   588→      !this.tileSolidAt(x * TILE + 8, y * TILE + 8) &&
   589→      !this.inExitZone(x, y);
   590→    if (ok(tx, ty)) return [tx, ty];
   591→    for (let r = 1; r <= 8; r++) {
   592→      for (let dy = -r; dy <= r; dy++) {
   593→        for (let dx = -r; dx <= r; dx++) {
   594→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   595→          if (ok(tx + dx, ty + dy)) return [tx + dx, ty + dy];
   596→        }
   597→      }
   598→    }
   599→    return [tx, ty]; // sin sitio libre: comportamiento anterior
   600→  }

   602→  boxFree(x: number, y: number, w: number, h: number): boolean {
   603→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   604→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   605→  }

   607→  moveEntity(e: { x: number; y: number; w: number; h: number }, dx: number, dy: number) {
   608→    if (dx !== 0 && this.boxFree(e.x + dx, e.y, e.w, e.h)) e.x += dx;
   609→    if (dy !== 0 && this.boxFree(e.x, e.y + dy, e.w, e.h)) e.y += dy;
   610→  }

   650→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   651→    const p = this.player;
   652→    if (!p) return null;
   653→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   654→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   655→      const d = Math.hypot(x - p.x, y - p.y);
   656→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   657→    };
   658→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   659→    for (const ch of this.map.chests) {
   660→      if (this.openedChests.has(ch.id)) continue;
   661→      if (ch.needPast && this.epoch !== 'pasado') continue;
   662→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   663→    }
   664→    for (const pr of this.map.props) {
   665→      if (pr.needPast && this.epoch !== 'pasado') continue;
   666→      if (pr.needPresent && this.epoch !== 'presente') continue;
   667→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   668→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   669→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   670→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   671→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   672→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   673→      else if (pr.kind === 'lamp' && !this.flags[pr.id]) consider(px, py, 'lamp', 'Encender el Farol del Recuerdo', () => this.lightLamp(pr.id), 28);
   674→    }
   675→    for (const ec of this.map.echoes) {
   676→      if (this.takenEchoes.has(ec.id)) continue;
   677→      consider(ec.x * TILE + 8, ec.y * TILE + 8, 'echo', 'Escuchar eco menor', () => this.takeEchoMinor(ec.id, ec.title, ec.text), 26);
   678→    }
   679→    return best;
   680→  }

   820→    if (!this.dlgNode) return;

   822→    if (this.dlgNode.action) this.applyAction(this.dlgNode.action);
   823→    if (this.dlgNode.onEnd) this.applyAction(this.dlgNode.onEnd);
   824→    if (this.dlgNode.options && this.dlgNode.options.length > 0) {
   825→      const opt = this.dlgNode.options[this.dlgSel];
   826→      recordDialogueTone(this, opt);

   828→      if (opt.next) { this.openDialogue(opt.next); return; }
   829→      this.closeDialogue();
   830→      return;
   831→    }
   832→    if (this.dlgNode.next) { this.openDialogue(this.dlgNode.next); return; }
   833→    this.closeDialogue();

  1147→  private onKeyDown = (e: KeyboardEvent) => {
  1148→    const k = e.key.toLowerCase();
  1149→    // no interceptar teclas cuando se escribe en un input DOM (nombre del Portador)
  1150→    const t = e.target as HTMLElement | null;
  1151→    const isDomInput = !!t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable);
  1152→    if (!isDomInput && [' ', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(k)) e.preventDefault();
  1153→    if (e.repeat) return;
  1154→    this.keys.add(k);
  1155→    audio.resume();
  1156→
  1157→    if (this.state === 'title') {
  1158→      if (k === 'enter' || k === 'e' || k === ' ') { /* botones por click; enter = nueva partida */ }
  1159→    } else if (this.state === 'intro') {
  1160→      if (k === 'e' || k === ' ' || k === 'enter') {
  1161→        this.introIdx++;
  1162→        audio.sfx('blip');
  1163→        if (this.introIdx >= 3) this.startPlay();
  1164→      }
  1165→    } else if (this.state === 'dialogue') {
  1166→      if (k === 'e' || k === ' ' || k === 'enter') this.advanceDialogue();
  1167→      else if (k === 'arrowup' || k === 'w') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + this.dlgNode.options.length - 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
  1168→      else if (k === 'arrowdown' || k === 's') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
  1169→    } else if (this.state === 'play') {
  1170→      if (k === ' ' && this.player && this.player.rollT <= 0 && this.player.attackT <= 0 && this.player.sta >= 20) {
  1171→        this.rollQueued = true; // 1 pulsación = 1 voltereta (no en cadena)
  1172→      }
  1173→      if (k === 'e') this.tryInteract();
  1174→      else if (k === 'q') this.epochSwitch();
  1175→      else if (k === 'escape' || k === 'm') { this.setState('pause'); audio.sfx('uiOpen'); }
  1176→      else if (k === 'f') this.drinkPotion();
  1177→      else if (k === 't' && e.shiftKey) {
  1178→        // accesibilidad / QoL: esperar hasta el alba
  1179→        this.dayT = 0.22;
  1180→        this.toast('Esperas junto al camino hasta el alba...', '#ffe86a');
  1181→        audio.sfx('save');
  1182→      }
  1183→      else if (['1', '2', '3', '4'].includes(k)) this.useSkill(parseInt(k, 10) - 1);
  1184→      else if (k === 'k') { this.setState('skills'); audio.sfx('uiOpen'); }
  1185→    } else if (this.state === 'pause') {
  1186→      if (k === 'escape' || k === 'm') this.setState('play');
  1187→    } else if (this.state === 'skills') {
  1188→      if (k === 'escape' || k === 'k' || k === 'm') { this.setState('play'); audio.sfx('uiOpen'); }

  1609→  damagePlayer(dmg: number, fromX: number, fromY: number) {
  1610→    const p = this.player!;
  1611→    // balanceador (12-c): embudo único de todo el daño enemigo (neutro en desafío)
  1612→    dmg = Math.max(1, Math.round(dmg * enemyStatMult(this).dmg));

  1639→    audio.sfx('hurt');
