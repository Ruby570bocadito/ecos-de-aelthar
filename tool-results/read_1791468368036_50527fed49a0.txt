   165→  start() {
   166→    if (this.running) return;
   167→    this.running = true;
   168→    this.lastTs = performance.now();
   169→    const loop = (ts: number) => {
   170→      if (!this.running) return;
   171→      try {
   172→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   173→        this.lastTs = ts;
   174→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   175→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   176→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   177→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   178→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   179→        this.globalT += dt;
   180→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   181→        drawGame(this);
   182→        this.loopError = null;
   183→      } catch (err) {
   184→        this.loopError = err instanceof Error ? (err.stack ?? String(err)) : String(err);
   185→        console.error('[EcosAelthar loop]', err);
   186→        try {
   187→          this.ctx.fillStyle = '#000';
   188→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   189→          this.ctx.font = '12px monospace';
   190→          this.ctx.fillStyle = '#ff7060';
   191→          this.ctx.textAlign = 'left';
   192→          this.ctx.textBaseline = 'top';
   193→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   194→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   195→        } catch { /* noop */ }
   196→      }
   197→      this.raf = requestAnimationFrame(loop);
   198→    };
   199→    this.raf = requestAnimationFrame(loop);
   200→  }
   201→
   202→  stop() {
   203→    this.running = false;
   204→    cancelAnimationFrame(this.raf);
   205→    audio.stopMusic();
   206→  }

   250→  continueGame() {
   251→    const raw = (() => { try { return localStorage.getItem('ecos-aelthar-save'); } catch { return null; } })();
   252→    if (!raw) return;
   253→    let d: SaveData;

   257→    } catch {
   258→      // save corrupto: nunca dejar el juego muerto — se descarta y se puede empezar de nuevo
   259→      try { localStorage.removeItem('ecos-aelthar-save'); } catch { /* noop */ }
   260→      this.toast('La partida guardada estaba dañada y no pudo recuperarse.', '#ff8060');
   261→      return;
   262→    }
   263→    const p = d.player;
   264→    const maxHp = p.maxHp;
   265→    this.player = {
   266→      kind: 'player', name: p.name, discipline: p.discipline,
   267→      x: d.x, y: d.y, w: 10, h: 8, vx: 0, vy: 0, dir: 'down',
   268→      hp: Math.max(1, p.hp), maxHp, sprite: p.discipline === 'alba' ? 'hero_alba' : 'hero_tejedor',
   269→      anim: 0, moving: false,
   270→      level: p.level, xp: p.xp, sta: p.maxSta, maxSta: p.maxSta, res: p.res, maxRes: 100,
   271→      attrs: { ...p.attrs }, points: p.points,
   272→      gold: p.gold, weaponPlus: p.weaponPlus, potions: p.potions, cds: [0, 0, 0, 0],
   273→      iframes: 0, parryT: 0, parryFx: 0, attackT: 0, combo: 0,

   281→    this.flags = { ...d.flags };

   293→    // (ni sólido ni dentro de una zona de salida); si no, conserva el tile

   333→  loadMap(id: MapId, tx: number, ty: number) {
   334→    // memoria del jefe: si sales de la cripta con el Guardián herido, no se
   335→    // regenera al volver (cierra el exploit de reseteo de combate)
   336→    // memoria del jefe (generalizada Acto II): si sales de un mapa con su JEFE vivo, no se regenera
   337→    if (this.bossRef && !this.bossRef.dead) {
   338→      this.flags[`bossHp_${this.mapId}`] = this.bossRef.hp;
   339→      if (this.mapId === 'cripta') this.flags.bossHp = this.bossRef.hp; // compat con saves antiguos
   340→    }
   341→    this.mapId = id;
   342→    this.map = MAPS[id];
   343→    this.rows = mapRows(this.map);
   344→    this.buildGround();
   345→    if (this.player) {
   346→      // defensa anti-softlock: nunca aterrizar sobre un tile sólido ni dentro
   347→      // de una zona de salida del destino (provocaba el bucle cripta↔bosque)
   348→      const [sx, sy] = this.findSafeTile(tx, ty);
   349→      this.player.x = sx * TILE + 8;
   350→      this.player.y = sy * TILE + 8;
   351→    }
   352→    this.exitCd = 0.9;
   353→    this.spawnEnemies();
   354→    // restaura la vida memorizada del Guardián si la partida sigue en curso
   355→    const savedBossHp = typeof this.flags[`bossHp_${id}`] === 'number' ? (this.flags[`bossHp_${id}`] as number)
   356→      : id === 'cripta' && typeof this.flags.bossHp === 'number' ? (this.flags.bossHp as number) : null;
   357→    if (savedBossHp !== null) {
   358→      const boss = this.enemies.find(e => BOSS_DEFEAT_FLAG[e.etype] !== undefined);
   359→      if (boss) boss.hp = Math.max(1, Math.min(boss.maxHp, savedBossHp));
   360→    }
   361→    this.spawnNpcs();
   362→    this.projectiles = []; this.waves = []; this.telegraphs = [];
   363→    this.bossRef = null; this.bossActive = false;
   364→    audio.setCombat(false);
   365→    this.visitedMaps[id] = true;
   366→    if (id === 'bosque' && !this.flags.visitedBosque) {
   367→      this.flags.visitedBosque = true;
   368→      if (this.questIdx === 2 && this.questStep === 0) this.questAdvance();
   369→    }
   370→    if (id === 'cripta' && !this.flags.visitedCripta) {
   371→      this.flags.visitedCripta = true;
   372→      if (this.questIdx === 3 && this.questStep === 0) this.questAdvance();
   373→    }
   374→    // Acto II: primera visita a los mapas nuevos avanza el objetivo de viaje
   375→    if ((id === 'costa' || id === 'aldea' || id === 'cumbres') && !this.flags[`visited_${id}`]) {
   376→      this.flags[`visited_${id}`] = true;
   377→      if (id === 'costa' && this.questIdx === 5 && this.questStep === 0) this.questAdvance();
   378→      if (id === 'aldea' && this.questIdx === 7 && this.questStep === 0) this.questAdvance();
   379→      if (id === 'cumbres' && this.questIdx === 8 && this.questStep === 0) this.questAdvance();
   380→    }
   381→    this.mapTitleT = 2.6;
   382→    this.updateCamera(true);
   383→  }

   385→  buildGround() {
   386→    const { w, h } = this.map;
   387→    const mk = (ep: Epoch) => {
   388→      const c = document.createElement('canvas');
   389→      c.width = w * TILE; c.height = h * TILE;
   390→      const x = c.getContext('2d')!;
   391→      x.imageSmoothingEnabled = false;
   392→      // suelo
   393→      for (let ty = 0; ty < h; ty++) {
   394→        for (let tx = 0; tx < w; tx++) {
   395→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   396→          // Acto II: arena/nieve/hielo y suelos por bioma los dibuja la expansión; fallback genérico
   397→          if (!drawExpansionTile(x, ch, tx, ty, this.mapId)) drawTile(x, ch, tx, ty, this.mapId, 0);
   398→        }
   399→      }
   400→      // objetos altos por orden de fila
   401→      for (let ty = 0; ty < h; ty++) {
   402→        for (let tx = 0; tx < w; tx++) {
   403→          const ch = tileAt(this.map, this.rows, tx, ty, ep);
   404→          if (SOLID_CHARS.has(ch) && (ch === 't' || ch === 'p')) {
   405→            if (!drawExpansionTallTile(x, ch, tx, ty, this.mapId)) drawTallTile(x, ch, tx, ty, this.mapId);
   406→          }
   407→        }
   408→      }
   409→      return c;
   410→    };
   411→    this.groundCanvas = mk('presente');
   412→    this.groundPastCanvas = this.map.epochDiffs.length ? mk('pasado') : this.groundCanvas;
   413→    // minimapa
   414→    const mini = document.createElement('canvas');
   415→    mini.width = w; mini.height = h;
   416→    const mx = mini.getContext('2d')!;
   417→    for (let ty = 0; ty < h; ty++) {
   418→      for (let tx = 0; tx < w; tx++) {
   419→        const ch = tileAt(this.map, this.rows, tx, ty, 'presente');
   420→        mx.fillStyle =
   421→          ch === 's' ? '#d8c07a' : ch === 'S' ? '#dfe8f2' : ch === 'i' ? '#a8cfe8' :
   422→          ch === '~' ? '#3a6a9a' : ch === '=' ? '#b89a6a' :
   423→          ch === ':' ? '#6a6a7a' : ch === 't' || ch === 'p' ? '#24512a' :
   424→          ch === '#' || ch === 'H' || ch === 'r' ? '#7a7a8a' :
   425→          ch === 'n' ? '#7ea49a' : '#4a8a44';
   426→        mx.fillRect(tx, ty, 1, 1);
   427→      }
   428→    }
   429→    this.miniCanvas = mini;
   430→  }

   432→  spawnEnemies() {
   433→    this.enemies = [];
   434→    for (const s of this.map.spawns) {
   435→      if (s.needPast && this.epoch !== 'pasado') continue;
   436→      if (s.needPresent && this.epoch !== 'presente') continue;
   437→      const defFlag = BOSS_DEFEAT_FLAG[s.type];

   485→  /** ¿(tx,ty) está dentro de alguna zona de salida del mapa actual? */
   486→  private inExitZone(tx: number, ty: number): boolean {
   487→    return this.map.exits.some(ex => {
   488→      if (ex.needPast && this.epoch !== 'pasado') return false;
   489→      return tx >= ex.x && tx < ex.x + ex.w && ty >= ex.y && ty < ex.y + ex.h;
   490→    });
   491→  }

   493→  /**
   494→   * Busca el tile más cercano (búsqueda por anillos) que sea transitable y
   495→   * NO esté dentro de una zona de salida. Garantiza que ningún viaje aterrice
   496→   * en un muro ni en un teletransporte (bucle cripta↔bosque, P0-1).
   497→   */
   498→  private findSafeTile(tx: number, ty: number): [number, number] {
   499→    const ok = (x: number, y: number) =>
   500→      x >= 0 && y >= 0 && x < this.map.w && y < this.map.h &&
   501→      !this.tileSolidAt(x * TILE + 8, y * TILE + 8) &&
   502→      !this.inExitZone(x, y);
   503→    if (ok(tx, ty)) return [tx, ty];
   504→    for (let r = 1; r <= 8; r++) {
   505→      for (let dy = -r; dy <= r; dy++) {
   506→        for (let dx = -r; dx <= r; dx++) {
   507→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   508→          if (ok(tx + dx, ty + dy)) return [tx + dx, ty + dy];
   509→        }
   510→      }
   511→    }
   512→    return [tx, ty]; // sin sitio libre: comportamiento anterior
   513→  }

   515→  boxFree(x: number, y: number, w: number, h: number): boolean {
   516→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   517→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   518→  }

   522→    if (dy !== 0 && this.boxFree(e.x, e.y + dy, e.w, e.h)) e.y += dy;

   561→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   562→    const p = this.player;
   563→    if (!p) return null;
   564→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   565→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   566→      const d = Math.hypot(x - p.x, y - p.y);
   567→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   568→    };
   569→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   570→    for (const ch of this.map.chests) {
   571→      if (this.openedChests.has(ch.id)) continue;
   572→      if (ch.needPast && this.epoch !== 'pasado') continue;
   573→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   574→    }
   575→    for (const pr of this.map.props) {
   576→      if (pr.needPast && this.epoch !== 'pasado') continue;
   577→      if (pr.needPresent && this.epoch !== 'presente') continue;
   578→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   579→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   580→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   581→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   582→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   583→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   584→      else if (pr.kind === 'lamp' && !this.flags[pr.id]) consider(px, py, 'lamp', 'Encender el Farol del Recuerdo', () => this.lightLamp(pr.id), 28);
   585→    }
   586→    for (const ec of this.map.echoes) {
   587→      if (this.takenEchoes.has(ec.id)) continue;
   588→      consider(ec.x * TILE + 8, ec.y * TILE + 8, 'echo', 'Escuchar eco menor', () => this.takeEchoMinor(ec.id, ec.title, ec.text), 26);
   589→    }
   590→    return best;
   591→  }

   824→      }
   825→      case action === 'end_demo': {

   827→        this.endStats =
   828→          `Nivel ${p.level} · ${p.kills} enemigos derrotados · ${p.gold} coronas\n` +

   839→      default:

   949→  private onKeyDown = (e: KeyboardEvent) => {
   950→    const k = e.key.toLowerCase();
   951→    // no interceptar teclas cuando se escribe en un input DOM (nombre del Portador)
   952→    const t = e.target as HTMLElement | null;
   953→    const isDomInput = !!t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable);
   954→    if (!isDomInput && [' ', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(k)) e.preventDefault();
   955→    if (e.repeat) return;
   956→    this.keys.add(k);
   957→    audio.resume();
   958→
   959→    if (this.state === 'title') {
   960→      if (k === 'enter' || k === 'e' || k === ' ') { /* botones por click; enter = nueva partida */ }
   961→    } else if (this.state === 'intro') {
   962→      if (k === 'e' || k === ' ' || k === 'enter') {
   963→        this.introIdx++;
   964→        audio.sfx('blip');
   965→        if (this.introIdx >= 3) this.startPlay();
   966→      }
   967→    } else if (this.state === 'dialogue') {
   968→      if (k === 'e' || k === ' ' || k === 'enter') this.advanceDialogue();
   969→      else if (k === 'arrowup' || k === 'w') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + this.dlgNode.options.length - 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
   970→      else if (k === 'arrowdown' || k === 's') { if (this.dlgNode?.options) { this.dlgSel = (this.dlgSel + 1) % this.dlgNode.options.length; audio.sfx('blip'); } }
   971→    } else if (this.state === 'play') {
   972→      if (k === ' ' && this.player && this.player.rollT <= 0 && this.player.attackT <= 0 && this.player.sta >= 20) {
   973→        this.rollQueued = true; // 1 pulsación = 1 voltereta (no en cadena)
   974→      }
   975→      if (k === 'e') this.tryInteract();
   976→      else if (k === 'q') this.epochSwitch();
   977→      else if (k === 'escape' || k === 'm') { this.setState('pause'); audio.sfx('uiOpen'); }
   978→      else if (k === 'f') this.drinkPotion();
   979→      else if (k === 't' && e.shiftKey) {
   980→        // accesibilidad / QoL: esperar hasta el alba
   981→        this.dayT = 0.22;
   982→        this.toast('Esperas junto al camino hasta el alba...', '#ffe86a');
   983→        audio.sfx('save');
   984→      }
   985→      else if (['1', '2', '3', '4'].includes(k)) this.useSkill(parseInt(k, 10) - 1);
   986→    } else if (this.state === 'pause') {
   987→      if (k === 'escape' || k === 'm') this.setState('play');
   988→    } else if (this.state === 'dead') {
   989→      if (k === 'e' || k === 'enter') this.respawn();
   990→    } else if (this.state === 'end') {
   991→      if (k === 'enter' || k === 'e') this.setState('title');
   992→    }
   993→  };

  1201→    const p = this.player!;

  1205→    for (const e of this.enemies) {
  1206→      if (e.dead) continue;

  1214→  aoeHit(x: number, y: number, r: number, dmg: number, element: Element, stun: boolean) {
  1215→    for (const e of this.enemies) {
  1216→      if (e.dead) continue;
  1217→      if (Math.hypot(e.x - x, e.y - y) < r + e.w / 2) {
  1218→        this.damageEnemy(e, dmg, element, 60, Math.sign(e.x - x), Math.sign(e.y - y));

  1224→  chainLightning(x: number, y: number, dmg: number, jumps: number, nx: number, ny: number) {
  1225→    let cx = x + nx * 14, cy = y + ny * 14;
  1226→    const hitSet = new Set<Enemy>();
  1227→    for (let j = 0; j < jumps; j++) {
  1228→      let best: Enemy | null = null, bd = 90;
  1229→      for (const e of this.enemies) {
  1230→        if (e.dead || hitSet.has(e)) continue;
  1231→        const d = Math.hypot(e.x - cx, e.y - cy);
  1232→        if (d < bd) { bd = d; best = e; }
  1233→      }
  1234→      if (!best) break;
  1235→      hitSet.add(best);
  1236→      this.damageEnemy(best, dmg * (1 - j * 0.2), 'rayo', 20, 0, 0);
  1237→      this.lightningFx(cx, cy, best.x, best.y);
  1238→      cx = best.x; cy = best.y;
  1239→    }

  1242→  lightningFx(x0: number, y0: number, x1: number, y1: number) {

  1254→  damageEnemy(e: Enemy, dmg: number, element: Element, kb: number, kbx = 0, kby = 0) {
  1255→    if (e.invulT !== undefined && e.invulT > 0) {
  1256→      // fase intangible (espectro desvaneciéndose / sirena sumergida): el golpe atraviesa

  1260→    }
  1261→    if (e.dead) return;
  1262→    const def = ENEMY_DEFS[e.etype];
  1263→    e.aggro = true;
  1264→    let final = dmg;
  1265→    let crit = false;
  1266→    const p = this.player!;
  1267→    if (Math.random() * 100 < Math.min(40, 5 + p.attrs.des * 2)) { final *= 2; crit = true; }
  1268→    if (def.weakTo !== 'ninguno' && element === def.weakTo) final *= 1.5;
  1269→    if (e.ai === 'aturdido' && e.etype === 'guardian') final *= 1.5;
  1270→    final = Math.max(1, Math.round(final));
  1271→    e.hp -= final;
  1272→    e.hitFlash = 0.12;
  1273→    // quiebre
  1274→    if (e.maxSta > 0) {
  1275→      e.sta -= final * 1.1;
  1276→      if (e.sta <= 0 && e.ai !== 'aturdido') {
  1277→        e.sta = 0;
  1278→        e.ai = 'aturdido';
  1279→        e.aiT = 4;
  1280→        this.toast('¡QUEBRADO! El enemigo queda vulnerable', '#ffe86a');
  1281→        this.floatAt(e.x, e.y - 20, 'QUEBRADO', '#ffe86a', 7);
  1282→        audio.sfx('roar');
  1283→      }
  1284→    }
  1285→    // estados
  1286→    // estados: se REFRESCAN en vez de apilarse sin tope (el quemado infinito
  1287→    // multiplicaba el DPS y trivializaba el quiebre del jefe)
  1288→    if (element === 'fuego') {
  1289→      const b = e.statuses.find(s => s.kind === 'quemado');
  1290→      if (b) { b.t = Math.min(6, b.t + 3); b.power = Math.min(8, b.power + 2); }
  1291→      else e.statuses.push({ kind: 'quemado', t: 3, power: 4 });
  1292→    }
  1293→    if (element === 'hielo') {
  1294→      const c = e.statuses.find(s => s.kind === 'congelado');
  1295→      if (c) c.t = Math.min(5, c.t + 2.5);
  1296→      else e.statuses.push({ kind: 'congelado', t: 2.5, power: 0.5 });
  1297→    }
  1298→    // reacciones elementales
  1299→    if (element === 'hielo' && e.statuses.some(s => s.kind === 'quemado' && s.t > 1)) {
  1300→      this.aoeHit(e.x, e.y, 30, 12, 'ninguno', false);
  1301→      this.floatAt(e.x, e.y - 16, 'VAPOR', '#d0f0f8', 6);
  1302→      this.burst(e.x, e.y - 6, '#d0f0f8', 12);
  1303→    }
  1304→    // knockback
  1305→    if (kb > 0) this.moveEntity(e, kbx * kb * 0.06, kby * kb * 0.06);
  1306→    // feedback
  1307→    this.floatAt(e.x + (Math.random() - 0.5) * 8, e.y - 14, `${final}`, crit ? '#ffd24a' : '#fff');
  1308→    if (crit) this.floatAt(e.x, e.y - 22, '¡CRÍTICO!', '#ffd24a', 6);
  1309→    this.burst(e.x, e.y - 4, crit ? '#ffd24a' : '#f0e8e0', crit ? 10 : 5);
  1310→    audio.sfx(crit ? 'crit' : 'hit');
  1311→    this.hitStop = crit ? 0.09 : 0.04;
  1312→    // resonancia
  1313→    const espMult = 1 + p.attrs.esp * 0.1;
  1314→    p.res = Math.min(p.maxRes, p.res + 6 * espMult);
  1315→    if (e.hp <= 0) this.killEnemy(e);
  1316→  }

  1373→  gainXp(xp: number) {
  1374→    const p = this.player!;
  1375→    p.xp += xp;
  1376→    let next = this.xpNext(p.level);
  1377→    while (p.xp >= next) {
  1378→      p.xp -= next;
  1379→      p.level++;
  1380→      p.maxHp += 7;
  1381→      p.hp = p.maxHp;
  1382→      p.points += 3;
  1383→      audio.sfx('levelup');
  1384→      this.toast(`¡Nivel ${p.level}! +3 puntos de atributo (menú > Estado)`, '#ffe86a');
  1385→      this.burst(p.x, p.y - 8, '#ffe86a', 20, 70);
  1386→      next = this.xpNext(p.level);
  1387→    }
  1388→  }

  1394→  damagePlayer(dmg: number, fromX: number, fromY: number) {
  1395→    const p = this.player!;
  1396→    if (p.iframes > 0 || p.rollT > 0 || this.state !== 'play') return;
  1397→    if (p.parryT > 0) {
  1398→      // ¡parada perfecta!
  1399→      audio.sfx('parry');
  1400→      p.res = Math.min(p.maxRes, p.res + 20);
  1401→      p.parryFx = 0.4;
  1402→      this.hitStop = 0.12;
  1403→      this.burst(p.x + (fromX - p.x) * 0.3, p.y - 8 + (fromY - p.y) * 0.3, '#fff8c0', 14, 100);
  1404→      this.floatAt(p.x, p.y - 22, '¡PARADA!', '#fff8c0', 7);
  1405→      // aturde al atacante cercano
  1406→      let best: Enemy | null = null, bd = 34;
  1407→      for (const e of this.enemies) {
  1408→        if (e.dead) continue;
  1409→        const d = Math.hypot(e.x - p.x, e.y - p.y);
  1410→        if (d < bd) { bd = d; best = e; }
  1411→      }
  1412→      if (best) { best.ai = 'aturdido'; best.aiT = 1.6; best.windup = 0; }
  1413→      p.parryT = 0;
  1414→      return;
  1415→    }
  1416→    const red = Math.min(0.5, p.attrs.vig * 0.01);
  1417→    const final = Math.max(1, Math.round(dmg * (1 - red)));
  1418→    p.hp -= final;
  1419→    p.iframes = 0.6;
  1420→    p.lastHitT = 0.3;
  1421→    this.shake = 4;
  1422→    audio.sfx('hurt');
  1423→    this.floatAt(p.x, p.y - 16, `-${final}`, '#ff7060');
  1424→    const dx = p.x - fromX, dy = p.y - fromY;
  1425→    const l = Math.max(1, Math.hypot(dx, dy));
  1426→    this.moveEntity(p, (dx / l) * 8, (dy / l) * 8);
  1427→    if (p.hp <= 0) this.playerDied();
  1428→  }
