    58→  return Math.hypot(bx - ax, by - ay);

    73→/** Color de chispa según el contexto: arma del Portador (acero/dorado
    74→ *  cargado) o elemento del golpe si viene de habilidad. */
    75→function sparkColorFor(p: Player, e: Enemy): string {
    76→  if (p.attackT > 0) return p.chargedHit ? '#ffe86a' : '#f5f2ea';
    77→  if (e.statuses.some(s => s.kind === 'quemado')) return '#ff9040';
    78→  if (e.statuses.some(s => s.kind === 'congelado')) return '#a0e8ff';
    79→  return '#e8e2d4';
    80→}

    82→/** Escanea enemigos buscando golpes recién conectados y suelta chispas
    83→ *  direccionales opuestas al swing (+ destello dorado si fue crítico:
    84→ *  el motor fija hitStop 0.09 en crítico frente a 0.04 normal).
    85→ *  Debe correr ANTES del bucle de enemigos (hitFlash aún sin decaer). */
    86→function updateCombatFx(g: Game): void {
    87→  fxF++;
    88→  const p = g.player;
    89→  if (!p) return;
    90→  for (const e of g.enemies) {
    91→    if (e.dead) continue;
    92→    const seen = fxHitSeen.get(e) ?? 0;
    93→    fxHitSeen.set(e, e.hitFlash);
    94→    if (e.hitFlash <= seen + 5e-4 || e.hitFlash <= 0.06) continue; // no es golpe nuevo
    95→    if (fxF - (fxOwnFx.get(e) ?? -99) <= 2) continue;              // ya tiene chispas propias
    96→    // rebote del impacto: chispas en dirección opuesta al swing
    97→    combatSparks(g, e.x, e.y - 4, p.x - e.x, p.y - e.y, sparkColorFor(p, e), 5, 70);
    98→    if (g.hitStop > 0.05) critGlint(g, e.x, e.y - 6);
    99→  }
   100→}

   102→export function updateGame(g: Game, dt: number) {
   103→  // cosmética siempre
   104→  if (g.state === 'dialogue' && g.dlgNode) g.dlgCharT += dt * 45;
   105→  g.mapTitleT = Math.max(0, g.mapTitleT - dt);
   106→  g.epochFx = Math.max(0, g.epochFx - dt);
   107→  g.shake = Math.max(0, g.shake - dt * 22);
   108→  for (const t of g.toasts) t.t -= dt;
   109→  g.toasts = g.toasts.filter(t => t.t > 0);
   110→  for (const f of g.floats) { f.t -= dt; f.y += f.vy * dt; }
   111→  g.floats = g.floats.filter(f => f.t > 0);
   112→  for (const p of g.particles) { p.t -= dt; p.x += p.vx * dt; p.y += p.vy * dt; p.vy += p.grav * dt; }
   113→  g.particles = g.particles.filter(p => p.t > 0);
   114→  if (g.state !== 'play') { g.updateCamera(); return; }
   115→
   116→  const p = g.player;
   117→  if (!p) return;
   118→  p.playTime += dt;
   119→  g.dayT = (g.dayT + dt / 240) % 1;
   120→  if (p.buffT) { p.buffT -= dt; if (p.buffT <= 0) p.buffT = undefined; }
   121→
   122→  // ---------------- fades y transición de mapa ----------------
   123→  // enfriamiento de salidas: tras cargar un mapa se ignoran sus zonas de
   124→  // salida un instante (defensa extra anti-bucle de teletransporte)
   125→  if (g.exitCd > 0) g.exitCd -= dt;
   126→  if (g.fadeDir !== 0) {
   127→    g.fadeT += g.fadeDir * dt * 2.4;
   128→    if (g.fadeT >= 1 && g.fadeDir > 0) {
   129→      g.fadeT = 1;
   130→      if (g.pendingMap) {
   131→        const { to, tx, ty } = g.pendingMap;
   132→        g.pendingMap = null;
   133→        g.loadMap(to, tx, ty);
   134→        audio.playTrack(g.map.music);
   135→        g.save();
   136→        g.toast('Autoguardado', '#8ef0ff');
   137→      }
   138→      g.fadeDir = -1;
   139→    }
   140→    if (g.fadeT <= 0 && g.fadeDir < 0) { g.fadeT = 0; g.fadeDir = 0; }
   141→    if (g.fadeDir !== 0) { g.updateCamera(); return; }
   142→  }
   143→
   144→  // ---------------- input del jugador ----------------
   145→  const k = g.keys;
   146→  let mx = 0, my = 0;
   147→  if (k.has('a') || k.has('arrowleft')) mx -= 1;
   148→  if (k.has('d') || k.has('arrowright')) mx += 1;
   149→  if (k.has('w') || k.has('arrowup')) my -= 1;
   150→  if (k.has('s') || k.has('arrowdown')) my += 1;
   151→  const mlen = Math.hypot(mx, my) || 1;
   152→
   153→  p.sta = Math.min(p.maxSta, p.sta + (p.rollT > 0 ? 0 : 26) * dt);
   154→  if (p.iframes > 0) p.iframes -= dt;
   155→  if (p.parryT > 0) p.parryT -= dt;
   156→  if (p.parryFx > 0) p.parryFx -= dt;
   157→  if (p.lastHitT > 0) p.lastHitT -= dt;
   158→  for (let i = 0; i < p.cds.length; i++) if (p.cds[i] > 0) p.cds[i] -= dt;
   159→
   160→  // knockback suave del jugador (ondas del jefe, slams) — contrato fxcore
   161→  stepKnockback(g, p, dt);
   162→
   163→  // carga de ataque
   164→  if (p.charging) {
   165→    p.chargeT += dt;
   166→    if (p.chargeT > 0.35 && Math.random() < 0.4) {
   167→      g.particles.push({ x: p.x + (Math.random() - 0.5) * 14, y: p.y - 10, vx: 0, vy: -30, t: 0.3, maxT: 0.3, color: '#ffe86a', size: 1.5, grav: 0 });
   168→    }
   169→  }
   170→
   171→  if (p.rollT > 0) {
   172→    // rodar
   173→    p.rollT -= dt;
   174→    const rd = (p as unknown as { rollDir?: Dir }).rollDir ?? p.dir;
   175→    const [rx, ry] = DIRS[rd];
   176→    const spd = 168;
   177→    g.moveEntity(p, rx * spd * dt, ry * spd * dt);
   178→    p.moving = true;
   179→    p.anim += dt * 1.4;
   180→    if (Math.random() < 0.5) g.particles.push({ x: p.x, y: p.y, vx: 0, vy: 0, t: 0.25, maxT: 0.25, color: '#d8dce4', size: 2, grav: 0 });
   181→  } else if (p.attackT > 0) {
   182→    p.attackT -= dt;
   183→    p.moving = false;
   184→    p.anim += dt;
   185→  } else {
   186→    // HIELO RESBALADIZO (Acto II, 9-a): caminar sobre el lago helado de las
   187→    // Cumbres (tile 'i') conserva la INERCIA. Fuera de hielo, control normal
   188→    // 1:1 (intacto). Nunca aplica con voltereta (otra rama) ni knockback.
   189→    const im = iceMem.get(p) ?? { vx: 0, vy: 0 };
   190→    iceMem.set(p, im);
   191→    const kbActive = (p.kbVx ?? 0) !== 0 || (p.kbVy ?? 0) !== 0;
   192→    const onIce = !kbActive &&
   193→      tileAt(g.map, g.rows, Math.floor(p.x / TILE), Math.floor((p.y + p.h / 2) / TILE), g.epoch) === 'i';
   194→    const spd = 74;
   195→    if (onIce) {
   196→      if (mx !== 0 || my !== 0) {
   197→        // al pisar hielo con velocidad, se parte de la inercia real del frame anterior
   198→        if (im.vx === 0 && im.vy === 0) { im.vx = plVX; im.vy = plVY; }
   199→        // aceleración reducida (×0.55): la velocidad persigue la objetivo más despacio
   200→        const k = 1 - Math.pow(0.55, dt * 60);
   201→        im.vx += ((mx / mlen) * spd - im.vx) * k;
   202→        im.vy += ((my / mlen) * spd - im.vy) * k;
   203→      } else {
   204→        // sin input: conserva la última velocidad con decaimiento 0.90^(dt·60)
   205→        // hasta bajar de 5 px/s (deslizamiento suave, no frenazo seco)
   206→        const dec = Math.pow(0.9, dt * 60);
   207→        im.vx *= dec;
   208→        im.vy *= dec;
   209→        if (im.vx * im.vx + im.vy * im.vy < 25) { im.vx = 0; im.vy = 0; }
   210→      }
   211→      if (im.vx !== 0 || im.vy !== 0) {
   212→        g.moveEntity(p, im.vx * dt, im.vy * dt);
   213→        p.moving = true;
   214→        p.anim += dt * ((mx !== 0 || my !== 0) ? 1 : 0.7); // derrapa sin input
   215→        if ((mx !== 0 || my !== 0) && (!p.charging || p.chargeT < 0.2)) {
   216→          if (Math.abs(mx) > Math.abs(my)) p.dir = mx > 0 ? 'right' : 'left';
   217→          else p.dir = my > 0 ? 'down' : 'up';
   218→        }
   219→      } else {
   220→        p.moving = false;
   221→        p.anim += dt * 0.4;
   222→      }
   223→    } else {
   224→      // control normal (tiles no helados): la memoria de hielo se resetea
   225→      im.vx = 0;
   226→      im.vy = 0;
   227→      if (mx !== 0 || my !== 0) {
   228→        g.moveEntity(p, (mx / mlen) * spd * dt, (my / mlen) * spd * dt);
   229→        p.moving = true;
   230→        p.anim += dt;
   231→        if (!p.charging || p.chargeT < 0.2) {
   232→          if (Math.abs(mx) > Math.abs(my)) p.dir = mx > 0 ? 'right' : 'left';
   233→          else p.dir = my > 0 ? 'down' : 'up';
   234→        }
   235→      } else {
   236→        p.moving = false;
   237→        p.anim += dt * 0.4;
   238→      }
   239→    }
   240→  }
   241→
   242→  // velocidad real del Portador (9-a): desplazamiento de este frame / dt.
   243→  // La usan la inercia del hielo (semilla al pisarlo) y la ventisca del
   244→  // gólem (getPortadorVel). Saltos de teleporte (>160 px) → velocidad 0.
   245→  if (plLastOk) {
   246→    const ddx = p.x - plLastX, ddy = p.y - plLastY;
   247→    if (ddx * ddx + ddy * ddy < 160 * 160) {
   248→      plVX = ddx / Math.max(1e-4, dt);
   249→      plVY = ddy / Math.max(1e-4, dt);
   250→      const vm = Math.hypot(plVX, plVY);
   251→      if (vm > 240) { plVX *= 240 / vm; plVY *= 240 / vm; } // tope (rodar+retroceso)
   252→    } else { plVX = 0; plVY = 0; }
   253→  }
   254→  plLastX = p.x; plLastY = p.y; plLastOk = true;
   255→
   256→  // ---------------- ventana de combo · golpe recién liberado · REMATE ----------------
   257→  // FX de impacto (8-c): detecta golpes conectados el frame anterior
   258→  // (input/parcelas) y suelta chispas; antes del bucle de enemigos.
   259→  updateCombatFx(g);
   260→
   261→  const prevAttackT = lastAttackT.get(p) ?? 0;
   262→  const attackStarted = p.attackT > 0 && prevAttackT <= 0;
   263→  lastAttackT.set(p, p.attackT);
   264→  // p.chargedHit es del motor (la usa render para el arco cargado); se limpia al
   265→  // terminar el golpe para que el siguiente ataque no herede el flag.
   266→  if (p.attackT <= 0 && p.chargedHit) p.chargedHit = false;
   267→
   268→  // ventana de combo: se mantiene mientras encadenas; al caducar el combo se reinicia
   269→  if (p.attackT > 0) {
   270→    p.comboT = 1.2;
   271→  } else if ((p.comboT ?? 0) > 0) {
   272→    p.comboT = (p.comboT ?? 0) - dt;
   273→    if ((p.comboT ?? 0) <= 0) { p.comboT = 0; p.combo = 0; }
   274→  }
   275→
   276→  if (attackStarted) {
   277→    // EMBESTIDA (8-c): el peso del tajo empuja al Portador un paso en su
   278→    // dirección. applyKnockback + stepKnockback (ya en marcha, línea 100)
   279→    // mueven con g.moveEntity → respeta tiles; fuerza suave que decae sola.
   280→    if (p.rollT <= 0) {
   281→      const [lgx, lgy] = DIRS[p.dir];
   282→      applyKnockback(p, lgx, lgy, p.chargedHit ? 88 : 62);
   283→    }
   284→    // impacto cargado: onda de empuje frontal (el peso del Portador)
   285→    if (p.chargedHit) {
   286→      const [fdx, fdy] = DIRS[p.dir];
   287→      const cx2 = p.x + fdx * 20, cy2 = p.y + fdy * 20;
   288→      for (const e of g.enemies) {
   289→        if (e.dead) continue;
   290→        if (Math.hypot(e.x - cx2, e.y - cy2) < 36 + e.w / 2) {
   291→          applyKnockback(e, e.x - p.x, e.y - p.y, 200);
   292→        }
   293→      }
   294→    }
   295→    // REMATE: enemigo quebrado cerca + golpe recién liberado → bonus único por quebrado
   296→    for (const e of g.enemies) {
   297→      if (e.dead || e.ai !== 'aturdido' || e.maxSta <= 0) continue;
   298→      if (finisherUsed.has(e)) continue;
   299→      if (dist(p.x, p.y, e.x, e.y) < 30) {
   300→        finisherUsed.add(e);
   301→        const [rdx, rdy] = DIRS[p.dir];
   302→        g.damageEnemy(e, playerMeleeDmg(p) * 0.6, 'ninguno', 90, rdx, rdy);
   303→        // chispas del REMATE (8-c): doradas, opuestas al swing; marcado para
   304→        // que updateCombatFx no las duplique al ver el hitFlash subir
   305→        fxOwnFx.set(e, fxF);
   306→        combatSparks(g, e.x, e.y - 4, -rdx, -rdy, '#ffe86a', 6, 90);
   307→        critGlint(g, e.x, e.y - 6);
   308→        g.floatAt(e.x, e.y - 26, '¡REMATE!', '#ffe86a', 12);
   309→        addShake(g, 4);
   310→        requestSlowmo(g, 0.2);
   311→        p.res = Math.min(p.maxRes, p.res + 15);
   312→        // sfx 'break' es nuevo del contrato (lo añade 3-a); audio.sfx no tiene
   313→        // default: nombre desconocido = no-op seguro.
   314→        audio.sfx('break');
   315→        break;
   316→      }
   317→    }
   318→  }
   319→
   320→  // esquiva (detección de borde: la pulsación se pone en cola en keydown;
   321→  // mantener Espacio ya no encadena volteretas con invulnerabilidad continua)
   322→  if (g.rollQueued && p.rollT <= 0 && p.attackT <= 0 && p.sta >= 20) {
   323→    g.rollQueued = false;
   324→    p.rollT = 0.3;
   325→    p.iframes = 0.34;
   326→    p.sta -= 20;
   327→    let rd: Dir = p.dir;
   328→    if (mx !== 0 || my !== 0) rd = Math.abs(mx) > Math.abs(my) ? (mx > 0 ? 'right' : 'left') : (my > 0 ? 'down' : 'up');
   329→    (p as unknown as { rollDir?: Dir }).rollDir = rd;
   330→    // soplo de esquiva (8-c): anillo de partículas que se abre con la voltereta
   331→    dodgeRing(g, p.x, p.y);
   332→    audio.sfx('dodge');
   333→  }
   334→
   335→  // impacto del ataque (instantáneo en la liberación, ver releaseCharge)
   336→
   337→  // ---------------- recoger oro perdido ----------------
   338→  for (let i = g.deadGolds.length - 1; i >= 0; i--) {
   339→    const dgl = g.deadGolds[i];
   340→    if (dgl.map !== g.mapId) continue;
   341→    if (dist(p.x, p.y, dgl.x, dgl.y) < 14) {
   342→      p.gold += dgl.amount;
   343→      g.deadGolds.splice(i, 1);
   344→      audio.sfx('coin');
   345→      g.floatAt(dgl.x, dgl.y - 10, `+${dgl.amount}`, '#f0c84a');
   346→      g.toast(`Recuperas tu eco de oro (+${dgl.amount})`, '#f0c84a');
   347→    }
   348→  }
   349→
   350→  // ---------------- salidas de mapa ----------------
   351→  if (g.fadeDir === 0 && g.exitCd <= 0) {
   352→    const ptx = Math.floor(p.x / TILE), pty = Math.floor(p.y / TILE);
   353→    for (const ex of g.map.exits) {
   354→      if (ex.needPast && g.epoch !== 'pasado') continue;
   355→      if (ptx >= ex.x && ptx < ex.x + ex.w && pty >= ex.y && pty < ex.y + ex.h) {
   356→        g.fadeTo(ex.to, ex.tx, ex.ty);
   357→        audio.sfx('echo');
   358→        break;
   359→      }
   360→    }
   361→  }
   362→
   363→  // ---------------- watchers de contenido (baratos, O(1) por frame) ----------------
   364→  // q2_done: Brisa da por completada la misión de los lobos → Ilwen aparece en el Bosque
   365→  // (la bandera no la fijaba nadie y su showFlag dejaba a la elfa invisible para siempre)
   366→  if (g.questIdx >= 2 && !g.flags.q2_done) g.flags.q2_done = true;
   367→  // Memoria II: tras recuperar el Eco de la Voz, la casa junto al río aflora al pisar el Bosque
   368→  if (g.flags.ecoVoz && g.mapId === 'bosque' && !g.flags.mem_casa) {
   369→    g.flags.mem_casa = true;
   370→    if (!(p.memories ?? []).includes('mem_casa')) g.applyAction('memory_mem_casa');
   371→  }
   372→  // Memoria III: la madre sin rostro aflora al vencer al Guardián Hueco
   373→  if (g.flags.guardianDefeated && !g.flags.mem_madre) {
   374→    g.flags.mem_madre = true;
   375→    if (!(p.memories ?? []).includes('mem_madre')) g.applyAction('memory_mem_madre');
   376→  }
   377→
   378→  // ---------------- compañera ----------------
   379→  if (g.companion) updateCompanion(g, dt);
   380→
   381→  // ---------------- enemigos ----------------
   382→  let anyAggro = false;
   383→  for (const e of g.enemies) {
   384→    if (e.dead) {
   385→      // jefes del Acto II (9-a) + jefes 14-a: última salva visual en cuanto mueren,
   386→      // ANTES de que el filtro de abajo los retire del array
   387→      if (e.etype === 'sirena' || e.etype === 'golem' || e.etype === 'vult' || e.etype === 'coro') expansionDeathFx(g, e);
   388→      continue;
   389→    }

   647→function updateEnemy(g: Game, e: Enemy, dt: number) {
   648→  const p = g.player!;
   649→  const def = ENEMY_DEFS[e.etype];
   650→  if (e.hitFlash > 0) e.hitFlash -= dt;
   651→  if (e.spawnGuard && e.spawnGuard > 0) { e.spawnGuard -= dt; return; }
   652→
   653→  // REMATE disponible al entrar en aturdimiento/quebrado (una vez por periodo)
   654→  const prevStun = wasAturdido.get(e) ?? false;
   655→  if (e.ai === 'aturdido' && !prevStun) finisherUsed.delete(e);
   656→  wasAturdido.set(e, e.ai === 'aturdido');
   657→
   658→  // knockback suave (proyectiles, cargas, ondas) — contrato fxcore
   659→  stepKnockback(g, e, dt);
   660→
   661→  // estados
   662→  for (let i = e.statuses.length - 1; i >= 0; i--) {
   663→    const s = e.statuses[i];
   664→    s.t -= dt;
   665→    if (s.kind === 'quemado') {
   666→      e.hp -= s.power * dt;
   667→      if (Math.random() < 0.15) g.particles.push({ x: e.x + (Math.random() - 0.5) * 8, y: e.y - 6, vx: 0, vy: -24, t: 0.3, maxT: 0.3, color: '#ff9040', size: 1.5, grav: 0 });
   668→      if (e.hp <= 0) { g.killEnemy(e); return; }
   669→    }
   670→    if (s.t <= 0) e.statuses.splice(i, 1);
   671→  }
   672→  // decae la marca de Ilwen (el icono lo dibuja 3-a)
   673→  if (e.marked !== undefined) {
   674→    e.marked -= dt;
   675→    if (e.marked <= 0) e.marked = undefined;
   676→  }
   677→  if (e.hp < e.maxHp) e.aggro = true;
   678→
   679→  // Acto II: decae la fase intangible y delega en el cerebro propio si lo maneja
   680→  if (e.invulT !== undefined && e.invulT > 0) e.invulT -= dt;
   681→  if (EXPANSION_TYPES.has(e.etype) && expansionTick(g, e, dt, def)) return;
   682→
   683→  const d = dist(e.x, e.y, p.x, p.y);
   684→  const nightMult = isNight(g) ? 1.3 : 1;
   685→  const aggroR = def.aggroR * nightMult * (e.etype === 'guardian' ? (g.bossActive ? 99 : 1) : 1);
   686→  if (!e.aggro && d < aggroR && g.state === 'play') {
   687→    e.aggro = true;
   688→    if (e.etype === 'guardian') {
   689→      // primera vez que entra en aggro: banner del jefe (render lo dibuja)
   690→      if (!g.flags.bossIntro) {
   691→        g.flags.bossIntro = true;
   692→        g.bossBannerT = 3.2;
   693→        g.bossBannerText = 'GUARDIÁN HUECO';
   694→        g.bossBannerSub = 'Custodio del Eco de la Voz';
   695→        addFlash(g, '#7ee8ff', 0.25);
   696→        // sfx 'banner' es nuevo del contrato (3-a); no-op seguro hasta entonces
   697→        audio.sfx('banner');
   698→      }
   699→    }
   700→    else audio.sfx('blip');
   701→  }
   702→  e.atkCd -= dt;
   703→  e.anim += dt * (e.ai === 'persigue' ? 1.4 : 0.6);
   704→
   705→  if (e.ai === 'aturdido') {
   706→    e.aiT -= dt;
   707→    e.moving = false;
   708→    if (e.aiT <= 0) { e.ai = 'persigue'; if (e.maxSta > 0) e.sta = e.maxSta; }
   709→    return;
   710→  }
   711→
   712→  switch (e.ai) {
   713→    case 'patrulla': {
   714→      e.aiT -= dt;
   715→      if (e.aiT <= 0) {
   716→        e.aiT = 1.5 + Math.random() * 2;
   717→        e.patrolAngle = Math.random() * Math.PI * 2;
   718→      }
   719→      // si se aleja de su hogar, vuelve; si no, deambula
   720→      const far = Math.hypot(e.x - e.homeX, e.y - e.homeY) > 60;
   721→      const ang = far ? Math.atan2(e.homeY - e.y, e.homeX - e.x) : e.patrolAngle;
   722→      const vx = Math.cos(ang) * 22, vy = Math.sin(ang) * 22;
   723→      const before = { x: e.x, y: e.y };
   724→      g.moveEntity(e, vx * dt * speedMult(e), vy * dt * speedMult(e));
   725→      if (Math.abs(e.x - before.x) < 0.01 && Math.abs(e.y - before.y) < 0.01) {
   726→        e.patrolAngle += Math.PI * (0.5 + Math.random());
   727→      }
   728→      e.moving = true;
   729→      e.dir = Math.cos(ang) > 0 ? 'right' : 'left';
   730→      if (e.aggro) e.ai = 'persigue';
   731→      break;
   732→    }
   733→    case 'persigue': {
   734→      if (!e.aggro) { e.ai = 'patrulla'; break; }
   735→      if (d > aggroR * 2.4) { e.aggro = false; e.ai = 'patrulla'; break; }
   736→      const dx = (p.x - e.x) / (d || 1), dy = (p.y - e.y) / (d || 1);
   737→      const spd = def.speed * speedMult(e);
   738→      if (d > def.atkR * 0.8) {
   739→        g.moveEntity(e, dx * spd * dt, dy * spd * dt);
   740→        e.moving = true;
   741→        e.dir = dx > 0 ? 'right' : 'left';
   742→      } else e.moving = false;
   743→      if (d <= def.atkR && e.atkCd <= 0) {
   744→        e.ai = 'carga';
   745→        e.windup = def.windup * (e.etype === 'guardian' && e.phase >= 3 ? 0.7 : 1);
   746→        e.telegraphKind = undefined;
   747→      }
   748→      // jefe: acciones especiales
   749→      if (e.etype === 'guardian') guardianBrain(g, e, dt, d);
   750→      break;
   751→    }
   752→    case 'carga': {
   753→      e.windup -= dt;
   754→      e.moving = false;
   755→      // mirar al jugador
   756→      e.dir = p.x > e.x ? 'right' : 'left';
   757→      if (e.windup <= 0) {
   758→        if (e.telegraphKind === 'onda') {
   759→          // onda expansiva del Guardián
   760→          g.waves.push({ x: e.x, y: e.y, r: 10, maxR: 150, speed: 130, dmg: 13 * e.phase, hit: false });
   761→          audio.sfx('slam');
   762→          g.shake = 5;
   763→          e.telegraphKind = undefined;
   764→          e.ai = 'recupera';
   765→          e.aiT = 0.6;
   766→          e.atkCd = 2.6;
   767→        } else if (e.etype === 'guardian') {
   768→          // golpe en área
   769→          e.ai = 'ataca';
   770→          e.aiT = 0.22;
   771→          audio.sfx('slam');
   772→          if (d < 46) {
   773→            // el slam del Guardián arrea al Portador
   774→            if (p.parryT <= 0 && p.iframes <= 0 && p.rollT <= 0) applyKnockback(p, p.x - e.x, p.y - e.y, 260);
   775→            g.damagePlayer(def.dmg * e.phase, e.x, e.y);
   776→          }
   777→          g.burst(e.x, e.y, '#8a84a8', 14, 80);
   778→          g.shake = 5;
   779→        } else {
   780→          // estampida del lobo / mandoble del esqueleto
   781→          e.ai = 'ataca';
   782→          e.aiT = 0.22;
   783→          const dx = (p.x - e.x) / (d || 1), dy = (p.y - e.y) / (d || 1);
   784→          g.moveEntity(e, dx * 14, dy * 14);
   785→          if (d < def.atkR + 10) {
   786→            g.damagePlayer(def.dmg, e.x, e.y);
   787→          }
   788→        }
   789→      }
   790→      break;
   791→    }
   792→    case 'ataca': {
   793→      e.aiT -= dt;
   794→      if (e.aiT <= 0) {
   795→        e.ai = 'recupera';
   796→        e.aiT = 0.45;
   797→        e.atkCd = def.atkCd * (0.85 + Math.random() * 0.3);
   798→      }
   799→      break;
   800→    }
   801→    case 'recupera': {
   802→      e.aiT -= dt;
   803→      e.moving = false;
   804→      if (e.aiT <= 0) e.ai = e.aggro ? 'persigue' : 'patrulla';
   805→      break;
   806→    }
   807→    default: break;
   808→  }
   809→
   810→  // huida de lobos heridos
   811→  if (e.etype === 'lobo' && e.hp < e.maxHp * 0.25 && e.ai !== 'huye' && Math.random() < 0.01) {
   812→    e.ai = 'huye'; e.aiT = 2.2;
   813→  }
   814→  if (e.ai === 'huye') {
   815→    e.aiT -= dt;
   816→    const dx = (e.x - p.x) / (d || 1), dy = (e.y - p.y) / (d || 1);
   817→    g.moveEntity(e, dx * def.speed * 1.2 * dt, dy * def.speed * 1.2 * dt);
   818→    if (e.aiT <= 0) e.ai = 'persigue';
   819→  }
   820→}

   822→function guardianBrain(g: Game, e: Enemy, dt: number, d: number) {
   823→  const p = g.player!;
   824→  const hpPct = e.hp / e.maxHp;
   825→  const newPhase = hpPct > 0.6 ? 1 : hpPct > 0.3 ? 2 : 3;
   826→  if (newPhase !== e.phase) {
   827→    e.phase = newPhase;
   828→    audio.sfx('roar');
   829→    g.toast(`El Guardián cambia de fase (${e.phase}/3)`, '#7ee8ff');
   830→    addShake(g, 5);
   831→    addFlash(g, e.phase === 3 ? '#e8a0ff' : '#7ee8ff', 0.2);
   832→    requestSlowmo(g, 0.25);
   833→    if (e.phase === 2) {
   834→      // invoca sombras
   835→      for (let i = 0; i < 2; i++) {
   836→        const s = g.makeEnemy('sombra', e.x + (i === 0 ? -24 : 24), e.y + 12, 0, 'boss');
   837→        s.aggro = true;
   838→        g.enemies.push(s);
   839→      }
   840→      g.toast('El Guardián llama a sombras sin rostro', '#b48fff');
   841→    }
   842→  }
   843→  e.sumT -= dt;
   844→  if (e.atkCd <= 0) {
   845→    // elegir acción: slam a distancia, onda en fase 2+, swipe si muy cerca
   846→    if (d < 40) {
   847→      e.ai = 'carga';
   848→      e.windup = 0.55;
   849→      e.atkCd = 2.2;
   850→    } else if (e.phase >= 2 && Math.random() < 0.5) {
   851→      // onda expansiva
   852→      e.windup = 0.9;
   853→      e.ai = 'carga';
   854→      e.telegraphKind = 'onda';
   855→      e.atkCd = 3.2;
   856→    } else {
   857→      // slam telegrafiado sobre el jugador
   858→      g.telegraphs.push({ x: p.x, y: p.y, r: 34, t: 0.85, maxT: 0.85, dmg: 16 * e.phase, kind: 'slam' });
   859→      e.atkCd = 2.4 / (e.phase >= 3 ? 1.4 : 1);
   860→      e.ai = 'recupera';
   861→      e.aiT = 0.5;
   862→    }
   863→  }
   864→}
