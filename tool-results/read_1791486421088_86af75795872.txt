    57→/**
    58→ * Ajusta el buffer del juego al aspecto de la ventana para eliminar el
    59→ * letterbox (la "barra negra" abajo/arriba). Mantiene 540px de alto base
    60→ * a 16:9 exacto; pantallas más anchas ganan vista lateral (hasta 1600px)
    61→ * y más cuadradas ganan vista vertical (hasta 800px). Clamp de seguridad
    62→ * para aspectos extremos (móvil vertical): ahí el letterbox es aceptable.
    63→ */
    64→export function fitViewToWindow(winW: number, winH: number): void {
    65→  const a = winW / Math.max(1, winH);
    66→  let vw = Math.round(540 * a);
    67→  let vh = 540;
    68→  if (vw < 840) { vw = 840; vh = Math.round(vw / a); }
    69→  else if (vw > 1600) { vw = 1600; vh = Math.round(vw / a); }
    70→  vh = Math.max(460, Math.min(800, vh));
    71→  vw = Math.max(840, Math.min(1600, Math.round(vh * a)));
    72→  if (vw !== VIEW_W || vh !== VIEW_H) { VIEW_W = vw; VIEW_H = vh; }
    73→}

   266→    if (this.questIdx < 5 || this.questIdx > 9) return;
   267→    // q7 (idx 6): 0 avistada · 1 derrotada · 2 Eco de las Mareas tomado
   268→    if (this.questIdx === 6) {
   269→      if (this.questStep === 0 && this.flags.sirenaDefeated) this.questAdvance();
   270→      if (this.questIdx === 6 && this.questStep === 1 && this.flags.sirenaDefeated) this.questAdvance();
   271→      if (this.questIdx === 6 && this.questStep === 2 && this.flags.ecoMareas) this.questAdvance();
   272→    }
   273→    // q8 (idx 7): 1 los 3 faroles encendidos (orden-independiente) · 2 Eco de los Nombres
   274→    if (this.questIdx === 7) {
   275→      if (this.questStep === 1 &&
   276→          (['lamp1', 'lamp2', 'lamp3'] as const).filter(l => this.flags[l]).length >= 3) this.questAdvance();
   277→      if (this.questIdx === 7 && this.questStep === 2 && this.flags.ecoNombres) this.questAdvance();
   278→    }
   279→    // q9 (idx 8): 1 gólem derrotado · 2 Eco de las Cumbres tomado
   280→    if (this.questIdx === 8) {
   281→      if (this.questStep === 1 && this.flags.golemDefeated) this.questAdvance();
   282→      if (this.questIdx === 8 && this.questStep === 2 && this.flags.ecoCumbres) this.questAdvance();
   283→    }
   284→  }

   288→  start() {
   289→    if (this.running) return;
   290→    this.running = true;
   291→    this.lastTs = performance.now();
   292→    const loop = (ts: number) => {
   293→      if (!this.running) return;
   294→      try {
   295→        let dt = Math.min(0.05, (ts - this.lastTs) / 1000);
   296→        this.lastTs = ts;
   297→        if (this.flashT > 0) this.flashT = Math.max(0, this.flashT - dt);
   298→        if (this.bossBannerT > 0) this.bossBannerT = Math.max(0, this.bossBannerT - dt);
   299→        if (this.memoryReveal) { this.memoryReveal.t -= dt; if (this.memoryReveal.t <= 0) this.memoryReveal = null; }
   300→        if (this.hitStop > 0) { this.hitStop -= dt; dt *= 0.12; }
   301→        if (this.slowmoT > 0) { this.slowmoT -= dt; dt *= 0.35; }
   302→        this.globalT += dt;
   303→        if (this.state === 'play' || this.state === 'dialogue') this.update(dt);
   304→        drawGame(this);
   305→        this.loopError = null;
   306→      } catch (err) {
   307→        this.loopError = err instanceof Error ? (err.stack ?? String(err)) : String(err);
   308→        console.error('[EcosAelthar loop]', err);
   309→        try {
   310→          this.ctx.fillStyle = '#000';
   311→          this.ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   312→          this.ctx.font = '12px monospace';
   313→          this.ctx.fillStyle = '#ff7060';
   314→          this.ctx.textAlign = 'left';
   315→          this.ctx.textBaseline = 'top';
   316→          this.ctx.fillText('Error: ' + String(err).slice(0, 90), 20, 20);
   317→          this.ctx.fillText((err instanceof Error ? (err.stack ?? '') : '').slice(0, 600).replace(/\n/g, ' | '), 20, 44);
   318→        } catch { /* noop */ }
   319→      }
   320→      this.raf = requestAnimationFrame(loop);
   321→    };
   322→    this.raf = requestAnimationFrame(loop);
   323→  }
   324→
   325→  stop() {
   326→    this.running = false;
   327→    cancelAnimationFrame(this.raf);
   328→    audio.stopMusic();
   329→  }

   373→  continueGame() {
   374→    const raw = (() => { try { return localStorage.getItem('ecos-aelthar-save'); } catch { return null; } })();
   375→    if (!raw) return;
   376→    let d: SaveData;
   377→    try {

   379→      if (!d || !d.player || typeof d.x !== 'number' || typeof d.y !== 'number' || !MAPS[d.map]) throw new Error('forma inválida');
   380→    } catch {
   381→      // save corrupto: nunca dejar el juego muerto — se descarta y se puede empezar de nuevo
   382→      try { localStorage.removeItem('ecos-aelthar-save'); } catch { /* noop */ }
   383→      this.toast('La partida guardada estaba dañada y no pudo recuperarse.', '#ff8060');
   384→      return;
   385→    }
   386→    const p = d.player;
   387→    const maxHp = p.maxHp;
   388→    this.player = {
   389→      kind: 'player', name: p.name, discipline: p.discipline,
   390→      x: d.x, y: d.y, w: 10, h: 8, vx: 0, vy: 0, dir: 'down',
   391→      hp: Math.max(1, p.hp), maxHp, sprite: p.discipline === 'alba' ? 'hero_alba' : 'hero_tejedor',
   392→      anim: 0, moving: false,
   393→      level: p.level, xp: p.xp, sta: p.maxSta, maxSta: p.maxSta, res: p.res, maxRes: 100,
   394→      attrs: { ...p.attrs }, points: p.points,
   395→      gold: p.gold, weaponPlus: p.weaponPlus, potions: p.potions, cds: [0, 0, 0, 0],
   396→      iframes: 0, parryT: 0, parryFx: 0, attackT: 0, combo: 0,
   397→      chargeT: 0, charging: false, rollT: 0, lastHitT: 0,
   398→      hasEcho: p.hasEcho, kills: p.kills, deaths: p.deaths,
   399→      repGuardianes: p.repGuardianes, playTime: p.playTime,
   400→      tones: p.tones ?? { empatico: 0, pragmatico: 0, sarcastico: 0, amenazante: 0 },
   401→      memories: p.memories ?? [],
   402→      repFacciones: p.repFacciones ?? { guardianes: 0, orden: 0, circulo: 0, liga: 0 },
   403→    };
   404→    this.flags = { ...d.flags };
   405→    this.questIdx = d.questIdx; this.questStep = d.questStep;
   406→    this.openedChests = new Set(d.openedChests);
   407→    this.takenEchoes = new Set(d.takenEchoes);
   408→    this.deadGolds = d.deadGolds ?? [];
   409→    this.epoch = d.epoch === 'pasado' && p.hasEcho ? 'pasado' : 'presente';
   410→    this.companion = d.companion ? this.makeCompanion() : null;
   411→    this.visitedMaps = { lunaris: true };
   412→    if (this.flags.visitedBosque) this.visitedMaps.bosque = true;
   413→    if (this.flags.visitedCripta) this.visitedMaps.cripta = true;
   414→    this.loadMap(d.map, Math.floor(d.x / TILE), Math.floor(d.y / TILE));
   415→    // restaura la posición fina del guardado SOLO si su tile es seguro
   416→    // (ni sólido ni dentro de una zona de salida); si no, conserva el tile

   456→  loadMap(id: MapId, tx: number, ty: number) {
   457→    // memoria del jefe: si sales de la cripta con el Guardián herido, no se
   458→    // regenera al volver (cierra el exploit de reseteo de combate)
   459→    // memoria del jefe (generalizada Acto II): si sales de un mapa con su JEFE vivo, no se regenera
   460→    if (this.bossRef && !this.bossRef.dead) {
   461→      this.flags[`bossHp_${this.mapId}`] = this.bossRef.hp;
   462→      if (this.mapId === 'cripta') this.flags.bossHp = this.bossRef.hp; // compat con saves antiguos
   463→    }
   464→    this.mapId = id;
   465→    this.map = MAPS[id];
   466→    this.rows = mapRows(this.map);
   467→    this.buildGround();
   468→    if (this.player) {
   469→      // defensa anti-softlock: nunca aterrizar sobre un tile sólido ni dentro
   470→      // de una zona de salida del destino (provocaba el bucle cripta↔bosque)
   471→      const [sx, sy] = this.findSafeTile(tx, ty);
   472→      this.player.x = sx * TILE + 8;
   473→      this.player.y = sy * TILE + 8;
   474→      // 14-b (auditoría de movimiento): el estado transitorio de combate NO
   475→      // viaja de mapa — los iframes de un golpe recibido antes del viaje
   476→      // daban invulnerabilidad gratis al aterrizar, y una voltereta/carga en
   477→      // curso al iniciar el fundido seguía viva al llegar (roll a ciegas en
   478→      // el mapa nuevo, posible encallamiento contra muros desconocidos).
   479→      this.player.iframes = 0;
   480→      this.player.rollT = 0;
   481→      this.player.charging = false;
   482→      this.player.chargeT = 0;
   483→      this.player.parryT = 0;
   484→      this.player.lastHitT = 0;
   485→      this.rollQueued = false;
   486→    }
   487→    this.exitCd = 0.9;
   488→    this.spawnEnemies();
   489→    // restaura la vida memorizada del Guardián si la partida sigue en curso
   490→    const savedBossHp = typeof this.flags[`bossHp_${id}`] === 'number' ? (this.flags[`bossHp_${id}`] as number)
   491→      : id === 'cripta' && typeof this.flags.bossHp === 'number' ? (this.flags.bossHp as number) : null;
   492→    if (savedBossHp !== null) {
   493→      const boss = this.enemies.find(e => BOSS_DEFEAT_FLAG[e.etype] !== undefined);
   494→      if (boss) boss.hp = Math.max(1, Math.min(boss.maxHp, savedBossHp));
   495→    }
   496→    this.spawnNpcs();
   497→    this.projectiles = []; this.waves = []; this.telegraphs = [];
   498→    this.bossRef = null; this.bossActive = false;
   499→    audio.setCombat(false);
   500→    this.visitedMaps[id] = true;
   501→    if (id === 'bosque' && !this.flags.visitedBosque) {
   502→      this.flags.visitedBosque = true;
   503→      if (this.questIdx === 2 && this.questStep === 0) this.questAdvance();
   504→    }
   505→    if (id === 'cripta' && !this.flags.visitedCripta) {
   506→      this.flags.visitedCripta = true;
   507→      if (this.questIdx === 3 && this.questStep === 0) this.questAdvance();
   508→    }
   509→    // Acto II: pisar cada mapa nuevo completa su objetivo de viaje. Sin guard
   510→    // de "primera visita": si el jugador llega ANTES de aceptar la misión, el
   511→    // avance debe funcionar igual al volver (condición idempotente por estado).
   512→    if (id === 'costa' && this.questIdx === 5 && this.questStep === 0) this.questAdvance();
   513→    if (id === 'aldea' && this.questIdx === 7 && this.questStep === 0) this.questAdvance();
   514→    if (id === 'cumbres' && this.questIdx === 8 && this.questStep === 0) this.questAdvance();

   520→  buildGround() {

   522→    const mk = (ep: Epoch) => {
   523→      const c = document.createElement('canvas');

   564→          ch === '#' || ch === 'H' || ch === 'r' ? '#7a7a8a' :
   565→          ch === 'n' ? '#7ea49a' : '#4a8a44';

   622→  // ---------------- Compañera (14-b: escolta robusta) ----------------
   623→  // El seguimiento, el ciclo de flechas elementales, la marca y la Lluvia de
   624→  // estrellas siguen en update.ts (congelado). Este tick AÑADE lo que faltaba:
   625→  // (a) nunca se pierde (teletransporte >12 tiles / atasco / caja en muro),
   626→  // (b) apoyo de combate propio contra el enemigo aggro más cercano al
   627→  //     Portador (prioriza marcados) con cadencia propia de 2.4 s,
   628→  // (c) regeneración extra fuera de combate (+8 hp/s — no muere por su cuenta;
   629→  //     el regen base de 2 hp/s ya existía en update.ts),
   630→  // (d) avisos tácticos ocasionales cuando te rodean 2+ enemigos (cd 25 s).
   631→  private companionTick(dt: number) {
   632→    const c = this.companion, p = this.player;
   633→    if (!c || !p || c.downT > 0) return; // derribada: la gestiona update.ts
   634→    const mem = compMem.get(c) ?? { lastX: c.x, lastY: c.y, stuckT: 0, coverCd: 0.5, lineCd: 14, lineIdx: 0 };
   635→    compMem.set(c, mem);
   636→    const d = Math.hypot(p.x - c.x, p.y - c.y);
   637→    // (a) escolta: a >12 tiles (384 px) reaparece a tu lado; si su caja quedó
   638→    // dentro de un sólido (sin pathfinding detrás de un muro), también.
   639→    if (d > 12 * TILE || !this.boxFree(c.x, c.y, c.w, c.h)) {
   640→      this.companionTeleport(c, p);
   641→      mem.lastX = c.x; mem.lastY = c.y;
   642→      return;
   643→    }
   644→    // anti-atasco: quería seguirte (d>40) y lleva 1.2 s sin avanzar → teleporte suave
   645→    const moved = Math.hypot(c.x - mem.lastX, c.y - mem.lastY);
   646→    mem.lastX = c.x; mem.lastY = c.y;
   647→    if (d > 40 && moved < 0.6) mem.stuckT += dt; else mem.stuckT = 0;
   648→    if (mem.stuckT >= 1.2) {
   649→      this.companionTeleport(c, p);
   650→      mem.stuckT = 0;
   651→      return;
   652→    }
   653→    // (b) apoyo de combate con cadencia propia: flecha de cobertura SIN
   654→    // elementos (los estados los pone el ciclo de update.ts, aquí no se
   655→    // duplican) dirigida al aggro más cercano AL PORTADOR.
   656→    mem.coverCd -= dt;
   657→    if (mem.coverCd <= 0 && this.state === 'play') {
   658→      let best: Enemy | null = null, bd = 190;
   659→      for (const e of this.enemies) {
   660→        if (e.dead || !e.aggro) continue;
   661→        const score = Math.hypot(e.x - p.x, e.y - p.y) - ((e.marked ?? 0) > 0 ? 60 : 0);
   662→        if (score < bd) { bd = score; best = e; }
   663→      }
   664→      if (best) {
   665→        mem.coverCd = 2.4;
   666→        const dx = best.x - c.x, dy = best.y - 4 - (c.y - 6);
   667→        const l = Math.max(1, Math.hypot(dx, dy));
   668→        this.projectiles.push({
   669→          x: c.x, y: c.y - 6, vx: (dx / l) * 190, vy: (dy / l) * 190, t: 1.2,
   670→          dmg: 6 + p.level * 1.2, element: 'ninguno', from: 'companion',
   671→          sprite: 'p_arrow', radius: 3, pierce: 0,
   672→        });
   673→        audio.sfx('companionShot');
   674→      }
   675→    }
   676→    // (c) fuera de combate regenera más rápido (el vínculo la mantiene entera)
   677→    const inCombat = this.enemies.some(e => !e.dead && e.aggro);
   678→    if (!inCombat && c.hp < c.maxHp) c.hp = Math.min(c.maxHp, c.hp + 8 * dt);
   679→    // (d) aviso táctico: 2+ enemigos aggro a <110 px del Portador, cd largo
   680→    mem.lineCd -= dt;
   681→    if (mem.lineCd <= 0 && inCombat && this.state === 'play') {
   682→      let near = 0;
   683→      for (const e of this.enemies) {
   684→        if (!e.dead && e.aggro && Math.hypot(e.x - p.x, e.y - p.y) < 110) near++;
   685→      }
   686→      if (near >= 2) {
   687→        mem.lineCd = 25;
   688→        this.floatAt(c.x, c.y - 24, COMPANION_LINES[mem.lineIdx % COMPANION_LINES.length], '#a8e8c8', 6);
   689→        mem.lineIdx++;
   690→      } else {
   691→        mem.lineCd = 4; // sin rodeo todavía: recheck pronto, sin spamear
   692→      }
   693→    }
   694→  }

   696→  /** Teletransporte de escolta: hueco libre alrededor del Portador (anillos). */
   697→  private companionTeleport(c: Companion, p: Player) {
   698→    let placed = false;
   699→    for (let r = 1; r <= 3 && !placed; r++) {
   700→      for (let i = 0; i < 8 * r && !placed; i++) {
   701→        const a = (i / (8 * r)) * Math.PI * 2;
   702→        const nx = p.x + Math.cos(a) * 18 * r;
   703→        const ny = p.y + Math.sin(a) * 18 * r;
   704→        if (this.boxFree(nx, ny, c.w, c.h)) { c.x = nx; c.y = ny; placed = true; }
   705→      }
   706→    }
   707→    if (!placed) { c.x = p.x - 14; c.y = p.y; } // sin hueco: cae a su sombra
   708→    c.kbVx = 0; c.kbVy = 0;
   709→    this.burst(c.x, c.y - 4, '#8ef0b0', 8, 40);
   710→  }

   713→   * 14-b (auditoría de movimiento): red anti-encallamiento. moveEntity solo
   714→   * acepta un eje si la caja COMPLETA queda libre: si el Portador acaba con
   715→   * su caja dentro de un tile sólido (residuo de knockback antiguo, un save
   716→   * previo a un fix, un edge de época o de un mapa), TODOS los ejes fallan y
   717→   * ningún input lo saca — encallamiento permanente. Este guard (O(1) si
   718→   * estás libre) lo recoloca al tile libre más cercano, misma red que
   719→   * findSafeTile (sin caer en zonas de salida para no viajar por accidente).
   720→   */
   721→  private antiStuck() {
   722→    const p = this.player;
   723→    // antes del primer loadMap (estado intro/título) no hay filas: no-op
   724→    if (!p || !this.rows.length || this.boxFree(p.x, p.y, p.w, p.h)) return;
   725→    const tx = Math.floor(p.x / TILE), ty = Math.floor(p.y / TILE);
   726→    for (let r = 1; r <= 6; r++) {
   727→      for (let dy = -r; dy <= r; dy++) {
   728→        for (let dx = -r; dx <= r; dx++) {
   729→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   730→          const nx = tx + dx, ny = ty + dy;
   731→          if (nx < 0 || ny < 0 || nx >= this.map.w || ny >= this.map.h) continue;
   732→          const wx = nx * TILE + 8, wy = ny * TILE + 8;
   733→          if (this.boxFree(wx, wy, p.w, p.h) && !this.inExitZone(nx, ny)) {
   734→            p.x = wx; p.y = wy;
   735→            this.floatAt(p.x, p.y - 20, 'El canto te aparta del muro', '#c8b0e8', 6);
   736→            return;
   737→          }
   738→        }
   739→      }
   740→    }

   749→  /** ¿(tx,ty) está dentro de alguna zona de salida del mapa actual? */
   750→  private inExitZone(tx: number, ty: number): boolean {
   751→    return this.map.exits.some(ex => {
   752→      if (ex.needPast && this.epoch !== 'pasado') return false;
   753→      return tx >= ex.x && tx < ex.x + ex.w && ty >= ex.y && ty < ex.y + ex.h;
   754→    });
   755→  }

   757→  /**
   758→   * Busca el tile más cercano (búsqueda por anillos) que sea transitable y
   759→   * NO esté dentro de una zona de salida. Garantiza que ningún viaje aterrice
   760→   * en un muro ni en un teletransporte (bucle cripta↔bosque, P0-1).
   761→   */
   762→  private findSafeTile(tx: number, ty: number): [number, number] {
   763→    const ok = (x: number, y: number) =>
   764→      x >= 0 && y >= 0 && x < this.map.w && y < this.map.h &&
   765→      !this.tileSolidAt(x * TILE + 8, y * TILE + 8) &&
   766→      !this.inExitZone(x, y);
   767→    if (ok(tx, ty)) return [tx, ty];
   768→    for (let r = 1; r <= 8; r++) {
   769→      for (let dy = -r; dy <= r; dy++) {
   770→        for (let dx = -r; dx <= r; dx++) {
   771→          if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
   772→          if (ok(tx + dx, ty + dy)) return [tx + dx, ty + dy];
   773→        }
   774→      }
   775→    }
   776→    return [tx, ty]; // sin sitio libre: comportamiento anterior
   777→  }

   779→  boxFree(x: number, y: number, w: number, h: number): boolean {
   780→    const x0 = x - w / 2, x1 = x + w / 2, y0 = y - h / 2 - 2, y1 = y + h / 2;
   781→    return !(this.tileSolidAt(x0, y0) || this.tileSolidAt(x1, y0) || this.tileSolidAt(x0, y1) || this.tileSolidAt(x1, y1));
   782→  }

   784→  moveEntity(e: { x: number; y: number; w: number; h: number }, dx: number, dy: number) {
   785→    // 14-b (armaduras): Placas del Canto — el peso de la armadura frena al
   786→    // Portador (−8%). moveEntity es la ÚNICA puerta por la que pasa TODO
   787→    // movimiento (caminar, rodar, empujones, Paso de Brisa): una sola fuente
   788→    // de verdad, sin tocar update.ts. Los enemigos y la compañera no pesan.

   832→  nearestInteract(): { kind: string; label: string; act: () => void; dist: number } | null {
   833→    const p = this.player;
   834→    if (!p) return null;
   835→    let best: { kind: string; label: string; act: () => void; dist: number } | null = null;
   836→    const consider = (x: number, y: number, kind: string, label: string, act: () => void, r = 30) => {
   837→      const d = Math.hypot(x - p.x, y - p.y);
   838→      if (d < r && (!best || d < best.dist)) best = { kind, label, act, dist: d };
   839→    };
   840→    for (const n of this.npcs) consider(n.x, n.y, 'npc', `Hablar con ${n.dispName}`, () => this.talkTo(n.nid), 34);
   841→    for (const ch of this.map.chests) {
   842→      if (this.openedChests.has(ch.id)) continue;
   843→      if (ch.needPast && this.epoch !== 'pasado') continue;
   844→      consider(ch.x * TILE + 8, ch.y * TILE + 8, 'chest', 'Abrir cofre', () => this.openChest(ch.id), 26);
   845→    }
   846→    for (const pr of this.map.props) {
   847→      if (pr.needPast && this.epoch !== 'pasado') continue;
   848→      if (pr.needPresent && this.epoch !== 'presente') continue;
   849→      const px = pr.x * TILE + 8, py = pr.y * TILE + 8;
   850→      if (pr.kind === 'sanctuary') consider(px, py, 'sanc', 'Santuario del Eco', () => this.openSanctuary(), 26);
   851→      else if (pr.kind === 'forge') consider(px, py, 'forge', 'Forja de Toln', () => this.talkTo('toln'), 28);
   852→      else if (pr.kind === 'fragment') consider(px, py, 'frag', 'Fragmento de Eco', () => this.openDialogue('voz_fragment'), 30);
   853→      else if (pr.kind === 'altarEcho') consider(px, py, 'altar', 'Altar del Eco', () => this.tryTakeEco(), 30);
   854→      else if (pr.kind === 'sign') consider(px, py, 'sign', 'Leer cartel', () => this.readSign(pr.label ?? ''), 28);
   855→      else if (pr.kind === 'lamp' && !this.flags[pr.id]) consider(px, py, 'lamp', 'Encender el Farol del Recuerdo', () => this.lightLamp(pr.id), 28);
   856→    }
   857→    for (const ec of this.map.echoes) {
   858→      if (this.takenEchoes.has(ec.id)) continue;
   859→      consider(ec.x * TILE + 8, ec.y * TILE + 8, 'echo', 'Escuchar eco menor', () => this.takeEchoMinor(ec.id, ec.title, ec.text), 26);
   860→    }
   861→    return best;
   862→  }

  1845→  damagePlayer(dmg: number, fromX: number, fromY: number) {
  1846→    const p = this.player!;
  1847→    // balanceador (12-c): embudo único de todo el daño enemigo (neutro en desafío)
  1848→    dmg = Math.max(1, Math.round(dmg * enemyStatMult(this).dmg));
  1849→    if (p.iframes > 0 || p.rollT > 0 || this.state !== 'play') return;
  1850→    if (p.parryT > 0) {
  1851→      // ¡parada perfecta!
  1852→      audio.sfx('parry');
  1853→      p.res = Math.min(p.maxRes, p.res + 20);
  1854→      p.parryFx = 0.4;
  1855→      this.hitStop = 0.12;
  1856→      this.burst(p.x + (fromX - p.x) * 0.3, p.y - 8 + (fromY - p.y) * 0.3, '#fff8c0', 14, 100);
  1857→      this.floatAt(p.x, p.y - 22, '¡PARADA!', '#fff8c0', 7);
  1858→      // aturde al atacante cercano
  1859→      let best: Enemy | null = null, bd = 34;
  1860→      for (const e of this.enemies) {
  1861→        if (e.dead) continue;
  1862→        const d = Math.hypot(e.x - p.x, e.y - p.y);
  1863→        if (d < bd) { bd = d; best = e; }
  1864→      }
  1865→      if (best) { best.ai = 'aturdido'; best.aiT = 1.6; best.windup = 0; }
  1866→      p.parryT = 0;
  1867→      return;
  1868→    }
  1869→    // 14-b: armadura activa — reduce DESPUÉS del balanceador y ANTES de la
  1870→    // Vigia (contrato armor.ts): dmg_final = max(1, round(bm·(1−red)·(1−vig)))
  1871→    const ared = armorReduction(this);
  1872→    const red = Math.min(0.5, p.attrs.vig * 0.01);
  1873→    const final = Math.max(1, Math.round(dmg * (1 - ared) * (1 - red)));
  1874→    p.hp -= final;
  1875→    p.iframes = 0.6;
  1876→    p.lastHitT = 0.3;
  1877→    // 14-b: Manto de Ecos — refleja 15% SOLO de contacto melé (el origen del
  1878→    // golpe coincide con el cuerpo de un enemigo); los proyectiles vuelan
  1879→    // desde lejos y NO reflejan (decisión documentada en armor.ts)
  1880→    const refl = armorActive(this)?.reflect;
  1881→    if (refl && final > 0) {
  1882→      const src = this.enemies.find(e => !e.dead
  1883→        && Math.abs(e.x - fromX) < e.w / 2 + 6 && Math.abs(e.y - fromY) < e.h / 2 + 8);
  1884→      if (src) {
  1885→        this.damageEnemy(src, Math.max(1, Math.round(final * refl)), 'ninguno', 0);
  1886→        this.burst(src.x, src.y - 6, '#ffe9a0', 8, 70);
  1887→      }
  1888→    }
  1889→    this.shake = 4;
  1890→    audio.sfx('hurt');
  1891→    this.floatAt(p.x, p.y - 16, `-${final}`, '#ff7060');
  1892→    const dx = p.x - fromX, dy = p.y - fromY;
  1893→    const l = Math.max(1, Math.hypot(dx, dy));
  1894→    this.moveEntity(p, (dx / l) * 8, (dy / l) * 8);
  1895→    if (p.hp <= 0) this.playerDied();
  1896→  }
